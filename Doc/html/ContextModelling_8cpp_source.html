60 , m_maxLog2TrDynamicRange (tu.cs->sps->getMaxLog2TrDynamicRange(m_chType))

71 , m_lastShiftY (0)

115void CoeffCodingContext::initSubblock( int SubsetId, bool sigGroupFlag )
116{
117 m_subSetId = SubsetId;
118 m_subSetPos = m_scanCG[m_subSetId].idx;
119 m_subSetPosY = m_subSetPos / m_widthInGroups;
120 m_subSetPosX = m_subSetPos - ( m_subSetPosY * m_widthInGroups );
121 m_minSubPos = m_subSetId << m_log2CGSize;
122 m_maxSubPos = m_minSubPos + ( 1 << m_log2CGSize ) - 1;
123 if( sigGroupFlag )
124 {
125 m_sigCoeffGroupFlag.set ( m_subSetPos );
126 }
127 unsigned CGPosY = m_subSetPosY;
128 unsigned CGPosX = m_subSetPosX;
129 unsigned sigRight = unsigned( ( CGPosX + 1 ) < m_widthInGroups ? m_sigCoeffGroupFlag[ m_subSetPos + 1 ] : false );
130 unsigned sigLower = unsigned( ( CGPosY + 1 ) < m_heightInGroups ? m_sigCoeffGroupFlag[ m_subSetPos + m_widthInGroups ] : false );
131 m_sigGroupCtxId = Ctx::SigCoeffGroup[m_chType]( sigRight | sigLower );
132 unsigned sigLeft = unsigned( CGPosX > 0 ? m_sigCoeffGroupFlag[m_subSetPos - 1 ] : false );
133 unsigned sigAbove = unsigned( CGPosY > 0 ? m_sigCoeffGroupFlag[m_subSetPos - m_widthInGroups] : false );
134 m_sigGroupCtxIdTS = Ctx::TsSigCoeffGroup( sigLeft + sigAbove );
135}

138unsigned DeriveCtx::CtxModeConsFlag( const CodingStructure& cs, Partitioner& partitioner )
139{
140 assert( partitioner.chType == CHANNEL_TYPE_LUMA );
141 const Position pos = partitioner.currArea().blocks[partitioner.chType];
142 const unsigned curSliceIdx = cs.slice->getIndependentSliceIdx();
143 const unsigned curTileIdx = cs.pps->getTileIdx( partitioner.currArea().lumaPos() );
144

153void DeriveCtx::CtxSplit( const CodingStructure& cs, Partitioner& partitioner, unsigned& ctxSpl, unsigned& ctxQt, unsigned& ctxHv, unsigned& ctxHorBt, unsigned& ctxVerBt, bool* _canSplit /*= nullptr */ )
154{
155 const Position pos = partitioner.currArea().blocks[partitioner.chType];
156 const unsigned curSliceIdx = cs.slice->getIndependentSliceIdx();
157 const unsigned curTileIdx = cs.pps->getTileIdx( partitioner.currArea().lumaPos() );
158
159 // get left depth
160 const CodingUnit* cuLeft = cs.getCURestricted( pos.offset( -1, 0 ), pos, curSliceIdx, curTileIdx, partitioner.chType );
161
162 // get above depth
163 const CodingUnit* cuAbove = cs.getCURestricted( pos.offset( 0, -1 ), pos, curSliceIdx, curTileIdx, partitioner.chType );
164
165 bool canSplit[6];
166
167 if( _canSplit == nullptr )
168 {
169 partitioner.canSplit( cs, canSplit[0], canSplit[1], canSplit[2], canSplit[3], canSplit[4], canSplit[5] );
170 }
171 else
172 {
173 memcpy( canSplit, _canSplit, 6 * sizeof( bool ) );
174 }
175
177 // CTX do split (0-8)
179 const unsigned widthCurr = partitioner.currArea().blocks[partitioner.chType].width;
180 const unsigned heightCurr = partitioner.currArea().blocks[partitioner.chType].height;
181
182 ctxSpl = 0;
183
184 if( cuLeft )
185 {
186 const unsigned heightLeft = cuLeft->blocks[partitioner.chType].height;
187 ctxSpl += ( heightLeft < heightCurr ? 1 : 0 );
188 }
189 if( cuAbove )
190 {
191 const unsigned widthAbove = cuAbove->blocks[partitioner.chType].width;
192 ctxSpl += ( widthAbove < widthCurr ? 1 : 0 );
193 }
194
195 unsigned numSplit = 0;
196 if (canSplit[1])
197 {
198 numSplit += 2;
199 }
200 if (canSplit[2])
201 {
202 numSplit += 1;
203 }
204 if (canSplit[3])
205 {
206 numSplit += 1;
207 }
208 if (canSplit[4])
209 {
210 numSplit += 1;
211 }
212 if (canSplit[5])
213 {
214 numSplit += 1;
215 }
216
217 if (numSplit > 0)
218 {
219 numSplit--;
220 }
221
222 ctxSpl += 3 * ( numSplit >> 1 );
223
225 // CTX is qt split (0-5)
227 ctxQt = ( cuLeft && cuLeft->qtDepth > partitioner.currQtDepth ) ? 1 : 0;
228 ctxQt += ( cuAbove && cuAbove->qtDepth > partitioner.currQtDepth ) ? 1 : 0;
229 ctxQt += partitioner.currQtDepth < 2 ? 0 : 3;
230
232 // CTX is ver split (0-4)
234 ctxHv = 0;
235
236 const unsigned numHor = ( canSplit[2] ? 1 : 0 ) + ( canSplit[4] ? 1 : 0 );
237 const unsigned numVer = ( canSplit[3] ? 1 : 0 ) + ( canSplit[5] ? 1 : 0 );
238
239 if( numVer == numHor )
240 {
241 const Area& area = partitioner.currArea().blocks[partitioner.chType];
242
243 const unsigned wAbove = cuAbove ? cuAbove->blocks[partitioner.chType].width : 1;
244 const unsigned hLeft = cuLeft ? cuLeft ->blocks[partitioner.chType].height : 1;
245
246 const unsigned depAbove = area.width / wAbove;
247 const unsigned depLeft = area.height / hLeft;
248
249 if (depAbove == depLeft || !cuLeft || !cuAbove)
250 {
251 ctxHv = 0;
252 }
253 else if (depAbove < depLeft)
254 {
255 ctxHv = 1;
256 }
257 else
258 {
259 ctxHv = 2;
260 }
261 }
262 else if( numVer < numHor )
263 {
264 ctxHv = 3;
265 }
266 else
267 {
268 ctxHv = 4;
269 }
270
272 // CTX is h/v bt (0-3)
274 ctxHorBt = ( partitioner.currMtDepth <= 1 ? 1 : 0 );
275 ctxVerBt = ( partitioner.currMtDepth <= 1 ? 3 : 2 );
276}

381void MergeCtx::setMmvdMergeCandiInfo(PredictionUnit& pu, int candIdx)
382{
383 const Slice &slice = *pu.cs->slice;
384 const int mvShift = MV_FRACTIONAL_BITS_DIFF;
385 const int refMvdCands[8] = { 1 << mvShift , 2 << mvShift , 4 << mvShift , 8 << mvShift , 16 << mvShift , 32 << mvShift, 64 << mvShift , 128 << mvShift };
386 int fPosGroup = 0;
387 int fPosBaseIdx = 0;
388 int fPosStep = 0;
389 int tempIdx = 0;
390 int fPosPosition = 0;
391 Mv tempMv[2];
392
393 tempIdx = candIdx;
394 fPosGroup = tempIdx / (MMVD_BASE_MV_NUM * MMVD_MAX_REFINE_NUM);
395 tempIdx = tempIdx - fPosGroup * (MMVD_BASE_MV_NUM * MMVD_MAX_REFINE_NUM);
396 fPosBaseIdx = tempIdx / MMVD_MAX_REFINE_NUM;
397 tempIdx = tempIdx - fPosBaseIdx * (MMVD_MAX_REFINE_NUM);
398 fPosStep = tempIdx / 4;
399 fPosPosition = tempIdx - fPosStep * (4);
400 int offset = refMvdCands[fPosStep];
401 if ( pu.cu->slice->getPicHeader()->getDisFracMMVD() )
402 {
403 offset <<= 2;
404 }
405 const int refList0 = mmvdBaseMv[fPosBaseIdx][0].refIdx;
406 const int refList1 = mmvdBaseMv[fPosBaseIdx][1].refIdx;
407
408 if ((refList0 != -1) && (refList1 != -1))
409 {
410 const int poc0 = slice.getRefPOC(REF_PIC_LIST_0, refList0);
411 const int poc1 = slice.getRefPOC(REF_PIC_LIST_1, refList1);
412 const int currPoc = slice.getPOC();
413 if (fPosPosition == 0)
414 {
415 tempMv[0] = Mv(offset, 0);
416 }
417 else if (fPosPosition == 1)
418 {
419 tempMv[0] = Mv(-offset, 0);
420 }
421 else if (fPosPosition == 2)
422 {
423 tempMv[0] = Mv(0, offset);
424 }
425 else
426 {
427 tempMv[0] = Mv(0, -offset);
428 }
429 if ((poc0 - currPoc) == (poc1 - currPoc))
430 {
431 tempMv[1] = tempMv[0];
432 }
433 else if (abs(poc1 - currPoc) > abs(poc0 - currPoc))
434 {
435 const int scale = PU::getDistScaleFactor(currPoc, poc0, currPoc, poc1);
436 tempMv[1] = tempMv[0];
437 const bool isL0RefLongTerm = slice.getRefPic(REF_PIC_LIST_0, refList0)->longTerm;
438 const bool isL1RefLongTerm = slice.getRefPic(REF_PIC_LIST_1, refList1)->longTerm;
439 if (isL0RefLongTerm || isL1RefLongTerm)
440 {
441 if ((poc1 - currPoc)*(poc0 - currPoc) > 0)
442 {
443 tempMv[0] = tempMv[1];
444 }
445 else
446 {
447 tempMv[0].set(-1 * tempMv[1].getHor(), -1 * tempMv[1].getVer());
448 }
449 }
450 else
451 tempMv[0] = tempMv[1].scaleMv(scale);
452 }
453 else
454 {
455 const int scale = PU::getDistScaleFactor(currPoc, poc1, currPoc, poc0);
456 const bool isL0RefLongTerm = slice.getRefPic(REF_PIC_LIST_0, refList0)->longTerm;
457 const bool isL1RefLongTerm = slice.getRefPic(REF_PIC_LIST_1, refList1)->longTerm;
458 if (isL0RefLongTerm || isL1RefLongTerm)
459 {
460 if ((poc1 - currPoc)*(poc0 - currPoc) > 0)
461 {
462 tempMv[1] = tempMv[0];
463 }
464 else
465 {
466 tempMv[1].set(-1 * tempMv[0].getHor(), -1 * tempMv[0].getVer());
467 }
468 }
469 else
470 tempMv[1] = tempMv[0].scaleMv(scale);
471 }
472
473 pu.interDir = 3;
474 pu.mv[REF_PIC_LIST_0] = mmvdBaseMv[fPosBaseIdx][0].mv + tempMv[0];
475 pu.refIdx[REF_PIC_LIST_0] = refList0;
476 pu.mv[REF_PIC_LIST_1] = mmvdBaseMv[fPosBaseIdx][1].mv + tempMv[1];
477 pu.refIdx[REF_PIC_LIST_1] = refList1;
478 }
479 else if (refList0 != -1)
480 {
481 if (fPosPosition == 0)
482 {
483 tempMv[0] = Mv(offset, 0);
484 }
485 else if (fPosPosition == 1)
486 {
487 tempMv[0] = Mv(-offset, 0);
488 }
489 else if (fPosPosition == 2)
490 {
491 tempMv[0] = Mv(0, offset);
492 }
493 else
494 {
495 tempMv[0] = Mv(0, -offset);
496 }
497 pu.interDir = 1;
498 pu.mv[REF_PIC_LIST_0] = mmvdBaseMv[fPosBaseIdx][0].mv + tempMv[0];
499 pu.refIdx[REF_PIC_LIST_0] = refList0;
500 pu.mv[REF_PIC_LIST_1] = Mv(0, 0);
501 pu.refIdx[REF_PIC_LIST_1] = -1;
502 }
503 else if (refList1 != -1)
504 {
505 if (fPosPosition == 0)
506 {
507 tempMv[1] = Mv(offset, 0);
508 }
509 else if (fPosPosition == 1)
510 {
511 tempMv[1] = Mv(-offset, 0);
512 }
513 else if (fPosPosition == 2)
514 {
515 tempMv[1] = Mv(0, offset);
516 }
517 else
518 {
519 tempMv[1] = Mv(0, -offset);
520 }
521 pu.interDir = 2;
522 pu.mv[REF_PIC_LIST_0] = Mv(0, 0);
523 pu.refIdx[REF_PIC_LIST_0] = -1;
524 pu.mv[REF_PIC_LIST_1] = mmvdBaseMv[fPosBaseIdx][1].mv + tempMv[1];
525 pu.refIdx[REF_PIC_LIST_1] = refList1;
526 }
527
528 pu.mmvdMergeFlag = true;
529 pu.mmvdMergeIdx = candIdx;
530 pu.mergeFlag = true;
531 pu.regularMergeFlag = true;
532 pu.mergeIdx = candIdx;
533 pu.mergeType = MRG_TYPE_DEFAULT_N;
534 pu.mvd[REF_PIC_LIST_0] = Mv();
535 pu.mvd[REF_PIC_LIST_1] = Mv();
536 pu.mvpIdx[REF_PIC_LIST_0] = NOT_VALID;
537 pu.mvpIdx[REF_PIC_LIST_1] = NOT_VALID;
538 pu.mvpNum[REF_PIC_LIST_0] = NOT_VALID;
539 pu.mvpNum[REF_PIC_LIST_1] = NOT_VALID;
540 pu.cu->imv = mmvdUseAltHpelIf[fPosBaseIdx] ? IMV_HPEL : 0;
541
542 pu.cu->BcwIdx = (interDirNeighbours[fPosBaseIdx] == 3) ? BcwIdx[fPosBaseIdx] : BCW_DEFAULT;
543
544 for (int refList = 0; refList < 2; refList++)
545 {
546 if (pu.refIdx[refList] >= 0)
547 {
548 pu.mv[refList].clipToStorageBitDepth();
549 }
550 }
551
552
553 PU::restrictBiPredMergeCandsOne(pu);
554}

556unsigned DeriveCtx::CtxMipFlag( const CodingUnit& cu )
