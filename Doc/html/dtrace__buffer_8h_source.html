50inline unsigned calcCheckSum( const int iWidth, const int iHeight, const Pel* p, const uint32_t stride, const int bitdepth )
51{
52 unsigned checksum = 0;
53 for( unsigned y = 0; y < iHeight; y++)
54 {
55 for( unsigned x = 0; x < iWidth; x++)
56 {
57 uint8_t xor_mask = (x & 0xff) ^ (y & 0xff) ^ (x >> 8) ^ (y >> 8);
58 checksum = (checksum + ((p[y*stride+x] & 0xff) ^ xor_mask)) & 0xffffffff;
59
60 if(bitdepth > 8)
61 {
62 checksum = (checksum + ((p[y*stride+x]>>8) ^ xor_mask)) & 0xffffffff;
63 }
64 }
65 }
66 return checksum;
67}

69inline unsigned calcCheckSum( const CPelBuf& buf, int bitdepth )
70{
71 return calcCheckSum( buf.width, buf.height, buf.buf, buf.stride, bitdepth );

148inline void dtraceCRC( CDTrace *trace_ctx, DTRACE_CHANNEL channel, const CodingStructure& cs, const CPelUnitBuf& pelUnitBuf, const Area* parea = NULL )
149{
150 const Area& area = parea ? *parea : cs.area.Y();
151 DTRACE( trace_ctx, channel, " CRC: %6lld %3d @(%4d,%4d) [%2dx%2d] ,Checksum(%x %x %x)\n",
152 DTRACE_GET_COUNTER( g_trace_ctx, channel ),
153 cs.slice->getPOC(),
154 area.x, area.y, area.width, area.height,
155 calcCheckSum( pelUnitBuf.bufs[COMPONENT_Y], cs.sps->getBitDepth (CHANNEL_TYPE_LUMA)),

170inline void dtraceMotField( CDTrace *trace_ctx, const PredictionUnit& pu )
171{
172 DTRACE( trace_ctx, D_MOT_FIELD, "PU %d,%d @ %d,%d\n", pu.lwidth(), pu.lheight(), pu.lx(), pu.ly() );
173 const CMotionBuf mb = pu.getMotionBuf();
174 for( uint32_t listIdx = 0; listIdx < 2; listIdx++ )
175 {
176 RefPicList eListIdx = RefPicList( listIdx );
177 for( int y = 0, i = 0; y < pu.lheight(); y += 4 )
178 {
179 for( int x = 0; x < pu.lwidth(); x += 4, i++ )
180 {
181 const MotionInfo &mi = mb.at( x >> 2, y >> 2 );
182 DTRACE( trace_ctx, D_MOT_FIELD, "%d,%d:%d ", mi.mv[eListIdx].getHor(), mi.mv[eListIdx].getVer(), mi.refIdx[eListIdx] );
183 }
184 DTRACE( trace_ctx, D_MOT_FIELD, "\n" );
185 }
186 DTRACE( trace_ctx, D_MOT_FIELD, "\n" );
187 }
