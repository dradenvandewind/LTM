151 void Rom::xInitScanArrays()
152 {
153 if( m_scansInitialized )
154 {
155 return;
156 }
157 ::memset( m_scanId2NbInfoSbbArray, 0, sizeof(m_scanId2NbInfoSbbArray) );
158 ::memset( m_scanId2NbInfoOutArray, 0, sizeof(m_scanId2NbInfoOutArray) );
159 ::memset( m_tuParameters, 0, sizeof(m_tuParameters) );
160
161 uint32_t raster2id[ MAX_CU_SIZE * MAX_CU_SIZE ];
162 ::memset(raster2id, 0, sizeof(raster2id));
163
164 for( int hd = 0; hd <= MAX_CU_DEPTH; hd++ )
165 {
166 for( int vd = 0; vd <= MAX_CU_DEPTH; vd++ )
167 {
168 if( (hd == 0 && vd <= 1) || (hd <= 1 && vd == 0) )
169 {
170 continue;
171 }
172 const uint32_t blockWidth = (1 << hd);
173 const uint32_t blockHeight = (1 << vd);
174 const uint32_t log2CGWidth = g_log2SbbSize[hd][vd][0];
175 const uint32_t log2CGHeight = g_log2SbbSize[hd][vd][1];
176 const uint32_t groupWidth = 1 << log2CGWidth;
177 const uint32_t groupHeight = 1 << log2CGHeight;
178 const uint32_t groupSize = groupWidth * groupHeight;
179 const CoeffScanType scanType = SCAN_DIAG;
180 const SizeType blkWidthIdx = gp_sizeIdxInfo->idxFrom( blockWidth );
181 const SizeType blkHeightIdx = gp_sizeIdxInfo->idxFrom( blockHeight );
182 const ScanElement * scanId2RP = g_scanOrder[SCAN_GROUPED_4x4][scanType][blkWidthIdx][blkHeightIdx];
183 NbInfoSbb*& sId2NbSbb = m_scanId2NbInfoSbbArray[hd][vd];
184 NbInfoOut*& sId2NbOut = m_scanId2NbInfoOutArray[hd][vd];
185 // consider only non-zero-out region
186 const uint32_t blkWidthNZOut = std::min<unsigned>( JVET_C0024_ZERO_OUT_TH, blockWidth );
187 const uint32_t blkHeightNZOut= std::min<unsigned>( JVET_C0024_ZERO_OUT_TH, blockHeight );
188 const uint32_t totalValues = blkWidthNZOut * blkHeightNZOut;
189
190 sId2NbSbb = new NbInfoSbb[ totalValues ];
191 sId2NbOut = new NbInfoOut[ totalValues ];
192
193 for( uint32_t scanId = 0; scanId < totalValues; scanId++ )
194 {
195 raster2id[scanId2RP[scanId].idx] = scanId;
196 }
197
198 for( unsigned scanId = 0; scanId < totalValues; scanId++ )
199 {
200 const int posX = scanId2RP[scanId].x;
201 const int posY = scanId2RP[scanId].y;
202 const int rpos = scanId2RP[scanId].idx;
203 {
204 //===== inside subband neighbours =====
205 NbInfoSbb& nbSbb = sId2NbSbb[ scanId ];
206 const int begSbb = scanId - ( scanId & (groupSize-1) ); // first pos in current subblock
207 int cpos[5];
208
209 cpos[0] = ( posX + 1 < blkWidthNZOut ? ( raster2id[rpos+1 ] < groupSize + begSbb ? raster2id[rpos+1 ] - begSbb : 0 ) : 0 );
210 cpos[1] = ( posX + 2 < blkWidthNZOut ? ( raster2id[rpos+2 ] < groupSize + begSbb ? raster2id[rpos+2 ] - begSbb : 0 ) : 0 );
211 cpos[2] = ( posX + 1 < blkWidthNZOut && posY + 1 < blkHeightNZOut ? ( raster2id[rpos+1+blockWidth] < groupSize + begSbb ? raster2id[rpos+1+blockWidth] - begSbb : 0 ) : 0 );
212 cpos[3] = ( posY + 1 < blkHeightNZOut ? ( raster2id[rpos+ blockWidth] < groupSize + begSbb ? raster2id[rpos+ blockWidth] - begSbb : 0 ) : 0 );
213 cpos[4] = ( posY + 2 < blkHeightNZOut ? ( raster2id[rpos+2*blockWidth] < groupSize + begSbb ? raster2id[rpos+2*blockWidth] - begSbb : 0 ) : 0 );
214
215 for( nbSbb.num = 0; true; )
216 {
217 int nk = -1;
218 for( int k = 0; k < 5; k++ )
219 {
220 if( cpos[k] != 0 && ( nk < 0 || cpos[k] < cpos[nk] ) )
221 {
222 nk = k;
223 }
224 }
225 if( nk < 0 )
226 {
227 break;
228 }
229 nbSbb.inPos[ nbSbb.num++ ] = uint8_t( cpos[nk] );
230 cpos[nk] = 0;
231 }
232 for( int k = nbSbb.num; k < 5; k++ )
233 {
234 nbSbb.inPos[k] = 0;
235 }
236 }
237 {
238 //===== outside subband neighbours =====
239 NbInfoOut& nbOut = sId2NbOut[ scanId ];
240 const int begSbb = scanId - ( scanId & (groupSize-1) ); // first pos in current subblock
241 int cpos[5];
242
243 cpos[0] = ( posX + 1 < blkWidthNZOut ? ( raster2id[rpos+1 ] >= groupSize + begSbb ? raster2id[rpos+1 ] : 0 ) : 0 );
244 cpos[1] = ( posX + 2 < blkWidthNZOut ? ( raster2id[rpos+2 ] >= groupSize + begSbb ? raster2id[rpos+2 ] : 0 ) : 0 );
245 cpos[2] = ( posX + 1 < blkWidthNZOut && posY + 1 < blkHeightNZOut ? ( raster2id[rpos+1+blockWidth] >= groupSize + begSbb ? raster2id[rpos+1+blockWidth] : 0 ) : 0 );
246 cpos[3] = ( posY + 1 < blkHeightNZOut ? ( raster2id[rpos+ blockWidth] >= groupSize + begSbb ? raster2id[rpos+ blockWidth] : 0 ) : 0 );
247 cpos[4] = ( posY + 2 < blkHeightNZOut ? ( raster2id[rpos+2*blockWidth] >= groupSize + begSbb ? raster2id[rpos+2*blockWidth] : 0 ) : 0 );
248
249 for( nbOut.num = 0; true; )
250 {
251 int nk = -1;
252 for( int k = 0; k < 5; k++ )
253 {
254 if( cpos[k] != 0 && ( nk < 0 || cpos[k] < cpos[nk] ) )
255 {
256 nk = k;
257 }
258 }
259 if( nk < 0 )
260 {
261 break;
262 }
263 nbOut.outPos[ nbOut.num++ ] = uint16_t( cpos[nk] );
264 cpos[nk] = 0;
265 }
266 for( int k = nbOut.num; k < 5; k++ )
267 {
268 nbOut.outPos[k] = 0;
269 }
270 nbOut.maxDist = ( scanId == 0 ? 0 : sId2NbOut[scanId-1].maxDist );
271 for( int k = 0; k < nbOut.num; k++ )
272 {
273 if( nbOut.outPos[k] > nbOut.maxDist )
274 {
275 nbOut.maxDist = nbOut.outPos[k];
276 }
277 }
278 }
279 }
280
281 // make it relative
282 for( unsigned scanId = 0; scanId < totalValues; scanId++ )
283 {
284 NbInfoOut& nbOut = sId2NbOut[scanId];
285 const int begSbb = scanId - ( scanId & (groupSize-1) ); // first pos in current subblock
286 for( int k = 0; k < nbOut.num; k++ )
287 {
288 CHECK(begSbb > nbOut.outPos[k], "Position must be past sub block begin");
289 nbOut.outPos[k] -= begSbb;
290 }
291 nbOut.maxDist -= scanId;
292 }
293
294 for( int chId = 0; chId < MAX_NUM_CHANNEL_TYPE; chId++ )
295 {
296 m_tuParameters[hd][vd][chId] = new TUParameters( *this, blockWidth, blockHeight, ChannelType(chId) );
297 }
298 }
299 }
300 m_scansInitialized = true;
301 }

303 void Rom::xUninitScanArrays()
304 {
305 if( !m_scansInitialized )
306 {
307 return;
308 }
309 for( int hd = 0; hd <= MAX_CU_DEPTH; hd++ )
310 {
311 for( int vd = 0; vd <= MAX_CU_DEPTH; vd++ )
312 {
313 NbInfoSbb*& sId2NbSbb = m_scanId2NbInfoSbbArray[hd][vd];
314 NbInfoOut*& sId2NbOut = m_scanId2NbInfoOutArray[hd][vd];
315 if( sId2NbSbb )
316 {
317 delete [] sId2NbSbb;
318 }
319 if( sId2NbOut )
320 {
321 delete [] sId2NbOut;
322 }
323 for( int chId = 0; chId < MAX_NUM_CHANNEL_TYPE; chId++ )
324 {
325 TUParameters*& tuPars = m_tuParameters[hd][vd][chId];
326 if( tuPars )
327 {
328 delete tuPars;
329 }
330 }
331 }
332 }
333 m_scansInitialized = false;

340 TUParameters::TUParameters( const Rom& rom, const unsigned width, const unsigned height, const ChannelType chType )
341 {
342 m_chType = chType;
343 m_width = width;
344 m_height = height;
345 const uint32_t nonzeroWidth = std::min<uint32_t>(JVET_C0024_ZERO_OUT_TH, m_width);
346 const uint32_t nonzeroHeight = std::min<uint32_t>(JVET_C0024_ZERO_OUT_TH, m_height);
347 m_numCoeff = nonzeroWidth * nonzeroHeight;
348 const int log2W = floorLog2( m_width );
349 const int log2H = floorLog2( m_height );
350 m_log2SbbWidth = g_log2SbbSize[ log2W ][ log2H ][0];
351 m_log2SbbHeight = g_log2SbbSize[ log2W ][ log2H ][1];
352 m_log2SbbSize = m_log2SbbWidth + m_log2SbbHeight;
353 m_sbbSize = ( 1 << m_log2SbbSize );
354 m_sbbMask = m_sbbSize - 1;
355 m_widthInSbb = nonzeroWidth >> m_log2SbbWidth;
356 m_heightInSbb = nonzeroHeight >> m_log2SbbHeight;
357 m_numSbb = m_widthInSbb * m_heightInSbb;
358 m_scanType = SCAN_DIAG;
359 SizeType hsbb = gp_sizeIdxInfo->idxFrom( m_widthInSbb );
360 SizeType vsbb = gp_sizeIdxInfo->idxFrom( m_heightInSbb );
361 SizeType hsId = gp_sizeIdxInfo->idxFrom( m_width );
362 SizeType vsId = gp_sizeIdxInfo->idxFrom( m_height );
363 m_scanSbbId2SbbPos = g_scanOrder [ SCAN_UNGROUPED ][ m_scanType ][ hsbb ][ vsbb ];
364 m_scanId2BlkPos = g_scanOrder [ SCAN_GROUPED_4x4 ][ m_scanType ][ hsId ][ vsId ];
365 m_scanId2NbInfoSbb = rom.getNbInfoSbb( log2W, log2H );
366 m_scanId2NbInfoOut = rom.getNbInfoOut( log2W, log2H );
367 m_scanInfo = new ScanInfo[ m_numCoeff ];
368 for( int scanIdx = 0; scanIdx < m_numCoeff; scanIdx++ )
369 {
370 xSetScanInfo( m_scanInfo[scanIdx], scanIdx );
371 }
372 }

375 void TUParameters::xSetScanInfo( ScanInfo& scanInfo, int scanIdx )
376 {
377 scanInfo.chType = m_chType;
378 scanInfo.tuWidth = m_width;
379 scanInfo.tuHeight = m_height;
380 scanInfo.sbbSize = m_sbbSize;
381 scanInfo.numSbb = m_numSbb;
382 scanInfo.scanIdx = scanIdx;
383 scanInfo.rasterPos = m_scanId2BlkPos[scanIdx].idx;
384 scanInfo.sbbPos = m_scanSbbId2SbbPos[scanIdx >> m_log2SbbSize].idx;
385 scanInfo.insidePos = scanIdx & m_sbbMask;
386 scanInfo.eosbb = ( scanInfo.insidePos == 0 );
387 scanInfo.spt = SCAN_ISCSBB;
388 if( scanInfo.insidePos == m_sbbMask && scanIdx > scanInfo.sbbSize && scanIdx < m_numCoeff - 1 )
389 scanInfo.spt = SCAN_SOCSBB;
390 else if( scanInfo.eosbb && scanIdx > 0 && scanIdx < m_numCoeff - m_sbbSize )
391 scanInfo.spt = SCAN_EOCSBB;
392 scanInfo.posX = m_scanId2BlkPos[scanIdx].x;
393 scanInfo.posY = m_scanId2BlkPos[scanIdx].y;
394 if( scanIdx )
395 {
396 const int nextScanIdx = scanIdx - 1;
397 const int diag = m_scanId2BlkPos[nextScanIdx].x + m_scanId2BlkPos[nextScanIdx].y;
398 if( m_chType == CHANNEL_TYPE_LUMA )
399 {
400 scanInfo.sigCtxOffsetNext = ( diag < 2 ? 8 : diag < 5 ? 4 : 0 );
401 scanInfo.gtxCtxOffsetNext = ( diag < 1 ? 16 : diag < 3 ? 11 : diag < 10 ? 6 : 1 );
402 }
403 else
404 {
405 scanInfo.sigCtxOffsetNext = ( diag < 2 ? 4 : 0 );
406 scanInfo.gtxCtxOffsetNext = ( diag < 1 ? 6 : 1 );
407 }
408 scanInfo.nextInsidePos = nextScanIdx & m_sbbMask;
409 scanInfo.nextNbInfoSbb = m_scanId2NbInfoSbb[ nextScanIdx ];
410 if( scanInfo.eosbb )
411 {
412 const int nextSbbPos = m_scanSbbId2SbbPos[nextScanIdx >> m_log2SbbSize].idx;
413 const int nextSbbPosY = nextSbbPos / m_widthInSbb;
414 const int nextSbbPosX = nextSbbPos - nextSbbPosY * m_widthInSbb;
415 scanInfo.nextSbbRight = ( nextSbbPosX < m_widthInSbb - 1 ? nextSbbPos + 1 : 0 );
416 scanInfo.nextSbbBelow = ( nextSbbPosY < m_heightInSbb - 1 ? nextSbbPos + m_widthInSbb : 0 );
417 }
418 }

472 void RateEstimator::xSetLastCoeffOffset( const FracBitsAccess& fracBitsAccess, const TUParameters& tuPars, const TransformUnit& tu, const ComponentID compID )
473 {
474 const ChannelType chType = ( compID == COMPONENT_Y ? CHANNEL_TYPE_LUMA : CHANNEL_TYPE_CHROMA );
475 int32_t cbfDeltaBits = 0;
476 if( compID == COMPONENT_Y && !CU::isIntra(*tu.cu) && !tu.depth )
477 {
478 const BinFracBits bits = fracBitsAccess.getFracBitsArray( Ctx::QtRootCbf() );
479 cbfDeltaBits = int32_t( bits.intBits[1] ) - int32_t( bits.intBits[0] );
480 }
481 else
482 {
483 BinFracBits bits;
484 bool prevLumaCbf = false;
485 bool lastCbfIsInferred = false;
486 bool useIntraSubPartitions = tu.cu->ispMode && isLuma(chType);
487 if( useIntraSubPartitions )
488 {
489 bool rootCbfSoFar = false;
490 bool isLastSubPartition = CU::isISPLast(*tu.cu, tu.Y(), compID);
491 uint32_t nTus = tu.cu->ispMode == HOR_INTRA_SUBPARTITIONS ? tu.cu->lheight() >> floorLog2(tu.lheight()) : tu.cu->lwidth() >> floorLog2(tu.lwidth());
492 if( isLastSubPartition )
493 {
494 TransformUnit* tuPointer = tu.cu->firstTU;
495 for( int tuIdx = 0; tuIdx < nTus - 1; tuIdx++ )
496 {
497 rootCbfSoFar |= TU::getCbfAtDepth(*tuPointer, COMPONENT_Y, tu.depth);
498 tuPointer = tuPointer->next;
499 }
500 if( !rootCbfSoFar )
501 {
502 lastCbfIsInferred = true;
503 }
504 }
505 if( !lastCbfIsInferred )
506 {
507 prevLumaCbf = TU::getPrevTuCbfAtDepth(tu, compID, tu.depth);
508 }
509 bits = fracBitsAccess.getFracBitsArray(Ctx::QtCbf[compID](DeriveCtx::CtxQtCbf(compID, prevLumaCbf, true)));
510 }
511 else
512 {
513 bits = fracBitsAccess.getFracBitsArray(Ctx::QtCbf[compID](DeriveCtx::CtxQtCbf(compID, tu.cbf[COMPONENT_Cb])));
514 }
515 cbfDeltaBits = lastCbfIsInferred ? 0 : int32_t(bits.intBits[1]) - int32_t(bits.intBits[0]);
516 }
517
518 static const unsigned prefixCtx[] = { 0, 0, 0, 3, 6, 10, 15, 21 };
519 uint32_t ctxBits [ LAST_SIGNIFICANT_GROUPS ];
520 for( unsigned xy = 0; xy < 2; xy++ )
521 {
522 int32_t bitOffset = ( xy ? cbfDeltaBits : 0 );
523 int32_t* lastBits = ( xy ? m_lastBitsY : m_lastBitsX );
524 const unsigned size = ( xy ? tuPars.m_height : tuPars.m_width );
525 const unsigned log2Size = ceilLog2( size );
526 const bool useYCtx = ( xy != 0 );
527 const CtxSet& ctxSetLast = ( useYCtx ? Ctx::LastY : Ctx::LastX )[ chType ];
528 const unsigned lastShift = ( compID == COMPONENT_Y ? (log2Size+1)>>2 : Clip3<unsigned>(0,2,size>>3) );
529 const unsigned lastOffset = ( compID == COMPONENT_Y ? ( prefixCtx[log2Size] ) : 0 );
530 uint32_t sumFBits = 0;
531 unsigned maxCtxId = g_uiGroupIdx[std::min<unsigned>(JVET_C0024_ZERO_OUT_TH, size) - 1];
532 for( unsigned ctxId = 0; ctxId < maxCtxId; ctxId++ )
533 {
534 const BinFracBits bits = fracBitsAccess.getFracBitsArray( ctxSetLast( lastOffset + ( ctxId >> lastShift ) ) );
535 ctxBits[ ctxId ] = sumFBits + bits.intBits[0] + ( ctxId>3 ? ((ctxId-2)>>1)<<SCALE_BITS : 0 ) + bitOffset;
536 sumFBits += bits.intBits[1];
537 }
538 ctxBits [ maxCtxId ] = sumFBits + ( maxCtxId>3 ? ((maxCtxId-2)>>1)<<SCALE_BITS : 0 ) + bitOffset;

652 inline int ceil_log2(uint64_t x)
653 {
654 static const uint64_t t[6] = { 0xFFFFFFFF00000000ull, 0x00000000FFFF0000ull, 0x000000000000FF00ull, 0x00000000000000F0ull, 0x000000000000000Cull, 0x0000000000000002ull };
655 int y = (((x & (x - 1)) == 0) ? 0 : 1);
656 int j = 32;
657 for( int i = 0; i < 6; i++)
658 {
659 int k = (((x & t[i]) == 0) ? 0 : j);
660 y += k;
661 x >>= k;
662 j >>= 1;
663 }
664 return y;

666 void Quantizer::initQuantBlock(const TransformUnit& tu, const ComponentID compID, const QpParam& cQP, const double lambda, int gValue = -1)
667 {
668 CHECKD( lambda <= 0.0, "Lambda must be greater than 0" );
669
670 const int qpDQ = cQP.Qp(tu.mtsIdx[compID] == MTS_SKIP) + 1;
671 const int qpPer = qpDQ / 6;
672 const int qpRem = qpDQ - 6 * qpPer;
673 const SPS& sps = *tu.cs->sps;
674 const CompArea& area = tu.blocks[ compID ];
675 const ChannelType chType = toChannelType( compID );
676 const int channelBitDepth = sps.getBitDepth( chType );
677 const int maxLog2TrDynamicRange = sps.getMaxLog2TrDynamicRange( chType );
678 const int nomTransformShift = getTransformShift( channelBitDepth, area.size(), maxLog2TrDynamicRange );
679 const bool clipTransformShift = ( tu.mtsIdx[compID] == MTS_SKIP && sps.getSpsRangeExtension().getExtendedPrecisionProcessingFlag());
680 const bool needsSqrt2ScaleAdjustment = TU::needsSqrt2Scale(tu, compID);
681 const int transformShift = ( clipTransformShift ? std::max<int>( 0, nomTransformShift ) : nomTransformShift ) + (needsSqrt2ScaleAdjustment?-1:0);
682 // quant parameters
683 m_QShift = QUANT_SHIFT - 1 + qpPer + transformShift;
684 m_QAdd = -( ( 3 << m_QShift ) >> 1 );
685 int invShift = IQUANT_SHIFT + 1 - qpPer - transformShift;
686 m_QScale = g_quantScales[needsSqrt2ScaleAdjustment?1:0][ qpRem ];
687 const unsigned qIdxBD = std::min<unsigned>( maxLog2TrDynamicRange + 1, 8*sizeof(Intermediate_Int) + invShift - IQUANT_SHIFT - 1 );
688 m_maxQIdx = ( 1 << (qIdxBD-1) ) - 4;
689 m_thresLast = TCoeff((int64_t(4) << m_QShift));
690 m_thresSSbb = TCoeff((int64_t(3) << m_QShift));
691 // distortion calculation parameters
692 const int64_t qScale = (gValue==-1) ? m_QScale : gValue;
693 const int nomDShift =
694 SCALE_BITS - 2 * (nomTransformShift + DISTORTION_PRECISION_ADJUSTMENT(channelBitDepth)) + m_QShift + (needsSqrt2ScaleAdjustment ? 1 : 0);
695 const double qScale2 = double( qScale * qScale );
696 const double nomDistFactor = ( nomDShift < 0 ? 1.0/(double(int64_t(1)<<(-nomDShift))*qScale2*lambda) : double(int64_t(1)<<nomDShift)/(qScale2*lambda) );
697 const int64_t pow2dfShift = (int64_t)( nomDistFactor * qScale2 ) + 1;
698 const int dfShift = ceil_log2( pow2dfShift );
699 m_DistShift = 62 + m_QShift - 2*maxLog2TrDynamicRange - dfShift;
700 m_DistAdd = (int64_t(1) << m_DistShift) >> 1;
701 m_DistStepAdd = (int64_t)( nomDistFactor * double(int64_t(1)<<(m_DistShift+m_QShift)) + .5 );
702 m_DistOrgFact = (int64_t)( nomDistFactor * double(int64_t(1)<<(m_DistShift+1 )) + .5 );
703 }

705 void Quantizer::dequantBlock( const TransformUnit& tu, const ComponentID compID, const QpParam& cQP, CoeffBuf& recCoeff, bool enableScalingLists, int* piDequantCoef) const
706 {
707
708 //----- set basic parameters -----
709 const CompArea& area = tu.blocks[ compID ];
710 const int numCoeff = area.area();
711 const SizeType hsId = gp_sizeIdxInfo->idxFrom( area.width );
712 const SizeType vsId = gp_sizeIdxInfo->idxFrom( area.height );
713 const CoeffScanType scanType = SCAN_DIAG;
714 const ScanElement *scan = g_scanOrder[SCAN_GROUPED_4x4][scanType][hsId][vsId];
715 const TCoeff* qCoeff = tu.getCoeffs( compID ).buf;
716 TCoeff* tCoeff = recCoeff.buf;
717
718 //----- reset coefficients and get last scan index -----
719 ::memset( tCoeff, 0, numCoeff * sizeof(TCoeff) );
720 int lastScanIdx = -1;
721 for( int scanIdx = numCoeff - 1; scanIdx >= 0; scanIdx-- )
722 {
723 if (qCoeff[scan[scanIdx].idx])
724 {
725 lastScanIdx = scanIdx;
726 break;
727 }
728 }
729 if( lastScanIdx < 0 )
730 {
731 return;
732 }
733
734 //----- set dequant parameters -----
735 const int qpDQ = cQP.Qp(tu.mtsIdx[compID] == MTS_SKIP) + 1;
736 const int qpPer = qpDQ / 6;
737 const int qpRem = qpDQ - 6 * qpPer;
738 const SPS& sps = *tu.cs->sps;
739 const ChannelType chType = toChannelType( compID );
740 const int channelBitDepth = sps.getBitDepth( chType );
741 const int maxLog2TrDynamicRange = sps.getMaxLog2TrDynamicRange( chType );
742 const TCoeff minTCoeff = -( 1 << maxLog2TrDynamicRange );
743 const TCoeff maxTCoeff = ( 1 << maxLog2TrDynamicRange ) - 1;
744 const int nomTransformShift = getTransformShift( channelBitDepth, area.size(), maxLog2TrDynamicRange );
745 const bool clipTransformShift = ( tu.mtsIdx[compID] == MTS_SKIP && sps.getSpsRangeExtension().getExtendedPrecisionProcessingFlag());
746 const bool needsSqrt2ScaleAdjustment = TU::needsSqrt2Scale(tu, compID);
747 const int transformShift = ( clipTransformShift ? std::max<int>( 0, nomTransformShift ) : nomTransformShift ) + (needsSqrt2ScaleAdjustment?-1:0);
748 Intermediate_Int shift = IQUANT_SHIFT + 1 - qpPer - transformShift + (enableScalingLists ? LOG2_SCALING_LIST_NEUTRAL_VALUE : 0);
749 Intermediate_Int invQScale = g_invQuantScales[needsSqrt2ScaleAdjustment?1:0][ qpRem ];
750 Intermediate_Int add = (shift < 0) ? 0 : ((1 << shift) >> 1);
751 //----- dequant coefficients -----
752 for( int state = 0, scanIdx = lastScanIdx; scanIdx >= 0; scanIdx-- )
753 {
754 const unsigned rasterPos = scan[scanIdx].idx;
755 const TCoeff& level = qCoeff[ rasterPos ];
756 if( level )
757 {
758 if (enableScalingLists)
759 invQScale = piDequantCoef[rasterPos];//scalingfactor*levelScale
760 if (shift < 0 && (enableScalingLists || scanIdx == lastScanIdx))
761 {
762 invQScale <<= -shift;
763 }
764 Intermediate_Int qIdx = ( level << 1 ) + ( level > 0 ? -(state>>1) : (state>>1) );
765 int64_t nomTCoeff = ((int64_t)qIdx * (int64_t)invQScale + add) >> ((shift < 0) ? 0 : shift);
766 tCoeff[rasterPos] = (TCoeff)Clip3<int64_t>(minTCoeff, maxTCoeff, nomTCoeff);
767 }
768 state = ( 32040 >> ((state<<2)+((level&1)<<1)) ) & 3; // the 16-bit value "32040" represent the state transition table
769 }
770 }

772 inline void Quantizer::preQuantCoeff(const TCoeff absCoeff, PQData *pqData, TCoeff quanCoeff) const
773 {
774 int64_t scaledOrg = int64_t( absCoeff ) * quanCoeff;
775 TCoeff qIdx = std::max<TCoeff>( 1, std::min<TCoeff>( m_maxQIdx, TCoeff( ( scaledOrg + m_QAdd ) >> m_QShift ) ) );
776 int64_t scaledAdd = qIdx * m_DistStepAdd - scaledOrg * m_DistOrgFact;
777 PQData& pq_a = pqData[ qIdx & 3 ];
778 pq_a.deltaDist = ( scaledAdd * qIdx + m_DistAdd ) >> m_DistShift;
779 pq_a.absLevel = ( ++qIdx ) >> 1;
780 scaledAdd += m_DistStepAdd;
781 PQData& pq_b = pqData[ qIdx & 3 ];
782 pq_b.deltaDist = ( scaledAdd * qIdx + m_DistAdd ) >> m_DistShift;
783 pq_b.absLevel = ( ++qIdx ) >> 1;
784 scaledAdd += m_DistStepAdd;
785 PQData& pq_c = pqData[ qIdx & 3 ];
786 pq_c.deltaDist = ( scaledAdd * qIdx + m_DistAdd ) >> m_DistShift;
787 pq_c.absLevel = ( ++qIdx ) >> 1;
788 scaledAdd += m_DistStepAdd;
789 PQData& pq_d = pqData[ qIdx & 3 ];
790 pq_d.deltaDist = ( scaledAdd * qIdx + m_DistAdd ) >> m_DistShift;
791 pq_d.absLevel = ( ++qIdx ) >> 1;
792 }

1549 }
