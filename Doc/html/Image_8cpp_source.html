126unsigned ImageDescription::plane_size(unsigned int plane) const {

131// Total number of bytes to store an image
132unsigned ImageDescription::byte_size() const {
133 unsigned r = 0;
134 for (unsigned p = 0; p < num_planes(); ++p)
135 r += plane_size(p);

140// Return byte offset of pixel row within buffer
141unsigned ImageDescription::row_offset(unsigned int plane, unsigned row) const {
142 assert(plane < num_planes());
143 assert(row < height_);
144
145 unsigned r = 0;
146 for (unsigned p = 0; p < plane; ++p)
147 r += plane_size(p);
148
149 return r + row * row_stride(plane);
150}

152// Return byte size of pixel row (may be less than stride due to alignment)
153unsigned ImageDescription::row_size(unsigned int plane) const {
154 assert(plane < num_planes());
155 const ImageDescInfo &info = imagedesc();
156
157 // Width of plane, rounded up to component units
158 return info.bytes * (width_ >> imagedesc().width_plane_shift[plane]);
159}
160
161// Return byte stride of plane
162unsigned ImageDescription::row_stride(unsigned int plane) const {
163 assert(plane < num_planes());
164 unsigned a = 1;
165
166 // Align stride to required power-of-2
167 assert(((a + 1) & a) == 0);
168 return (row_size(plane) + a) & ~a;
169}

205// Make a new image description with different bit depth
206ImageDescription ImageDescription::with_depth(unsigned depth) const {
207 unsigned fmt = (format_ - IMAGE_FORMAT_YUV420P8) % (IMAGE_FORMAT_YUV420P10 - IMAGE_FORMAT_YUV420P8);
208 fmt += IMAGE_FORMAT_YUV420P8;
209
210 switch (depth) {
211 case 8:
212 break;
213 case 10:
214 fmt += IMAGE_FORMAT_YUV420P10 - IMAGE_FORMAT_YUV420P8;
215 break;
216 case 12:
217 fmt += IMAGE_FORMAT_YUV420P12 - IMAGE_FORMAT_YUV420P8;
218 break;
219 case 14:
220 fmt += IMAGE_FORMAT_YUV420P14 - IMAGE_FORMAT_YUV420P8;
221 break;
222 case 16:
223 fmt += IMAGE_FORMAT_YUV420P16 - IMAGE_FORMAT_YUV420P8;
224 break;
225 default:
226 CHECK(0);
227 }
228
229 return ImageDescription(static_cast<ImageFormat>(fmt), width(), height());
230}
