57inline void dtracePicComp( DTRACE_CHANNEL channel, CodingStructure& cs, const CPelUnitBuf& pelUnitBuf, ComponentID compId )
58{
59 if( !g_trace_ctx ) return;
60 if( pelUnitBuf.chromaFormat == CHROMA_400 && compId != COMPONENT_Y ) return;
61 const Pel* piSrc = pelUnitBuf.bufs[compId].buf;
62 uint32_t uiStride = pelUnitBuf.bufs[compId].stride;
63 uint32_t uiWidth = pelUnitBuf.bufs[compId].width;
64 uint32_t uiHeight = pelUnitBuf.bufs[compId].height;
65 uint32_t uiChromaScaleX = getComponentScaleX( compId, pelUnitBuf.chromaFormat );
66 uint32_t uiChromaScaleY = getComponentScaleY( compId, pelUnitBuf.chromaFormat );
67
68 DTRACE ( g_trace_ctx, channel, "\n%s: poc = %d, size=%dx%d\n\n", g_trace_ctx->getChannelName(channel), cs.slice->getPOC(), uiWidth, uiHeight );
69 DTRACE_FRAME_BLOCKWISE( g_trace_ctx, channel, piSrc, uiStride, uiWidth, uiHeight, cs.sps->getMaxCUWidth() >> uiChromaScaleX, cs.sps->getMaxCUHeight() >> uiChromaScaleY);
70}

74inline void dtraceModeCost(CodingStructure &cs, double lambda)
75{
76 CHECK( cs.cus.size() != 1, "Only the cost for a single CU can be show with dtraceModeCost!" );
77
78 Distortion tempDist = cs.dist;
79
80#if OLD_RDCOST
81 uint64_t tempBits = cs.fracBits >> SCALE_BITS;
82 uint64_t tempCost = (uint64_t)(cs.dist + (double)tempBits * lambda);
83#else
84 uint64_t tempBits = cs.fracBits;
85 uint64_t tempCost = (uint64_t)cs.cost;
86#endif
87
88 if( cs.cost == MAX_DOUBLE )
89 {
90 tempCost = 0;
91 tempBits = 0;
92 tempDist = 0;
93 }
94
95 bool isIntra = CU::isIntra( *cs.cus.front() );
96 int intraModeL = isIntra ? cs.pus.front()->intraDir[0] : 0;
97 int intraModeC = isIntra ? cs.pus.front()->intraDir[1] : 0;
98 if (isIntra && intraModeC == DM_CHROMA_IDX)
99 intraModeC = 68;
100 int imvVal = 0;
101 imvVal = cs.cus[0]->imv;
102 DTRACE( g_trace_ctx, D_MODE_COST, "ModeCost: %6lld %3d @(%4d,%4d) [%2dx%2d] %d (qp%d,pm%d,skip%d,mrg%d,fruc%d,obmc%d,ic%d,imv%d,affn%d,%d,%d) tempCS = %lld (%d,%d)\n",
103 DTRACE_GET_COUNTER( g_trace_ctx, D_MODE_COST ),
104 cs.slice->getPOC(),
105 cs.area.Y().x, cs.area.Y().y,
106 cs.area.Y().width, cs.area.Y().height,
107 cs.cus[0]->qtDepth,
108 cs.cus[0]->qp,
109 cs.cus[0]->predMode,
110 cs.cus[0]->skip,
111 cs.pus[0]->mergeFlag,
112 0, 0,
113 imvVal,
114 0, 0,
115 intraModeL, intraModeC,
116 tempCost, tempBits, tempDist );
117}

119inline void dtraceBestMode(CodingStructure *&tempCS, CodingStructure *&bestCS, double lambda)
120{
121 bool bSplitCS = tempCS->cus.size() > 1 || bestCS->cus.size() > 1;
122 ChannelType chType = tempCS->cus.back()->chType;
123
124 // if the last CU does not align with the CS, we probably are at the edge
125 bSplitCS |= tempCS->cus.back()->blocks[chType].bottomRight() != tempCS->area.blocks[chType].bottomRight();
126
127 Distortion tempDist = tempCS->dist;
128
129#if OLD_RDCOST
130 uint64_t tempBits = tempCS->fracBits >> SCALE_BITS;
131 uint64_t bestBits = bestCS->fracBits >> SCALE_BITS;
132 uint64_t tempCost = (uint64_t)(tempCS->dist + (double)tempBits * lambda);
133 uint64_t bestCost = (uint64_t)(bestCS->dist + (double)bestBits * lambda);
134#else
135 uint64_t tempBits = tempCS->fracBits;
136 uint64_t bestBits = bestCS->fracBits;
137 uint64_t tempCost = (uint64_t)tempCS->cost;
138 uint64_t bestCost = (uint64_t)bestCS->cost;
139#endif
140
141 if( tempCS->cost == MAX_DOUBLE )
142 {
143 tempCost = 0;
144 tempBits = 0;
145 tempDist = 0;
146 }
147
148 bool isIntra = CU::isIntra( *tempCS->cus[0] );
149 int intraModeL = isIntra ? tempCS->pus[0]->intraDir[0] : 0;
150 int intraModeC = isIntra ? tempCS->pus[0]->intraDir[1] : 0;
151
152 if(!bSplitCS)
153 {
154 DTRACE( g_trace_ctx, D_BEST_MODE, "CheckModeCost: %6lld %3d @(%4d,%4d) [%2dx%2d] %d (%d,%d,%2d,%d,%d) tempCS = %lld (%d,%d), bestCS = %lld (%d,%d): --> choose %s\n",
155 DTRACE_GET_COUNTER( g_trace_ctx, D_BEST_MODE ),
156 tempCS->slice->getPOC(),
157 tempCS->area.Y().x, tempCS->area.Y().y,
158 tempCS->area.Y().width, tempCS->area.Y().height,
159 tempCS->cus[0]->qtDepth,
160 tempCS->cus[0]->qp,
161 tempCS->cus[0]->predMode,
162 tempCS->pus[0]->mergeFlag,
163 intraModeL, intraModeC,
164 tempCost, tempBits, tempDist,
165 bestCost, bestBits, bestCS->dist,
166 tempCS->cost < bestCS->cost ? "TEMP" : "BEST" );
167 }
168 else
169 {
170 DTRACE( g_trace_ctx, D_BEST_MODE, "CheckModeSplitCost: %6lld %3d @(%4d,%4d) [%2dx%2d] -------------------------- tempCS = %lld (%d,%d), bestCS = %lld (%d,%d): --> choose %s\n",
171 DTRACE_GET_COUNTER( g_trace_ctx, D_BEST_MODE ),
172 tempCS->slice->getPOC(),
173 tempCS->area.Y().x, tempCS->area.Y().y,
174 tempCS->area.Y().width, tempCS->area.Y().height,
175 tempCost, tempBits, tempDist,
176 bestCost, bestBits, bestCS->dist,
177 tempCS->cost < bestCS->cost ? "TEMP STRUCTURE" : "BEST STRUCTURE");
178 }
179}
