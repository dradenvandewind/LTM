102Encoder::Encoder(const ImageDescription &image_description, const Parameters &parameters) {
103 memset(&encoder_configuration_, 0, sizeof(encoder_configuration_));
104 memset(&configuration_, 0, sizeof(configuration_));
105
106 // Build default parameters
107 auto pb = Parameters::build();
108 Parameters defaults = parameter_defaults_global(parameters);
109
110 // Set base QP param as calculated before....
111 encoder_configuration_.base_qp = parameters["qp"].get<unsigned>(defaults);
112
113 update_global_configuration(parameters, defaults, image_description);
114
115 // Build image descriptions
116 src_image_description_ = image_description;
117 base_image_description_ = ImageDescription(image_description.format(), dimensions_.base_width(), dimensions_.base_height())
118 .with_depth(configuration_.global_configuration.base_depth);
119 intermediate_image_description_ = ImageDescription(
120 configuration_.global_configuration.level1_depth_flag ? src_image_description_.format() : base_image_description_.format(),
121 dimensions_.intermediate_width(), dimensions_.intermediate_height());
122
123 enhancement_image_description_ =
124 ImageDescription(image_description.format(), dimensions_.conformant_width(), dimensions_.conformant_height());
125}

127void Encoder::initialise_config(const Parameters &parameters, std::vector<std::unique_ptr<Image>> &src_image) {
128 // Build default parameters
129 auto pb = Parameters::build();
130 Parameters defaults = parameter_defaults_picture(parameters, src_image, encoder_configuration_.base_qp);
131
132 // Update Transform Type
133 if (!defaults["encoding_transform_type"].empty()) {
134 if (defaults["encoding_transform_type"].get<std::string>() == "dd") {
135 configuration_.global_configuration.transform_block_size = 2;
136 configuration_.global_configuration.num_residual_layers = 4;
137
138 // Update image descriptions (conformance window might change with new transform)
139 dimensions_.set(configuration_, src_image_description_.width(), src_image_description_.height());
140 configuration_.global_configuration.resolution_width = dimensions_.conformant_width();
141 configuration_.global_configuration.resolution_height = dimensions_.conformant_height();
142 base_image_description_ = base_image_description_.with_size(dimensions_.base_width(), dimensions_.base_height());
143 intermediate_image_description_ =
144 intermediate_image_description_.with_size(dimensions_.intermediate_width(), dimensions_.intermediate_height());
145 enhancement_image_description_ =
146 enhancement_image_description_.with_size(dimensions_.conformant_width(), dimensions_.conformant_height());
147
148 } else if (defaults["encoding_transform_type"].get<std::string>() == "dds") {
149 configuration_.global_configuration.transform_block_size = 4;
150 configuration_.global_configuration.num_residual_layers = 16;
151 }
152 }
153
154 update_sequence_configuration(parameters, defaults, src_image_description_);
155 update_encoder_configuration(parameters, defaults);
156 update_picture_configuration(parameters, defaults);
157
158 const unsigned step_width_2 = defaults["cq_step_width_loq_2"].get<unsigned>(MAX_STEP_WIDTH);
159 if (step_width_2 != (unsigned)MAX_STEP_WIDTH) {
160 configuration_.picture_configuration.step_width_loq[LOQ_LEVEL_2] =
161 clamp(step_width_2, (unsigned)MIN_STEP_WIDTH, (unsigned)MAX_STEP_WIDTH);
162 configuration_.picture_configuration.step_width_loq_orig[LOQ_LEVEL_2] =
163 clamp(step_width_2, (unsigned)MIN_STEP_WIDTH, (unsigned)MAX_STEP_WIDTH);
164 }
165 if (parameters["temporal_step_width_modifier"].empty()) {
166 configuration_.global_configuration.temporal_step_width_modifier =
167 defaults["temporal_step_width_modifier"].get<unsigned>(48);
168 }
169 if (parameters["encoding_upsample"].empty()) {
170 const Upsample upsample = defaults["encoding_upsample"].get_enum<Upsample>(Upsample_ModifiedCubic);
171 configuration_.global_configuration.upsample = upsample;
172 if (upsample == Upsample_AdaptiveCubic) {
173 defaults["upsampling_coefficients"].get_vector<unsigned>(
174 configuration_.global_configuration.upsampling_coefficients,
175 ARRAY_SIZE(configuration_.global_configuration.upsampling_coefficients));
176 }
177 }
178
179 // Initialize quantization matrix
180 std::fill_n(&quant_matrix_coeffs_[0][0][0], MAX_NUM_PLANES * MAX_NUM_LOQS * MAX_NUM_LAYERS, -1);
181
182 // Report final configurations
183 REPORT("#### #### encoding_transform_type %4d", configuration_.global_configuration.transform_block_size);
184 REPORT("#### #### num_processed_planes %4d", configuration_.global_configuration.num_processed_planes);
185 REPORT("#### #### dequant_offset_mode %4d", configuration_.picture_configuration.dequant_offset_mode);
186 REPORT("#### #### dequant_offset %4d", configuration_.picture_configuration.dequant_offset);
187 REPORT("#### #### temporal_cq_sw_multiplier %4d", encoder_configuration_.temporal_cq_sw_multiplier);
188 REPORT("#### #### temporal_step_width_modifier %4d", configuration_.global_configuration.temporal_step_width_modifier);
189 // REPORT("#### #### priority_mode %s", parameters["priority_mode"].get<std::string>().c_str());
190 REPORT("#### #### encoding_upsample %4d", configuration_.global_configuration.upsample);
191 REPORT("#### #### upsampling_coefficients %4d %4d %4d %4d", configuration_.global_configuration.upsampling_coefficients[0],
192 configuration_.global_configuration.upsampling_coefficients[1],
193 configuration_.global_configuration.upsampling_coefficients[2],
194 configuration_.global_configuration.upsampling_coefficients[3]);
195 REPORT("#### #### SW1 %4d", configuration_.picture_configuration.step_width_loq_orig[LOQ_LEVEL_1]);
196 REPORT("#### #### SW2 %4d", configuration_.picture_configuration.step_width_loq_orig[LOQ_LEVEL_2]);
197 REPORT("#### #### quant_matrix_mode %4d", configuration_.picture_configuration.quant_matrix_mode);
198 REPORT("#### #### qm_coefficient_1 %4d %4d %4d %4d", configuration_.picture_configuration.qm_coefficient_1_par[0],
199 configuration_.picture_configuration.qm_coefficient_1_par[1],
200 configuration_.picture_configuration.qm_coefficient_1_par[2],
201 configuration_.picture_configuration.qm_coefficient_1_par[3]);
202 REPORT("#### #### qm_coefficient_2 %4d %4d %4d %4d", configuration_.picture_configuration.qm_coefficient_2_par[0],
203 configuration_.picture_configuration.qm_coefficient_2_par[1],
204 configuration_.picture_configuration.qm_coefficient_2_par[2],
205 configuration_.picture_configuration.qm_coefficient_2_par[3]);
206}

313void Encoder::update_sequence_configuration(const Parameters &p, const Parameters &d, const ImageDescription &image_description) {
314 // Signalled
315 // clang-format off
316 configuration_.sequence_configuration.profile_idc = p["profile_idc"].get_enum<Profile>(Profile_Auto);
317 configuration_.sequence_configuration.level_idc = p["level_idc"].get<unsigned>(0);
318 configuration_.sequence_configuration.sublevel_idc = p["sublevel_idc"].get<unsigned>(1);
319 configuration_.sequence_configuration.conformance_window = p["conformance_window"].get<bool>(false);
320 configuration_.sequence_configuration.extended_profile_idc = p["extended_profile_idc"].get<unsigned>(0);
321 configuration_.sequence_configuration.extended_level_idc = p["extended_level_idc"].get<unsigned>(0);
322 configuration_.sequence_configuration.conf_win_left_offset = p["conf_win_left_offset"].get<unsigned>(0);
323 configuration_.sequence_configuration.conf_win_right_offset = p["conf_win_right_offset"].get<unsigned>(0);
324 configuration_.sequence_configuration.conf_win_top_offset = p["conf_win_top_offset"].get<unsigned>(0);
325 configuration_.sequence_configuration.conf_win_bottom_offset = p["conf_win_bottom_offset"].get<unsigned>(0);
326 // clang-format on
327
328 // Check for correct profile and level
329 if (configuration_.sequence_configuration.profile_idc == Profile_Auto) {
330 if (image_description.colourspace() == Colourspace_YUV422 || image_description.colourspace() == Colourspace_YUV444) {
331 configuration_.sequence_configuration.profile_idc = Profile_Main444;
332 INFO("Using Main 4:4:4 Profile with chroma_format_idc %u.", (unsigned)image_description.colourspace());
333 } else {
334 configuration_.sequence_configuration.profile_idc = Profile_Main;
335 INFO("Using Main Profile with chroma_format_idc %u.", (unsigned)image_description.colourspace());
336 }
337 } else if (configuration_.sequence_configuration.profile_idc == Profile_Main &&
338 (image_description.colourspace() == Colourspace_YUV422 || image_description.colourspace() == Colourspace_YUV444))
339 ERR("Colourspace not supported with selected profile (see Annex A.3)");
340 const unsigned fps = p["fps"].get<unsigned>(50);
341 const unsigned long output_sample_rate =
342 configuration_.global_configuration.resolution_width * configuration_.global_configuration.resolution_height * fps;
343 if (configuration_.sequence_configuration.level_idc == 0) {
344 // Choose minimum level automatically
345 if (output_sample_rate <= MAX_OUTPUT_RATE_LEVEL_1)
346 configuration_.sequence_configuration.level_idc = 1;
347 else if (output_sample_rate <= MAX_OUTPUT_RATE_LEVEL_2)
348 configuration_.sequence_configuration.level_idc = 2;
349 else if (output_sample_rate <= MAX_OUTPUT_RATE_LEVEL_3)
350 configuration_.sequence_configuration.level_idc = 3;
351 else if (output_sample_rate <= MAX_OUTPUT_RATE_LEVEL_4)
352 configuration_.sequence_configuration.level_idc = 4;
353 else
354 ERR("Cannot detect level automatically. Output sample rate not supported by any level. (see Annex A.4)");
355 INFO("Using Level %u.%u", configuration_.sequence_configuration.level_idc,
356 configuration_.sequence_configuration.sublevel_idc);
357 } else {
358 switch (configuration_.sequence_configuration.level_idc) {
359 case 1:
360 CHECK(output_sample_rate <= MAX_OUTPUT_RATE_LEVEL_1);
361 break;
362 case 2:
363 CHECK(output_sample_rate <= MAX_OUTPUT_RATE_LEVEL_2);
364 break;
365 case 3:
366 CHECK(output_sample_rate <= MAX_OUTPUT_RATE_LEVEL_3);
367 break;
368 case 4:
369 CHECK(output_sample_rate <= MAX_OUTPUT_RATE_LEVEL_4);
370 break;
371 default:
372 ERR("Not a supported Level (see Annex A.4)");
373 }
374 }
375
376 if (configuration_.global_configuration.resolution_width != image_description.width() ||
377 configuration_.global_configuration.resolution_height != image_description.height()) {
378 INFO("Resolution has been adjusted from %ux%u to %ux%u to conform.", image_description.width(), image_description.height(),
379 configuration_.global_configuration.resolution_width, configuration_.global_configuration.resolution_height);
380 configuration_.sequence_configuration.conformance_window = true;
381 configuration_.sequence_configuration.conf_win_right_offset =
382 (configuration_.global_configuration.resolution_width - image_description.width()) / 2;
383 configuration_.sequence_configuration.conf_win_bottom_offset =
384 (configuration_.global_configuration.resolution_height - image_description.height()) / 2;
385 }

430 if (configuration_.picture_configuration.dequant_offset == 0 &&
431 configuration_.picture_configuration.dequant_offset_mode == DequantOffset_Default) {
432 configuration_.picture_configuration.dequant_offset_signalled = false;
433 } else {
434 configuration_.picture_configuration.dequant_offset_signalled = true;
435 }
436
437 // first, set the QM to default values
438 const uint8_t *d2 = nullptr;
439 const uint8_t *d1 = nullptr;
440 const bool horizontal_only = (configuration_.global_configuration.scaling_mode[LOQ_LEVEL_2] == ScalingMode_1D ? true : false);
441 if (configuration_.global_configuration.num_residual_layers == 4) {
442 // 2x2 case
443 d1 = default_qm_coefficient_2x2[2]; // LOQ1
444 // 4x4 case
445 if (horizontal_only)
446 d2 = default_qm_coefficient_2x2[0]; // LOQ2 1D
447 else
448 d2 = default_qm_coefficient_2x2[1]; // LOQ2 2D
449 } else {
450 // 4x4 case
451 d1 = default_qm_coefficient_4x4[2]; // LOQ1
452 if (horizontal_only)
453 d2 = default_qm_coefficient_4x4[0]; // LOQ2 1D
454 else
455 d2 = default_qm_coefficient_4x4[1]; // LOQ2 2D

458 configuration_.picture_configuration.qm_coefficient_2[i] = d2[i];
459 configuration_.picture_configuration.qm_coefficient_1[i] = d1[i];
460 }
461
462 {
463 // set the QM to default or specific custom values
464 if (configuration_.picture_configuration.quant_matrix_mode == QuantMatrix_BothDefault) {
465 // parameters read from input (command line or config file) initialised to default
466 for (unsigned i = 0; i < configuration_.global_configuration.num_residual_layers; i++) {
467 configuration_.picture_configuration.qm_coefficient_2_par[i] = d2[i];
468 configuration_.picture_configuration.qm_coefficient_1_par[i] = d1[i];
469 }
470 } else if (configuration_.picture_configuration.quant_matrix_mode == QuantMatrix_SameAndCustom) {
471 p["qm_coefficient_2"].get_vector<unsigned>(configuration_.picture_configuration.qm_coefficient_2_par,
472 configuration_.global_configuration.num_residual_layers);
473 for (unsigned i = 0; i < configuration_.global_configuration.num_residual_layers; i++) {
474 configuration_.picture_configuration.qm_coefficient_1_par[i] =
475 configuration_.picture_configuration.qm_coefficient_2_par[i];
476 }
477 } else if (configuration_.picture_configuration.quant_matrix_mode == QuantMatrix_Level2CustomLevel1Default) {
478 p["qm_coefficient_2"].get_vector<unsigned>(configuration_.picture_configuration.qm_coefficient_2_par,
479 configuration_.global_configuration.num_residual_layers);
480 } else if (configuration_.picture_configuration.quant_matrix_mode == QuantMatrix_Level2DefaultLevel1Custom) {
481 p["qm_coefficient_1"].get_vector<unsigned>(configuration_.picture_configuration.qm_coefficient_1_par,
482 configuration_.global_configuration.num_residual_layers);
483 } else if (configuration_.picture_configuration.quant_matrix_mode == QuantMatrix_DifferentAndCustom) {
484 p["qm_coefficient_2"].get_vector<unsigned>(configuration_.picture_configuration.qm_coefficient_2_par,
485 configuration_.global_configuration.num_residual_layers);
486 p["qm_coefficient_1"].get_vector<unsigned>(configuration_.picture_configuration.qm_coefficient_1_par,
487 configuration_.global_configuration.num_residual_layers);
488 }
489 }
490
491 // third, save the QM in memory for the next round
492 for (unsigned i = 0; i < configuration_.global_configuration.num_residual_layers; i++) {
493 configuration_.picture_configuration.qm_coefficient_2_mem[i] = configuration_.picture_configuration.qm_coefficient_2[i];
494 configuration_.picture_configuration.qm_coefficient_1_mem[i] = configuration_.picture_configuration.qm_coefficient_1[i];
495 }

614void Encoder::encode_residuals(unsigned plane, unsigned loq, const Surface &residuals, Surface symbols[MAX_NUM_LAYERS],
615 Temporal_SWM swm_type, EncodingMode mode, const Surface &temporal_mask, const Surface &priority_type,
616 PriorityMBType priority_mb_type, const bool final, const Surface &pixel_sad) const {
618 //
619 Surface coefficients[MAX_NUM_LAYERS];
620 const bool horizontal_only = (configuration_.global_configuration.scaling_mode[loq] == ScalingMode_1D ? true : false);
621
622 if (!horizontal_only) {
623 // Horizontal and Vertical
624 //
625 if (configuration_.global_configuration.transform_block_size == 4) {
626 TransformDDS().process(residuals, EncodingMode::ENCODE_ALL, coefficients);
627 } else if (configuration_.global_configuration.transform_block_size == 2) {
628 TransformDD().process(residuals, EncodingMode::ENCODE_ALL, coefficients);
629 }
630 } else {
631 // Horizontal only
632 //
633 if (configuration_.global_configuration.transform_block_size == 4) {
634 TransformDDS_1D().process(residuals, EncodingMode::ENCODE_ALL, coefficients);
635 } else if (configuration_.global_configuration.transform_block_size == 2) {
636 TransformDD_1D().process(residuals, EncodingMode::ENCODE_ALL, coefficients);
637 }
638 }
639
640 if (final && loq == LOQ_LEVEL_1)
641 Surface::dump_layers(coefficients, format("enc_base_coeff_transform_output_P%1d", plane), transform_block_size());
642 else if (final && loq == LOQ_LEVEL_2)
643 Surface::dump_layers(coefficients, format("enc_full_coeff_transform_output_P%1d", plane), transform_block_size());
644
645 Surface coefficients_sad[MAX_NUM_LAYERS];
646 if (!pixel_sad.empty() && encoder_configuration_.sad_threshold != 0 && encoder_configuration_.sad_coeff_threshold != 0) {
647 StaticResiduals().process(coefficients, coefficients_sad, pixel_sad, num_residual_layers(),
648 configuration_.picture_configuration.step_width_loq[LOQ_LEVEL_2],
649 encoder_configuration_.sad_threshold, encoder_configuration_.sad_coeff_threshold);
650 if (final && loq == LOQ_LEVEL_2)
651 Surface::dump_layers(coefficients_sad, format("enc_full_coeff_SAD_output_P%1d", plane), transform_block_size());
652 } else {
653 for (unsigned layer = 0; layer < num_residual_layers(); layer++) {
654 const auto coeff = coefficients[layer].view_as<int16_t>();
655 coefficients_sad[layer] = Surface::build_from<int16_t>()
656 .generate(coefficients[layer].width(), coefficients[layer].height(),
657 [&](unsigned x, unsigned y) -> int16_t { return coeff.read(x, y); })
658 .finish();
659 }
660 }
661
662#if defined PRIORITY_COEF
663 Surface priority_coefficients[MAX_NUM_LAYERS];
664 if (priority_type.width() != 0 && priority_type.height() != 0) {
665 PriorityMap().ApplyPriorityCoefficients(coefficients_sad[0].width(), coefficients_sad[0].height(),
666 configuration_.global_configuration.transform_block_size, loq, coefficients_sad,
667 priority_coefficients, priority_type, pixel_sad,
668 configuration_.picture_configuration.step_width_loq[loq], priority_mb_type, mode);
669 if (final && loq == LOQ_LEVEL_1)
670 Surface::dump_layers(priority_coefficients, format("enc_base_coeff_PM_output_P%1d", plane), transform_block_size());
671 else if (final && loq == LOQ_LEVEL_2)
672 Surface::dump_layers(priority_coefficients, format("enc_full_coeff_PM_output_P%1d", plane), transform_block_size());
673 } else {
674 for (unsigned i = 0; i < configuration_.global_configuration.num_residual_layers; i++) {
675 const auto coef = coefficients_sad[i].view_as<int16_t>();
676 priority_coefficients[i] = Surface::build_from<int16_t>()
677 .generate(coefficients_sad[i].width(), coefficients_sad[i].height(),
678 [&](unsigned x, unsigned y) -> int16_t { return coef.read(x, y); })
679 .finish();
680 }
681 }
682#endif // defined PRIORITY_COEF
683
685 //
686 int32_t dirq_step_width[MAX_NUM_LAYERS][2] = {};
687 int32_t dirq_deadzone[MAX_NUM_LAYERS][2] = {};
688 unsigned passes = 1;
689
690 {
691 // XXX This looks to be replicated twice here and once in the decoder
692 unsigned step_width[2];
693 int16_t calculated_step_width = configuration_.picture_configuration.step_width_loq[loq];
694 if (loq == LOQ_LEVEL_2 && plane > 0) {
695 calculated_step_width =
696 clamp((int16_t)((calculated_step_width * configuration_.global_configuration.chroma_step_width_multiplier) >> 6),
697 (int16_t)MIN_STEP_WIDTH, (int16_t)MAX_STEP_WIDTH);
698 }
699 if (loq == LOQ_LEVEL_2) {
700 switch (swm_type) {
701 case Temporal_SWM::SWM_Disabled:
702 step_width[0] = calculated_step_width;
703 break;
704 case Temporal_SWM::SWM_Active:
705 step_width[0] =
706 clamp((int16_t)(calculated_step_width *
707 (1 - clamp(((float)configuration_.global_configuration.temporal_step_width_modifier / (float)255),
708 (float)0, (float)0.5))),
709 (int16_t)MIN_STEP_WIDTH, (int16_t)MAX_STEP_WIDTH);
710 break;
711 case Temporal_SWM::SWM_Dependent:
712 step_width[0] =
713 clamp((int16_t)(calculated_step_width *
714 (1 - clamp(((float)configuration_.global_configuration.temporal_step_width_modifier / (float)255),
715 (float)0, (float)0.5))),
716 (int16_t)MIN_STEP_WIDTH, (int16_t)MAX_STEP_WIDTH);
717 step_width[1] = calculated_step_width;
718 passes = 2;
719 break;
720 default:
721 CHECK(0);
722 break;
723 }
724 } else
725 step_width[0] = calculated_step_width;
726
727 for (unsigned pass = 0; pass < passes; pass++) {
728 for (unsigned layer = 0; layer < num_residual_layers(); ++layer) {
729 dirq_step_width[layer][pass] = find_dirq_step_width(step_width[pass], quant_matrix_coeffs_[plane][loq][layer]);
730 dirq_deadzone[layer][pass] = find_layer_deadzone(step_width[pass], dirq_step_width[layer][pass]);
731 }
732 }
733 }
734
736 //
737 for (unsigned layer = 0; layer < num_residual_layers(); ++layer) {
738
739 Surface syms;
740 if (passes == 1) {
741#if defined PRIORITY_COEF
742 syms = Quantize().process(priority_coefficients[layer], dirq_step_width[layer][0], dirq_deadzone[layer][0], pixel_sad,
743 transform_block_size(), encoder_configuration_.quant_reduced_deadzone);
744#else // defined PRIORITY_COEF
745 syms = Quantize().process(coefficients_sad[layer], dirq_step_width[layer][0], dirq_deadzone[layer][0], pixel_sad,
746 transform_block_size(), encoder_configuration_.quant_reduced_deadzone);
747#endif // defined PRIORITY_COEF
748 } else {
749#if defined PRIORITY_COEF
750 syms = Quantize_SWM().process(priority_coefficients[layer], transform_block_size(), dirq_step_width[layer],
751 dirq_deadzone[layer], temporal_mask, pixel_sad,
752 encoder_configuration_.quant_reduced_deadzone);
753#else // defined PRIORITY_COEF
754 syms = Quantize_SWM().process(coefficients_sad[layer], transform_block_size(), dirq_step_width[layer],
755 dirq_deadzone[layer], temporal_mask, pixel_sad,
756 encoder_configuration_.quant_reduced_deadzone);
757#endif // defined PRIORITY_COEF
758 }
759 if (is_user_data_layer(loq, layer)) {
761#if USER_DATA_EXTRACTION
762 FILE *file = fopen("userdata_enc.bin", "ab");
763 symbols[layer] = UserDataInsert().process(syms, encoder_configuration_.user_data_method,
764 configuration_.global_configuration.user_data_enabled, file);
765 fclose(file);
766#else
767 symbols[layer] = UserDataInsert().process(syms, encoder_configuration_.user_data_method,
768 configuration_.global_configuration.user_data_enabled);
769#endif
770 } else
771 symbols[layer] = syms;
772 }
773 if (final && loq == LOQ_LEVEL_1)
774 Surface::dump_layers(symbols, format("enc_base_coeff_quant_output_P%1d", plane), transform_block_size());
775 else if (final && loq == LOQ_LEVEL_2)
776 Surface::dump_layers(symbols, format("enc_full_coeff_quant_output_P%1d", plane), transform_block_size());
777
778 if (final && !pixel_sad.empty() && pixel_sad.get_dump_surfaces() && loq == LOQ_LEVEL_2) {
779 // Creation of debug maps
780
781 Surface priority_debug[MAX_NUM_LAYERS];
782 if (priority_type.width() != 0 && priority_type.height() != 0) {
783 PriorityMap().ApplyPriorityCoefficients(
784 coefficients[0].width(), coefficients[0].height(), configuration_.global_configuration.transform_block_size, loq,
785 coefficients, priority_debug, priority_type, pixel_sad, configuration_.picture_configuration.step_width_loq[loq],
786 priority_mb_type, mode);
787 } else {
788 for (unsigned i = 0; i < configuration_.global_configuration.num_residual_layers; i++) {
789 const auto coef = coefficients[i].view_as<int16_t>();
790 priority_debug[i] = Surface::build_from<int16_t>()
791 .generate(coefficients[i].width(), coefficients[i].height(),
792 [&](unsigned x, unsigned y) -> int16_t { return coef.read(x, y); })
793 .finish();
794 }
795 }
796
797 Surface syms_anchor[MAX_NUM_LAYERS], syms_no_sad[MAX_NUM_LAYERS];
798
799 for (unsigned layer = 0; layer < num_residual_layers(); ++layer) {
800 Surface syms_a, syms_b;
801 if (passes == 1) {
802 syms_a = Quantize().process(priority_coefficients[layer], dirq_step_width[layer][0], dirq_deadzone[layer][0],
803 pixel_sad, transform_block_size(), 5u);
804 syms_b = Quantize().process(priority_debug[layer], dirq_step_width[layer][0], dirq_deadzone[layer][0], pixel_sad,
805 transform_block_size(), 5u);
806 } else {
807 syms_a = Quantize_SWM().process(priority_coefficients[layer], transform_block_size(), dirq_step_width[layer],
808 dirq_deadzone[layer], temporal_mask, pixel_sad, 5u);
809 syms_b = Quantize_SWM().process(priority_debug[layer], transform_block_size(), dirq_step_width[layer],
810 dirq_deadzone[layer], temporal_mask, pixel_sad, 5u);
811 }
812 syms_anchor[layer] = syms_a;
813 syms_no_sad[layer] = syms_b;
814 }
815
816 const Surface static_residuals_debug =
817 Surface::build_from<int16_t>()
818 .generate(residuals.width(), residuals.height(),
819 [&](unsigned x, unsigned y) -> int16_t {
820 const unsigned layer =
821 (x % transform_block_size() + transform_block_size() * (y % transform_block_size()));
822 const auto syms_anchor_view = syms_anchor[layer].view_as<int16_t>();
823 const auto syms_view = symbols[layer].view_as<int16_t>();
824 const auto syms_no_sad_view = syms_no_sad[layer].view_as<int16_t>();
825
826 if (syms_no_sad_view.read(x / transform_block_size(), y / transform_block_size()) !=
827 syms_anchor_view.read(x / transform_block_size(), y / transform_block_size()))
828 return 20000;
829 else if (syms_anchor_view.read(x / transform_block_size(), y / transform_block_size()) !=
830 syms_view.read(x / transform_block_size(), y / transform_block_size()))
831 return -20000;
832 else
833 return 0;
834 })
835 .finish();
836 static_residuals_debug.dump(format("enc_full_coeff_SAD_changes_P%1d", plane));
837 }
838}

840Surface Encoder::decode_residuals(unsigned plane, unsigned loq, const Surface symbols[MAX_NUM_LAYERS], Temporal_SWM temp_type,
841 const Surface &temporal_mask) const {
842
843 Surface coefficients[MAX_NUM_LAYERS] = {};
844 int32_t dirq_step_width[MAX_NUM_LAYERS][2] = {};
845 int32_t invq_step_width[MAX_NUM_LAYERS][2] = {};
846 int32_t invq_offset[MAX_NUM_LAYERS][2] = {};
847 int32_t invq_deadzone[MAX_NUM_LAYERS][2] = {};
848 const bool horizontal_only = (configuration_.global_configuration.scaling_mode[loq] == ScalingMode_1D ? true : false);
849
850 // Apply temporal step width modifier, if required
851 unsigned passes = 1;
852 {
853 int16_t step_width[2];
854 int16_t calculated_step_width = configuration_.picture_configuration.step_width_loq[loq];
855 if (loq == LOQ_LEVEL_2 && plane > 0) {
856 calculated_step_width =
857 clamp((int16_t)((calculated_step_width * configuration_.global_configuration.chroma_step_width_multiplier) >> 6),
858 (int16_t)MIN_STEP_WIDTH, (int16_t)MAX_STEP_WIDTH);
859 }
860 if (loq == LOQ_LEVEL_2) {
861 switch (temp_type) {
862 case Temporal_SWM::SWM_Disabled:
863 step_width[0] = calculated_step_width;
864 break;
865 case Temporal_SWM::SWM_Active:
866 step_width[0] =
867 clamp((int16_t)(calculated_step_width *
868 (1 - clamp(((float)configuration_.global_configuration.temporal_step_width_modifier / (float)255),
869 (float)0, (float)0.5))),
870 (int16_t)MIN_STEP_WIDTH, (int16_t)MAX_STEP_WIDTH);
871 break;
872 case Temporal_SWM::SWM_Dependent:
873 step_width[0] =
874 clamp((int16_t)(calculated_step_width *
875 (1 - clamp(((float)configuration_.global_configuration.temporal_step_width_modifier / (float)255),
876 (float)0, (float)0.5))),
877 (int16_t)MIN_STEP_WIDTH, (int16_t)MAX_STEP_WIDTH);
878 step_width[1] = calculated_step_width;
879 passes = 2;
880 break;
881 default:
882 break;
883 }
884 } else {
885 step_width[0] = calculated_step_width;
886 }
887
888 for (unsigned pass = 0; pass < passes; pass++) {
889 for (unsigned layer = 0; layer < num_residual_layers(); ++layer) {
890 dirq_step_width[layer][pass] = find_dirq_step_width(step_width[pass], quant_matrix_coeffs_[plane][loq][layer]);
891 invq_offset[layer][pass] =
892 find_invq_offset(configuration_.picture_configuration, step_width[pass], dirq_step_width[layer][pass]);
893
894 invq_step_width[layer][pass] =
895 find_invq_step_width(configuration_.picture_configuration, dirq_step_width[layer][pass], invq_offset[layer][pass]);
896
897 invq_deadzone[layer][pass] = find_layer_deadzone(step_width[pass], invq_step_width[layer][pass]);
898 }
899 }
900 }
901
902 // Dequantize
903 for (unsigned layer = 0; layer < num_residual_layers(); ++layer) {
904 Surface syms;
905 if (is_user_data_layer(loq, layer))
906 syms = UserDataClear().process(symbols[layer], configuration_.global_configuration.user_data_enabled);
907 else
908 syms = symbols[layer];
909
910 if (passes == 1) {
911 coefficients[layer] = InverseQuantize().process(
912 syms, invq_step_width[layer][0],
