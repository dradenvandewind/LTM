49void roundAffineMv( int& mvx, int& mvy, int nShift )
50{
51 const int nOffset = 1 << (nShift - 1);
52 mvx = (mvx + nOffset - (mvx >= 0)) >> nShift;
53 mvy = (mvy + nOffset - (mvy >= 0)) >> nShift;
54}

58void clipMvInPic ( Mv& rcMv, const struct Position& pos, const struct Size& size, const class SPS& sps, const class PPS& pps )
59{
60 if (sps.getWrapAroundEnabledFlag())
61 {
62 wrapClipMv(rcMv, pos, size, &sps, &pps);
63 return;
64 }
65
66 int mvShift = MV_FRACTIONAL_BITS_INTERNAL;
67 int offset = 8;
68 int horMax = (pps.getPicWidthInLumaSamples() + offset - (int)pos.x - 1) << mvShift;
69 int horMin = (-(int)sps.getMaxCUWidth() - offset - (int)pos.x + 1) << mvShift;
70
71 int verMax = (pps.getPicHeightInLumaSamples() + offset - (int)pos.y - 1) << mvShift;
72 int verMin = (-(int)sps.getMaxCUHeight() - offset - (int)pos.y + 1) << mvShift;
73
74 rcMv.setHor(std::min(horMax, std::max(horMin, rcMv.getHor())));
75 rcMv.setVer(std::min(verMax, std::max(verMin, rcMv.getVer())));
76}

106bool wrapClipMv( Mv& rcMv, const Position& pos, const struct Size& size, const SPS *sps, const PPS *pps )
107{
108 bool wrapRef = true;
109 int iMvShift = MV_FRACTIONAL_BITS_INTERNAL;
110 int iOffset = 8;
111 int iHorMax = ( pps->getPicWidthInLumaSamples() + sps->getMaxCUWidth() - size.width + iOffset - (int)pos.x - 1 ) << iMvShift;
112 int iHorMin = ( -( int ) sps->getMaxCUWidth() - iOffset - ( int ) pos.x + 1 ) << iMvShift;
113
114 int iVerMax = ( pps->getPicHeightInLumaSamples() + iOffset - ( int ) pos.y - 1 ) << iMvShift;
115 int iVerMin = ( -( int ) sps->getMaxCUHeight() - iOffset - ( int ) pos.y + 1 ) << iMvShift;
116
117 const SubPic& curSubPic = pps->getSubPicFromPos( pos );
118 if( curSubPic.getTreatedAsPicFlag() )
119 {
120 iVerMax = ( ( curSubPic.getSubPicBottom() + 1 ) + iOffset - ( int ) pos.y - 1 ) << iMvShift;
121 iVerMin = ( -( int ) sps->getMaxCUHeight() - iOffset - ( ( int ) pos.y - curSubPic.getSubPicTop() ) + 1 ) << iMvShift;
122 }
123 int mvX = rcMv.getHor();
124
125 if(mvX > iHorMax)
126 {
127 mvX -= ( pps->getWrapAroundOffset() << iMvShift );
128 mvX = std::min( iHorMax, std::max( iHorMin, mvX ) );
129 wrapRef = false;
130 }
131 if(mvX < iHorMin)
132 {
133 mvX += ( pps->getWrapAroundOffset() << iMvShift );
134 mvX = std::min( iHorMax, std::max( iHorMin, mvX ) );
135 wrapRef = false;
136 }
137
138 rcMv.setHor( mvX );
139 rcMv.setVer( std::min( iVerMax, std::max( iVerMin, rcMv.getVer() ) ) );
140 return wrapRef;
141}
