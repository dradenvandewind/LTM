57def hash(filename, blocksize=65536):
58 """
59 Compute hash of named file.
60 """
61 hash = xxhash.xxh3_64()
62 with open(filename, "rb") as f:
63 for block in iter(lambda: f.read(blocksize), b""):
64 hash.update(block)
65 return hash.hexdigest()

85def parse_vooya_filename(name):
86 """
87 Extract vooya-ish image metadata from filename in form suitable for use as encoder parameters.
88 """
89 params = {}
90 for p in name.split("_"):
91 for n,r in IMAGE_NAME_PARAMS:
92 m = r.match(p)
93 if m:
94 params[n] = m.group(1)
95
96 if 'bit_depth' in params:
97 if 'format' not in params:
98 params['format'] = "420p"
99 params['format'] += params.pop('bit_depth')
100
101 if 'format' in params and params['format'].startswith('4'):
102 params['format'] = 'yuv' + params['format']
103
104 if 'width' in params:
105 params['width'] = int(params['width'])
106 if 'height' in params:
107 params['height'] = int(params['height'])
108 if 'fps' in params:
109 params['fps'] = int(params['fps'])
110
111 return params

392def run_tests(args):
393 """Read a JSON array and run test described by each entry."""
394
395 fails = 0
396
397 # Load tests
398 with open(args.test_file) as f:
399 tests = json.load(f)
400
401 if type(tests) != list:
402 print("Top level of tests JSON must be an array.", file=sys.stderr)
403 return 1
404
405 default_parameters = json.loads(args.parameters)
406
407 if args.sets:
408 test_set = set(args.sets.split(','))
409 else:
410 test_set = set()
411
412
413 test_jobs = []
414 for n,test in enumerate(tests):
415
416 # Single test
417 if args.single and test['name'] != args.single:
418 continue
419
420 # Sets of tests
421 if test_set:
422 if 'sets' not in test or test_set.isdisjoint(set(test['sets'])):
423 continue
424
425 test_jobs.append({
426 'number':n,
427 'test':test,
428 'encoder':os.path.abspath(args.encoder),
429 'decoder':os.path.abspath(args.decoder),
430 'input_dir':args.input_dir,
431 'base_dir':args.base_dir,
432 'default_parameters':default_parameters,
433 'label':args.label,
434 'update_checksums':args.updated_checksums != "",
435 'show_progress': args.progress,
436 'decode_only': args.decode_only})
437
438
439 if args.parallel == "0":
440 print(f"Running {len(test_jobs)} tests", file=sys.stderr)
441 # Run jobs serially in process
442 for j in test_jobs:
443 ok = test_encode_decode(**j)
444 if not ok:
445 fails += 1
446 else:
447 print(f"Running {len(test_jobs)} tests using {args.parallel} workers", file=sys.stderr)
448 # Run jobs in parallel in seperate processes
449 with concurrent.futures.ProcessPoolExecutor(max_workers=int(args.parallel)) as executor:
450 for ok,n,updated_test in executor.map(run_test, test_jobs):
451 if not ok:
452 fails += 1
453 if ok and args.updated_checksums:
454 # Collect updated data from subprocess
455 tests[n] = updated_test
456
457 if fails != 0:
458 print(f"Tests failed: {fails}", file=sys.stderr)
459
460
461 if args.updated_checksums:
462 with open(args.updated_checksums, "wt") as f:
463 json.dump(tests, f, indent=4)
464
465 return fails
