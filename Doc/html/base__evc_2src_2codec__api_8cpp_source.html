122static int32_t pull_image(CodecContext c, CodecImage* image, CodecMetadata *metadata, int8_t* eos, CodecError *error){
123 Context *context = (Context *)c;
124 if(error)
125 *error = 0;
126
127 switch(context->state_) {
128 case StatePushing:
129 // Still accumulating input stream
130 *eos = 0;
131 return 0;
132
133 case StateDecoding:
134 context->decoder_thread_ = std::thread([context] {
135 context->decode();
136 context->decoded_images_.push(nullptr);
137 });
138
139 context->state_ = StateFlushing;
140 /* Fall through */
141
142 case StateFlushing:
143 context->decoded_images_.pop(context->decoded_image_);
144 if(context->decoded_image_) {
145 *image = context->decoded_image_->image_;
146 *eos = 0;
147 return 1;
148 } else {
149 context->decoder_thread_.join();
150 *eos = 1;
151 return 0;
152 }
153 default:
154 ERR("Unexpected decoder state");
155 }
156 return 0;
157}

189void Context::output_begin(bool is16bit, uint32_t width, uint32_t height) {
190 output_frame_++;
191 size_t frame_size = (width * height + 2 * (width / 2 * height / 2)) * (is16bit ? 2 : 1);
192
193 CHECK(!writing_image_);
194
195 writing_image_ = std::shared_ptr<DecodedImage>(new DecodedImage());
196
197 writing_image_->buffer_.reserve(frame_size);
198
199 writing_image_->image_.bpp = is16bit ? 2 : 1;
200 writing_image_->image_.width_y = width;
201 writing_image_->image_.height_y = height;
202 writing_image_->image_.stride_y = width * writing_image_->image_.bpp;
203
204 writing_image_->image_.width_uv = width / 2;
205 writing_image_->image_.height_uv = height / 2;
206 writing_image_->image_.stride_uv = (width / 2) * writing_image_->image_.bpp;
207}

209size_t Context::output_write(const uint8_t *data, size_t size) {
210 CHECK(writing_image_);
211
212 writing_image_->buffer_.insert(writing_image_->buffer_.end(), data, data + size);
213 return size;

216void Context::output_end() {
217 CHECK(writing_image_);
218
219 writing_image_->image_.data_y = writing_image_->buffer_.data();
220 writing_image_->image_.data_u = writing_image_->image_.data_y + writing_image_->image_.stride_y * writing_image_->image_.height_y;
221 writing_image_->image_.data_v = writing_image_->image_.data_u + writing_image_->image_.stride_uv * writing_image_->image_.height_uv;
222
223 decoded_images_.push(writing_image_);
224 writing_image_ = nullptr;
225}
