46std::string format(const char *fmt, ...) {
47 char tmp[16 * 1024];
48
49 va_list args;
50 va_start(args, fmt);
51 vsnprintf(tmp, sizeof(tmp) - 1, fmt, args);
52 va_end(args);
53
54 return tmp;
55}

57std::string join(const std::vector<std::string> &list, const std::string &seperator, bool include_trailing) {
58 std::string r;
59
60 auto i = list.begin();
61 while (i != list.end()) {
62 r += *i;
63 ++i;
64 if (i != list.end() || include_trailing)
65 r += seperator;
66 }
67
68 return r;
69}

101std::string hex_dump(const uint8_t *data, uint64_t size, unsigned int offset) {
102 const int bytes_per_line = 16;
103 std::string r;
104
105 for (uint64_t l = 0; l < size; l += bytes_per_line) {
106
107 r += format(" /* %04x */ ", offset + l);
108 // Bytes
109 for (uint64_t b = 0; b < bytes_per_line; ++b) {
110 if (l + b < size)
111 r += format("0x%02x, ", data[l + b]);
112 else
113 r += format(" ");
114 }
115
116 r += format(" // ");
117
118 // Chars
119 for (uint64_t b = 0; b < bytes_per_line; ++b) {
120 if (l + b < size)
121 r += format("%c", isprint(data[l + b]) ? data[l + b] : '.');
122 }
123
124 r += format("\n");
125 }
126
127 return r;
128}
