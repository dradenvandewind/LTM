67void* cache_mem_align_malloc(int size, int alignSize)
68{
69 unsigned char *alignBuf;
70 unsigned char *buf = (unsigned char *)malloc(size + 2 * alignSize + sizeof(void **));
71 if (buf)
72 {
73 alignBuf = buf + alignSize + sizeof(void **);
74 alignBuf -= (intptr_t)alignBuf & (alignSize - 1);
75 *((void **)(alignBuf - sizeof(void **))) = buf;
76 return alignBuf;
77 }
78 return nullptr; // memory keep fail
79}

119int CacheModel::xCalcPower( int num )
120{
121 int power = -1;
122
123 for ( int i = 0 ; i < 32 ; i++ )
124 {
125 if ( num == (1 << i) )
126 {
127 power = i;
128 break;
129 }
130 }
131 if (power < 0)
132 {
133 THROW("non power of 2");
134 }
135 return power;
136}

138void CacheModel::xConfigure(const std::string& filename )
139{
140 po::Options opts;
141 opts.addOptions()
142 ("CacheEnable", m_cacheEnable, false, "Cache Enable" )
143 ("CacheLineSize", m_cacheLineSize, 128, "Cache line size")
144 ("NumCacheLine", m_numCacheLine, 32, "Number of cache line")
145 ("NumWay", m_numWay, 4, "Number of way")
146 ("CacheAddrMode", m_cacheAddrMode, 0, "Address mapping mode 0 : linear address 1 : 2D address")
147 ("BlkWidth", m_cacheBlkWidth, 32, "Block width in 2D address mode")
148 ("BlkHeight", m_cacheBlkHeight, 16, "Block height in 2D address mode")
149 ("FrameReport", m_frameReport, false, "Report in each frame" )
150 ;
151
152 po::setDefaults(opts);
153 po::parseConfigFile( opts, filename );
154
155 if ( m_cacheLineSize > CACHE_MEM_ALIGN_SIZE )
156 {
157 fprintf( stderr, "cache line size is bigger that memory alignment\n" );
158 fprintf( stderr, "This may lead mismatch among enviroments\n" );
159 }
160 if ( m_cacheAddrMode == CACHE_MODE_2D )
161 {
162 int blkSize = m_cacheBlkWidth * m_cacheBlkHeight;
163 if ( m_cacheLineSize % blkSize != 0 && blkSize % m_cacheLineSize ) {
164 THROW("CacheLineSize shall be multiple of BlkWidth x BlkHeight or BlkWidth x BlkHeight shall be multiple of CacheLineSize in 2D mode");
165 }
166 }
167}

169// initilize cache information such as size
170void CacheModel::create(const std::string& cacheCfgFileName)
171{
172 bool init = cacheCfgFileName == "";
173
174 if (cacheCfgFileName.length() > 1000)
175 {
176 THROW("config file name for cache model is too long. It shall be < 1000\n");
177 }
178 if ( init )
179 {
180 return; // no cache config
181 }
182 xConfigure(cacheCfgFileName);
183
184 if ( !m_cacheEnable )
185 {
186 return;
187 }
188
189 // set parameters
190 m_cacheSize = m_numCacheLine * m_numWay;
191 // calc address calculation parameter
192 m_shift = xCalcPower( m_cacheLineSize );
193 // keep memory
194 m_cacheAddr = new size_t [m_cacheSize];
195 m_cachePoc = new int [m_cacheSize];
196 m_cacheComp = new ComponentID [m_cacheSize];
197 m_available = new bool [m_cacheSize];
198 m_hitCount = new int [m_cacheSize];
199 // PLRU
200 m_treeDepth = xCalcPower( m_numWay );
201 m_treeStatus = new int [m_numCacheLine];
202 if ( m_cacheLineSize > 0 && m_numCacheLine > 0 && m_numWay > 0 )
203 {
204 m_cacheEnableFilter = true;
205 }
206}

340int CacheModel::xGetWayTreePLRU( int entry )
341{
342 int shift = 0;
343 int way = 0;
344 for ( int i = 0; i < m_treeDepth ; i++ )
345 {
346 int flag = (m_treeStatus[ entry ] >> shift) & 0x1;
347 shift = (shift << 1) + flag + 1;
348 way = (way << 1) | (flag ^ 0x1);
349 }
350 xUpdateCacheStatus( entry, way );
351
352 return way;
353}
354
355void CacheModel::xUpdatePLRUStatus( int entry, int way )
356{
357 int val = m_treeStatus[ entry ];
358 int shift = 0;
359
360 for ( int i = 0 ; i < m_treeDepth ; i++ )
361 {
362 int flag = (way >> (m_treeDepth - i - 1)) & 0x1;
363 val = (val & (~0 ^ (1 << shift))) | (flag << shift); // only set shift-th bit
364 shift = (shift << 1) + 2 - flag;
365 }
366
367 m_treeStatus[ entry ] = val;
368}

410size_t CacheModel::xMapAddress( size_t offset ) {
411
412 size_t ret;
413 size_t xInPic, yInPic, blkPosX, blkPosY, xInBlk, yInBlk;
414
415 switch ( m_cacheAddrMode ) {
416 case CACHE_MODE_1D : // diret mapping
417 return offset;
418 case CACHE_MODE_2D : // 2D address mapping
419 xInPic = offset % m_picWidth;
420 yInPic = offset / m_picWidth;
421 blkPosX = xInPic / m_cacheBlkWidth;
422 blkPosY = yInPic / m_cacheBlkHeight;
423 xInBlk = xInPic % m_cacheBlkWidth;
424 yInBlk = yInPic % m_cacheBlkHeight;
425 ret = m_picWidth * blkPosY * m_cacheBlkHeight;
426 ret += blkPosX * m_cacheBlkWidth * m_cacheBlkHeight;
427 ret += yInBlk * m_cacheBlkWidth;
428 ret += xInBlk;
429 return ret;
430 default :
431 THROW( "Unknown address mode " << m_cacheAddrMode );
432 return 0;
433 }
434}

436// check cache hit/miss
437void CacheModel::cacheAccess( const Pel *addr, const std::string& fileName, const int lineNum )
438{
439 if ( !m_cacheEnable || !m_cacheEnableFilter )
440 {
441 return;
442 }
443 bool hit = false;
444 size_t cacheAddr = xMapAddress( (size_t) (addr - m_base) ) >> m_shift;
445 int entry = (int) (cacheAddr % m_numCacheLine);
446 int pos = entry * m_numWay;
447 int way;
448
449 // check cache hit in each way
450 for ( way = 0 ; way < m_numWay ; way++ )
451 {
452 if ( xIsCacheHit( pos + way, cacheAddr ) ) {
453 hit = true;
454 break;
455 }
456 }
457#if JVET_J0090_MEMORY_BANDWITH_MEASURE_PRINT_ACCESS_INFO
458 if ( m_frameCount == JVET_J0090_MEMORY_BANDWITH_MEASURE_PRINT_FRAME )
459 {
460 fprintf( stdout, "%s %d:%p\n", fileName.c_str(), lineNum, addr );
461 }
462#endif
463
464 if ( !hit )
465 {
466 // read data from external memory
467 m_missHitCount++;
468 // update cache entry
469 xUpdateCache( entry, cacheAddr );
470 }
471 else
472 {
473 // update hit status
474 m_hitCount[ pos + way ] ++;
475 xUpdateCacheStatus( entry, way );
476 }
477 m_totalAccess++;
478}
