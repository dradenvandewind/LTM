96class ScanGenerator
97{
98private:
99 uint32_t m_line, m_column;
100 const uint32_t m_blockWidth, m_blockHeight;
101 const uint32_t m_stride;
102 const CoeffScanType m_scanType;
103
104public:
105 ScanGenerator(uint32_t blockWidth, uint32_t blockHeight, uint32_t stride, CoeffScanType scanType)
106 : m_line(0), m_column(0), m_blockWidth(blockWidth), m_blockHeight(blockHeight), m_stride(stride), m_scanType(scanType)
107 { }
108
109 uint32_t GetCurrentX() const { return m_column; }
110 uint32_t GetCurrentY() const { return m_line; }
111
112 uint32_t GetNextIndex(uint32_t blockOffsetX, uint32_t blockOffsetY)
113 {
114 const uint32_t rtn = ((m_line + blockOffsetY) * m_stride) + m_column + blockOffsetX;
115
116 //advance line and column to the next position
117 switch (m_scanType)
118 {
119 //------------------------------------------------
120
121 case SCAN_DIAG:
122
123 if ((m_column == m_blockWidth - 1) || (m_line == 0)) //if we reach the end of a rank, go diagonally down to the next one
124 {
125 m_line += m_column + 1;
126 m_column = 0;
127
128 if (m_line >= m_blockHeight) //if that takes us outside the block, adjust so that we are back on the bottom row
129 {
130 m_column += m_line - (m_blockHeight - 1);
131 m_line = m_blockHeight - 1;
132 }
133 }
134 else
135 {
136 m_column++;
137 m_line--;
138 }
139 break;
140
141 case SCAN_TRAV_HOR:
142 if (m_line % 2 == 0)
143 {
144 if (m_column == (m_blockWidth - 1))
145 {
146 m_line++;
147 m_column = m_blockWidth - 1;
148 }
149 else
150 {
151 m_column++;
152 }
153 }
154 else
155 {
156 if (m_column == 0)
157 {
158 m_line++;
159 m_column = 0;
160 }
161 else
162 {
163 m_column--;
164 }
165 }
166 break;
167
168 case SCAN_TRAV_VER:
169 if (m_column % 2 == 0)
170 {
171 if (m_line == (m_blockHeight - 1))
172 {
173 m_column++;
174 m_line = m_blockHeight - 1;
175 }
176 else
177 {
178 m_line++;
179 }
180 }
181 else
182 {
183 if (m_line == 0)
184 {
185 m_column++;
186 m_line = 0;
187 }
188 else
189 {
190 m_line--;
191 }
192 }
193 break;
194 //------------------------------------------------
195
196 default:
197
198 THROW("ERROR: Unknown scan type \"" << m_scanType << "\"in ScanGenerator::GetNextIndex");
199 break;
200 }
201
202 return rtn;
203 }

212int8_t getBcwWeight(uint8_t bcwIdx, uint8_t uhRefFrmList)
213{
214 // Weghts for the model: P0 + w * (P1 - P0) = (1-w) * P0 + w * P1
215 // Retuning 1-w for P0 or w for P1
216 return (uhRefFrmList == REF_PIC_LIST_0 ? g_BcwWeightBase - g_BcwWeights[bcwIdx] : g_BcwWeights[bcwIdx]);
217}

219void resetBcwCodingOrder(bool bRunDecoding, const CodingStructure &cs)
220{
221 // Form parsing order: { BCW_DEFAULT, BCW_DEFAULT+1, BCW_DEFAULT-1, BCW_DEFAULT+2, BCW_DEFAULT-2, ... }
222 g_BcwParsingOrder[0] = BCW_DEFAULT;
223 for (int i = 1; i <= (BCW_NUM >> 1); ++i)
224 {
225 g_BcwParsingOrder[2 * i - 1] = BCW_DEFAULT + (int8_t)i;
226 g_BcwParsingOrder[2 * i] = BCW_DEFAULT - (int8_t)i;
227 }
228
229 // Form encoding order
230 if (!bRunDecoding)
231 {
232 for (int i = 0; i < BCW_NUM; ++i)
233 {
234 g_BcwCodingOrder[(uint32_t)g_BcwParsingOrder[i]] = i;
235 }
236 }
237}

239uint32_t deriveWeightIdxBits(uint8_t bcwIdx) // Note: align this with TEncSbac::codeBcwIdx and TDecSbac::parseBcwIdx
240{
241 uint32_t numBits = 1;
242 uint8_t bcwCodingIdx = (uint8_t)g_BcwCodingOrder[bcwIdx];
243
244 if (BCW_NUM > 2 && bcwCodingIdx != 0)
245 {
246 uint32_t prefixNumBits = BCW_NUM - 2;
247 uint32_t step = 1;
248 uint8_t prefixSymbol = bcwCodingIdx;
249
250 // Truncated unary code
251 uint8_t idx = 1;
252 for (int ui = 0; ui < prefixNumBits; ++ui)
253 {
254 if (prefixSymbol == idx)
255 {
256 ++numBits;
257 break;
258 }
259 else
260 {
261 ++numBits;
262 idx += step;
263 }
264 }
265 }
266 return numBits;
267}

282void initROM()
283{
284 gp_sizeIdxInfo = new SizeIndexInfoLog2();
285 gp_sizeIdxInfo->init(MAX_CU_SIZE);
286
287
288 SizeIndexInfoLog2 sizeInfo;
289 sizeInfo.init(MAX_CU_SIZE);
290
291 // initialize scan orders
292 for (uint32_t blockHeightIdx = 0; blockHeightIdx < sizeInfo.numAllHeights(); blockHeightIdx++)
293 {
294 for (uint32_t blockWidthIdx = 0; blockWidthIdx < sizeInfo.numAllWidths(); blockWidthIdx++)
295 {
296 const uint32_t blockWidth = sizeInfo.sizeFrom(blockWidthIdx);
297 const uint32_t blockHeight = sizeInfo.sizeFrom(blockHeightIdx);
298 const uint32_t totalValues = blockWidth * blockHeight;
299
300 //--------------------------------------------------------------------------------------------------
301
302 //non-grouped scan orders
303
304 for (uint32_t scanTypeIndex = 0; scanTypeIndex < SCAN_NUMBER_OF_TYPES; scanTypeIndex++)
305 {
306 const CoeffScanType scanType = CoeffScanType(scanTypeIndex);
307 ScanElement * scan = nullptr;
308
309 if (blockWidthIdx < sizeInfo.numWidths() && blockHeightIdx < sizeInfo.numHeights())
310 {
311 scan = new ScanElement[totalValues];
312 }
313
314 g_scanOrder[SCAN_UNGROUPED][scanType][blockWidthIdx][blockHeightIdx] = scan;
315
316 if (scan == nullptr)
317 {
318 continue;
319 }
320
321 ScanGenerator fullBlockScan(blockWidth, blockHeight, blockWidth, scanType);
322
323 for (uint32_t scanPosition = 0; scanPosition < totalValues; scanPosition++)
324 {
325 const int rasterPos = fullBlockScan.GetNextIndex( 0, 0 );
326 const int posY = rasterPos / blockWidth;
327 const int posX = rasterPos - ( posY * blockWidth );
328
329 scan[scanPosition].idx = rasterPos;
330 scan[scanPosition].x = posX;
331 scan[scanPosition].y = posY;
332 }
333 }
334
335 //--------------------------------------------------------------------------------------------------
336
337 //grouped scan orders
338 const uint32_t* log2Sbb = g_log2SbbSize[floorLog2(blockWidth)][floorLog2(blockHeight)];
339 const uint32_t log2CGWidth = log2Sbb[0];
340 const uint32_t log2CGHeight = log2Sbb[1];
341
342 const uint32_t groupWidth = 1 << log2CGWidth;
343 const uint32_t groupHeight = 1 << log2CGHeight;
344 const uint32_t widthInGroups = std::min<unsigned>(JVET_C0024_ZERO_OUT_TH, blockWidth) >> log2CGWidth;
345 const uint32_t heightInGroups = std::min<unsigned>(JVET_C0024_ZERO_OUT_TH, blockHeight) >> log2CGHeight;
346
347 const uint32_t groupSize = groupWidth * groupHeight;
348 const uint32_t totalGroups = widthInGroups * heightInGroups;
349
350 for (uint32_t scanTypeIndex = 0; scanTypeIndex < SCAN_NUMBER_OF_TYPES; scanTypeIndex++)
351 {
352 const CoeffScanType scanType = CoeffScanType(scanTypeIndex);
353
354 ScanElement *scan = new ScanElement[totalValues];
355
356 g_scanOrder[SCAN_GROUPED_4x4][scanType][blockWidthIdx][blockHeightIdx] = scan;
357
358 if ( blockWidth > JVET_C0024_ZERO_OUT_TH || blockHeight > JVET_C0024_ZERO_OUT_TH )
359 {
360 for (uint32_t i = 0; i < totalValues; i++)
361 {
362 scan[i].idx = totalValues - 1;
363 scan[i].x = blockWidth - 1;
364 scan[i].y = blockHeight - 1;
365 }
366 }
367
368 ScanGenerator fullBlockScan(widthInGroups, heightInGroups, groupWidth, scanType);
369
370 for (uint32_t groupIndex = 0; groupIndex < totalGroups; groupIndex++)
371 {
372 const uint32_t groupPositionY = fullBlockScan.GetCurrentY();
373 const uint32_t groupPositionX = fullBlockScan.GetCurrentX();
374 const uint32_t groupOffsetX = groupPositionX * groupWidth;
375 const uint32_t groupOffsetY = groupPositionY * groupHeight;
376 const uint32_t groupOffsetScan = groupIndex * groupSize;
377
378 ScanGenerator groupScan(groupWidth, groupHeight, blockWidth, scanType);
379
380 for (uint32_t scanPosition = 0; scanPosition < groupSize; scanPosition++)
381 {
382 const int rasterPos = groupScan.GetNextIndex( groupOffsetX, groupOffsetY );
383 const int posY = rasterPos / blockWidth;
384 const int posX = rasterPos - ( posY * blockWidth );
385
386 scan[groupOffsetScan + scanPosition].idx = rasterPos;
387 scan[groupOffsetScan + scanPosition].x = posX;
388 scan[groupOffsetScan + scanPosition].y = posY;
389 }
390
391 fullBlockScan.GetNextIndex(0, 0);
392 }
393 }
394
395 //--------------------------------------------------------------------------------------------------
396 }
397 }

438void destroyROM()
439{
440 unsigned numWidths = gp_sizeIdxInfo->numAllWidths();
441 unsigned numHeights = gp_sizeIdxInfo->numAllHeights();
442
443 for (uint32_t groupTypeIndex = 0; groupTypeIndex < SCAN_NUMBER_OF_GROUP_TYPES; groupTypeIndex++)
444 {
445 for (uint32_t scanOrderIndex = 0; scanOrderIndex < SCAN_NUMBER_OF_TYPES; scanOrderIndex++)
446 {
447 for (uint32_t blockWidthIdx = 0; blockWidthIdx <= numWidths; blockWidthIdx++)
448 {
449 for (uint32_t blockHeightIdx = 0; blockHeightIdx <= numHeights; blockHeightIdx++)
450 {
451 delete[] g_scanOrder[groupTypeIndex][scanOrderIndex][blockWidthIdx][blockHeightIdx];
452 g_scanOrder[groupTypeIndex][scanOrderIndex][blockWidthIdx][blockHeightIdx] = nullptr;
453 }
454 }
455 }
456 }
457
458 delete gp_sizeIdxInfo;
459 gp_sizeIdxInfo = nullptr;
460
461 for( int modeIdx = 0; modeIdx < GEO_NUM_PARTITION_MODE; modeIdx++ )
462 {
463 delete[] g_GeoParams[modeIdx];
464 g_GeoParams[modeIdx] = nullptr;
465 }
466 delete[] g_GeoParams;
467 for( int i = 0; i < GEO_NUM_PRESTORED_MASK; i++ )
468 {
469 delete[] g_globalGeoWeights [i];
470 delete[] g_globalGeoEncSADmask[i];
471 g_globalGeoWeights [i] = nullptr;
472 g_globalGeoEncSADmask[i] = nullptr;
473 }
474}

718void initGeoTemplate()
719{
720 g_GeoParams = new int16_t*[GEO_NUM_PARTITION_MODE];
721 int modeIdx = 0;
722 for( int angleIdx = 0; angleIdx < GEO_NUM_ANGLES; angleIdx++ )
723 {
724 for( int distanceIdx = 0; distanceIdx < GEO_NUM_DISTANCES; distanceIdx++ )
725 {
726 if( (distanceIdx == 0 && angleIdx >= 16)
727 || ((distanceIdx == 2 || distanceIdx == 0) && (g_angle2mask[angleIdx] == 0 || g_angle2mask[angleIdx] == 5))
728 || g_angle2mask[angleIdx] == -1 )
729 {
730 continue;
731 }
732 g_GeoParams[modeIdx] = new int16_t[2];
733 g_GeoParams[modeIdx][0] = (int16_t)angleIdx;
734 g_GeoParams[modeIdx][1] = (int16_t)distanceIdx;
735 modeIdx++;
736 }
737 }
738 for (int angleIdx = 0; angleIdx < (GEO_NUM_ANGLES >> 2) + 1; angleIdx++)
739 {
740 if (g_angle2mask[angleIdx] == -1)
741 {
742 continue;
743 }
744 g_globalGeoWeights[g_angle2mask[angleIdx]] = new int16_t[GEO_WEIGHT_MASK_SIZE * GEO_WEIGHT_MASK_SIZE];
745 g_globalGeoEncSADmask[g_angle2mask[angleIdx]] = new Pel[GEO_WEIGHT_MASK_SIZE * GEO_WEIGHT_MASK_SIZE];
746
747 int distanceX = angleIdx;
748 int distanceY = (distanceX + (GEO_NUM_ANGLES >> 2)) % GEO_NUM_ANGLES;
749 int16_t rho = (g_Dis[distanceX] << (GEO_MAX_CU_LOG2+1)) + (g_Dis[distanceY] << (GEO_MAX_CU_LOG2 + 1));
750 static const int16_t maskOffset = (2*GEO_MAX_CU_SIZE - GEO_WEIGHT_MASK_SIZE) >> 1;
751 int index = 0;
752 for( int y = 0; y < GEO_WEIGHT_MASK_SIZE; y++ )
753 {
754 int16_t lookUpY = (((y + maskOffset) << 1) + 1) * g_Dis[distanceY];
755 for( int x = 0; x < GEO_WEIGHT_MASK_SIZE; x++, index++ )
756 {
757 int16_t sx_i = ((x + maskOffset) << 1) + 1;
758 int16_t weightIdx = sx_i * g_Dis[distanceX] + lookUpY - rho;
759 int weightLinearIdx = 32 + weightIdx;
760 g_globalGeoWeights[g_angle2mask[angleIdx]][index] = Clip3(0, 8, (weightLinearIdx + 4) >> 3);
761 g_globalGeoEncSADmask[g_angle2mask[angleIdx]][index] = weightIdx > 0 ? 1 : 0;
762 }
763 }
764 }
765
766 for( int hIdx = 0; hIdx < GEO_NUM_CU_SIZE; hIdx++ )
767 {
768 int16_t height = 1 << ( hIdx + GEO_MIN_CU_LOG2);
769 for( int wIdx = 0; wIdx < GEO_NUM_CU_SIZE; wIdx++ )
770 {
771 int16_t width = 1 << (wIdx + GEO_MIN_CU_LOG2);
772 for( int splitDir = 0; splitDir < GEO_NUM_PARTITION_MODE; splitDir++ )
773 {
774 int16_t angle = g_GeoParams[splitDir][0];
775 int16_t distance = g_GeoParams[splitDir][1];
776 int16_t offsetX = (GEO_WEIGHT_MASK_SIZE - width) >> 1;
777 int16_t offsetY = (GEO_WEIGHT_MASK_SIZE - height) >> 1;
778 if( distance > 0 )
779 {
780 if( angle % 16 == 8 || (angle % 16 != 0 && height >= width) )
781 {
782 offsetY += angle < 16 ? ((distance * (int32_t)height) >> 3) : -((distance * (int32_t)height) >> 3);
783 }
784 else
785 {
786 offsetX += angle < 16 ? ((distance * (int32_t)width) >> 3) : -((distance * (int32_t)width) >> 3);
787 }
788 }
789 g_weightOffset[splitDir][hIdx][wIdx][0] = offsetX;
790 g_weightOffset[splitDir][hIdx][wIdx][1] = offsetY;
791 }
792 }
793 }
794}
