239template<typename T>
240void AreaBuf<T>::fill(const T &val)
241{
242 if( sizeof( T ) == 1 )
243 {
244 if( width == stride )
245 {
246 ::memset( buf, reinterpret_cast< const signed char& >( val ), width * height * sizeof( T ) );
247 }
248 else
249 {
250 T* dest = buf;
251 size_t line = width * sizeof( T );
252
253 for( unsigned y = 0; y < height; y++ )
254 {
255 ::memset( dest, reinterpret_cast< const signed char& >( val ), line );
256
257 dest += stride;
258 }
259 }
260 }
261 else if( T( 0 ) == val )
262 {
263 if( width == stride )
264 {
265 ::memset( buf, 0, width * height * sizeof( T ) );
266 }
267 else
268 {
269 T* dest = buf;
270 size_t line = width * sizeof( T );
271
272 for( unsigned y = 0; y < height; y++ )
273 {
274 ::memset( dest, 0, line );
275
276 dest += stride;
277 }
278 }
279 }
280 else
281 {
282 T* dest = buf;
283
284#define FILL_INC dest += stride
285#define FILL_OP( ADDR ) dest[ADDR] = val
286
287 SIZE_AWARE_PER_EL_OP( FILL_OP, FILL_INC );
288
289#undef FILL_INC
290#undef FILL_OP
291 }

315template<typename T>
316void AreaBuf<T>::copyFrom( const AreaBuf<const T> &other )
317{
318#if !defined(__GNUC__) || __GNUC__ > 5
319 static_assert( std::is_trivially_copyable<T>::value, "Type T is not trivially_copyable" );
320#endif
321
322 CHECK( width != other.width, "Incompatible size" );
323 CHECK( height != other.height, "Incompatible size" );
324
325 if( buf == other.buf )
326 {
327 return;
328 }
329
330 if( width == stride && stride == other.stride )
331 {
332 memcpy( buf, other.buf, width * height * sizeof( T ) );
333 }
334 else
335 {
336 T* dst = buf;
337 const T* src = other.buf;
338 const unsigned srcStride = other.stride;
339
340 for( unsigned y = 0; y < height; y++ )
341 {
342 memcpy( dst, src, width * sizeof( T ) );
343
344 dst += stride;
345 src += srcStride;
346 }
347 }

419template<typename T>
420void AreaBuf<T>::removeWeightHighFreq(const AreaBuf<T>& other, const bool bClip, const ClpRng& clpRng, const int8_t bcwWeight)
421{
422 const int8_t bcwWeightOther = g_BcwWeightBase - bcwWeight;
423 const int8_t log2WeightBase = g_BcwLog2WeightBase;
424
425 const Pel* src = other.buf;
426 const int srcStride = other.stride;
427
428 Pel* dst = buf;
429 const int dstStride = stride;
430
431#if ENABLE_SIMD_OPT_BCW
432 if(!bClip)
433 {
434 if(!(width & 7))
435 g_pelBufOP.removeWeightHighFreq8(dst, dstStride, src, srcStride, width, height, 16, bcwWeight);
436 else if(!(width & 3))
437 g_pelBufOP.removeWeightHighFreq4(dst, dstStride, src, srcStride, width, height, 16, bcwWeight);
438 else
439 CHECK(true, "Not supported");
440 }
441 else
442 {
443#endif
444 Intermediate_Int normalizer = ((1 << 16) + (bcwWeight > 0 ? (bcwWeight >> 1) : -(bcwWeight >> 1))) / bcwWeight;
445 Intermediate_Int weight0 = normalizer << log2WeightBase;
446 Intermediate_Int weight1 = bcwWeightOther * normalizer;
447#define REM_HF_INC \
448 src += srcStride; \
449 dst += dstStride; \
450
451#define REM_HF_OP_CLIP( ADDR ) dst[ADDR] = ClipPel<T>( T((dst[ADDR]*weight0 - src[ADDR]*weight1 + (1<<15))>>16), clpRng )
452#define REM_HF_OP( ADDR ) dst[ADDR] = T((dst[ADDR]*weight0 - src[ADDR]*weight1 + (1<<15))>>16)
453
454 if(bClip)
455 {
456 SIZE_AWARE_PER_EL_OP(REM_HF_OP_CLIP, REM_HF_INC);
457 }
458 else
459 {
460 SIZE_AWARE_PER_EL_OP(REM_HF_OP, REM_HF_INC);
461 }
462
463#undef REM_HF_INC
464#undef REM_HF_OP
465#undef REM_HF_OP_CLIP
466#if ENABLE_SIMD_OPT_BCW
467 }
468#endif
469}

533template<typename T>
534void AreaBuf<T>::extendBorderPel(unsigned marginX, unsigned marginY)
535{
536 T* p = buf;
537 int h = height;
538 int w = width;
539 int s = stride;
540
541 CHECK((w + 2 * marginX) > s, "Size of buffer too small to extend");
542 // do left and right margins
543 for (int y = 0; y < h; y++)
544 {
545 for (int x = 0; x < marginX; x++)
546 {
547 *(p - marginX + x) = p[0];
548 p[w + x] = p[w - 1];
549 }
550 p += s;
551 }
552
553 // p is now the (0,height) (bottom left of image within bigger picture
554 p -= (s + marginX);
555 // p is now the (-margin, height-1)
556 for (int y = 0; y < marginY; y++)
557 {
558 ::memcpy(p + (y + 1) * s, p, sizeof(T) * (w + (marginX << 1)));
559 }
560
561 // p is still (-marginX, height-1)
562 p -= ((h - 1) * s);
563 // p is now (-marginX, 0)
564 for (int y = 0; y < marginY; y++)
565 {
566 ::memcpy(p - (y + 1) * s, p, sizeof(T) * (w + (marginX << 1)));
567 }
568}

648{
649 int64_t acc = 0;
650
651 CHECK( width != other.width, "Incompatible size" );
652 CHECK( height != other.height, "Incompatible size" );
653
654 const T* src1 = buf;
655 const T* src2 = other.buf;
656
657#define MEAN_DIFF_INC \
658 src1 += stride; \
659 src2 += other.stride; \
660
661#define MEAN_DIFF_OP(ADDR) acc += src1[ADDR] - src2[ADDR]

664
665#undef MEAN_DIFF_INC
666#undef MEAN_DIFF_OP

668 return T( acc / area() );
