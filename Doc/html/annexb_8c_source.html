136int get_annex_b_NALU (VideoParameters *p_Vid, NALU_t *nalu, ANNEXB_t *annex_b)
137{
138 int i;
139 int info2 = 0, info3 = 0, pos = 0;
140 int StartCodeFound = 0;
141 int LeadingZero8BitsCount = 0;
142 byte *pBuf = annex_b->Buf;
143
144 if (annex_b->nextstartcodebytes != 0)
145 {
146 for (i=0; i<annex_b->nextstartcodebytes-1; i++)
147 {
148 (*pBuf++) = 0;
149 pos++;
150 }
151 (*pBuf++) = 1;
152 pos++;
153 }
154 else
155 {
156 while(!annex_b->is_eof)
157 {
158 pos++;
159 if ((*(pBuf++)= getfbyte(annex_b))!= 0)
160 break;
161 }
162 }
163 if(annex_b->is_eof == TRUE)
164 {
165 if(pos==0)
166 {
167 return 0;
168 }
169 else
170 {
171 printf( "get_annex_b_NALU can't read start code\n");
172 return -1;
173 }
174 }
175
176 if(*(pBuf - 1) != 1 || pos < 3)
177 {
178 printf ("get_annex_b_NALU: no Start Code at the beginning of the NALU, return -1\n");
179 return -1;
180 }
181
182 if (pos == 3)
183 {
184 nalu->startcodeprefix_len = 3;
185 }
186 else
187 {
188 LeadingZero8BitsCount = pos - 4;
189 nalu->startcodeprefix_len = 4;
190 }
191
192 //the 1st byte stream NAL unit can has leading_zero_8bits, but subsequent ones are not
193 //allowed to contain it since these zeros(if any) are considered trailing_zero_8bits
194 //of the previous byte stream NAL unit.
195 if(!annex_b->IsFirstByteStreamNALU && LeadingZero8BitsCount > 0)
196 {
197 printf ("get_annex_b_NALU: The leading_zero_8bits syntax can only be present in the first byte stream NAL unit, return -1\n");
198 return -1;
199 }
200
201 LeadingZero8BitsCount = pos;
202 annex_b->IsFirstByteStreamNALU = 0;
203
204 while (!StartCodeFound)
205 {
206 if (annex_b->is_eof == TRUE)
207 {
208 pBuf -= 2;
209 while(*(pBuf--)==0)
210 pos--;
211
212 nalu->len = (pos - 1) - LeadingZero8BitsCount;
213 memcpy (nalu->buf, annex_b->Buf + LeadingZero8BitsCount, nalu->len);
214 nalu->forbidden_bit = (*(nalu->buf) >> 7) & 1;
215 nalu->nal_reference_idc = (NalRefIdc) ((*(nalu->buf) >> 5) & 3);
216 nalu->nal_unit_type = (NaluType) ((*(nalu->buf)) & 0x1f);
217 annex_b->nextstartcodebytes = 0;
218
219 // printf ("get_annex_b_NALU, eof case: pos %d nalu->len %d, nalu->reference_idc %d, nal_unit_type %d \n", pos, nalu->len, nalu->nal_reference_idc, nalu->nal_unit_type);
220
221#if TRACE
222 fprintf (p_Dec->p_trace, "\n\nLast NALU in File\n\n");
223 fprintf (p_Dec->p_trace, "Annex B NALU w/ %s startcode, len %d, forbidden_bit %d, nal_reference_idc %d, nal_unit_type %d\n\n",
224 nalu->startcodeprefix_len == 4?"long":"short", nalu->len, nalu->forbidden_bit, nalu->nal_reference_idc, nalu->nal_unit_type);
225 fflush (p_Dec->p_trace);
226#endif
227 return (pos - 1);
228 }
229
230 pos++;
231 *(pBuf ++) = getfbyte(annex_b);
232 info3 = FindStartCode(pBuf - 4, 3);
233 if(info3 != 1)
234 {
235 info2 = FindStartCode(pBuf - 3, 2);
236 StartCodeFound = info2 & 0x01;
237 }
238 else
239 StartCodeFound = 1;
240 }
241
242 // Here, we have found another start code (and read length of startcode bytes more than we should
243 // have. Hence, go back in the file
244 if(info3 == 1) //if the detected start code is 00 00 01, trailing_zero_8bits is sure not to be present
245 {
246 pBuf -= 5;
247 while(*(pBuf--) == 0)
248 pos--;
249 annex_b->nextstartcodebytes = 4;
250 }
251 else if (info2 == 1)
252 annex_b->nextstartcodebytes = 3;
253 else
254 {
255 printf(" Panic: Error in next start code search \n");
256 return -1;
257 }
258
259 pos -= annex_b->nextstartcodebytes;
260
261 // Here the leading zeros(if any), Start code, the complete NALU, trailing zeros(if any)
262 // and the next start code is in the Buf.
263 // The size of Buf is pos - rewind, pos are the number of bytes excluding the next
264 // start code, and (pos) - LeadingZero8BitsCount
265 // is the size of the NALU.
266
267 nalu->len = pos - LeadingZero8BitsCount;
268 fast_memcpy (nalu->buf, annex_b->Buf + LeadingZero8BitsCount, nalu->len);
269 nalu->forbidden_bit = (*(nalu->buf) >> 7) & 1;
270 nalu->nal_reference_idc = (NalRefIdc) ((*(nalu->buf) >> 5) & 3);
271 nalu->nal_unit_type = (NaluType) ((*(nalu->buf)) & 0x1f);
272 nalu->lost_packets = 0;
273
274 //printf ("get_annex_b_NALU, regular case: pos %d nalu->len %d, nalu->reference_idc %d, nal_unit_type %d \n", pos, nalu->len, nalu->nal_reference_idc, nalu->nal_unit_type);
275#if TRACE
276 fprintf (p_Dec->p_trace, "\n\nAnnex B NALU w/ %s startcode, len %d, forbidden_bit %d, nal_reference_idc %d, nal_unit_type %d\n\n",
277 nalu->startcodeprefix_len == 4?"long":"short", nalu->len, nalu->forbidden_bit, nalu->nal_reference_idc, nalu->nal_unit_type);
278 fflush (p_Dec->p_trace);
279#endif
280
281 return (pos);
