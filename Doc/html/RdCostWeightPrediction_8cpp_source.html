58Distortion RdCostWeightPrediction::xGetSADw( const DistParam &rcDtParam )
59{
60 const Pel *piOrg = rcDtParam.org.buf;
61 const Pel *piCur = rcDtParam.cur.buf;
62 const int iCols = rcDtParam.org.width;
63 const int iStrideCur = rcDtParam.cur.stride;
64 const int iStrideOrg = rcDtParam.org.stride;
65 const ComponentID compID = rcDtParam.compID;
66
67 CHECK( compID >= MAX_NUM_COMPONENT, "Invalid component" );
68
69 const WPScalingParam &wpCur = rcDtParam.wpCur[compID];
70
71 const int w0 = wpCur.w;
72 const int offset = wpCur.offset;
73 const int shift = wpCur.shift;
74 const int round = wpCur.round;
75 const int distortionShift = DISTORTION_PRECISION_ADJUSTMENT(rcDtParam.bitDepth);
76
77 Distortion uiSum = 0;
78
79#if !U0040_MODIFIED_WEIGHTEDPREDICTION_WITH_BIPRED_AND_CLIPPING
80 for(int iRows = rcDtParam.org.height; iRows != 0; iRows-- )
81 {
82 for (int n = 0; n < iCols; n++ )
83 {
84 const Pel pred = ( (w0*piCur[n] + round) >> shift ) + offset ;
85
86 uiSum += abs( piOrg[n] - pred );
87 }
88 if (rcDtParam.maximumDistortionForEarlyExit < ( uiSum >> distortionShift))
89 {
90 return uiSum >> distortionShift;
91 }
92 piOrg += iStrideOrg;
93 piCur += iStrideCur;
94 }
95
96 //rcDtParam.compID = MAX_NUM_COMPONENT; // reset for DEBUG (assert test)
97#else
98 ClpRng clpRng; // this just affects the cost
99 clpRng.min = 0;
100 clpRng.max = (1 << rcDtParam.bitDepth) - 1;
101
102 // Default weight
103 if (w0 == 1 << shift)
104 {
105 // no offset
106 if (offset == 0)
107 {
108 for(int iRows = rcDtParam.org.height; iRows != 0; iRows-- )
109 {
110 for (int n = 0; n < iCols; n++ )
111 {
112 uiSum += abs( piOrg[n] - piCur[n] );
113 }
114 if (rcDtParam.maximumDistortionForEarlyExit < ( uiSum >> distortionShift))
115 {
116 return uiSum >> distortionShift;
117 }
118 piOrg += iStrideOrg;
119 piCur += iStrideCur;
120 }
121 }
122 else
123 {
124 // Lets not clip for the bipredictive case since clipping should be done after
125 // combining both elements. Unfortunately the code uses the suboptimal "subtraction"
126 // method, which is faster but introduces the clipping issue (therefore Bipred is suboptimal).
127 if (rcDtParam.isBiPred)
128 {
129 for(int iRows = rcDtParam.org.height; iRows != 0; iRows-- )
130 {
131 for (int n = 0; n < iCols; n++ )
132 {
133 uiSum += abs( piOrg[n] - (piCur[n] + offset) );
134 }
135 if (rcDtParam.maximumDistortionForEarlyExit < ( uiSum >> distortionShift))
136 {
137 return uiSum >> distortionShift;
138 }
139
140 piOrg += iStrideOrg;
141 piCur += iStrideCur;
142 }
143 }
144 else
145 {
146 for(int iRows = rcDtParam.org.height; iRows != 0; iRows-- )
147 {
148 for (int n = 0; n < iCols; n++ )
149 {
150 const Pel pred = ClipPel( piCur[n] + offset, clpRng);
151
152 uiSum += abs( piOrg[n] - pred );
153 }
154 if (rcDtParam.maximumDistortionForEarlyExit < ( uiSum >> distortionShift))
155 {
156 return uiSum >> distortionShift;
157 }
158 piOrg += iStrideOrg;
159 piCur += iStrideCur;
160 }
161 }
162 }
163 }
164 else
165 {
166 // Lets not clip for the bipredictive case since clipping should be done after
167 // combining both elements. Unfortunately the code uses the suboptimal "subtraction"
168 // method, which is faster but introduces the clipping issue (therefore Bipred is suboptimal).
169 if (rcDtParam.isBiPred)
170 {
171 for(int iRows = rcDtParam.org.height; iRows != 0; iRows-- )
172 {
173 for (int n = 0; n < iCols; n++ )
174 {
175 const Pel pred = ( (w0*piCur[n] + round) >> shift ) + offset ;
176 uiSum += abs( piOrg[n] - pred );
177 }
178 if (rcDtParam.maximumDistortionForEarlyExit < ( uiSum >> distortionShift))
179 {
180 return uiSum >> distortionShift;
181 }
182
183 piOrg += iStrideOrg;
184 piCur += iStrideCur;
185 }
186 }
187 else
188 {
189 if (offset == 0)
190 {
191 for(int iRows = rcDtParam.org.height; iRows != 0; iRows-- )
192 {
193 for (int n = 0; n < iCols; n++ )
194 {
195 const Pel pred = ClipPel( (w0*piCur[n] + round) >> shift, clpRng);
196
197 uiSum += abs( piOrg[n] - pred );
198 }
199 if (rcDtParam.maximumDistortionForEarlyExit < ( uiSum >> distortionShift))
200 {
201 return uiSum >> distortionShift;
202 }
203 piOrg += iStrideOrg;
204 piCur += iStrideCur;
205 }
206 }
207 else
208 {
209 for(int iRows = rcDtParam.org.height; iRows != 0; iRows-- )
210 {
211 for (int n = 0; n < iCols; n++ )
212 {
213 const Pel pred = ClipPel(((w0*piCur[n] + round) >> shift ) + offset, clpRng) ;
214
215 uiSum += abs( piOrg[n] - pred );
216 }
217 if (rcDtParam.maximumDistortionForEarlyExit < ( uiSum >> distortionShift))
218 {
219 return uiSum >> distortionShift;
220 }
221 piOrg += iStrideOrg;
222 piCur += iStrideCur;
223 }
224 }
225 }
226 }
227 //rcDtParam.compIdx = MAX_NUM_COMPONENT; // reset for DEBUG (assert test)
228#endif
229
230 return uiSum >> distortionShift;
231}

241Distortion RdCostWeightPrediction::xGetSSEw( const DistParam &rcDtParam )
242{
243 const Pel *piOrg = rcDtParam.org.buf;
244 const Pel *piCur = rcDtParam.cur.buf;
245 const int iCols = rcDtParam.org.width;
246 const int iStrideCur = rcDtParam.cur.stride;
247 const int iStrideOrg = rcDtParam.org.stride;
248 const ComponentID compID = rcDtParam.compID;
249
250 CHECK( rcDtParam.subShift != 0, "Subshift not supported" ); // NOTE: what is this protecting?
251 CHECK( compID >= MAX_NUM_COMPONENT, "Invalid channel" );
252
253 const WPScalingParam &wpCur = rcDtParam.wpCur[compID];
254 const int w0 = wpCur.w;
255 const int offset = wpCur.offset;
256 const int shift = wpCur.shift;
257 const int round = wpCur.round;
258 const uint32_t distortionShift = DISTORTION_PRECISION_ADJUSTMENT(rcDtParam.bitDepth) << 1;
259
260 Distortion sum = 0;
261
262#if !U0040_MODIFIED_WEIGHTEDPREDICTION_WITH_BIPRED_AND_CLIPPING
263 for(int iRows = rcDtParam.org.height ; iRows != 0; iRows-- )
264 {
265 for (int n = 0; n < iCols; n++ )
266 {
267 const Pel pred = ( (w0*piCur[n] + round) >> shift ) + offset ;
268 const Pel residual = piOrg[n] - pred;
269 sum += ( Distortion(residual) * Distortion(residual) ) >> distortionShift;
270 }
271 piOrg += iStrideOrg;
272 piCur += iStrideCur;
273 }
274
275 //rcDtParam.compIdx = MAX_NUM_COMPONENT; // reset for DEBUG (assert test)
276#else
277 if (rcDtParam.isBiPred)
278 {
279 for(int iRows = rcDtParam.org.height ; iRows != 0; iRows-- )
280 {
281 for (int n = 0; n < iCols; n++ )
282 {
283 const Pel pred = ( (w0*piCur[n] + round) >> shift ) + offset ;
284 const Pel residual = piOrg[n] - pred;
285 sum += ( Distortion(residual) * Distortion(residual) ) >> distortionShift;
286 }
287 piOrg += iStrideOrg;
288 piCur += iStrideCur;
289 }
290 }
291 else
292 {
293 ClpRng clpRng; // this just affects the cost - fix this later
294 clpRng.min = 0;
295 clpRng.max = (1 << rcDtParam.bitDepth) - 1;
296
297 for(int iRows = rcDtParam.org.height; iRows != 0; iRows-- )
298 {
299 for (int n = 0; n < iCols; n++ )
300 {
301 const Pel pred = ClipPel(( (w0*piCur[n] + round) >> shift ) + offset, clpRng) ;
302 const Pel residual = piOrg[n] - pred;
303 sum += ( Distortion(residual) * Distortion(residual) ) >> distortionShift;
304 }
305 piOrg += iStrideOrg;
306 piCur += iStrideCur;
307 }
308 }
309
310 //rcDtParam.compIdx = MAX_NUM_COMPONENT; // reset for DEBUG (assert test)
311#endif
312
313 return sum;
314}

321Distortion xCalcHADs2x2w( const WPScalingParam &wpCur, const Pel *piOrg, const Pel *piCur, int iStrideOrg, int iStrideCur, int iStep )
322{
323 const int round = wpCur.round;
324 const int shift = wpCur.shift;
325 const int offset = wpCur.offset;
326 const int w0 = wpCur.w;
327
328 Distortion satd = 0;
329 TCoeff diff[4];
330 TCoeff m[4];
331
332 Pel pred;
333
334 pred = ( (w0*piCur[0*iStep ] + round) >> shift ) + offset ;
335 diff[0] = piOrg[0 ] - pred;
336 pred = ( (w0*piCur[1*iStep ] + round) >> shift ) + offset ;
337 diff[1] = piOrg[1 ] - pred;
338 pred = ( (w0*piCur[0*iStep + iStrideCur] + round) >> shift ) + offset ;
339 diff[2] = piOrg[iStrideOrg ] - pred;
340 pred = ( (w0*piCur[1*iStep + iStrideCur] + round) >> shift ) + offset ;
341 diff[3] = piOrg[iStrideOrg + 1] - pred;
342
343 m[0] = diff[0] + diff[2];
344 m[1] = diff[1] + diff[3];
345 m[2] = diff[0] - diff[2];
346 m[3] = diff[1] - diff[3];
347
348 satd += abs(m[0] + m[1]);
349 satd += abs(m[0] - m[1]);
350 satd += abs(m[2] + m[3]);
351 satd += abs(m[2] - m[3]);
352
353 return satd;
354}

358Distortion xCalcHADs4x4w( const WPScalingParam &wpCur, const Pel *piOrg, const Pel *piCur, int iStrideOrg, int iStrideCur, int iStep )
359{
360 const int round = wpCur.round;
361 const int shift = wpCur.shift;
362 const int offset = wpCur.offset;
363 const int w0 = wpCur.w;
364
365 Distortion satd = 0;
366 TCoeff diff[16];
367 TCoeff m[16];
368 TCoeff d[16];
369
370
371 for(int k = 0; k < 16; k+=4 )
372 {
373 Pel pred;
374 pred = ( (w0*piCur[0*iStep] + round) >> shift ) + offset ;
375 diff[k+0] = piOrg[0] - pred;
376 pred = ( (w0*piCur[1*iStep] + round) >> shift ) + offset ;
377 diff[k+1] = piOrg[1] - pred;
378 pred = ( (w0*piCur[2*iStep] + round) >> shift ) + offset ;
379 diff[k+2] = piOrg[2] - pred;
380 pred = ( (w0*piCur[3*iStep] + round) >> shift ) + offset ;
381 diff[k+3] = piOrg[3] - pred;
382
383 piCur += iStrideCur;
384 piOrg += iStrideOrg;
385 }
386
387 /*===== hadamard transform =====*/
388 m[ 0] = diff[ 0] + diff[12];
389 m[ 1] = diff[ 1] + diff[13];
390 m[ 2] = diff[ 2] + diff[14];
391 m[ 3] = diff[ 3] + diff[15];
392 m[ 4] = diff[ 4] + diff[ 8];
393 m[ 5] = diff[ 5] + diff[ 9];
394 m[ 6] = diff[ 6] + diff[10];
395 m[ 7] = diff[ 7] + diff[11];
396 m[ 8] = diff[ 4] - diff[ 8];
397 m[ 9] = diff[ 5] - diff[ 9];
398 m[10] = diff[ 6] - diff[10];
399 m[11] = diff[ 7] - diff[11];
400 m[12] = diff[ 0] - diff[12];
401 m[13] = diff[ 1] - diff[13];
402 m[14] = diff[ 2] - diff[14];
403 m[15] = diff[ 3] - diff[15];
404
405 d[ 0] = m[ 0] + m[ 4];
406 d[ 1] = m[ 1] + m[ 5];
407 d[ 2] = m[ 2] + m[ 6];
408 d[ 3] = m[ 3] + m[ 7];
409 d[ 4] = m[ 8] + m[12];
410 d[ 5] = m[ 9] + m[13];
411 d[ 6] = m[10] + m[14];
412 d[ 7] = m[11] + m[15];
413 d[ 8] = m[ 0] - m[ 4];
414 d[ 9] = m[ 1] - m[ 5];
415 d[10] = m[ 2] - m[ 6];
416 d[11] = m[ 3] - m[ 7];
417 d[12] = m[12] - m[ 8];
418 d[13] = m[13] - m[ 9];
419 d[14] = m[14] - m[10];
420 d[15] = m[15] - m[11];
421
422 m[ 0] = d[ 0] + d[ 3];
423 m[ 1] = d[ 1] + d[ 2];
424 m[ 2] = d[ 1] - d[ 2];
425 m[ 3] = d[ 0] - d[ 3];
426 m[ 4] = d[ 4] + d[ 7];
427 m[ 5] = d[ 5] + d[ 6];
428 m[ 6] = d[ 5] - d[ 6];
429 m[ 7] = d[ 4] - d[ 7];
430 m[ 8] = d[ 8] + d[11];
431 m[ 9] = d[ 9] + d[10];
432 m[10] = d[ 9] - d[10];
433 m[11] = d[ 8] - d[11];
434 m[12] = d[12] + d[15];
435 m[13] = d[13] + d[14];
436 m[14] = d[13] - d[14];
437 m[15] = d[12] - d[15];
438
439 d[ 0] = m[ 0] + m[ 1];
440 d[ 1] = m[ 0] - m[ 1];
441 d[ 2] = m[ 2] + m[ 3];
442 d[ 3] = m[ 3] - m[ 2];
443 d[ 4] = m[ 4] + m[ 5];
444 d[ 5] = m[ 4] - m[ 5];
445 d[ 6] = m[ 6] + m[ 7];
446 d[ 7] = m[ 7] - m[ 6];
447 d[ 8] = m[ 8] + m[ 9];
448 d[ 9] = m[ 8] - m[ 9];
449 d[10] = m[10] + m[11];
450 d[11] = m[11] - m[10];
451 d[12] = m[12] + m[13];
452 d[13] = m[12] - m[13];
453 d[14] = m[14] + m[15];
454 d[15] = m[15] - m[14];
455
456 for (int k=0; k<16; ++k)
457 {
458 satd += abs(d[k]);
459 }
460 satd = ((satd+1)>>1);
461
462 return satd;
463}

467Distortion xCalcHADs8x8w( const WPScalingParam &wpCur, const Pel *piOrg, const Pel *piCur, int iStrideOrg, int iStrideCur, int iStep )
468{
469 Distortion sad=0;
470 TCoeff diff[64], m1[8][8], m2[8][8], m3[8][8];
471 int iStep2 = iStep<<1;
472 int iStep3 = iStep2 + iStep;
473 int iStep4 = iStep3 + iStep;
474 int iStep5 = iStep4 + iStep;
475 int iStep6 = iStep5 + iStep;
476 int iStep7 = iStep6 + iStep;
477 const int round = wpCur.round;
478 const int shift = wpCur.shift;
479 const int offset = wpCur.offset;
480 const int w0 = wpCur.w;
481
482 Pel pred;
483
484 for(int k = 0; k < 64; k+=8 )
485 {
486 pred = ( (w0*piCur[ 0] + round) >> shift ) + offset ;
487 diff[k+0] = piOrg[0] - pred;
488 pred = ( (w0*piCur[iStep ] + round) >> shift ) + offset ;
489 diff[k+1] = piOrg[1] - pred;
490 pred = ( (w0*piCur[iStep2] + round) >> shift ) + offset ;
491 diff[k+2] = piOrg[2] - pred;
492 pred = ( (w0*piCur[iStep3] + round) >> shift ) + offset ;
493 diff[k+3] = piOrg[3] - pred;
494 pred = ( (w0*piCur[iStep4] + round) >> shift ) + offset ;
495 diff[k+4] = piOrg[4] - pred;
496 pred = ( (w0*piCur[iStep5] + round) >> shift ) + offset ;
497 diff[k+5] = piOrg[5] - pred;
498 pred = ( (w0*piCur[iStep6] + round) >> shift ) + offset ;
499 diff[k+6] = piOrg[6] - pred;
500 pred = ( (w0*piCur[iStep7] + round) >> shift ) + offset ;
501 diff[k+7] = piOrg[7] - pred;
502
503 piCur += iStrideCur;
504 piOrg += iStrideOrg;
505 }
506
507 //horizontal
508 for (int j=0; j < 8; j++)
509 {
510 const int jj = j << 3;
511 m2[j][0] = diff[jj ] + diff[jj+4];
512 m2[j][1] = diff[jj+1] + diff[jj+5];
513 m2[j][2] = diff[jj+2] + diff[jj+6];
514 m2[j][3] = diff[jj+3] + diff[jj+7];
515 m2[j][4] = diff[jj ] - diff[jj+4];
516 m2[j][5] = diff[jj+1] - diff[jj+5];
517 m2[j][6] = diff[jj+2] - diff[jj+6];
518 m2[j][7] = diff[jj+3] - diff[jj+7];
519
520 m1[j][0] = m2[j][0] + m2[j][2];
521 m1[j][1] = m2[j][1] + m2[j][3];
522 m1[j][2] = m2[j][0] - m2[j][2];
523 m1[j][3] = m2[j][1] - m2[j][3];
524 m1[j][4] = m2[j][4] + m2[j][6];
525 m1[j][5] = m2[j][5] + m2[j][7];
526 m1[j][6] = m2[j][4] - m2[j][6];
527 m1[j][7] = m2[j][5] - m2[j][7];
528
529 m2[j][0] = m1[j][0] + m1[j][1];
530 m2[j][1] = m1[j][0] - m1[j][1];
531 m2[j][2] = m1[j][2] + m1[j][3];
532 m2[j][3] = m1[j][2] - m1[j][3];
533 m2[j][4] = m1[j][4] + m1[j][5];
534 m2[j][5] = m1[j][4] - m1[j][5];
535 m2[j][6] = m1[j][6] + m1[j][7];
536 m2[j][7] = m1[j][6] - m1[j][7];
537 }
538
539 //vertical
540 for (int i=0; i < 8; i++)
541 {
542 m3[0][i] = m2[0][i] + m2[4][i];
543 m3[1][i] = m2[1][i] + m2[5][i];
544 m3[2][i] = m2[2][i] + m2[6][i];
545 m3[3][i] = m2[3][i] + m2[7][i];
546 m3[4][i] = m2[0][i] - m2[4][i];
547 m3[5][i] = m2[1][i] - m2[5][i];
548 m3[6][i] = m2[2][i] - m2[6][i];
549 m3[7][i] = m2[3][i] - m2[7][i];
550
551 m1[0][i] = m3[0][i] + m3[2][i];
552 m1[1][i] = m3[1][i] + m3[3][i];
553 m1[2][i] = m3[0][i] - m3[2][i];
554 m1[3][i] = m3[1][i] - m3[3][i];
555 m1[4][i] = m3[4][i] + m3[6][i];
556 m1[5][i] = m3[5][i] + m3[7][i];
557 m1[6][i] = m3[4][i] - m3[6][i];
558 m1[7][i] = m3[5][i] - m3[7][i];
559
560 m2[0][i] = m1[0][i] + m1[1][i];
561 m2[1][i] = m1[0][i] - m1[1][i];
562 m2[2][i] = m1[2][i] + m1[3][i];
563 m2[3][i] = m1[2][i] - m1[3][i];
564 m2[4][i] = m1[4][i] + m1[5][i];
565 m2[5][i] = m1[4][i] - m1[5][i];
566 m2[6][i] = m1[6][i] + m1[7][i];
567 m2[7][i] = m1[6][i] - m1[7][i];
568 }
569
570 for (int j=0; j < 8; j++)
571 {
572 for (int i=0; i < 8; i++)
573 {
574 sad += (abs(m2[j][i]));
575 }
576 }
577
578 sad=((sad+2)>>2);
579
580 return sad;
581}

585Distortion RdCostWeightPrediction::xGetHADsw( const DistParam &rcDtParam )
586{
587 const Pel *piOrg = rcDtParam.org.buf;
588 const Pel *piCur = rcDtParam.cur.buf;
589 const int iRows = rcDtParam.org.height;
590 const int iCols = rcDtParam.org.width;
591 const int iStrideCur = rcDtParam.cur.stride;
592 const int iStrideOrg = rcDtParam.org.stride;
593 const int iStep = rcDtParam.step;
594 const ComponentID compIdx = rcDtParam.compID;
595 CHECK(compIdx>=MAX_NUM_COMPONENT, "Invalid component");
596 const WPScalingParam &wpCur = rcDtParam.wpCur[compIdx];
597
598 Distortion uiSum = 0;
599
600
601 if( ( iRows % 8 == 0) && (iCols % 8 == 0) )
602 {
603 const int iOffsetOrg = iStrideOrg<<3;
604 const int iOffsetCur = iStrideCur<<3;
