42static inline size_t PathLastSlash(const std::string &path) {
43 size_t p = path.rfind('/');
44#if WIN32
45 size_t p1 = path.rfind('\\');
46 if ((p1 != std::string::npos) && ((p1 > p) || (p == std::string::npos)))
47 p = p1;
48#endif
49 return p;
50}

53bool IEquals(const std::string &a, const std::string &b) {
54 if (a.length() != b.length()) {
55 return false;
56 }
57
58 const char *aptr = a.c_str();
59 const char *bptr = b.c_str();
60
61 while (*aptr && *bptr) {
62 if (std::toupper(*aptr) != std::toupper(*bptr)) {
63 return false;
64 }
65
66 aptr++;
67 bptr++;
68 }
69
70 return true;
71}

95std::string PathDirectory(const std::string &path) {

104std::string PathFile(const std::string &path) {
105 size_t p = PathLastSlash(path);
106
107 if (p == std::string::npos)
108 return path;
109
110 return path.substr(p + 1, path.length() - p);
111}

113std::string PathExtension(const std::string &path) {
114 size_t p = path.rfind('.');
115 std::string res;
116
117 if (p != std::string::npos) {
118 res = path.substr(p + 1, res.length() - p);
119 }
120
121 return ToLower(res);
122}

124const std::string &PathNormalise(std::string &path, bool bDirectory) {
125 if (path.length() == 0)
126 return path;
127
128 // Convert backslashes to forward.
129 std::replace(path.begin(), path.end(), '\\', '/');
130
131 // Cap with a slash
132 if (bDirectory) {
133 if (path[path.length() - 1] != '/') {
134 path.append("/");
135 }
136 }
137
138 return path;
139}
