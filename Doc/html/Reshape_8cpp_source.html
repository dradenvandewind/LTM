63void Reshape::createDec(int bitDepth)
64{

66 m_reshapeLUTSize = 1 << m_lumaBD;
67 m_initCW = m_reshapeLUTSize / PIC_CODE_CW_BINS;
68 if (m_fwdLUT.empty())

70 if (m_invLUT.empty())

74 if (m_inputPivot.empty())
75 m_inputPivot.resize(PIC_CODE_CW_BINS + 1, 0);
76 if (m_fwdScaleCoef.empty())
77 m_fwdScaleCoef.resize(PIC_CODE_CW_BINS, 1 << FP_PREC);
78 if (m_invScaleCoef.empty())
79 m_invScaleCoef.resize(PIC_CODE_CW_BINS, 1 << FP_PREC);
80 if (m_reshapePivot.empty())
81 m_reshapePivot.resize(PIC_CODE_CW_BINS + 1, 0);
82 if (m_chromaAdjHelpLUT.empty())
83 m_chromaAdjHelpLUT.resize(PIC_CODE_CW_BINS, 1<<CSCALE_FP_PREC);

106int Reshape::calculateChromaAdjVpduNei(TransformUnit &tu, const CompArea &areaY)
107{
108 CodingStructure &cs = *tu.cs;
109 int xPos = areaY.lumaPos().x;
110 int yPos = areaY.lumaPos().y;
111 int ctuSize = cs.sps->getCTUSize();
112 int numNeighbor = std::min(64, ctuSize);
113 int numNeighborLog = floorLog2(numNeighbor);
114 if (ctuSize == 128)
115 {
116 xPos = xPos / 64 * 64;
117 yPos = yPos / 64 * 64;
118 }
119 else
120 {
121 xPos = xPos / ctuSize * ctuSize;
122 yPos = yPos / ctuSize * ctuSize;
123 }
124
125 if (isVPDUprocessed(xPos, yPos) && !cs.pcv->isEncoder)
126 {
127 return getChromaScale();
128 }
129 else
130 {
131 setVPDULoc(xPos, yPos);
132 Position topLeft(xPos, yPos);
133 CodingUnit *topLeftLuma;
134 const CodingUnit *cuAbove, *cuLeft;
135 if (CS::isDualITree(cs) && cs.slice->getSliceType() == I_SLICE)
136 {
137 topLeftLuma = tu.cs->picture->cs->getCU(topLeft, CHANNEL_TYPE_LUMA);
138 cuAbove = cs.picture->cs->getCURestricted(topLeftLuma->lumaPos().offset(0, -1), *topLeftLuma, CHANNEL_TYPE_LUMA);
139 cuLeft = cs.picture->cs->getCURestricted(topLeftLuma->lumaPos().offset(-1, 0), *topLeftLuma, CHANNEL_TYPE_LUMA);
140 }
141 else
142 {
143 topLeftLuma = cs.getCU(topLeft, CHANNEL_TYPE_LUMA);
144 cuAbove = cs.getCURestricted(topLeftLuma->lumaPos().offset(0, -1), *topLeftLuma, CHANNEL_TYPE_LUMA);
145 cuLeft = cs.getCURestricted(topLeftLuma->lumaPos().offset(-1, 0), *topLeftLuma, CHANNEL_TYPE_LUMA);
146 }
147
148 xPos = topLeftLuma->lumaPos().x;
149 yPos = topLeftLuma->lumaPos().y;
150
151 CompArea lumaArea = CompArea(COMPONENT_Y, tu.chromaFormat, topLeftLuma->lumaPos(), topLeftLuma->lumaSize(), true);
152 PelBuf piRecoY = cs.picture->getRecoBuf(lumaArea);
153 int strideY = piRecoY.stride;
154 int chromaScale = (1 << CSCALE_FP_PREC);
155 int lumaValue = -1;
156
157 Pel* recSrc0 = piRecoY.bufAt(0, 0);
158 const uint32_t picH = tu.cs->picture->lheight();
159 const uint32_t picW = tu.cs->picture->lwidth();
160 const Pel valueDC = 1 << (tu.cs->sps->getBitDepth(CHANNEL_TYPE_LUMA) - 1);
161 int32_t recLuma = 0;
162 int pelnum = 0;
163 if (cuLeft != nullptr)
164 {
165 for (int i = 0; i < numNeighbor; i++)
166 {
167 int k = (yPos + i) >= picH ? (picH - yPos - 1) : i;
168 recLuma += recSrc0[-1 + k * strideY];
169 pelnum++;
170 }
171 }
172 if (cuAbove != nullptr)
173 {
174 for (int i = 0; i < numNeighbor; i++)
175 {
176 int k = (xPos + i) >= picW ? (picW - xPos - 1) : i;
177 recLuma += recSrc0[-strideY + k];
178 pelnum++;
179 }
180 }
181 if (pelnum == numNeighbor)
182 {
183 lumaValue = (recLuma + (1 << (numNeighborLog - 1))) >> numNeighborLog;
184 }
185 else if (pelnum == (numNeighbor << 1))
186 {
187 lumaValue = (recLuma + (1 << numNeighborLog)) >> (numNeighborLog + 1);
188 }
189 else
190 {
191 CHECK(pelnum != 0, "");
192 lumaValue = valueDC;
193 }
194 chromaScale = calculateChromaAdj(lumaValue);
195 setChromaScale(chromaScale);
196 return(chromaScale);
197 }

203int Reshape::getPWLIdxInv(int lumaVal)
204{
205 int idxS = 0;
206 for (idxS = m_sliceReshapeInfo.reshaperModelMinBinIdx; (idxS <= m_sliceReshapeInfo.reshaperModelMaxBinIdx); idxS++)
207 {
208 if (lumaVal < m_reshapePivot[idxS + 1]) break;
209 }
210 return std::min(idxS, PIC_CODE_CW_BINS-1);
211}

240void Reshape::constructReshaper()
241{
242 int pwlFwdLUTsize = PIC_CODE_CW_BINS;
243 int pwlFwdBinLen = m_reshapeLUTSize / PIC_CODE_CW_BINS;
244
245 for (int i = 0; i < m_sliceReshapeInfo.reshaperModelMinBinIdx; i++)
246 m_binCW[i] = 0;
247 for (int i = m_sliceReshapeInfo.reshaperModelMaxBinIdx + 1; i < PIC_CODE_CW_BINS; i++)
248 m_binCW[i] = 0;
249 for (int i = m_sliceReshapeInfo.reshaperModelMinBinIdx; i <= m_sliceReshapeInfo.reshaperModelMaxBinIdx; i++)
250 m_binCW[i] = (uint16_t)(m_sliceReshapeInfo.reshaperModelBinCWDelta[i] + (int)m_initCW);
251
252 for (int i = 0; i < pwlFwdLUTsize; i++)
253 {
254 m_reshapePivot[i + 1] = m_reshapePivot[i] + m_binCW[i];
255 m_inputPivot[i + 1] = m_inputPivot[i] + m_initCW;
256 m_fwdScaleCoef[i] = ((int32_t)m_binCW[i] * (1 << FP_PREC) + (1 << (floorLog2(pwlFwdBinLen) - 1))) >> floorLog2(pwlFwdBinLen);
257 if (m_binCW[i] == 0)
258 {
259 m_invScaleCoef[i] = 0;
260 m_chromaAdjHelpLUT[i] = 1 << CSCALE_FP_PREC;
261 }
262 else
263 {
264 m_invScaleCoef[i] = (int32_t)(m_initCW * (1 << FP_PREC) / m_binCW[i]);
265 m_chromaAdjHelpLUT[i] = (int32_t)(m_initCW * (1 << FP_PREC) / ( m_binCW[i] + m_sliceReshapeInfo.chrResScalingOffset ) );
266 }
267 }
268 for (int lumaSample = 0; lumaSample < m_reshapeLUTSize; lumaSample++)
269 {
270 int idxY = lumaSample / m_initCW;
271 int tempVal = m_reshapePivot[idxY] + ((m_fwdScaleCoef[idxY] * (lumaSample - m_inputPivot[idxY]) + (1 << (FP_PREC - 1))) >> FP_PREC);
272 m_fwdLUT[lumaSample] = Clip3((Pel)0, (Pel)((1 << m_lumaBD) - 1), (Pel)(tempVal));
273
274 int idxYInv = getPWLIdxInv(lumaSample);
275 int invSample = m_inputPivot[idxYInv] + ((m_invScaleCoef[idxYInv] * (lumaSample - m_reshapePivot[idxYInv]) + (1 << (FP_PREC - 1))) >> FP_PREC);
276 m_invLUT[lumaSample] = Clip3((Pel)0, (Pel)((1 << m_lumaBD) - 1), (Pel)(invSample));
277 }
278}
