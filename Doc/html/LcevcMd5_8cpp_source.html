166void lcevc_md5_update(LCEVC_MD5 *md5, void *buf_t, u32 len) {
167 u8 *buf;
168 u32 i, idx, part_len;
169
170 buf = (u8 *)buf_t;
171
172 idx = (u32)((md5->bits[0] >> 3) & 0x3f);
173
174 md5->bits[0] += (len << 3);
175 if (md5->bits[0] < (len << 3)) {
176 (md5->bits[1])++;
177 }
178
179 md5->bits[1] += (len >> 29);
180 part_len = 64 - idx;
181
182 if (len >= part_len) {
183 lcevc_mcpy(md5->msg + idx, buf, part_len);
184 lcevc_md5_trans(md5->h, (u32 *)md5->msg);
185
186 for (i = part_len; i + 63 < len; i += 64) {
187 lcevc_md5_trans(md5->h, (u32 *)(buf + i));
188 }
189 idx = 0;
190 } else {
191 i = 0;
192 }
193
194 if (len - i > 0) {
195 lcevc_mcpy(md5->msg + idx, buf + i, len - i);
196 }
197}

201void lcevc_md5_update_16(LCEVC_MD5 *md5, void *buf_t, u32 len) {
202 u16 *buf;
203 u32 i, idx, part_len, j;
204 u8 t[512];
205
206 buf = (u16 *)buf_t;
207 idx = (u32)((md5->bits[0] >> 3) & 0x3f);
208
209 len = len * 2;
210 for (j = 0; j < len; j += 2) {
211 t[j] = (u8)(*(buf));
212 t[j + 1] = *(buf) >> 8;
213 buf++;
214 }
215
216 md5->bits[0] += (len << 3);
217 if (md5->bits[0] < (len << 3)) {
218 (md5->bits[1])++;
219 }
220
221 md5->bits[1] += (len >> 29);
222 part_len = 64 - idx;
223
224 if (len >= part_len) {
225 lcevc_mcpy(md5->msg + idx, t, part_len);
226 lcevc_md5_trans(md5->h, (u32 *)md5->msg);
227
228 for (i = part_len; i + 63 < len; i += 64) {
229 lcevc_md5_trans(md5->h, (u32 *)(t + i));
230 }
231 idx = 0;
232 } else {
233 i = 0;
234 }
235
236 if (len - i > 0) {
237 lcevc_mcpy(md5->msg + idx, t + i, len - i);
238 }

243void lcevc_md5_finish(LCEVC_MD5 *md5, u8 digest[16]) {
244 u8 *pos;
245 int cnt;
246
247 cnt = (md5->bits[0] >> 3) & 0x3F;
248 pos = md5->msg + cnt;
249 *pos++ = 0x80;
250 cnt = 64 - 1 - cnt;
251
252 if (cnt < 8) {
253 lcevc_mset(pos, 0, cnt);
254 lcevc_md5_trans(md5->h, (u32 *)md5->msg);
255 lcevc_mset(md5->msg, 0, 56);
256 } else {
257 lcevc_mset(pos, 0, cnt - 8);
258 }
259
260 lcevc_mcpy((md5->msg + 14 * sizeof(u32)), &md5->bits[0], sizeof(u32));
261 lcevc_mcpy((md5->msg + 15 * sizeof(u32)), &md5->bits[1], sizeof(u32));
262
263 lcevc_md5_trans(md5->h, (u32 *)md5->msg);
264 lcevc_mcpy(digest, md5->h, 16);
265 lcevc_mset(md5, 0, sizeof(LCEVC_MD5));
266}

291int lcevc_md5_imgb(LCEVC_IMGB *imgb, u8 digest[N_C][16]) {
292 LCEVC_MD5 md5[N_C];
293 int i, j;
294
295 for (i = 0; i < imgb->np; i++) {
296 lcevc_md5_init(&md5[i]);
297
298 // fprintf(stdout, "p %4d - h %4d - w %4d - s %4d \n", i, imgb->h[i], imgb->w[i], imgb->s[i]); fflush(stdout);
299
300 for (j = imgb->y[i]; j < imgb->h[i]; j++) {
301 lcevc_md5_update(&md5[i], ((u8 *)imgb->a[i]) + j * imgb->s[i] + imgb->x[i], imgb->w[i] * 2);
302 }
303
304 lcevc_md5_finish(&md5[i], digest[i]);
305 }
306
307 return LCEVC_OK;
308}
