95// Push encoded data into decoder
96//
97// To avoid messing with JM any more than necessary, this cheats a bit and saves the entire input bitstream
98// until EOS, then starts the JM decoder with the whole stream.
99//
100static int32_t push_packet(CodecContext c, const uint8_t *data, size_t length, CodecMetadata metadata, int8_t eos,
101 CodecError *error) {
102 Context *context = (Context *)c;
103 if (error)
104 *error = 0;
105
106 switch (context->state_) {
107 case StatePushing:
108 if (!eos) {
109 CHECK(data);
110 CHECK(length >= 3);
111 // Fixup NALU start code - if first is 3 bytes, make it 4
112 if (data[0] == '\0' && data[1] == '\0' && data[2] == '\1') {
113 char const zero = '\0';
114 context->base_bitstream_.push_back(0);
115 }
116
117 // Accumulate AUs into a temporary buffer
118 context->base_bitstream_.insert(context->base_bitstream_.end(), data, data + length);
119 } else {
121 context->state_ = StateDecoding;
122 }
123 return 0;
124
125 default:
126 ERR("Unexpected decoder state");
127 }
128
129 return 0;
130}

132static bool extract_image(DecodedPicList *pic, CodecImage *codec_image) {
133
134 // Is there a picture?
135 if (!pic || !pic->pY || !pic->pU || !pic->pV) {
136 return false;
137 }
138
139 if (!pic->bValid) {
140 INFO("Picture not marked valid?");
141 return false;
142 }
143
144 CHECK(codec_image);
145 codec_image->bpp = ((pic->iBitDepth + 7) >> 3);
146 codec_image->width_y = pic->iWidth;
147 codec_image->height_y = pic->iHeight;
148 codec_image->stride_y = pic->iYBufStride;
149
150 codec_image->width_uv = (pic->iYUVFormat != YUV444) ? (pic->iWidth / 2) : pic->iWidth;
151 codec_image->height_uv = (pic->iYUVFormat == YUV420) ? (pic->iHeight / 2) : pic->iHeight;
152 codec_image->stride_uv = pic->iUVBufStride;
153
154 codec_image->data_y = pic->pY;
155 codec_image->data_u = pic->pU;
156 codec_image->data_v = pic->pV;
157
158 return true;
159}

161static int32_t pull_image(CodecContext c, CodecImage *image, CodecMetadata *metadata, int8_t *eos, CodecError *error) {
162 Context *context = (Context *)c;
163 if (error)
164 *error = 0;
165
166 switch (context->state_) {
167 case StatePushing:
168 // Still accumulating input stream
169 *eos = 0;
170 return 0;
171
172 case StateDecoding:
173 context->decoder_thread_ = std::thread([context] {
174 context->flush();
175 context->decoded_images_.push(nullptr);
176 });
177
178 context->state_ = StateFlushing;
179 /* Fall through */
180
181 case StateFlushing:
182 context->decoded_images_.pop(context->decoded_image_);
183 if (context->decoded_image_) {
184 *image = context->decoded_image_->image_;
185 *eos = 0;
186 return 1;
187 } else {
188 if (context->decoder_thread_.joinable())
189 context->decoder_thread_.join();
190 *eos = 1;
191 return 0;
192 }
193 default:
194 ERR("Unexpected decoder state");
195 }
196
197 return 0;
198}

202void Context::flush() {
203
204 // Poke bitstream into our JM hooks
205 inject_annex_b_bitstream(base_bitstream_.data(), (unsigned)base_bitstream_.size());
206
207 InputParameters input_parameters;
208 memset(&input_parameters, 0, sizeof(input_parameters));
209 strcpy(input_parameters.infile, "");
210 strcpy(input_parameters.outfile, "");
211 strcpy(input_parameters.reffile, "");
212
213 const char *const decoder_args[] = {"ldecod"};
214 ParseCommand(&input_parameters, ARRAY_SIZE(decoder_args), (char **)decoder_args);
215
216 // Open decoder
217 if (OpenDecoder(&input_parameters) != DEC_OPEN_NOERR) {
218 ERR("Cannot open base decoder");
219 }
220
221 p_Dec->p_Vid->p_out = -2;
222
223 // Run decoder
224 //
225 DecodedPicList *pDecPicList = 0;
226 int frame_count = 0;
227
228 int r = 0;
229 do {
230 r = DecodeOneFrame(&pDecPicList);
231
232 if (r == DEC_EOS || r == DEC_SUCCEED) {
233 decode_enhanced_frames(pDecPicList, false);
234 frame_count++;
235 } else {
236 ERR("Error in decoding process: 0x%x", r);
237 }
238 } while ((r == DEC_SUCCEED) && ((p_Dec->p_Inp->iDecFrmNum == 0) || (frame_count < p_Dec->p_Inp->iDecFrmNum)));
239
240 FinitDecoder(&pDecPicList);
241
242 // Write last frame
243 decode_enhanced_frames(pDecPicList, true);
244
245 CloseDecoder();
246}

250void Context::decode_enhanced_frames(DecodedPicList *pic, bool all_frames) {
251 // Check picture and plane pointers - valid flag seems to be wrong sometimes!?
252 if (!pic || !pic->pY || !pic->pU || !pic->pV) {
253 return;
254 }
255
256 if (!pic->bValid) {
257 INFO("Picture not marked valid?");
258 }
259
260 do {
261 auto output_image = std::shared_ptr<DecodedImage>(new DecodedImage());
262
263 output_image->image_.bpp = ((pic->iBitDepth + 7) >> 3);
264 output_image->image_.width_y = pic->iWidth;
265 output_image->image_.height_y = pic->iHeight;
266 output_image->image_.stride_y = pic->iYBufStride;
267
268 output_image->image_.width_uv = (pic->iYUVFormat != YUV444) ? (pic->iWidth / 2) : pic->iWidth;
269 output_image->image_.height_uv = (pic->iYUVFormat == YUV420) ? (pic->iHeight / 2) : pic->iHeight;
270 output_image->image_.stride_uv = pic->iUVBufStride;
271
272 const size_t plane_size_y = output_image->image_.height_y * output_image->image_.stride_y;
273 const size_t plane_size_uv = output_image->image_.height_uv * output_image->image_.stride_uv;
274
275 output_image->buffer_.reserve(plane_size_y + 2 * plane_size_uv);
276
277 memcpy(output_image->buffer_.data(), pic->pY, plane_size_y);
278 memcpy(output_image->buffer_.data() + plane_size_y, pic->pU, plane_size_uv);
279 memcpy(output_image->buffer_.data() + plane_size_y + plane_size_uv, pic->pV, plane_size_uv);
280
281 output_image->image_.data_y = output_image->buffer_.data();
282 output_image->image_.data_u = output_image->image_.data_y + plane_size_y;
283 output_image->image_.data_v = output_image->image_.data_u + plane_size_uv;
284
285 decoded_images_.push(output_image);
286 // Mark as used and move to next
287 pic->bValid = 0;
288 pic = pic->pNext;
289 } while (pic != NULL && pic->bValid && all_frames);
290}
