112void OutputBitstream::write ( uint32_t uiBits, uint32_t uiNumberOfBits )
113{
114 CHECK( uiNumberOfBits > 32, "Number of bits is exceeds '32'" );
115 CHECK( uiNumberOfBits != 32 && (uiBits & (~0 << uiNumberOfBits)) != 0, "Unsupported parameters" );
116
117 /* any modulo 8 remainder of num_total_bits cannot be written this time,
118 * and will be held until next time. */
119 uint32_t num_total_bits = uiNumberOfBits + m_num_held_bits;
120 uint32_t next_num_held_bits = num_total_bits % 8;
121
122 /* form a byte aligned word (write_bits), by concatenating any held bits
123 * with the new bits, discarding the bits that will form the next_held_bits.
124 * eg: H = held bits, V = n new bits /---- next_held_bits
125 * len(H)=7, len(V)=1: ... ---- HHHH HHHV . 0000 0000, next_num_held_bits=0
126 * len(H)=7, len(V)=2: ... ---- HHHH HHHV . V000 0000, next_num_held_bits=1
127 * if total_bits < 8, the value of v_ is not used */
128 uint8_t next_held_bits = uiBits << (8 - next_num_held_bits);
129
130 if (!(num_total_bits >> 3))
131 {
132 /* insufficient bits accumulated to write out, append new_held_bits to
133 * current held_bits */
134 /* NB, this requires that v only contains 0 in bit positions {31..n} */
135 m_held_bits |= next_held_bits;
136 m_num_held_bits = next_num_held_bits;
137 return;
138 }
139
140 /* topword serves to justify held_bits to align with the msb of uiBits */
141 uint32_t topword = (uiNumberOfBits - next_num_held_bits) & ~((1 << 3) -1);
142 uint32_t write_bits = (m_held_bits << topword) | (uiBits >> next_num_held_bits);
143
144 switch (num_total_bits >> 3)
145 {
146 case 4: m_fifo.push_back(write_bits >> 24);
147 case 3: m_fifo.push_back(write_bits >> 16);
148 case 2: m_fifo.push_back(write_bits >> 8);
149 case 1: m_fifo.push_back(write_bits);
150 }
151
152 m_held_bits = next_held_bits;
153 m_num_held_bits = next_num_held_bits;
154}

180{
181 uint32_t uiNumBits = pcSubstream->getNumberOfWrittenBits();
182
183 const vector<uint8_t>& rbsp = pcSubstream->getFIFO();
184 for (vector<uint8_t>::const_iterator it = rbsp.begin(); it != rbsp.end();)
185 {
186 write(*it++, 8);
187 }
188 if (uiNumBits&0x7)
189 {
190 write(pcSubstream->getHeldBits()>>(8-(uiNumBits&0x7)), uiNumBits&0x7);

200int OutputBitstream::countStartCodeEmulations()
201{
202 uint32_t cnt = 0;
203 vector<uint8_t>& rbsp = getFIFO();
204 for (vector<uint8_t>::iterator it = rbsp.begin(); it != rbsp.end();)
205 {
206 vector<uint8_t>::iterator found = it;
207 do
208 {
209 // find the next emulated 00 00 {00,01,02,03}
210 // NB, end()-1, prevents finding a trailing two byte sequence
211 found = search_n(found, rbsp.end()-1, 2, 0);
212 found++;
213 // if not found, found == end, otherwise found = second zero byte
214 if (found == rbsp.end())
215 {
216 break;
217 }
218 if (*(++found) <= 3)
219 {
220 break;
221 }
222 } while (true);
223 it = found;
224 if (found != rbsp.end())
225 {
226 cnt++;
227 }
228 }
229 return cnt;
230}

240void InputBitstream::pseudoRead ( uint32_t uiNumberOfBits, uint32_t& ruiBits )
241{
242 uint32_t saved_num_held_bits = m_num_held_bits;
243 uint8_t saved_held_bits = m_held_bits;
244 uint32_t saved_fifo_idx = m_fifo_idx;
245
246 uint32_t num_bits_to_read = min(uiNumberOfBits, getNumBitsLeft());
247 read(num_bits_to_read, ruiBits);
248 ruiBits <<= (uiNumberOfBits - num_bits_to_read);
249
250 m_fifo_idx = saved_fifo_idx;
251 m_held_bits = saved_held_bits;
252 m_num_held_bits = saved_num_held_bits;

256void InputBitstream::read (uint32_t uiNumberOfBits, uint32_t& ruiBits)
257{
258 CHECK( uiNumberOfBits > 32, "Too many bits read" );
259
260 m_numBitsRead += uiNumberOfBits;
261
262 /* NB, bits are extracted from the MSB of each byte. */
263 uint32_t retval = 0;
264 if (uiNumberOfBits <= m_num_held_bits)
265 {
266 /* n=1, len(H)=7: -VHH HHHH, shift_down=6, mask=0xfe
267 * n=3, len(H)=7: -VVV HHHH, shift_down=4, mask=0xf8
268 */
269 retval = m_held_bits >> (m_num_held_bits - uiNumberOfBits);
270 retval &= ~(0xff << uiNumberOfBits);
271 m_num_held_bits -= uiNumberOfBits;
272 ruiBits = retval;
273 return;
274 }
275
276 /* all num_held_bits will go into retval
277 * => need to mask leftover bits from previous extractions
278 * => align retval with top of extracted word */
279 /* n=5, len(H)=3: ---- -VVV, mask=0x07, shift_up=5-3=2,
280 * n=9, len(H)=3: ---- -VVV, mask=0x07, shift_up=9-3=6 */
281 uiNumberOfBits -= m_num_held_bits;
282 retval = m_held_bits & ~(0xff << m_num_held_bits);
283 retval <<= uiNumberOfBits;
284
285 /* number of whole bytes that need to be loaded to form retval */
286 /* n=32, len(H)=0, load 4bytes, shift_down=0
287 * n=32, len(H)=1, load 4bytes, shift_down=1
288 * n=31, len(H)=1, load 4bytes, shift_down=1+1
289 * n=8, len(H)=0, load 1byte, shift_down=0
290 * n=8, len(H)=3, load 1byte, shift_down=3
291 * n=5, len(H)=1, load 1byte, shift_down=1+3
292 */
293 uint32_t aligned_word = 0;
294 uint32_t num_bytes_to_load = (uiNumberOfBits - 1) >> 3;
295 CHECK(m_fifo_idx + num_bytes_to_load >= m_fifo.size(), "Exceeded FIFO size");
296
297 switch (num_bytes_to_load)
298 {
299 case 3: aligned_word = m_fifo[m_fifo_idx++] << 24;
300 case 2: aligned_word |= m_fifo[m_fifo_idx++] << 16;
301 case 1: aligned_word |= m_fifo[m_fifo_idx++] << 8;
302 case 0: aligned_word |= m_fifo[m_fifo_idx++];
303 }
304
305 /* resolve remainder bits */
306 uint32_t next_num_held_bits = (32 - uiNumberOfBits) % 8;
307
308 /* copy required part of aligned_word into retval */
309 retval |= aligned_word >> next_num_held_bits;
310
311 /* store held bits */
312 m_num_held_bits = next_num_held_bits;
313 m_held_bits = aligned_word;
314
315 ruiBits = retval;
316}

359InputBitstream *InputBitstream::extractSubstream( uint32_t uiNumBits )
360{
361 uint32_t uiNumBytes = uiNumBits/8;
362 InputBitstream *pResult = new InputBitstream;
363
364 std::vector<uint8_t> &buf = pResult->getFifo();
365 buf.reserve((uiNumBits+7)>>3);
366
367 if (m_num_held_bits == 0)
368 {
369 std::size_t currentOutputBufferSize=buf.size();
370 const uint32_t uiNumBytesToReadFromFifo = std::min<uint32_t>(uiNumBytes, (uint32_t)m_fifo.size() - m_fifo_idx);
371 buf.resize(currentOutputBufferSize+uiNumBytes);
372 memcpy(&(buf[currentOutputBufferSize]), &(m_fifo[m_fifo_idx]), uiNumBytesToReadFromFifo); m_fifo_idx+=uiNumBytesToReadFromFifo;
373 if (uiNumBytesToReadFromFifo != uiNumBytes)
374 {
375 memset(&(buf[currentOutputBufferSize+uiNumBytesToReadFromFifo]), 0, uiNumBytes - uiNumBytesToReadFromFifo);
376 }
377 }
378 else
379 {
380 for (uint32_t ui = 0; ui < uiNumBytes; ui++)
381 {
382 uint32_t uiByte;
383 read(8, uiByte);
384 buf.push_back(uiByte);
385 }
386 }
387 if (uiNumBits&0x7)
388 {
389 uint32_t uiByte = 0;
390 read(uiNumBits&0x7, uiByte);
391 uiByte <<= 8-(uiNumBits&0x7);
392 buf.push_back(uiByte);
393 }
394 return pResult;
395}

397uint32_t InputBitstream::readByteAlignment()
398{
399 uint32_t code = 0;
400 read( 1, code );
401 CHECK(code != 1, "Code is not '1'");
402
403 uint32_t numBits = getNumBitsUntilByteAligned();
404 if(numBits)
405 {
406 CHECK(numBits > getNumBitsLeft(), "More bits available than left");
407 read( numBits, code );
408 CHECK(code != 0, "Code not '0'");
409 }
410 return numBits+1;
411}
