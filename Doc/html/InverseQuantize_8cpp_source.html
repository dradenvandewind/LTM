63int32_t find_quant_matrix_coeff(const PictureConfiguration &picture_configuration, unsigned num_layers, bool horizontal_only,
64 unsigned loq, unsigned layer, bool is_idr, int32_t previous) {
65
66 const uint8_t *d = nullptr;
67 if (num_layers == 4) {
68 // 2x2 case
69 if (loq == LOQ_LEVEL_1) {
70 d = default_qm_coefficient_2x2[2]; // LOQ1
71 } else {
72 if (horizontal_only)
73 d = default_qm_coefficient_2x2[0]; // LOQ2 1D
74 else
75 d = default_qm_coefficient_2x2[1]; // LOQ2 2D
76 }
77 } else {
78 // 4x4 case
79 if (loq == LOQ_LEVEL_1) {
80 d = default_qm_coefficient_4x4[2]; // LOQ1
81 } else {
82 if (horizontal_only)
83 d = default_qm_coefficient_4x4[0]; // LOQ2 1D
84 else
85 d = default_qm_coefficient_4x4[1]; // LOQ2 2D
86 }
87 }
88
89 switch (picture_configuration.quant_matrix_mode) {
90 case QuantMatrix_BothPrevious:
91 if (is_idr)
92 return (int32_t)d[layer];
93 else
94 return (previous == -1) ? (int32_t)d[layer] : previous;
95
96 case QuantMatrix_BothDefault:
97 return (int32_t)d[layer];
98
99 case QuantMatrix_SameAndCustom:
100 return picture_configuration.qm_coefficient_2[layer];
101
102 case QuantMatrix_Level2CustomLevel1Default:
103 if (loq == LOQ_LEVEL_2)
104 return picture_configuration.qm_coefficient_2[layer];
105 else
106 return d[layer];
107
108 case QuantMatrix_Level2DefaultLevel1Custom:
109 if (loq == LOQ_LEVEL_1)
110 return picture_configuration.qm_coefficient_1[layer];
111 else
112 return d[layer];
113
114 case QuantMatrix_DifferentAndCustom:
115 if (loq == LOQ_LEVEL_2)
116 return picture_configuration.qm_coefficient_2[layer];
117 else
118 return picture_configuration.qm_coefficient_1[layer];
119
120 default:
121 CHECK(0);
122 return 0;
123 }
124}

131int32_t find_dirq_step_width(int32_t orig_step_width, int32_t quant_matrix_coeff) {
132 int64_t dirq_step_width_64 = quant_matrix_coeff;
133 dirq_step_width_64 = (dirq_step_width_64 * orig_step_width) + (1 << 16);
134 dirq_step_width_64 = clamp(dirq_step_width_64, (int64_t)0, (int64_t)(3 << 16));
135 dirq_step_width_64 = (dirq_step_width_64 * orig_step_width) >> 16;
136 int32_t dirq_step_width_32 = (int32_t)clamp(dirq_step_width_64, (int64_t)MIN_STEP_WIDTH, (int64_t)MAX_STEP_WIDTH);
137 return dirq_step_width_32;
138}

140int32_t find_invq_offset(const PictureConfiguration &picture_configuration, int32_t orig_step_width, int32_t dirq_step_width) {
141 const bool dequant_offset_signalled = picture_configuration.dequant_offset_signalled;
142 const DequantOffset dequant_offset_mode = picture_configuration.dequant_offset_mode;
143 const int32_t dequant_offset = picture_configuration.dequant_offset;
144 int32_t dequant_offset_return = 0;
145 int32_t log_orig_sw = 0;
146 int32_t log_dirq_sw = 0;
147 int64_t dequant_offset64 = 0;
148
149 if (dequant_offset_signalled) {
150 if (dequant_offset == 0) {
151 dequant_offset_return = dequant_offset;
152 } else {
153 log_orig_sw = static_cast<int32_t>(((-1) * Cconst) * log(dirq_step_width));
154 log_dirq_sw = static_cast<int32_t>(((1) * Cconst) * log(orig_step_width));
155
156 if (dequant_offset_mode == DequantOffset_ConstOffset) {
157 dequant_offset64 = (dequant_offset << 9);
158 } else if (dequant_offset_mode == DequantOffset_Default) {
159 dequant_offset64 = (dequant_offset << 11);
160 }
161 dequant_offset64 = ((log_orig_sw + log_dirq_sw + dequant_offset64) * dirq_step_width);
162 dequant_offset_return = static_cast<int32_t>(dequant_offset64 >> 16);
163 }
164 }
165
166 return dequant_offset_return;
167}

169int32_t find_invq_step_width(const PictureConfiguration &picture_configuration, int32_t dirq_step_width, int32_t invq_offset) {
170 const bool dequant_offset_signalled = picture_configuration.dequant_offset_signalled;
171 const DequantOffset dequant_offset_mode = picture_configuration.dequant_offset_mode;
172 const int32_t dequant_offset = picture_configuration.dequant_offset;
173 int32_t invq_step_width = 0;
174
175 if (dequant_offset_signalled == false) {
176 int64_t modi_step_width = (int64_t)(Dconst - Cconst * log(dirq_step_width));
177 modi_step_width = (modi_step_width * dirq_step_width * dirq_step_width);
178#if defined __OPT_DIVISION__
179 modi_step_width = (modi_step_width >> 31);
180#else
181 modi_step_width = (modi_step_width / 32768) >> 16;
182#endif
183 invq_step_width = (int32_t)clamp(dirq_step_width + modi_step_width, (int64_t)MIN_STEP_WIDTH, (int64_t)MAX_STEP_WIDTH);
184 } else {
185 if (dequant_offset_mode == DequantOffset_ConstOffset) {
186 invq_step_width = dirq_step_width;
187 } else if (dequant_offset_mode == DequantOffset_Default) {
188 int64_t modi_step_width = (invq_offset * dirq_step_width);
189#if defined __OPT_DIVISION__
190 modi_step_width = (modi_step_width >> 15);
191#else
192 modi_step_width = modi_step_width / 32768;
193#endif
194 invq_step_width = (int32_t)clamp(dirq_step_width + modi_step_width, (int64_t)MIN_STEP_WIDTH, (int64_t)MAX_STEP_WIDTH);
195 }
196 }
197
198 return invq_step_width;
199}

201// Derive the appropriate dead_zone from current configuration
202//
203int32_t find_layer_deadzone(int32_t orig_step_width, int32_t step_width) {
204
205 int64_t deadzone;
206
207 if (orig_step_width > 16) {
208 deadzone = (((Aconst * step_width) + Bconst) >> 1);
209 deadzone = ((1 << 16) - deadzone);
210 deadzone = (deadzone * step_width) >> 16;
211 } else {
212 deadzone = (orig_step_width >> 1);
213 }
214
215 return (int32_t)deadzone;
216}

218int32_t find_invq_applied_offset(const PictureConfiguration &picture_configuration, int32_t invq_offset, int32_t layer_deadzone) {
219 const bool dequant_offset_signalled = picture_configuration.dequant_offset_signalled;
220 const DequantOffset dequant_offset_mode = picture_configuration.dequant_offset_mode;
221
222 if ((dequant_offset_signalled == false) || (dequant_offset_signalled == true && dequant_offset_mode == DequantOffset_Default))
223 return (int32_t)((-1) * layer_deadzone);
224 else if (dequant_offset_signalled == true && dequant_offset_mode == DequantOffset_ConstOffset)
225 return (int32_t)(invq_offset - layer_deadzone);
226 else {
227 CHECK(0);
228 return 0;
229 }

232Surface InverseQuantize::process(const Surface &src_plane, int32_t layer_step_width, int32_t applied_dequant_offset) {
233
234 const struct Context {
235 int32_t layer_step_width;
236 int32_t applied_dequant_offset;
237 SurfaceView<int16_t> src;
238 } ctx = {layer_step_width, applied_dequant_offset, src_plane.view_as<int16_t>()};
239
240#if defined __OPT_MATRIX__
241 auto dest = Surface::build_from<int16_t>();
242 dest.reserve(src_plane.width(), src_plane.height());
243 for (unsigned y = 0; y < src_plane.height(); ++y) {
244 const int16_t *__restrict psrc = ctx.src.data(0, y);
245 int16_t *__restrict pdst = dest.data(0, y);
246 for (unsigned x = 0; x < src_plane.width(); ++x) {
247 // int16_t c = ctx.src.read(x, y);
248 int16_t c = *psrc++;
249#if defined __QUANT_MULTIPLY__
250 int16_t sign = (c > 0) ? 1 : ((c < 0) ? -1 : 0);
251 *pdst++ = clamp_int16(c * ctx.layer_step_width + sign * ctx.applied_dequant_offset);
252#else
253 int16_t out = 0;
254 if (c > 0)
255 out = (int16_t)(c * ctx.layer_step_width + ctx.applied_dequant_offset);
256 else if (c < 0)
257 out = (int16_t)(c * ctx.layer_step_width - ctx.applied_dequant_offset);
258 // dest.write(x, y, out);
259 *pdst++ = clamp_int16(out);
260#endif
261 }
262 }
263 return dest.finish();
264#else
265 return Surface::build_from<int16_t>()
266 .generate<Context>(
267 src_plane.width(), src_plane.height(),
268 [](unsigned x, unsigned y, const Context &ctx) -> int16_t {
269 int16_t c = ctx.src.read(x, y);
270
271 if (c > 0)
272 return clamp_int16(c * ctx.layer_step_width + ctx.applied_dequant_offset);
273 else if (c < 0)
274 return clamp_int16(c * ctx.layer_step_width - ctx.applied_dequant_offset);
275 else
276 return 0;
277 },
278 ctx)
279 .finish();
280#endif
281}

283Surface InverseQuantize_SWM::process(const Surface &src_plane, unsigned transform_block_size, int32_t *layer_step_width,
284 int32_t *applied_dequant_offset, const Surface &temporal_map) {
285
286 const struct Context {
287 int32_t *layer_step_width;
288 int32_t *applied_dequant_offset;
289 const SurfaceView<int16_t> src;
290 const SurfaceView<uint8_t> mask;
291 } ctx = {layer_step_width, applied_dequant_offset, src_plane.view_as<int16_t>(), temporal_map.view_as<uint8_t>()};
292
293#if defined __OPT_MATRIX__
294 auto dest = Surface::build_from<int16_t>();
295 dest.reserve(src_plane.width(), src_plane.height());
296 for (unsigned y = 0; y < src_plane.height(); ++y) {
297 const int16_t *__restrict psrc = ctx.src.data(0, y);
298 int16_t *__restrict pdst = dest.data(0, y);
299 for (unsigned x = 0; x < src_plane.width(); ++x) {
300 // int16_t c = ctx.src.read(x, y);
301 int16_t c = *psrc++;
302 int16_t out = 0;
303 if (ctx.mask.read(x, y) == TEMPORAL_PRED) {
304#if defined __QUANT_MULTIPLY__
305 int16_t sign = (c > 0) ? 1 : ((c < 0) ? -1 : 0);
306 *pdst++ = (c * ctx.layer_step_width[0] + sign * ctx.applied_dequant_offset[0]);
307#else
308 if (c > 0)
309 out = clamp_int16(c * ctx.layer_step_width[0] + ctx.applied_dequant_offset[0]);
310 else if (c < 0)
311 out = clamp_int16(c * ctx.layer_step_width[0] - ctx.applied_dequant_offset[0]);
312#endif
313 } else {
314#if defined __QUANT_MULTIPLY__
315 int16_t sign = (c > 0) ? 1 : ((c < 0) ? -1 : 0);
316 *pdst++ = (c * ctx.layer_step_width[1] + sign * ctx.applied_dequant_offset[1]);
317#else
318 if (c > 0)
319 out = clamp_int16(c * ctx.layer_step_width[1] + ctx.applied_dequant_offset[1]);
320 else if (c < 0)
321 out = clamp_int16(c * ctx.layer_step_width[1] - ctx.applied_dequant_offset[1]);
322#endif
323 }
324 // dest.write(x, y, out);
325 *pdst++ = out;
326 }
327 }
328 return dest.finish();
329#else
330 return Surface::build_from<int16_t>()
331 .generate<Context>(
332 src_plane.width(), src_plane.height(),
333 [](unsigned x, unsigned y, const Context &ctx) -> int16_t {
334 int16_t c = ctx.src.read(x, y);
335 int16_t out = 0;
336 if (ctx.mask.read(x, y) == TEMPORAL_PRED) {
337 if (c > 0)
338 out = clamp_int16(c * ctx.layer_step_width[0] + ctx.applied_dequant_offset[0]);
339 else if (c < 0)
340 out = clamp_int16(c * ctx.layer_step_width[0] - ctx.applied_dequant_offset[0]);
341 } else {
342 if (c > 0)
343 out = clamp_int16(c * ctx.layer_step_width[1] + ctx.applied_dequant_offset[1]);
344 else if (c < 0)
345 out = clamp_int16(c * ctx.layer_step_width[1] - ctx.applied_dequant_offset[1]);
346 }
347 return out;
348 },
349 ctx)
350 .finish();
351#endif
352}
