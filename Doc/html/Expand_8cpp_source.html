47// Expands a uint8 source surface - adding borders to each side
48//
49Surface ExpandU8::process(const Surface &plane, unsigned width, unsigned height, unsigned offset_x, unsigned offset_y) {
50 const auto src = plane.view_as<uint8_t>();
51 const unsigned src_width = src.width();
52 const unsigned src_height = src.height();
53
54 return Surface::build_from<uint8_t>()
55 .generate(width, height,
56 [&](unsigned x, unsigned y) -> int8_t {
57 return src.read(clamp(x - offset_x, 0u, src_width - 1), clamp(y - offset_y, 0u, src_height - 1));
58 })
59 .finish();
60}

77// Add any necessary padding to image to make it conformant
78//
79Image ExpandImage(const Image &src, const ImageDescription &description) {
80 // Check if image is already the right size?
81 if (src.description() == description)
82 return src;
83
84 vector<Surface> surfaces;
85 for (unsigned p = 0; p < src.description().num_planes(); ++p) {
86 switch (src.description().bit_depth()) {
87 case 8:
88 surfaces.push_back(ExpandU8().process(src.plane(p), description.width(p), description.height(p), 0, 0));
89 break;
90 case 10:
91 case 12:
92 case 14:
93 case 16:
94 surfaces.push_back(ExpandU16().process(src.plane(p), description.width(p), description.height(p), 0, 0));
95 break;
96 default:
97 CHECK(0);
98 return Image();
99 }
100 }
101
102 return Image("expanded-" + src.name(), description, src.timestamp(), surfaces);
103}
