86template<int signedMode> std::pair<int64_t,int64_t> fwdTransformCbCr( const PelBuf &resCb, const PelBuf &resCr, PelBuf& resC1, PelBuf& resC2 )
87{
88 const Pel* cb = resCb.buf;
89 const Pel* cr = resCr.buf;
90 Pel* c1 = resC1.buf;
91 Pel* c2 = resC2.buf;
92 int64_t d1 = 0;
93 int64_t d2 = 0;
94 for( SizeType y = 0; y < resCb.height; y++, cb += resCb.stride, cr += resCr.stride, c1 += resC1.stride, c2 += resC2.stride )
95 {
96 for( SizeType x = 0; x < resCb.width; x++ )
97 {
98 int cbx = cb[x], crx = cr[x];
99 if ( signedMode == 1 )
100 {
101 c1[x] = Pel( ( 4*cbx + 2*crx ) / 5 );
102 d1 += square( cbx - c1[x] ) + square( crx - (c1[x]>>1) );
103 }
104 else if ( signedMode == -1 )
105 {
106 c1[x] = Pel( ( 4*cbx - 2*crx ) / 5 );
107 d1 += square( cbx - c1[x] ) + square( crx - (-c1[x]>>1) );
108 }
109 else if ( signedMode == 2 )
110 {
111 c1[x] = Pel( ( cbx + crx ) / 2 );
112 d1 += square( cbx - c1[x] ) + square( crx - c1[x] );
113 }
114 else if ( signedMode == -2 )
115 {
116 c1[x] = Pel( ( cbx - crx ) / 2 );
117 d1 += square( cbx - c1[x] ) + square( crx + c1[x] );
118 }
119 else if ( signedMode == 3 )
120 {
121 c2[x] = Pel( ( 4*crx + 2*cbx ) / 5 );
122 d1 += square( cbx - (c2[x]>>1) ) + square( crx - c2[x] );
123 }
124 else if ( signedMode == -3 )
125 {
126 c2[x] = Pel( ( 4*crx - 2*cbx ) / 5 );
127 d1 += square( cbx - (-c2[x]>>1) ) + square( crx - c2[x] );
128 }
129 else
130 {
131 d1 += square( cbx );
132 d2 += square( crx );
133 }
134 }
135 }
136 return std::make_pair(d1,d2);
137}

139template<int signedMode> void invTransformCbCr( PelBuf &resCb, PelBuf &resCr )
140{
141 Pel* cb = resCb.buf;
142 Pel* cr = resCr.buf;
143 for( SizeType y = 0; y < resCb.height; y++, cb += resCb.stride, cr += resCr.stride )
144 {
145 for( SizeType x = 0; x < resCb.width; x++ )
146 {
147 if (signedMode == 1)
148 {
149 cr[x] = cb[x] >> 1;
150 }
151 else if (signedMode == -1)
152 {
153 cr[x] = -cb[x] >> 1;
154 }
155 else if (signedMode == 2)
156 {
157 cr[x] = cb[x];
158 }
159 else if (signedMode == -2)
160 {
161 // non-normative clipping to prevent 16-bit overflow
162 cr[x] = (cb[x] == -32768 && sizeof(Pel) == 2) ? 32767 : -cb[x];
163 }
164 else if (signedMode == 3)
165 {
166 cb[x] = cr[x] >> 1;
167 }
168 else if (signedMode == -3)
169 {
170 cb[x] = -cr[x] >> 1;
171 }
172 }
173 }

247void TrQuant::fwdLfnstNxN( TCoeff* src, TCoeff* dst, const uint32_t mode, const uint32_t index, const uint32_t size, int zeroOutSize )
248{
249 const int8_t* trMat = ( size > 4 ) ? g_lfnst8x8[ mode ][ index ][ 0 ] : g_lfnst4x4[ mode ][ index ][ 0 ];
250 const int trSize = ( size > 4 ) ? 48 : 16;
251 TCoeff coef;
252 TCoeff* out = dst;
253 assert( index < 3 );
254
255 for( int j = 0; j < zeroOutSize; j++ )
256 {
257 TCoeff* srcPtr = src;
258 const int8_t* trMatTmp = trMat;
259 coef = 0;
260 for( int i = 0; i < trSize; i++ )
261 {
262 coef += *srcPtr++ * *trMatTmp++;
263 }
264 *out++ = ( coef + 64 ) >> 7;
265 trMat += trSize;
266 }
267
268 std::fill_n( out, trSize - zeroOutSize, 0 );
269}

271void TrQuant::invLfnstNxN( TCoeff* src, TCoeff* dst, const uint32_t mode, const uint32_t index, const uint32_t size, int zeroOutSize, const int maxLog2TrDynamicRange )
272{
273 const TCoeff outputMinimum = -( 1 << maxLog2TrDynamicRange );
274 const TCoeff outputMaximum = ( 1 << maxLog2TrDynamicRange ) - 1;
275 const int8_t* trMat = ( size > 4 ) ? g_lfnst8x8[ mode ][ index ][ 0 ] : g_lfnst4x4[ mode ][ index ][ 0 ];
276 const int trSize = ( size > 4 ) ? 48 : 16;
277 TCoeff resi;
278 TCoeff* out = dst;
279 assert( index < 3 );
280
281 for( int j = 0; j < trSize; j++ )
282 {
283 resi = 0;
284 const int8_t* trMatTmp = trMat;
285 TCoeff* srcPtr = src;
286 for( int i = 0; i < zeroOutSize; i++ )
287 {
288 resi += *srcPtr++ * *trMatTmp;
289 trMatTmp += trSize;
290 }
291 *out++ = Clip3<TCoeff>( outputMinimum, outputMaximum, ( resi + 64 ) >> 7 );
292 trMat++;
293 }
294}

322void TrQuant::xInvLfnst( const TransformUnit &tu, const ComponentID compID )
323{
324 const int maxLog2TrDynamicRange = tu.cs->sps->getMaxLog2TrDynamicRange(toChannelType(compID));
325 const CompArea& area = tu.blocks[ compID ];
326 const uint32_t width = area.width;
327 const uint32_t height = area.height;
328 const uint32_t lfnstIdx = tu.cu->lfnstIdx;
329 if( lfnstIdx && tu.mtsIdx[compID] != MTS_SKIP && (tu.cu->isSepTree() ? true : isLuma(compID)) )
330 {
331 const bool whge3 = width >= 8 && height >= 8;
332 const ScanElement * scan = whge3 ? g_coefTopLeftDiagScan8x8[ gp_sizeIdxInfo->idxFrom( width ) ] : g_scanOrder[ SCAN_GROUPED_4x4 ][ SCAN_DIAG ][ gp_sizeIdxInfo->idxFrom( width ) ][ gp_sizeIdxInfo->idxFrom( height ) ];
333 uint32_t intraMode = PU::getFinalIntraMode( *tu.cs->getPU( area.pos(), toChannelType( compID ) ), toChannelType( compID ) );
334
335 if( PU::isLMCMode( tu.cs->getPU( area.pos(), toChannelType( compID ) )->intraDir[ toChannelType( compID ) ] ) )
336 {
337 intraMode = PU::getCoLocatedIntraLumaMode( *tu.cs->getPU( area.pos(), toChannelType( compID ) ) );
338 }
339 if (PU::isMIP(*tu.cs->getPU(area.pos(), toChannelType(compID)), toChannelType(compID)))
340 {
341 intraMode = PLANAR_IDX;
342 }
343 CHECK( intraMode >= NUM_INTRA_MODE - 1, "Invalid intra mode" );
344
345 if( lfnstIdx < 3 )
346 {
347 intraMode = getLFNSTIntraMode( PU::getWideAngle( tu, intraMode, compID ) );
348#if RExt__DECODER_DEBUG_TOOL_STATISTICS
349 CodingStatistics::IncrementStatisticTool( CodingStatisticsClassType { STATS__TOOL_LFNST, width, height, compID } );
350#endif
351 bool transposeFlag = getTransposeFlag( intraMode );
352 const int sbSize = whge3 ? 8 : 4;
353 bool tu4x4Flag = ( width == 4 && height == 4 );
354 bool tu8x8Flag = ( width == 8 && height == 8 );
355 TCoeff* lfnstTemp;
356 TCoeff* coeffTemp;
357 int y;
358 lfnstTemp = m_tempInMatrix; // inverse spectral rearrangement
359 coeffTemp = m_tempCoeff;
360 TCoeff *dst = lfnstTemp;
361
362 const ScanElement *scanPtr = scan;
363 for (y = 0; y < 16; y++)
364 {
365 *dst++ = coeffTemp[scanPtr->idx];
366 scanPtr++;
367 }
368
369 invLfnstNxN(m_tempInMatrix, m_tempOutMatrix, g_lfnstLut[intraMode], lfnstIdx - 1, sbSize,
370 (tu4x4Flag || tu8x8Flag) ? 8 : 16, maxLog2TrDynamicRange);
371 lfnstTemp = m_tempOutMatrix; // inverse spectral rearrangement
372
373 if (transposeFlag)
374 {
375 if (sbSize == 4)
376 {
377 for (y = 0; y < 4; y++)
378 {
379 coeffTemp[0] = lfnstTemp[0];
380 coeffTemp[1] = lfnstTemp[4];
381 coeffTemp[2] = lfnstTemp[8];
382 coeffTemp[3] = lfnstTemp[12];
383 lfnstTemp++;
384 coeffTemp += width;
385 }
386 }
387 else // ( sbSize == 8 )
388 {
389 for (y = 0; y < 8; y++)
390 {
391 coeffTemp[0] = lfnstTemp[0];
392 coeffTemp[1] = lfnstTemp[8];
393 coeffTemp[2] = lfnstTemp[16];
394 coeffTemp[3] = lfnstTemp[24];
395 if (y < 4)
396 {
397 coeffTemp[4] = lfnstTemp[32];
398 coeffTemp[5] = lfnstTemp[36];
399 coeffTemp[6] = lfnstTemp[40];
400 coeffTemp[7] = lfnstTemp[44];
401 }
402 lfnstTemp++;
403 coeffTemp += width;
404 }
405 }
406 }
407 else
408 {
409 for (y = 0; y < sbSize; y++)
410 {
411 uint32_t uiStride = (y < 4) ? sbSize : 4;
412 ::memcpy(coeffTemp, lfnstTemp, uiStride * sizeof(TCoeff));
413 lfnstTemp += uiStride;
414 coeffTemp += width;
415 }
416 }
417 }
418 }

422{
423 const CompArea& area = tu.blocks[ compID ];
424 const uint32_t width = area.width;
425 const uint32_t height = area.height;

428 {
429 const bool whge3 = width >= 8 && height >= 8;
430 const ScanElement * scan = whge3 ? g_coefTopLeftDiagScan8x8[ gp_sizeIdxInfo->idxFrom( width ) ] : g_scanOrder[ SCAN_GROUPED_4x4 ][ SCAN_DIAG ][ gp_sizeIdxInfo->idxFrom( width ) ][ gp_sizeIdxInfo->idxFrom( height ) ];
431 uint32_t intraMode = PU::getFinalIntraMode( *tu.cs->getPU( area.pos(), toChannelType( compID ) ), toChannelType( compID ) );
432
433 if( PU::isLMCMode( tu.cs->getPU( area.pos(), toChannelType( compID ) )->intraDir[ toChannelType( compID ) ] ) )
434 {
435 intraMode = PU::getCoLocatedIntraLumaMode( *tu.cs->getPU( area.pos(), toChannelType( compID ) ) );
436 }
437 if (PU::isMIP(*tu.cs->getPU(area.pos(), toChannelType(compID)), toChannelType(compID)))
438 {
439 intraMode = PLANAR_IDX;
440 }
441 CHECK( intraMode >= NUM_INTRA_MODE - 1, "Invalid intra mode" );
442
443 if( lfnstIdx < 3 )
444 {
445 intraMode = getLFNSTIntraMode( PU::getWideAngle( tu, intraMode, compID ) );
446
447 bool transposeFlag = getTransposeFlag( intraMode );
448 const int sbSize = whge3 ? 8 : 4;
449 bool tu4x4Flag = ( width == 4 && height == 4 );
450 bool tu8x8Flag = ( width == 8 && height == 8 );
451 TCoeff* lfnstTemp;
452 TCoeff* coeffTemp;
453 TCoeff * tempCoeff = loadTr ? m_mtsCoeffs[tu.mtsIdx[compID]] : m_tempCoeff;
454
455 int y;
456 lfnstTemp = m_tempInMatrix; // forward low frequency non-separable transform
457 coeffTemp = tempCoeff;
458
459 if (transposeFlag)
460 {
461 if (sbSize == 4)
462 {
463 for (y = 0; y < 4; y++)
464 {
465 lfnstTemp[0] = coeffTemp[0];
466 lfnstTemp[4] = coeffTemp[1];
467 lfnstTemp[8] = coeffTemp[2];
468 lfnstTemp[12] = coeffTemp[3];
469 lfnstTemp++;
470 coeffTemp += width;
471 }
472 }
473 else // ( sbSize == 8 )
474 {
475 for (y = 0; y < 8; y++)
476 {
477 lfnstTemp[0] = coeffTemp[0];
478 lfnstTemp[8] = coeffTemp[1];
479 lfnstTemp[16] = coeffTemp[2];
480 lfnstTemp[24] = coeffTemp[3];
481 if (y < 4)
482 {
483 lfnstTemp[32] = coeffTemp[4];
484 lfnstTemp[36] = coeffTemp[5];
485 lfnstTemp[40] = coeffTemp[6];
486 lfnstTemp[44] = coeffTemp[7];
487 }
488 lfnstTemp++;
489 coeffTemp += width;
490 }
491 }
492 }
493 else
494 {
495 for (y = 0; y < sbSize; y++)
496 {
497 uint32_t uiStride = (y < 4) ? sbSize : 4;
498 ::memcpy(lfnstTemp, coeffTemp, uiStride * sizeof(TCoeff));
499 lfnstTemp += uiStride;
500 coeffTemp += width;
501 }
502 }
503
504 fwdLfnstNxN(m_tempInMatrix, m_tempOutMatrix, g_lfnstLut[intraMode], lfnstIdx - 1, sbSize,
505 (tu4x4Flag || tu8x8Flag) ? 8 : 16);
506
507 lfnstTemp = m_tempOutMatrix; // forward spectral rearrangement
508 coeffTemp = tempCoeff;
509 int lfnstCoeffNum = (sbSize == 4) ? sbSize * sbSize : 48;
510
511 const ScanElement *scanPtr = scan;
512
513 for (y = 0; y < lfnstCoeffNum; y++)
514 {
515 coeffTemp[scanPtr->idx] = *lfnstTemp++;
516 scanPtr++;
517 }
518 }
519 }

567std::vector<int> TrQuant::selectICTCandidates( const TransformUnit &tu, CompStorage* resCb, CompStorage* resCr )
568{
569 CHECK( !resCb[0].valid() || !resCr[0].valid(), "standard components are not valid" );
570
571 if( !CU::isIntra( *tu.cu ) )
572 {
573 int cbfMask = 3;
574 resCb[cbfMask].create( tu.blocks[COMPONENT_Cb] );
575 resCr[cbfMask].create( tu.blocks[COMPONENT_Cr] );
576 fwdTransformICT( tu, resCb[0], resCr[0], resCb[cbfMask], resCr[cbfMask], cbfMask );
577 std::vector<int> cbfMasksToTest;
578 cbfMasksToTest.push_back( cbfMask );
579 return cbfMasksToTest;
580 }
581
582 std::pair<int64_t,int64_t> pairDist[4];
583 for( int cbfMask = 0; cbfMask < 4; cbfMask++ )
584 {
585 if( cbfMask )
586 {
587 CHECK( resCb[cbfMask].valid() || resCr[cbfMask].valid(), "target components for cbfMask=" << cbfMask << " are already present" );
588 resCb[cbfMask].create( tu.blocks[COMPONENT_Cb] );
589 resCr[cbfMask].create( tu.blocks[COMPONENT_Cr] );
590 }
591 pairDist[cbfMask] = fwdTransformICT( tu, resCb[0], resCr[0], resCb[cbfMask], resCr[cbfMask], cbfMask );
592 }
593
594 std::vector<int> cbfMasksToTest;
595 int64_t minDist1 = std::min<int64_t>( pairDist[0].first, pairDist[0].second );
596 int64_t minDist2 = std::numeric_limits<int64_t>::max();
597 int cbfMask1 = 0;

730void TrQuant::xT( const TransformUnit &tu, const ComponentID &compID, const CPelBuf &resi, CoeffBuf &dstCoeff, const int width, const int height )

734 const int TRANSFORM_MATRIX_SHIFT = g_transformMatrixShift[TRANSFORM_FORWARD];
735 const uint32_t transformWidthIndex = floorLog2(width ) - 1; // nLog2WidthMinus1, since transform start from 2-point
736 const uint32_t transformHeightIndex = floorLog2(height) - 1; // nLog2HeightMinus1, since transform start from 2-point

748 {
749 skipWidth = width - 4;
750 skipHeight = height - 4;

772 {
773 for( int x = 0; x < width; x++ )

775 block[( y * width ) + x] = resiBuf[( y * resiStride ) + x];

780 {
781 const int shift_1st = ((floorLog2(width )) + bitDepth + TRANSFORM_MATRIX_SHIFT) - maxLog2TrDynamicRange + COM16_C806_TRANS_PREC;

788 fastFwdTrans[trTypeVer][transformHeightIndex](tmp, dstCoeff.buf, shift_2nd, width, skipWidth, skipHeight);
789 }
790 else if( height == 1 ) //1-D horizontal transform
791 {
792 const int shift = ((floorLog2(width )) + bitDepth + TRANSFORM_MATRIX_SHIFT) - maxLog2TrDynamicRange + COM16_C806_TRANS_PREC;
793 CHECK( shift < 0, "Negative shift" );
794 CHECKD( ( transformWidthIndex < 0 ), "There is a problem with the width." );
795 fastFwdTrans[trTypeHor][transformWidthIndex]( block, dstCoeff.buf, shift, 1, 0, skipWidth );
796 }
797 else //if (iWidth == 1) //1-D vertical transform
798 {
799 int shift = ( ( floorLog2(height) ) + bitDepth + TRANSFORM_MATRIX_SHIFT ) - maxLog2TrDynamicRange + COM16_C806_TRANS_PREC;
800 CHECK( shift < 0, "Negative shift" );
801 CHECKD( ( transformHeightIndex < 0 ), "There is a problem with the height." );
802 fastFwdTrans[trTypeVer][transformHeightIndex]( block, dstCoeff.buf, shift, 1, 0, skipHeight );
803 }

904void TrQuant::transformNxN( TransformUnit& tu, const ComponentID& compID, const QpParam& cQP, std::vector<TrMode>* trModes, const int maxCand )
905{
906 CodingStructure &cs = *tu.cs;
907 const CompArea &rect = tu.blocks[compID];
908 const uint32_t width = rect.width;
909 const uint32_t height = rect.height;
910
911 const CPelBuf resiBuf = cs.getResiBuf(rect);
912
913 CHECK( cs.sps->getMaxTbSize() < width, "Unsupported transformation size" );
914
915 int pos = 0;
916 std::vector<TrCost> trCosts;
917 std::vector<TrMode>::iterator it = trModes->begin();
918 const double facBB[] = { 1.2, 1.3, 1.3, 1.4, 1.5 };
919 while( it != trModes->end() )
920 {
921 tu.mtsIdx[compID] = it->first;
922 CoeffBuf tempCoeff( m_mtsCoeffs[tu.mtsIdx[compID]], rect);
923 if( tu.noResidual )
924 {
925 int sumAbs = 0;
926 trCosts.push_back( TrCost( sumAbs, pos++ ) );
927 it++;
928 continue;
929 }
930
931 if ( tu.mtsIdx[compID] == MTS_SKIP )
932 {
933 xTransformSkip( tu, compID, resiBuf, tempCoeff.buf );
934 }
935 else
936 {
937 xT( tu, compID, resiBuf, tempCoeff, width, height );
938 }
939
940 TCoeff sumAbs = 0;
941 for( int pos = 0; pos < width*height; pos++ )
942 {
943 sumAbs += abs( tempCoeff.buf[pos] );
944 }
945
946 double scaleSAD=1.0;
947 if ( tu.mtsIdx[compID] == MTS_SKIP && ((floorLog2(width) + floorLog2(height)) & 1) == 1)
948 {
949 scaleSAD=1.0/1.414213562; // compensate for not scaling transform skip coefficients by 1/sqrt(2)
950 }
951 if (tu.mtsIdx[compID] == MTS_SKIP)
952 {
953 int trShift = getTransformShift(tu.cu->slice->getSPS()->getBitDepth(toChannelType(compID)), rect.size(),
954 tu.cu->slice->getSPS()->getMaxLog2TrDynamicRange(toChannelType(compID)));
955 scaleSAD *= pow(2, trShift);
956 }
957
958 trCosts.push_back( TrCost( int(std::min<double>(sumAbs*scaleSAD, std::numeric_limits<int>::max())), pos++ ) );
959 it++;
960 }
961
962 int numTests = 0;
963 std::vector<TrCost>::iterator itC = trCosts.begin();
964 const double fac = facBB[std::max(0, floorLog2(std::max(width, height)) - 2)];
965 const double thr = fac * trCosts.begin()->first;
966 const double thrTS = trCosts.begin()->first;
967 while( itC != trCosts.end() )
968 {
969 const bool testTr = itC->first <= ( itC->second == 1 ? thrTS : thr ) && numTests <= maxCand;
970 trModes->at( itC->second ).second = testTr;
971 numTests += testTr;
972 itC++;
973 }
974}
