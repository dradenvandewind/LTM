51ParametersBuilder &ParametersBuilder::set_json(const std::string &json_str) {
52 json params;
53 try {
54 params = json::parse(json_str);
55 } catch (const std::exception &e) {
56 ERR("Cannot parse parameters: %s", e.what());
57 }
58
59 if (!params.is_object()) {
60 ERR("Parameters should be a json object.");
61 return *this;
62 }
63
64 for (const auto &p : params.items()) {
65 if (p.value().is_number()) {
66 set(p.key(), p.value().get<double>());
67 REPORT("parameter %s NUM %8.3f", p.key().c_str(), p.value().get<double>());
68 } else if (p.value().is_boolean()) {
69 set(p.key(), p.value().get<bool>());
70 REPORT("parameter %s BOOL %8d", p.key().c_str(), p.value().get<bool>());
71 } else if (p.value().is_string()) {
72 set(p.key(), p.value().get<std::string>());
73 REPORT("parameter %s STR %s", p.key().c_str(), p.value().get<std::string>().c_str());
74 } else {
75 WARN("Parameter %s should be bool, number, or string", p.key().c_str());
76 }
77 }
78
79 return *this;
80}

88const ParameterRef Parameters::operator[](const std::string &name) const {

90
91 if (i == parameters_.end())
