116void
117ProfileLevelTierFeatures::extractPTLInformation(const SPS &sps)
118{
119 const ProfileTierLevel &spsPtl =*(sps.getProfileTierLevel());
120
121 m_pProfile = nullptr;
122 m_pLevelTier = nullptr;
123 m_tier = spsPtl.getTierFlag();
124
125 // Identify the profile from the profile Idc, and possibly other constraints.
126 for(int32_t i=0; validProfiles[i].profile != Profile::NONE; i++)
127 {
128 if (spsPtl.getProfileIdc() == validProfiles[i].profile)
129 {
130 m_pProfile = &(validProfiles[i]);
131 break;
132 }
133 }
134
135 if (m_pProfile != nullptr)
136 {
137 // Now identify the level:
138 const LevelTierFeatures *pLTF = m_pProfile->pLevelTiersListInfo;
139 const Level::Name spsLevelName = spsPtl.getLevelIdc();
140 if (spsLevelName!=Level::LEVEL15_5 || m_pProfile->canUseLevel15p5)
141 {
142 for(int i=0; pLTF[i].level!=Level::NONE; i++)
143 {
144 if (pLTF[i].level == spsLevelName)
145 {
146 m_pLevelTier = &(pLTF[i]);
147 }
148 }
149 }
150 }
151}

163uint32_t ProfileLevelTierFeatures::getMaxDpbSize( uint32_t picSizeMaxInSamplesY ) const
164{
165 const uint32_t maxDpbPicBuf = 8;
166 uint32_t maxDpbSize;
167
168 if (m_pLevelTier->level == Level::LEVEL15_5)
169 {
170 // maxDpbSize is unconstrained in this case
171 maxDpbSize = std::numeric_limits<uint32_t>::max();
172 }
173 else if (2 * picSizeMaxInSamplesY <= m_pLevelTier->maxLumaPs)
174 {
175 maxDpbSize = 2 * maxDpbPicBuf;
176 }
177 else if (3 * picSizeMaxInSamplesY <= 2 * m_pLevelTier->maxLumaPs)
178 {
179 maxDpbSize = 3 * maxDpbPicBuf / 2;
180 }
181 else
182 {
183 maxDpbSize = maxDpbPicBuf;
184 }
185
186 return maxDpbSize;
187}
