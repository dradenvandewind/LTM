187{
188public:
189 TUIntraSubPartitioner(Partitioner& _initialState)
190 {
191 //we copy the input partitioner data
192 m_partStack.push_back(PartLevel(TU_NO_ISP, { _initialState.currArea() }));
193
194 currDepth = _initialState.currDepth;
195 currQtDepth = _initialState.currQtDepth;

197 currBtDepth = _initialState.currBtDepth;
198 currMtDepth = _initialState.currMtDepth;
199 chType = _initialState.chType;
200#if _DEBUG
201 m_currArea = _initialState.currArea();
202#endif
203 treeType = _initialState.treeType;
204 modeType = _initialState.modeType;
205 }
