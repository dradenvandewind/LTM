109 unsigned sigCtxIdAbs( int scanPos, const TCoeff* coeff, const int state )
110 {
111 const uint32_t posY = m_scan[scanPos].y;
112 const uint32_t posX = m_scan[scanPos].x;
113 const TCoeff* pData = coeff + posX + posY * m_width;
114 const int diag = posX + posY;
115 int numPos = 0;
116 TCoeff sumAbs = 0;
117#define UPDATE(x) {TCoeff a=abs(x);sumAbs+=std::min(4+(a&1),a);numPos+=int(!!a);}
118 if( posX < m_width-1 )
119 {
120 UPDATE( pData[1] );
121 if( posX < m_width-2 )
122 {
123 UPDATE( pData[2] );
124 }
125 if( posY < m_height-1 )
126 {
127 UPDATE( pData[m_width+1] );
128 }
129 }
130 if( posY < m_height-1 )
131 {
132 UPDATE( pData[m_width] );
133 if( posY < m_height-2 )
134 {
135 UPDATE( pData[m_width<<1] );
136 }
137 }
138#undef UPDATE
139
140
141 int ctxOfs = int(std::min<TCoeff>((sumAbs+1)>>1, 3)) + ( diag < 2 ? 4 : 0 );
142
143 if( m_chType == CHANNEL_TYPE_LUMA )
144 {
145 ctxOfs += diag < 5 ? 4 : 0;
146 }
147
148 m_tmplCpDiag = diag;
149 m_tmplCpSum1 = sumAbs - numPos;
150 return m_sigFlagCtxSet[std::max( 0, state-1 )]( ctxOfs );
151 }

153 uint8_t ctxOffsetAbs()
154 {
155 int offset = 0;
156 if( m_tmplCpDiag != -1 )
157 {
158 offset = int(std::min<TCoeff>( m_tmplCpSum1, 4 )) + 1;
159 offset += ( !m_tmplCpDiag ? ( m_chType == CHANNEL_TYPE_LUMA ? 15 : 5 ) : m_chType == CHANNEL_TYPE_LUMA ? m_tmplCpDiag < 3 ? 10 : ( m_tmplCpDiag < 10 ? 5 : 0 ) : 0 );
160 }
161 return uint8_t(offset);
162 }

253 unsigned signCtxIdAbsTS(int scanPos, const TCoeff* coeff, int bdpcm)
254 {
255 const uint32_t posY = m_scan[scanPos].y;
256 const uint32_t posX = m_scan[scanPos].x;
257 const TCoeff* pData = coeff + posX + posY * m_width;
258
259 int rightSign = 0, belowSign = 0;
260 unsigned signCtx = 0;
261
262 if (posX > 0)
263 {
264 rightSign = sgn(pData[-1]);
265 }
266 if (posY > 0)
267 {
268 belowSign = sgn(pData[-(int)m_width]);
269 }
270
271 if ((rightSign == 0 && belowSign == 0) || ((rightSign*belowSign) < 0))
272 {
273 signCtx = 0;
274 }
275 else if (rightSign >= 0 && belowSign >= 0)
276 {
277 signCtx = 1;
278 }
279 else
280 {
281 signCtx = 2;
282 }
283 if (bdpcm)
284 {
285 signCtx += 3;
286 }
287 return m_tsSignFlagCtxSet(signCtx);
288 }

308 int deriveModCoeff(int rightPixel, int belowPixel, TCoeff absCoeff, int bdpcm = 0)
309 {
310
311 if (absCoeff == 0)
312 return 0;
313 int pred1, absBelow = abs(belowPixel), absRight = abs(rightPixel);
314
315 int absCoeffMod = int(absCoeff);
316
317 if (bdpcm == 0)
318 {
319 pred1 = std::max(absBelow, absRight);
320
321 if (absCoeffMod == pred1)
322 {
323 absCoeffMod = 1;
324 }
325 else
326 {
327 absCoeffMod = absCoeffMod < pred1 ? absCoeffMod + 1 : absCoeffMod;
328 }
329 }
330
331 return(absCoeffMod);
332 }
333
334 TCoeff decDeriveModCoeff(int rightPixel, int belowPixel, TCoeff absCoeff)
335 {
336
337 if (absCoeff == 0)
338 return 0;
339
340 int pred1, absBelow = abs(belowPixel), absRight = abs(rightPixel);
341 pred1 = std::max(absBelow, absRight);
342
343 TCoeff absCoeffMod;
344
345 if (absCoeff == 1 && pred1 > 0)
346 {
347 absCoeffMod = pred1;
348 }
349 else
350 {
351 absCoeffMod = absCoeff - (absCoeff <= pred1);
352 }
353 return(absCoeffMod);
354 }
