67//
68// Default Buffer implementation that just uses a std::vector<>
69//
70class BufferVector : public Buffer {
71public:
72 ~BufferVector() override {}
73
74 BufferVector(const uint8_t *data, unsigned size) : bytes_(data, data + size) {}
75
76 BufferVector(unsigned size) : bytes_(size, 0) {}
77
78 void map_read(unsigned offset, unsigned size, const uint8_t *&mapped_data, unsigned &mapped_size) const override {
79 assert(offset <= bytes_.size());
80 assert(offset + size <= bytes_.size());
81
82 mapped_data = bytes_.data() + offset;
83 mapped_size = size;
84 }
85
86 void map_write(unsigned offset, unsigned size, uint8_t *&mapped_data, unsigned &mapped_size) override {
87 assert(offset <= bytes_.size());
88 assert(offset + size <= bytes_.size());
89
90 mapped_data = bytes_.data() + offset;
91 mapped_size = size;
92 }
93
94 void unmap() const override {}
95
96private:
97 std::vector<uint8_t> bytes_;
98};

107//
108// Create page aligned buffers
109//
110
111class BufferAligned : public Buffer {
112 const unsigned ALIGNMENT = 64;
113
114public:
115 BufferAligned(const uint8_t *data, unsigned size) : size_(size) {
116 size = (size + (ALIGNMENT - 1)) & ~(ALIGNMENT - 1);
117#ifdef _WIN32
118 bytes_ = _aligned_malloc(size, ALIGNMENT);
119#else
120 bytes_ = aligned_alloc(ALIGNMENT, size);
121#endif
122 memcpy(bytes_, data, size);
123 }
124
125 BufferAligned(unsigned size) : size_(size) {
126 size = (size + (ALIGNMENT - 1)) & ~(ALIGNMENT - 1);
127#ifdef _WIN32
128 bytes_ = _aligned_malloc(size, ALIGNMENT);
129#else
130 bytes_ = aligned_alloc(ALIGNMENT, size);
131#endif
132 }
133
134 ~BufferAligned() override {
135 if (bytes_) {
136#ifdef _WIN32
137 _aligned_free(bytes_);
138#else
139 free(bytes_);
140#endif
141 bytes_ = nullptr;
142 }
143 }
144
145 void map_read(unsigned offset, unsigned size, const uint8_t *&mapped_data, unsigned &mapped_size) const override {
146 assert(offset <= size_);
147 assert(offset + size <= size_);
148
149 mapped_data = (uint8_t *)bytes_;
150 mapped_size = size;
151 }
152
153 void map_write(unsigned offset, unsigned size, uint8_t *&mapped_data, unsigned &mapped_size) override {
154 assert(offset <= size_);
155 assert(offset + size <= size_);
156
157 mapped_data = (uint8_t *)bytes_;
158 mapped_size = size;
159 }
160
161 void unmap() const override {}
162
163private:
164 size_t size_ = 0;
165 void *bytes_ = 0;
166};
