55void CompArea::xRecalcLumaToChroma()
56{
57 const uint32_t csx = getComponentScaleX(compID, chromaFormat);
58 const uint32_t csy = getComponentScaleY(compID, chromaFormat);
59
60 x >>= csx;
61 y >>= csy;
62 width >>= csx;
63 height >>= csy;
64}

160bool UnitArea::contains(const UnitArea& other) const
161{
162 bool ret = true;
163 bool any = false;
164
165 for( const auto &blk : other.blocks )
166 {
167 if( blk.valid() && blocks[blk.compID].valid() )
168 {
169 ret &= blocks[blk.compID].contains( blk );
170 any = true;
171 }
172 }
173
174 return any && ret;
175}
176
177bool UnitArea::contains( const UnitArea& other, const ChannelType chType ) const
178{
179 bool ret = true;
180 bool any = false;
181
182 for( const auto &blk : other.blocks )
183 {
184 if( toChannelType( blk.compID ) == chType && blk.valid() && blocks[blk.compID].valid() )
185 {
186 ret &= blocks[blk.compID].contains( blk );
187 any = true;
188 }
189 }
190
191 return any && ret;
192}

380const bool CodingUnit::checkCCLMAllowed() const
381{
382 bool allowCCLM = false;
383
384 if( !CS::isDualITree( *cs ) ) //single tree I slice or non-I slice (Note: judging chType is no longer equivalent to checking dual-tree I slice since the local dual-tree is introduced)
385 {
386 allowCCLM = true;
387 }
388 else if( slice->getSPS()->getCTUSize() <= 32 ) //dual tree, CTUsize < 64
389 {
390 allowCCLM = true;
391 }
392 else //dual tree, CTU size 64 or 128
393 {
394 int depthFor64x64Node = slice->getSPS()->getCTUSize() == 128 ? 1 : 0;
395 const PartSplit cuSplitTypeDepth1 = CU::getSplitAtDepth( *this, depthFor64x64Node );
396 const PartSplit cuSplitTypeDepth2 = CU::getSplitAtDepth( *this, depthFor64x64Node + 1 );
397
398 //allow CCLM if 64x64 chroma tree node uses QT split or HBT+VBT split combination
399 if( cuSplitTypeDepth1 == CU_QUAD_SPLIT || (cuSplitTypeDepth1 == CU_HORZ_SPLIT && cuSplitTypeDepth2 == CU_VERT_SPLIT) )
400 {
401 if( chromaFormat == CHROMA_420 )
402 {
403 CHECK( !(blocks[COMPONENT_Cb].width <= 16 && blocks[COMPONENT_Cb].height <= 16), "chroma cu size shall be <= 16x16 for YUV420 format" );
404 }
405 allowCCLM = true;
406 }
407 //allow CCLM if 64x64 chroma tree node uses NS (No Split) and becomes a chroma CU containing 32x32 chroma blocks
408 else if( cuSplitTypeDepth1 == CU_DONT_SPLIT )
409 {
410 if( chromaFormat == CHROMA_420 )
411 {
412 CHECK( !(blocks[COMPONENT_Cb].width == 32 && blocks[COMPONENT_Cb].height == 32), "chroma cu size shall be 32x32 for YUV420 format" );
413 }
414 allowCCLM = true;
415 }
416 //allow CCLM if 64x32 chroma tree node uses NS and becomes a chroma CU containing 32x16 chroma blocks
417 else if( cuSplitTypeDepth1 == CU_HORZ_SPLIT && cuSplitTypeDepth2 == CU_DONT_SPLIT )
418 {
419 if( chromaFormat == CHROMA_420 )
420 {
421 CHECK( !(blocks[COMPONENT_Cb].width == 32 && blocks[COMPONENT_Cb].height == 16), "chroma cu size shall be 32x16 for YUV420 format" );
422 }
423 allowCCLM = true;
424 }
425
426 //further check luma conditions
427 if( allowCCLM )
428 {
429 //disallow CCLM if luma 64x64 block uses BT or TT or NS with ISP
430 const Position lumaRefPos( chromaPos().x << getComponentScaleX( COMPONENT_Cb, chromaFormat ), chromaPos().y << getComponentScaleY( COMPONENT_Cb, chromaFormat ) );
431 const CodingUnit* colLumaCu = cs->picture->cs->getCU( lumaRefPos, CHANNEL_TYPE_LUMA );
432
433 if( colLumaCu->lwidth() < 64 || colLumaCu->lheight() < 64 ) //further split at 64x64 luma node
434 {
435 const PartSplit cuSplitTypeDepth1Luma = CU::getSplitAtDepth( *colLumaCu, depthFor64x64Node );
436 CHECK( !(cuSplitTypeDepth1Luma >= CU_QUAD_SPLIT && cuSplitTypeDepth1Luma <= CU_TRIV_SPLIT), "split mode shall be BT, TT or QT" );
437 if( cuSplitTypeDepth1Luma != CU_QUAD_SPLIT )
438 {
439 allowCCLM = false;
440 }
441 }
442 else if( colLumaCu->lwidth() == 64 && colLumaCu->lheight() == 64 && colLumaCu->ispMode ) //not split at 64x64 luma node and use ISP mode
443 {
444 allowCCLM = false;
445 }
446 }
447 }
448
449 return allowCCLM;
450}

452const uint8_t CodingUnit::checkAllowedSbt() const
453{
454 if( !slice->getSPS()->getUseSBT() )
455 {
456 return 0;
457 }
458
459 //check on prediction mode
460 if (predMode == MODE_INTRA || predMode == MODE_IBC || predMode == MODE_PLT ) //intra, palette or IBC
461 {
462 return 0;
463 }
464 if( firstPU->ciipFlag )
465 {
466 return 0;
467 }
468
469 uint8_t sbtAllowed = 0;
470 int cuWidth = lwidth();
471 int cuHeight = lheight();
472 bool allow_type[NUMBER_SBT_IDX];
473 memset( allow_type, false, NUMBER_SBT_IDX * sizeof( bool ) );
474
475 //parameter
476 int maxSbtCUSize = cs->sps->getMaxTbSize();
477 int minSbtCUSize = 1 << ( MIN_CU_LOG2 + 1 );
478
479 //check on size
480 if( cuWidth > maxSbtCUSize || cuHeight > maxSbtCUSize )
481 {
482 return 0;
483 }
484
485 allow_type[SBT_VER_HALF] = cuWidth >= minSbtCUSize;
486 allow_type[SBT_HOR_HALF] = cuHeight >= minSbtCUSize;
487 allow_type[SBT_VER_QUAD] = cuWidth >= ( minSbtCUSize << 1 );
488 allow_type[SBT_HOR_QUAD] = cuHeight >= ( minSbtCUSize << 1 );
489
490 for( int i = 0; i < NUMBER_SBT_IDX; i++ )
491 {
492 sbtAllowed += (uint8_t)allow_type[i] << i;
493 }
494
495 return sbtAllowed;
496}

498uint8_t CodingUnit::getSbtTuSplit() const
499{
500 uint8_t sbtTuSplitType = 0;
501
502 switch( getSbtIdx() )
503 {
504 case SBT_VER_HALF: sbtTuSplitType = ( getSbtPos() == SBT_POS0 ? 0 : 1 ) + SBT_VER_HALF_POS0_SPLIT; break;
505 case SBT_HOR_HALF: sbtTuSplitType = ( getSbtPos() == SBT_POS0 ? 0 : 1 ) + SBT_HOR_HALF_POS0_SPLIT; break;
506 case SBT_VER_QUAD: sbtTuSplitType = ( getSbtPos() == SBT_POS0 ? 0 : 1 ) + SBT_VER_QUAD_POS0_SPLIT; break;
507 case SBT_HOR_QUAD: sbtTuSplitType = ( getSbtPos() == SBT_POS0 ? 0 : 1 ) + SBT_HOR_QUAD_POS0_SPLIT; break;
508 default: assert( 0 ); break;
509 }
510
511 assert( sbtTuSplitType <= SBT_HOR_QUAD_POS1_SPLIT && sbtTuSplitType >= SBT_VER_HALF_POS0_SPLIT );
512 return sbtTuSplitType;
513}
