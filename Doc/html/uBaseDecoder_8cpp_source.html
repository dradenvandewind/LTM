73uint32_t GetBitCountFromMax(uint32_t maxValue) {
74 // Bit count is determined by ceil(log2(maxRepresentation);
75 // Because log2 is c99 which is not so widely available we can use standard log with a divide.
76 // log2(x) = logY(x) / logY(2)
77 //
78 // Alternatively we want to find the highest bit set in maxValue.
79 static const float logBase = log(2.0f);
80 float log2Value = log(static_cast<float>(maxValue)) / logBase;
81 return static_cast<uint32_t>(ceil(log2Value));
82}

84static uint32_t uOffsetForNalUnitHeader(const uint8_t *nal) {
85 if (memcmp(nal, kNalHeader, kNalHeaderLength) == 0)
86 return kNalHeaderLength;

88 return kNalHeaderLength + 1;
89 return 0;

102void BaseDecoder::Unencapsulate(const uint8_t *nalIn, uint32_t nalLength, DataBuffer &out) {
103 out.resize(nalLength);
104
105 const uint8_t *nalEnd = nalIn + nalLength;
106 uint8_t *outPtr = out.data();
107
108 while (nalIn < nalEnd - 3) {
109 if (memcmp(nalIn, kNalEscape, kNalEscapeLength) == 0) {
110 nalIn += 3;
111 *outPtr++ = 0;
112 *outPtr++ = 0;
113 } else {
114 *outPtr++ = *nalIn;
115 nalIn++;
116 }
117 }
118
119 // Appending remaining data.
120 uint32_t remaining = static_cast<uint32_t>(nalEnd - nalIn);
121 if (remaining) {
122 memcpy(outPtr, nalIn, remaining);
123 outPtr += remaining;
124 }
125
126 // Shrink to correct size.
127 uint32_t newLength = static_cast<uint32_t>(outPtr - out.data());
128 out.resize(newLength);
129}

131uint32_t BaseDecoder::ReadBits(uint8_t numBits) {
132 assert(numBits <= 32);
133
134 uint32_t res = 0;
135
136 while (numBits) {
137 if (m_remainingBits < numBits) {
138 res <<= m_remainingBits; // Make room
139 res |= (m_currentBits >> (8 - m_remainingBits)); // Push in the bits we want
140 numBits -= m_remainingBits; // Reduce number of bits remaining to read
141 m_byteOffset++; // Move to next byte for next load
142
143 if (m_byteOffset >= m_currentNalPayload.size()) // Safety.
144 {
145 INFO("ERROR -- NalParsing error, out of bytes to read from");
146 throw std::runtime_error("NalParsing error, out of bytes to read from");
147 }
148
149 m_currentBits = *(m_currentNalPayload.data() + m_byteOffset); // Load in new byte
150 m_remainingBits = 8; // Reset remaining bits
151 } else {
152 res <<= numBits; // Make room
153 res |= (m_currentBits >> (8 - numBits)); // Push in the bits we want
154 m_currentBits <<= numBits; // Update current bits
155 m_remainingBits -= numBits; // Reduce number of bits remaining to read
156 numBits = 0; // Done.
157 }
158 }
159
160 return res;
161}

163uint32_t BaseDecoder::ReadUE() {
164 uint32_t digitCount = 0;
165 uint32_t bit = ReadBits(1);
166 uint32_t res = 0;
167
168 while (bit == 0) {
169 digitCount += 1;
170 bit = ReadBits(1);
171 }
172
173 res = bit;
174
175 while (digitCount > 0) {
176 digitCount -= 1;
177 res <<= 1;
178 res |= ReadBits(1);
179 }
180
181 return res - 1;
182}

184int32_t BaseDecoder::ReadSE() {
185 uint32_t ue = ReadUE();
186
187 // Even UE are negative.
188 // Odd are positive.
189
190 if (ue & 0x01) {
191 return static_cast<int32_t>((ue + 1) >> 1);
192 } else {
193 // Negate, and on even boundary so no need for the addition.
194 return -static_cast<int32_t>(ue >> 1);
195 }
196}
