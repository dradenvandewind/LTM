86// Get image from frame position
87Image YUVReader::read(unsigned position, uint64_t timestamp) const {
88 set_position(position);
89
90 std::vector<Surface> surfaces;
91
92 for (unsigned p = 0; p < image_description_.num_planes(); ++p) {
93 auto b = Surface::build_from<int8_t>();
94 b.reserve_bpp(image_description_.width(p), image_description_.height(p), image_description_.byte_depth(),
95 image_description_.row_stride(p));
96 if (image_description_.rows_are_contiguous(p)) {
97 if (fread(b.data(), image_description_.plane_size(p), 1, file_.get()) != 1)
98 ERR("Cannot read YUV file");
99 } else {
100 for (unsigned y = 0; y < image_description_.height(p); ++y) {
101 if (fread(b.data(0, y), image_description_.row_size(p), 1, file_.get()) != 1)
102 ERR("Cannot read YUV file");
103 }
104 }
105 surfaces.push_back(b.finish());
106 }
107
108 return Image(format("%s:%d", name_.c_str(), position_), image_description_, timestamp, surfaces);
109}

111// Parse the picture details from the filename
112//
113// Uses roughly the same conventions as Vooya and YUVDeluxe
114//
115// Returns a description - if format == PF_NONE, then no format was recognised
116// If a rate is recognized, and pRate != NULL, then the frame rate will be stored through that pointer
117//
118
119// Map of format names to internal version
120static const struct {
121 const char *name;
122 int bits;
123 ImageFormat format;
124} KnownFormats[] = {
125 {"420", -1, IMAGE_FORMAT_YUV420P8}, {"420p", -1, IMAGE_FORMAT_YUV420P8}, {"p420", -1, IMAGE_FORMAT_YUV420P8},
126 {"yuv", -1, IMAGE_FORMAT_YUV420P8},
127
128 {"420", 8, IMAGE_FORMAT_YUV420P8}, {"420p", 8, IMAGE_FORMAT_YUV420P8}, {"p420", 8, IMAGE_FORMAT_YUV420P8},
129 {"yuv", 8, IMAGE_FORMAT_YUV420P8},
130
131 {"420", 10, IMAGE_FORMAT_YUV420P10}, {"420p", 10, IMAGE_FORMAT_YUV420P10}, {"p420", 10, IMAGE_FORMAT_YUV420P10},
132 {"yuv", 10, IMAGE_FORMAT_YUV420P10},
133
134 {"y", 8, IMAGE_FORMAT_Y8}, {"y", 10, IMAGE_FORMAT_Y10}, {"y", 16, IMAGE_FORMAT_Y16},

138static ImageDescription ParseYUVFilename(const string &name, float *pRate) {
139
140 ImageFormat image_format = IMAGE_FORMAT_NONE;
141 unsigned width = 0;
142 unsigned height = 0;
143
144 // Parse filename for picture description
145 //
146 vector<string> parts;
147 split(parts, name, ("-_."));
148
149 static const regex dimensions_re("([0-9]+)x([0-9]+)"); // Size
150 static const regex fps_re("([0-9]+)(fps|hz)"); // Hz
151 static const regex bits_re("([0-9]+)(bits?|bpp)"); // Bit depth
152 static const regex format_re("(420|420p|p420|422|p422|422p|yuv|yuyv|y|)"); // Format
153
154 string format;
155 int bits = -1;
156
157 for (auto const p : parts) {
158 cmatch m;
159 const string lp = lowercase(p);
160
161 if (regex_match(lp.c_str(), m, dimensions_re)) {
162 width = stoi(m[1]);
163 height = stoi(m[2]);
164 }
165
166 if (regex_match(p.c_str(), m, fps_re)) {
167 if (pRate)
168 *pRate = stof(m[1].str());
169 }
170
171 if (regex_match(p.c_str(), m, bits_re)) {
172 bits = stoi(m[1].str());
173 }
174
175 if (regex_match(p.c_str(), m, format_re)) {
176 if (format.empty())
177 format = lowercase(m[1].str());
178 }
179 }
180
181 for (size_t i = 0; i < ARRAY_SIZE(KnownFormats); ++i)
182 if (format == KnownFormats[i].name && bits == KnownFormats[i].bits)
183 image_format = KnownFormats[i].format;
184
185 return ImageDescription(image_format, width, height);
186}

188// Open file for reading, given explicit format
189//
190std::unique_ptr<YUVReader> CreateYUVReader(const std::string &name, const ImageDescription &description, unsigned rate) {
191 // Is file there?
192 const uintmax_t fileSize = file_size(name);
193 if (fileSize == static_cast<uintmax_t>(-1))
194 return 0;
195
196 // Figure length
197 unsigned length = static_cast<unsigned>(fileSize / description.byte_size());
198
199 if (length == 0)
200 ERR("YUV file is too small");
201
202 UniquePtrFile yuvFile(std::fopen(name.c_str(), "rb"));
203
204 if (yuvFile)
205 return unique_ptr<YUVReader>(new YUVReader(name, description, length, (float)rate, yuvFile.release(), fileSize));
206
207 ERR("Cannot open YUV file");
208 return 0;
209}

211// Open file for reading, format is inferred from filename
212//
213unique_ptr<YUVReader> CreateYUVReader(const string &name) {
214 // Does it have a sensible name?
215 float rate = 25.0f;
216 ImageDescription description = ParseYUVFilename(name, &rate);
217 if (description.format() == IMAGE_FORMAT_NONE)
218 ERR("Cannot parse YUV filename");
219
220 return CreateYUVReader(name, description, (unsigned)rate);
221}
