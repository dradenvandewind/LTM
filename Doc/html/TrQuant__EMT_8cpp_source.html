51void fastForwardDCT2_B2(const TCoeff *src, TCoeff *dst, int shift, int line, int iSkipLine, int iSkipLine2)
52{
53 int j;
54 TCoeff E, O;
55 TCoeff add = (shift > 0) ? (1 << (shift - 1)) : 0;
56
57 const TMatrixCoeff *iT = g_trCoreDCT2P2[TRANSFORM_FORWARD][0];
58
59 TCoeff *pCoef = dst;
60 const int reducedLine = line - iSkipLine;
61 for (j = 0; j<reducedLine; j++)
62 {
63 /* E and O */
64 E = src[0] + src[1];
65 O = src[0] - src[1];
66
67 dst[0] = (iT[0] * E + add) >> shift;
68 dst[line] = (iT[2] * O + add) >> shift;
69
70
71 src += 2;
72 dst++;
73 }
74 if (iSkipLine)
75 {
76 dst = pCoef + reducedLine;
77 for (j = 0; j<2; j++)
78 {
79 memset(dst, 0, sizeof(TCoeff)*iSkipLine);
80 dst += line;
81 }
82 }

85void fastInverseDCT2_B2(const TCoeff *src, TCoeff *dst, int shift, int line, int iSkipLine, int iSkipLine2, const TCoeff outputMinimum, const TCoeff outputMaximum)
86{
87 int j;
88 TCoeff E, O;
89 TCoeff add = TCoeff(1) << (shift - 1);
90
91 const TMatrixCoeff *iT = g_trCoreDCT2P2[TRANSFORM_INVERSE][0];
92
93 const int reducedLine = line - iSkipLine;
94 for (j = 0; j<reducedLine; j++)
95 {
96 /* Utilizing symmetry properties to the maximum to minimize the number of multiplications */
97 E = iT[0] * (src[0] + src[line]);
98 O = iT[2] * (src[0] - src[line]);
99
100 /* Combining even and odd terms at each hierarchy levels to calculate the final spatial domain vector */
101 dst[0] = Clip3<TCoeff>(outputMinimum, outputMaximum, (E + add) >> shift);
102 dst[1] = Clip3<TCoeff>(outputMinimum, outputMaximum, (O + add) >> shift);
103
104 src++;
105 dst += 2;
106 }
107 if (iSkipLine)
108 {
109 memset(dst, 0, (iSkipLine << 1) * sizeof(TCoeff));

165{
166 int j;
167 TCoeff E[2], O[2];
168 TCoeff add = TCoeff(1) << ( shift - 1 );
169
170 const TMatrixCoeff *iT = g_trCoreDCT2P4[TRANSFORM_INVERSE][0];
171
172 const int reducedLine = line - iSkipLine;
173 for( j = 0; j < reducedLine; j++ )
174 {
175 /* Utilizing symmetry properties to the maximum to minimize the number of multiplications */
176 O[0] = iT[1 * 4 + 0] * src[line] + iT[3 * 4 + 0] * src[3 * line];
177 O[1] = iT[1 * 4 + 1] * src[line] + iT[3 * 4 + 1] * src[3 * line];
178 E[0] = iT[0 * 4 + 0] * src[ 0] + iT[2 * 4 + 0] * src[2 * line];
179 E[1] = iT[0 * 4 + 1] * src[ 0] + iT[2 * 4 + 1] * src[2 * line];
180
181 /* Combining even and odd terms at each hierarchy levels to calculate the final spatial domain vector */
182 dst[0] = Clip3<TCoeff>( outputMinimum, outputMaximum, ( E[0] + O[0] + add ) >> shift );
183 dst[1] = Clip3<TCoeff>( outputMinimum, outputMaximum, ( E[1] + O[1] + add ) >> shift );
184 dst[2] = Clip3<TCoeff>( outputMinimum, outputMaximum, ( E[1] - O[1] + add ) >> shift );
185 dst[3] = Clip3<TCoeff>( outputMinimum, outputMaximum, ( E[0] - O[0] + add ) >> shift );
186
187 src++;
188 dst += 4;
189 }
190 if( iSkipLine )
191 {
192 memset( dst, 0, ( iSkipLine << 2 ) * sizeof( TCoeff ) );
193 }

197template< int uiTrSize >
198inline void _fastInverseMM( const TCoeff *src, TCoeff *dst, int shift, int line, int iSkipLine, int iSkipLine2, const TCoeff outputMinimum, const TCoeff outputMaximum, const TMatrixCoeff* iT )
199{
200 const TCoeff rnd_factor = TCoeff(1) << (shift - 1);
201 const int reducedLine = line - iSkipLine;
202 const int cutoff = uiTrSize - iSkipLine2;
203
204 for( int i = 0; i<reducedLine; i++ )
205 {
206 for( int j = 0; j<uiTrSize; j++ )
207 {
208 TCoeff iSum = 0;
209 for( int k = 0; k<cutoff; k++)
210 {
211 iSum += src[k*line + i] * iT[k*uiTrSize + j];
212 }
213 dst[i*uiTrSize + j] = Clip3<TCoeff>(outputMinimum, outputMaximum, (iSum + rnd_factor) >> shift);
214 }
215 }
216
217 if (iSkipLine)
218 {
219 memset(dst + (reducedLine*uiTrSize), 0, (iSkipLine*uiTrSize) * sizeof(TCoeff));
220 }
221}

224template< int uiTrSize >
225inline void _fastForwardMM( const TCoeff *src, TCoeff *dst, int shift, int line, int iSkipLine, int iSkipLine2, const TMatrixCoeff* tc )
226{
227 const TCoeff rnd_factor = TCoeff(1) << (shift - 1);
228 const int reducedLine = line - iSkipLine;
229 const int cutoff = uiTrSize - iSkipLine2;
230 TCoeff *pCoef;
231
232 for( int i = 0; i<reducedLine; i++ )
233 {
234 pCoef = dst;
235 const TMatrixCoeff* iT = tc;
236 for( int j = 0; j<cutoff; j++ )
237 {
238 TCoeff iSum = 0;
239 for( int k = 0; k<uiTrSize; k++ )
240 {
241 iSum += src[k] * iT[k];
242 }
243 pCoef[i] = (iSum + rnd_factor) >> shift;
244 pCoef += line;
245 iT += uiTrSize;
246 }
247 src += uiTrSize;
248 }
249
250 if( iSkipLine )
251 {
252 pCoef = dst + reducedLine;
253 for( int j = 0; j<cutoff; j++ )
254 {
255 memset(pCoef, 0, sizeof(TCoeff) * iSkipLine);
256 pCoef += line;
257 }
258 }
259
260 if( iSkipLine2 )
261 {
262 pCoef = dst + line*cutoff;
263 memset(pCoef, 0, sizeof(TCoeff) * line * iSkipLine2);
264 }
265}
