49Surface Quantize::process(const Surface &src_plane, int32_t dirq_step_width, int32_t deadzone, const Surface &pixel_sad_plane,
50 unsigned transform_block_size, unsigned threshold) {
51
52 if (pixel_sad_plane.empty() || threshold == 5) {
53 // Regular Quantization
54 struct Context {
55 const SurfaceView<int16_t> src;
56 int32_t step_width;
57 int32_t deadzone;
58 } context = {
59 src_plane.view_as<int16_t>(),
60 dirq_step_width,
61 deadzone,
62 };
63
64 return Surface::build_from<int16_t>()
65 .generate<Context>(
66 src_plane.width(), src_plane.height(),
67 [](unsigned x, unsigned y, const Context &c) -> int16_t {
68 const int16_t in = c.src.read(x, y);
69
70 int16_t sign = ((in > 0) ? 1 : ((in < 0) ? (-1) : 0));
71 int16_t out = clamp((int16_t)(sign * (std::max(0, (((sign * in + c.deadzone) / c.step_width))))),
72 (int16_t)-8192, (int16_t)8191);
73
74 return out;
75 },
76 context)
77 .finish();
78 } else {
79 // Motion-adaptive Quantization
80 CHECK(threshold > 0 && threshold < 5);
81 struct Context {
82 const SurfaceView<int16_t> src;
83 const SurfaceView<int16_t> pixel_sad;
84 int32_t step_width;
85 int32_t deadzone;
86 unsigned transform_block_size;
87 unsigned threshold;
88 } context = {
89 src_plane.view_as<int16_t>(),
90 pixel_sad_plane.view_as<int16_t>(),
91 dirq_step_width,
92 deadzone,
93 transform_block_size,
94 threshold,
95 };
96
97 return Surface::build_from<int16_t>()
98 .generate<Context>(
99 src_plane.width(), src_plane.height(),
100 [](unsigned x, unsigned y, const Context &c) -> int16_t {
101 const int16_t in = c.src.read(x, y);
102 const int16_t sad = c.pixel_sad.read(x, y);
103
104 int16_t sign = ((in > 0) ? 1 : ((in < 0) ? (-1) : 0));
105 int16_t out = (sign * (std::max(0, (((sign * in + c.deadzone) / c.step_width)))));
106
107 if (sad > (c.transform_block_size == 4 ? 200 : 100))
108 return clamp(out, (int16_t)-8192, (int16_t)8191);
109 else {
110 int16_t r_deadzone = ((int16_t)c.threshold * c.deadzone) / 5;
111 int16_t reduction = sign * (std::min(1, std::max(0, (sign * in + r_deadzone) / c.step_width)));
112 int16_t correction = sign * (std::min(1, std::max(0, (sign * in + c.deadzone) / c.step_width)));
113
114 out = out + reduction - correction;
115 return clamp(out, (int16_t)-8192, (int16_t)8191);
116 }
117 },
118 context)
119 .finish();
120 }
121}

123Surface Quantize_SWM::process(const Surface &src_plane, unsigned transform_block_size, int32_t *dirq_step_width, int32_t *deadzone,
124 const Surface &temporal_mask, const Surface &pixel_sad_plane, unsigned threshold) {
125
126 if (pixel_sad_plane.empty() || threshold == 5) {
127 // Regular Quantization
128 struct Context {
129 const SurfaceView<int16_t> src;
130 const SurfaceView<uint8_t> mask;
131 unsigned d;
132 int32_t *step_width;
133 int32_t *deadzone;
134 } context = {
135 src_plane.view_as<int16_t>(), temporal_mask.view_as<uint8_t>(), 32 / transform_block_size, dirq_step_width, deadzone,
136 };
137
138 return Surface::build_from<int16_t>()
139 .generate<Context>(
140 src_plane.width(), src_plane.height(),
141 [](unsigned x, unsigned y, const Context &c) -> int16_t {
142 const uint8_t tile_refresh = c.mask.read((x / c.d) * c.d, (y / c.d) * c.d);
143 const int16_t in = c.src.read(x, y);
144 int16_t out;
145
146 int16_t sign = ((in > 0) ? 1 : ((in < 0) ? (-1) : 0));
147 if (tile_refresh == TEMPORAL_PRED)
148 out = clamp((int16_t)(sign * (std::max(0, (((sign * in + c.deadzone[0]) / c.step_width[0]))))),
149 (int16_t)-8192, (int16_t)8191);
150 else
151 out = clamp((int16_t)(sign * (std::max(0, (((sign * in + c.deadzone[1]) / c.step_width[1]))))),
152 (int16_t)-8192, (int16_t)8191);
153
154 return out;
155 },
156 context)
157 .finish();
158 } else {
159 // Motion-adaptive Quantization
160 CHECK(threshold > 0 && threshold < 5);
161 struct Context {
162 const SurfaceView<int16_t> src;
163 const SurfaceView<int16_t> pixel_sad;
164 const SurfaceView<uint8_t> mask;
165 unsigned d;
166 int32_t *step_width;
167 int32_t *deadzone;
168 unsigned threshold;
169 } context = {
170 src_plane.view_as<int16_t>(),
171 pixel_sad_plane.view_as<int16_t>(),
172 temporal_mask.view_as<uint8_t>(),
173 32 / transform_block_size,
174 dirq_step_width,
175 deadzone,
176 threshold,
177 };
178
179 return Surface::build_from<int16_t>()
180 .generate<Context>(
181 src_plane.width(), src_plane.height(),
182 [](unsigned x, unsigned y, const Context &c) -> int16_t {
183 const uint8_t tile_refresh = c.mask.read((x / c.d) * c.d, (y / c.d) * c.d);
184 const int16_t in = c.src.read(x, y);
185 const int16_t sad = c.pixel_sad.read(x, y);
186 int16_t out;
187
188 int16_t sign = ((in > 0) ? 1 : ((in < 0) ? (-1) : 0));
189 if (tile_refresh == TEMPORAL_PRED)
190 out = (sign * (std::max(0, (((sign * in + c.deadzone[0]) / c.step_width[0])))));
191 else
192 out = (sign * (std::max(0, (((sign * in + c.deadzone[1]) / c.step_width[1])))));
193
194 if (sad > (c.d == 8 ? 200 : 100))
195 return clamp(out, (int16_t)-8192, (int16_t)8191);
196 else {
197 if (tile_refresh == TEMPORAL_PRED) {
198 int16_t r_deadzone = ((int16_t)c.threshold * c.deadzone[0]) / 5;
199 int16_t reduction = sign * (std::min(1, std::max(0, (sign * in + r_deadzone) / c.step_width[0])));
200 int16_t correction = sign * (std::min(1, std::max(0, (sign * in + c.deadzone[0]) / c.step_width[0])));
201 out = out + reduction - correction;
202 } else {
203 int16_t r_deadzone = ((int16_t)c.threshold * c.deadzone[1]) / 5;
204 int16_t reduction = sign * (std::min(1, std::max(0, (sign * in + r_deadzone) / c.step_width[1])));
205 int16_t correction = sign * (std::min(1, std::max(0, (sign * in + c.deadzone[1]) / c.step_width[1])));
206 out = out + reduction - correction;
207 }
208 return clamp(out, (int16_t)-8192, (int16_t)8191);
209 }
210 },
211 context)
212 .finish();
213 }
