41 // IntegralConstant - used to generate a compile time value constant that can be used for
42 // conditions checking.
43 // ~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~
44
45 template<class T, T v> struct IntegralConstant { static const T Value = v; };
46 typedef IntegralConstant<bool, true> TrueType;

105 // ~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~
106 // Low bit mask - Generates an integer with the lowest N contiguous bits set of the supplied
107 // integer type.
108 // ~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~
109
110 template<typename INTEGER_TYPE, uint32_t Bits>
111 struct LowBitMask
112 {
113 VNStaticAssert(Bits <= (sizeof(INTEGER_TYPE) * 8));
114 static const INTEGER_TYPE Value = (1 << Bits) - 1;
115 };
116
117 // ~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~
118 // MinValue - Generates the minimum value represented by the lowest N contiguous bits,
119 // accounting for a signed bit when INTEGER_TYPE is signed.
120 // ~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~
121
122 template<typename INTEGER_TYPE, uint32_t Bits>
123 struct MinValue
124 {
125 static const INTEGER_TYPE Value = IsSigned<INTEGER_TYPE>::Value ? ~(LowBitMask<INTEGER_TYPE, Bits>::Value) : 0;
126 };
127
128 // ~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~
129 // MaxValue - Generates the maximum value represented by the lowest N contiguous bits.
130 // ~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~
131
132 template<typename INTEGER_TYPE, uint32_t Bits>
133 struct MaxValue
134 {
135 static const INTEGER_TYPE Value = LowBitMask<INTEGER_TYPE, Bits>::Value;
136 };
