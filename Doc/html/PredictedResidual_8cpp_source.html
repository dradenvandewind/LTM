44// Sum each 2x2 block of pels
45//
46Surface PredictedResidualSum::process(const Surface &src_plane) {
47
48 typedef SurfaceView<int16_t> Context;
49
50 const Context ctx = src_plane.view_as<int16_t>();
51
52#if defined __OPT_MATRIX__
53 auto dest = Surface::build_from<int32_t>();
54 dest.reserve(src_plane.width() / 2, src_plane.height() / 2);
55 int32_t *__restrict pdst = dest.data(0, 0);
56 for (unsigned y = 0; y < src_plane.height() / 2; ++y) {
57 const int16_t *__restrict psrc0 = ctx.data(0, 2 * y + 0);
58 const int16_t *__restrict psrc1 = ctx.data(0, 2 * y + 1);
59 for (unsigned x = 0; x < src_plane.width() / 2; ++x) {
60#if 1
61 const int32_t sum = *(psrc0 + 0) + *(psrc0 + 1) + *(psrc1 + 0) + *(psrc1 + 1);
62 psrc0 += 2;
63 psrc1 += 2;
64 *pdst++ = sum;
65#else
66 const int32_t sum = *psrc0++ + *psrc0++ + *psrc1++ + *psrc1++;
67 *pdst++ = sum;
68#endif
69 }
70 }
71 return dest.finish();
72#else
73 return Surface::build_from<int32_t>()
74 .generate<Context>(
75 src_plane.width() / 2, src_plane.height() / 2,
76 [](unsigned x, unsigned y, const Context &ctx) -> int32_t {
77 return ctx.read(x * 2 + 0, y * 2 + 0) + ctx.read(x * 2 + 0, y * 2 + 1) + ctx.read(x * 2 + 1, y * 2 + 0) +
78 ctx.read(x * 2 + 1, y * 2 + 1);
79 },
80 ctx)
81 .finish();
82#endif
83}

87Surface PredictedResidualSum_1D::process(const Surface &src_plane) {

89
90 return Surface::build_from<int32_t>()
91 .generate(src.width() / 2, src.height(),

96// Produce adjusted 2x2 upsampled layer that averages to the base layer
97Surface PredictedResidualAdjust::process(const Surface &base_plane, const Surface &enhanced_plane, const Surface &sum_plane) {
98 const struct Context {
99 SurfaceView<int16_t> base;
100 SurfaceView<int16_t> enhanced;
101 SurfaceView<int32_t> sum;
102 } ctx = {base_plane.view_as<int16_t>(), enhanced_plane.view_as<int16_t>(), sum_plane.view_as<int32_t>()};
103
104#if defined __OPT_MATRIX__
105 auto dest = Surface::build_from<int16_t>();
106 dest.reserve(enhanced_plane.width(), enhanced_plane.height());
107 const int16_t *__restrict pbas = ctx.base.data(0, 0);
108 const int32_t *__restrict psum = ctx.sum.data(0, 0);
109 for (unsigned y = 0; y < enhanced_plane.height() / 2; ++y) {
110 const int16_t *__restrict psrc0 = ctx.enhanced.data(0, 2 * y + 0);
111 const int16_t *__restrict psrc1 = ctx.enhanced.data(0, 2 * y + 1);
112 int16_t *__restrict pdst0 = dest.data(0, 2 * y + 0);
113 int16_t *__restrict pdst1 = dest.data(0, 2 * y + 1);
114 for (unsigned x = 0; x < enhanced_plane.width() / 2; ++x) {
115 const int32_t adjust = *pbas++ - ((*psum++ + 2) >> 2);
116#if 1
117 *(pdst0 + 0) = clamp(*(psrc0 + 0) + adjust, -32767, 32767);
118 *(pdst0 + 1) = clamp(*(psrc0 + 1) + adjust, -32767, 32767);
119 *(pdst1 + 0) = clamp(*(psrc1 + 0) + adjust, -32767, 32767);
120 *(pdst1 + 1) = clamp(*(psrc1 + 1) + adjust, -32767, 32767);
121 psrc0 += 2;
122 psrc1 += 2;
123 pdst0 += 2;
124 pdst1 += 2;
125#else
126 *pdst0++ = clamp(*psrc0++ + adjust, -32767, 32767);
127 *pdst0++ = clamp(*psrc0++ + adjust, -32767, 32767);
128 *pdst1++ = clamp(*psrc1++ + adjust, -32767, 32767);
129 *pdst1++ = clamp(*psrc1++ + adjust, -32767, 32767);
130#endif
131 }
132 }
133 return dest.finish();
134#else
135 return Surface::build_from<int16_t>()
136 .generate<Context>(
137 enhanced_plane.width(), enhanced_plane.height(),
138 [](unsigned x, unsigned y, const Context &ctx) -> int16_t {
139 // Calculate 4*delta to add to enhanced pel
140 const int32_t adjust = ctx.base.read(x / 2, y / 2) - ((ctx.sum.read(x / 2, y / 2) + 2) >> 2);
141 return clamp(ctx.enhanced.read(x, y) + adjust, -32767, 32767);
142 },
143 ctx)
144 .finish();
145#endif
146}

148// Produce adjusted 2x1 upsampled layer that averages to the base layer
149Surface PredictedResidualAdjust_1D::process(const Surface &base_plane, const Surface &enhanced_plane, const Surface &sum_plane) {
150 const auto base = base_plane.view_as<int16_t>();
151 const auto enhanced = enhanced_plane.view_as<int16_t>();
152 const auto sum = sum_plane.view_as<int32_t>();
153
154 return Surface::build_from<int16_t>()
155 .generate(enhanced.width(), enhanced.height(),
156 [&](unsigned x, unsigned y) -> int16_t {
157 // Calculate 2*delta to add to enhanced pel
158 const int32_t adjust = base.read(x / 2, y) - ((sum.read(x / 2, y) + 1) >> 1);
159 return clamp(enhanced.read(x, y) + adjust, -32767, 32767);
160 })
161 .finish();
162}
