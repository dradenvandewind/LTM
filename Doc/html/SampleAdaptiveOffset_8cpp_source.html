150void SampleAdaptiveOffset::invertQuantOffsets(ComponentID compIdx, int typeIdc, int typeAuxInfo, int* dstOffsets, int* srcOffsets)
151{
152 int codedOffset[MAX_NUM_SAO_CLASSES];
153
154 ::memcpy(codedOffset, srcOffsets, sizeof(int)*MAX_NUM_SAO_CLASSES);
155 ::memset(dstOffsets, 0, sizeof(int)*MAX_NUM_SAO_CLASSES);
156
157 if(typeIdc == SAO_TYPE_START_BO)
158 {
159 for(int i=0; i< 4; i++)
160 {
161 dstOffsets[(typeAuxInfo+ i)%NUM_SAO_BO_CLASSES] = codedOffset[(typeAuxInfo+ i)%NUM_SAO_BO_CLASSES]*(1<<m_offsetStepLog2[compIdx]);
162 }
163 }
164 else //EO
165 {
166 for(int i=0; i< NUM_SAO_EO_CLASSES; i++)
167 {
168 dstOffsets[i] = codedOffset[i] *(1<<m_offsetStepLog2[compIdx]);
169 }
170 CHECK(dstOffsets[SAO_CLASS_EO_PLAIN] != 0, "EO offset is not '0'"); //keep EO plain offset as zero
171 }

175int SampleAdaptiveOffset::getMergeList(CodingStructure& cs, int ctuRsAddr, SAOBlkParam* blkParams, SAOBlkParam* mergeList[NUM_SAO_MERGE_TYPES])
176{
177 const PreCalcValues& pcv = *cs.pcv;
178
179 int ctuX = ctuRsAddr % pcv.widthInCtus;
180 int ctuY = ctuRsAddr / pcv.widthInCtus;
181 const CodingUnit& cu = *cs.getCU(Position(ctuX*pcv.maxCUWidth, ctuY*pcv.maxCUHeight), CH_L);
182 int mergedCTUPos;
183 int numValidMergeCandidates = 0;
184
185 for(int mergeType=0; mergeType< NUM_SAO_MERGE_TYPES; mergeType++)
186 {
187 SAOBlkParam* mergeCandidate = NULL;
188
189 switch(mergeType)
190 {
191 case SAO_MERGE_ABOVE:
192 {
193 if(ctuY > 0)
194 {
195 mergedCTUPos = ctuRsAddr- pcv.widthInCtus;
196 if(cs.getCURestricted(Position(ctuX*pcv.maxCUWidth, (ctuY-1)*pcv.maxCUHeight), cu, cu.chType))
197 {
198 mergeCandidate = &(blkParams[mergedCTUPos]);
199 }
200 }
201 }
202 break;
203 case SAO_MERGE_LEFT:
204 {
205 if(ctuX > 0)
206 {
207 mergedCTUPos = ctuRsAddr- 1;
208 if(cs.getCURestricted(Position((ctuX-1)*pcv.maxCUWidth, ctuY*pcv.maxCUHeight), cu, cu.chType))
209 {
210 mergeCandidate = &(blkParams[mergedCTUPos]);
211 }
212 }
213 }
214 break;
215 default:
216 {
217 THROW("not a supported merge type");
218 }
219 }
220
221 mergeList[mergeType]=mergeCandidate;
222 if (mergeCandidate != NULL)
223 {
224 numValidMergeCandidates++;
225 }
226 }
227
228 return numValidMergeCandidates;
229}

232void SampleAdaptiveOffset::reconstructBlkSAOParam(SAOBlkParam& recParam, SAOBlkParam* mergeList[NUM_SAO_MERGE_TYPES])
233{
234 const int numberOfComponents = m_numberOfComponents;
235 for(int compIdx = 0; compIdx < numberOfComponents; compIdx++)
236 {
237 const ComponentID component = ComponentID(compIdx);
238 SAOOffset& offsetParam = recParam[component];
239
240 if(offsetParam.modeIdc == SAO_MODE_OFF)
241 {
242 continue;
243 }
244
245 switch(offsetParam.modeIdc)
246 {
247 case SAO_MODE_NEW:
248 {
249 invertQuantOffsets(component, offsetParam.typeIdc, offsetParam.typeAuxInfo, offsetParam.offset, offsetParam.offset);
250 }
251 break;
252 case SAO_MODE_MERGE:
253 {
254 SAOBlkParam* mergeTarget = mergeList[offsetParam.typeIdc];
255 CHECK(mergeTarget == NULL, "Merge target does not exist");
256
257 offsetParam = (*mergeTarget)[component];
258 }
259 break;
260 default:
261 {
262 THROW("Not a supported mode");
263 }
264 }
265 }
266}

269{
270 for(uint32_t compIdx = 0; compIdx < MAX_NUM_COMPONENT; compIdx++)
271 {
272 m_picSAOEnabled[compIdx] = false;

274

276

278 {
279 SAOBlkParam* mergeList[NUM_SAO_MERGE_TYPES] = { NULL };
280 getMergeList(cs, ctuRsAddr, saoBlkParams, mergeList);
281

283
284 for(uint32_t compIdx = 0; compIdx < numberOfComponents; compIdx++)
285 {
286 if(saoBlkParams[ctuRsAddr][compIdx].modeIdc != SAO_MODE_OFF)
287 {
288 m_picSAOEnabled[compIdx] = true;
289 }
290 }
291 }

295void SampleAdaptiveOffset::offsetBlock(const int channelBitDepth, const ClpRng& clpRng, int typeIdx, int* offset
296 , const Pel* srcBlk, Pel* resBlk, int srcStride, int resStride, int width, int height
297 , bool isLeftAvail, bool isRightAvail, bool isAboveAvail, bool isBelowAvail, bool isAboveLeftAvail, bool isAboveRightAvail, bool isBelowLeftAvail, bool isBelowRightAvail
298 , bool isCtuCrossedByVirtualBoundaries, int horVirBndryPos[], int verVirBndryPos[], int numHorVirBndry, int numVerVirBndry
299 )
300{
301 int x,y, startX, startY, endX, endY, edgeType;
302 int firstLineStartX, firstLineEndX, lastLineStartX, lastLineEndX;
303 int8_t signLeft, signRight, signDown;
304
305 const Pel* srcLine = srcBlk;
306 Pel* resLine = resBlk;
307
308 switch(typeIdx)
309 {
310 case SAO_TYPE_EO_0:
311 {
312 offset += 2;
313 startX = isLeftAvail ? 0 : 1;
314 endX = isRightAvail ? width : (width -1);
315 for (y=0; y< height; y++)
316 {
317 signLeft = (int8_t)sgn(srcLine[startX] - srcLine[startX-1]);
318 for (x=startX; x< endX; x++)
319 {
320 signRight = (int8_t)sgn(srcLine[x] - srcLine[x+1]);
321 if (isCtuCrossedByVirtualBoundaries && isProcessDisabled(x, y, numVerVirBndry, 0, verVirBndryPos, horVirBndryPos))
322 {
323 signLeft = -signRight;
324 continue;
325 }
326 edgeType = signRight + signLeft;
327 signLeft = -signRight;
328
329 resLine[x] = ClipPel<int>( srcLine[x] + offset[edgeType], clpRng);
330 }
331 srcLine += srcStride;
332 resLine += resStride;
333 }
334
335 }
336 break;
337 case SAO_TYPE_EO_90:
338 {
339 offset += 2;
340 int8_t *signUpLine = &m_signLineBuf1[0];
341
342 startY = isAboveAvail ? 0 : 1;
343 endY = isBelowAvail ? height : height-1;
344 if (!isAboveAvail)
345 {
346 srcLine += srcStride;
347 resLine += resStride;
348 }
349
350 const Pel* srcLineAbove= srcLine- srcStride;
351 for (x=0; x< width; x++)
352 {
353 signUpLine[x] = (int8_t)sgn(srcLine[x] - srcLineAbove[x]);
354 }
355
356 const Pel* srcLineBelow;
357 for (y=startY; y<endY; y++)
358 {
359 srcLineBelow= srcLine+ srcStride;
360
361 for (x=0; x< width; x++)
362 {
363 signDown = (int8_t)sgn(srcLine[x] - srcLineBelow[x]);
364 if (isCtuCrossedByVirtualBoundaries && isProcessDisabled(x, y, 0, numHorVirBndry, verVirBndryPos, horVirBndryPos))
365 {
366 signUpLine[x] = -signDown;
367 continue;
368 }
369 edgeType = signDown + signUpLine[x];
370 signUpLine[x]= -signDown;
371
372 resLine[x] = ClipPel<int>(srcLine[x] + offset[edgeType], clpRng);
373 }
374 srcLine += srcStride;
375 resLine += resStride;
376 }
377
378 }
379 break;
380 case SAO_TYPE_EO_135:
381 {
382 offset += 2;
383 int8_t *signUpLine, *signDownLine, *signTmpLine;
384
385 signUpLine = &m_signLineBuf1[0];
386 signDownLine= &m_signLineBuf2[0];
387
388 startX = isLeftAvail ? 0 : 1 ;
389 endX = isRightAvail ? width : (width-1);
390
391 //prepare 2nd line's upper sign
392 const Pel* srcLineBelow= srcLine+ srcStride;
393 for (x=startX; x< endX+1; x++)
394 {
395 signUpLine[x] = (int8_t)sgn(srcLineBelow[x] - srcLine[x- 1]);
396 }
397
398 //1st line
399 const Pel* srcLineAbove= srcLine- srcStride;
400 firstLineStartX = isAboveLeftAvail ? 0 : 1;
401 firstLineEndX = isAboveAvail? endX: 1;
402 for(x= firstLineStartX; x< firstLineEndX; x++)
403 {
404 if (isCtuCrossedByVirtualBoundaries && isProcessDisabled(x, 0, numVerVirBndry, numHorVirBndry, verVirBndryPos, horVirBndryPos))
405 {
406 continue;
407 }
408 edgeType = sgn(srcLine[x] - srcLineAbove[x- 1]) - signUpLine[x+1];
409
410 resLine[x] = ClipPel<int>( srcLine[x] + offset[edgeType], clpRng);
411 }
412 srcLine += srcStride;
413 resLine += resStride;
414
415
416 //middle lines
417 for (y= 1; y< height-1; y++)
418 {
419 srcLineBelow= srcLine+ srcStride;
420
421 for (x=startX; x<endX; x++)
422 {
423 signDown = (int8_t)sgn(srcLine[x] - srcLineBelow[x+ 1]);
424 if (isCtuCrossedByVirtualBoundaries && isProcessDisabled(x, y, numVerVirBndry, numHorVirBndry, verVirBndryPos, horVirBndryPos))
425 {
426 signDownLine[x + 1] = -signDown;
427 continue;
428 }
429 edgeType = signDown + signUpLine[x];
430 resLine[x] = ClipPel<int>( srcLine[x] + offset[edgeType], clpRng);
431
432 signDownLine[x+1] = -signDown;
433 }
434 signDownLine[startX] = (int8_t)sgn(srcLineBelow[startX] - srcLine[startX-1]);
435
436 signTmpLine = signUpLine;
437 signUpLine = signDownLine;
438 signDownLine = signTmpLine;
439
440 srcLine += srcStride;
441 resLine += resStride;
442 }
443
444 //last line
445 srcLineBelow= srcLine+ srcStride;
446 lastLineStartX = isBelowAvail ? startX : (width -1);
447 lastLineEndX = isBelowRightAvail ? width : (width -1);
448 for(x= lastLineStartX; x< lastLineEndX; x++)
449 {
450 if (isCtuCrossedByVirtualBoundaries && isProcessDisabled(x, height - 1, numVerVirBndry, numHorVirBndry, verVirBndryPos, horVirBndryPos))
451 {
452 continue;
453 }
454 edgeType = sgn(srcLine[x] - srcLineBelow[x+ 1]) + signUpLine[x];
455 resLine[x] = ClipPel<int>( srcLine[x] + offset[edgeType], clpRng);
456
457 }
458 }
459 break;
460 case SAO_TYPE_EO_45:
461 {
462 offset += 2;
463 int8_t *signUpLine = &m_signLineBuf1[1];
464
465 startX = isLeftAvail ? 0 : 1;
466 endX = isRightAvail ? width : (width -1);
467
468 //prepare 2nd line upper sign
469 const Pel* srcLineBelow= srcLine+ srcStride;
470 for (x=startX-1; x< endX; x++)
471 {
472 signUpLine[x] = (int8_t)sgn(srcLineBelow[x] - srcLine[x+1]);
473 }
474
475
476 //first line
477 const Pel* srcLineAbove= srcLine- srcStride;
478 firstLineStartX = isAboveAvail ? startX : (width -1 );
479 firstLineEndX = isAboveRightAvail ? width : (width-1);
480 for(x= firstLineStartX; x< firstLineEndX; x++)
481 {
482 if (isCtuCrossedByVirtualBoundaries && isProcessDisabled(x, 0, numVerVirBndry, numHorVirBndry, verVirBndryPos, horVirBndryPos))
483 {
484 continue;
485 }
486 edgeType = sgn(srcLine[x] - srcLineAbove[x+1]) -signUpLine[x-1];
487 resLine[x] = ClipPel<int>(srcLine[x] + offset[edgeType], clpRng);
488 }
489 srcLine += srcStride;
490 resLine += resStride;
491
492 //middle lines
493 for (y= 1; y< height-1; y++)
494 {
495 srcLineBelow= srcLine+ srcStride;
496
497 for(x= startX; x< endX; x++)
498 {
499 signDown = (int8_t)sgn(srcLine[x] - srcLineBelow[x-1]);
500 if (isCtuCrossedByVirtualBoundaries && isProcessDisabled(x, y, numVerVirBndry, numHorVirBndry, verVirBndryPos, horVirBndryPos))
501 {
502 signUpLine[x - 1] = -signDown;
503 continue;
504 }
505 edgeType = signDown + signUpLine[x];
506 resLine[x] = ClipPel<int>(srcLine[x] + offset[edgeType], clpRng);
507 signUpLine[x-1] = -signDown;
508 }
509 signUpLine[endX-1] = (int8_t)sgn(srcLineBelow[endX-1] - srcLine[endX]);
510 srcLine += srcStride;
511 resLine += resStride;
512 }
513
514 //last line
515 srcLineBelow= srcLine+ srcStride;
516 lastLineStartX = isBelowLeftAvail ? 0 : 1;
517 lastLineEndX = isBelowAvail ? endX : 1;
518 for(x= lastLineStartX; x< lastLineEndX; x++)
519 {
520 if (isCtuCrossedByVirtualBoundaries && isProcessDisabled(x, height - 1, numVerVirBndry, numHorVirBndry, verVirBndryPos, horVirBndryPos))
521 {
522 continue;
523 }
524 edgeType = sgn(srcLine[x] - srcLineBelow[x-1]) + signUpLine[x];
525 resLine[x] = ClipPel<int>(srcLine[x] + offset[edgeType], clpRng);
526
527 }
528 }
529 break;
530 case SAO_TYPE_BO:
531 {
532 const int shiftBits = channelBitDepth - NUM_SAO_BO_CLASSES_LOG2;
533 for (y=0; y< height; y++)
534 {
535 for (x=0; x< width; x++)
536 {
537 resLine[x] = ClipPel<int>(srcLine[x] + offset[srcLine[x] >> shiftBits], clpRng );
538 }
539 srcLine += srcStride;
540 resLine += resStride;
541 }
542 }
543 break;
544 default:
545 {
546 THROW("Not a supported SAO types\n");
547 }
548 }
549}

551void SampleAdaptiveOffset::offsetCTU( const UnitArea& area, const CPelUnitBuf& src, PelUnitBuf& res, SAOBlkParam& saoblkParam, CodingStructure& cs)
552{
553 const uint32_t numberOfComponents = getNumberValidComponents( area.chromaFormat );
554 bool bAllOff=true;
555 for( uint32_t compIdx = 0; compIdx < numberOfComponents; compIdx++)
556 {
557 if (saoblkParam[compIdx].modeIdc != SAO_MODE_OFF)
558 {
559 bAllOff=false;
560 }
561 }
562 if (bAllOff)
563 {
564 return;
565 }
566
567 bool isLeftAvail, isRightAvail, isAboveAvail, isBelowAvail, isAboveLeftAvail, isAboveRightAvail, isBelowLeftAvail, isBelowRightAvail;
568
569 //block boundary availability
570 deriveLoopFilterBoundaryAvailibility(cs, area.Y(), isLeftAvail,isRightAvail,isAboveAvail,isBelowAvail,isAboveLeftAvail,isAboveRightAvail,isBelowLeftAvail,isBelowRightAvail);
571
572 const size_t lineBufferSize = area.Y().width + 1;
573 if (m_signLineBuf1.size() < lineBufferSize)
574 {
575 m_signLineBuf1.resize(lineBufferSize);
576 m_signLineBuf2.resize(lineBufferSize);
577 }
578
579 int numHorVirBndry = 0, numVerVirBndry = 0;
580 int horVirBndryPos[] = { -1,-1,-1 };
581 int verVirBndryPos[] = { -1,-1,-1 };
582 int horVirBndryPosComp[] = { -1,-1,-1 };
583 int verVirBndryPosComp[] = { -1,-1,-1 };
584 bool isCtuCrossedByVirtualBoundaries = isCrossedByVirtualBoundaries(area.Y().x, area.Y().y, area.Y().width, area.Y().height, numHorVirBndry, numVerVirBndry, horVirBndryPos, verVirBndryPos, cs.picHeader );
585 for(int compIdx = 0; compIdx < numberOfComponents; compIdx++)
586 {
587 const ComponentID compID = ComponentID(compIdx);
588 const CompArea& compArea = area.block(compID);
589 SAOOffset& ctbOffset = saoblkParam[compIdx];
590
591 if(ctbOffset.modeIdc != SAO_MODE_OFF)
592 {
593 int srcStride = src.get(compID).stride;
594 const Pel* srcBlk = src.get(compID).bufAt(compArea);
595 int resStride = res.get(compID).stride;
596 Pel* resBlk = res.get(compID).bufAt(compArea);
597 for (int i = 0; i < numHorVirBndry; i++)
598 {
599 horVirBndryPosComp[i] = (horVirBndryPos[i] >> vtm::getComponentScaleY(compID, area.chromaFormat)) - compArea.y;
600 }
601 for (int i = 0; i < numVerVirBndry; i++)
602 {
603 verVirBndryPosComp[i] = (verVirBndryPos[i] >> vtm::getComponentScaleX(compID, area.chromaFormat)) - compArea.x;
604 }
605
606 offsetBlock( cs.sps->getBitDepth(toChannelType(compID)),
607 cs.slice->clpRng(compID),
608 ctbOffset.typeIdc, ctbOffset.offset
609 , srcBlk, resBlk, srcStride, resStride, compArea.width, compArea.height
610 , isLeftAvail, isRightAvail
611 , isAboveAvail, isBelowAvail
612 , isAboveLeftAvail, isAboveRightAvail
613 , isBelowLeftAvail, isBelowRightAvail
614 , isCtuCrossedByVirtualBoundaries, horVirBndryPosComp, verVirBndryPosComp, numHorVirBndry, numVerVirBndry
615 );
616 }
617 } //compIdx
618}

620void SampleAdaptiveOffset::SAOProcess( CodingStructure& cs, SAOBlkParam* saoBlkParams
621 )
622{
623 CHECK(!saoBlkParams, "No parameters present");
624
625 xReconstructBlkSAOParams(cs, saoBlkParams);
626
627 const uint32_t numberOfComponents = getNumberValidComponents(cs.area.chromaFormat);
628 bool bAllDisabled = true;
629 for (uint32_t compIdx = 0; compIdx < numberOfComponents; compIdx++)
630 {
631 if (m_picSAOEnabled[compIdx])
632 {
633 bAllDisabled = false;
634 }
635 }
636 if (bAllDisabled)
637 {
638 return;
639 }
640
641 const PreCalcValues& pcv = *cs.pcv;
642 PelUnitBuf rec = cs.getRecoBuf();
643 m_tempBuf.copyFrom( rec );
644
645 int ctuRsAddr = 0;
646 for( uint32_t yPos = 0; yPos < pcv.lumaHeight; yPos += pcv.maxCUHeight )
647 {
648 for( uint32_t xPos = 0; xPos < pcv.lumaWidth; xPos += pcv.maxCUWidth )
649 {
650 const uint32_t width = (xPos + pcv.maxCUWidth > pcv.lumaWidth) ? (pcv.lumaWidth - xPos) : pcv.maxCUWidth;
651 const uint32_t height = (yPos + pcv.maxCUHeight > pcv.lumaHeight) ? (pcv.lumaHeight - yPos) : pcv.maxCUHeight;
652 const UnitArea area( cs.area.chromaFormat, Area(xPos , yPos, width, height) );
653
654 offsetCTU( area, m_tempBuf, rec, cs.picture->getSAO()[ctuRsAddr], cs);
655 ctuRsAddr++;
656 }
657 }
658

670void SampleAdaptiveOffset::deriveLoopFilterBoundaryAvailibility(CodingStructure& cs, const Position &pos,
671 bool& isLeftAvail,
672 bool& isRightAvail,
673 bool& isAboveAvail,
674 bool& isBelowAvail,
675 bool& isAboveLeftAvail,
676 bool& isAboveRightAvail,
677 bool& isBelowLeftAvail,
678 bool& isBelowRightAvail
679 ) const
680{
681 const int width = cs.pcv->maxCUWidth;
682 const int height = cs.pcv->maxCUHeight;
683 const CodingUnit* cuCurr = cs.getCU(pos, CH_L);
684 const CodingUnit* cuLeft = cs.getCU(pos.offset(-width, 0), CH_L);
685 const CodingUnit* cuRight = cs.getCU(pos.offset(width, 0), CH_L);
686 const CodingUnit* cuAbove = cs.getCU(pos.offset(0, -height), CH_L);
687 const CodingUnit* cuBelow = cs.getCU(pos.offset(0, height), CH_L);
688 const CodingUnit* cuAboveLeft = cs.getCU(pos.offset(-width, -height), CH_L);
689 const CodingUnit* cuAboveRight = cs.getCU(pos.offset(width, -height), CH_L);
690 const CodingUnit* cuBelowLeft = cs.getCU(pos.offset(-width, height), CH_L);
691 const CodingUnit* cuBelowRight = cs.getCU(pos.offset(width, height), CH_L);
692
693 // check cross slice flags
694 const bool isLoopFilterAcrossSlicePPS = cs.pps->getLoopFilterAcrossSlicesEnabledFlag();
695 if (!isLoopFilterAcrossSlicePPS)
696 {
697 isLeftAvail = (cuLeft == NULL) ? false : CU::isSameSlice(*cuCurr, *cuLeft);
698 isAboveAvail = (cuAbove == NULL) ? false : CU::isSameSlice(*cuCurr, *cuAbove);
699 isRightAvail = (cuRight == NULL) ? false : CU::isSameSlice(*cuCurr, *cuRight);
700 isBelowAvail = (cuBelow == NULL) ? false : CU::isSameSlice(*cuCurr, *cuBelow);
701 isAboveLeftAvail = (cuAboveLeft == NULL) ? false : CU::isSameSlice(*cuCurr, *cuAboveLeft);
702 isAboveRightAvail = (cuAboveRight == NULL) ? false : CU::isSameSlice(*cuCurr, *cuAboveRight);
703 isBelowLeftAvail = (cuBelowLeft == NULL) ? false : CU::isSameSlice(*cuCurr, *cuBelowLeft);
704 isBelowRightAvail = (cuBelowRight == NULL) ? false : CU::isSameSlice(*cuCurr, *cuBelowRight);
705 }
706 else
707 {
708 isLeftAvail = (cuLeft != NULL);
709 isAboveAvail = (cuAbove != NULL);
710 isRightAvail = (cuRight != NULL);
711 isBelowAvail = (cuBelow != NULL);
712 isAboveLeftAvail = (cuAboveLeft != NULL);
713 isAboveRightAvail = (cuAboveRight != NULL);
714 isBelowLeftAvail = (cuBelowLeft != NULL);
715 isBelowRightAvail = (cuBelowRight != NULL);
716 }
717
718 // check cross tile flags
719 const bool isLoopFilterAcrossTilePPS = cs.pps->getLoopFilterAcrossTilesEnabledFlag();
720 if (!isLoopFilterAcrossTilePPS)
721 {
722 isLeftAvail = (!isLeftAvail) ? false : CU::isSameTile(*cuCurr, *cuLeft);
723 isAboveAvail = (!isAboveAvail) ? false : CU::isSameTile(*cuCurr, *cuAbove);
724 isRightAvail = (!isRightAvail) ? false : CU::isSameTile(*cuCurr, *cuRight);
725 isBelowAvail = (!isBelowAvail) ? false : CU::isSameTile(*cuCurr, *cuBelow);
726 isAboveLeftAvail = (!isAboveLeftAvail) ? false : CU::isSameTile(*cuCurr, *cuAboveLeft);
727 isAboveRightAvail = (!isAboveRightAvail) ? false : CU::isSameTile(*cuCurr, *cuAboveRight);
728 isBelowLeftAvail = (!isBelowLeftAvail) ? false : CU::isSameTile(*cuCurr, *cuBelowLeft);
729 isBelowRightAvail = (!isBelowRightAvail) ? false : CU::isSameTile(*cuCurr, *cuBelowRight);
730 }
731
732 // check cross subpic flags
733 const SubPic& curSubPic = cs.pps->getSubPicFromCU(*cuCurr);
734 if (!curSubPic.getloopFilterAcrossEnabledFlag())
735 {
736 isLeftAvail = (!isLeftAvail) ? false : CU::isSameSubPic(*cuCurr, *cuLeft);
737 isAboveAvail = (!isAboveAvail) ? false : CU::isSameSubPic(*cuCurr, *cuAbove);
738 isRightAvail = (!isRightAvail) ? false : CU::isSameSubPic(*cuCurr, *cuRight);
739 isBelowAvail = (!isBelowAvail) ? false : CU::isSameSubPic(*cuCurr, *cuBelow);
740 isAboveLeftAvail = (!isAboveLeftAvail) ? false : CU::isSameSubPic(*cuCurr, *cuAboveLeft);
741 isAboveRightAvail = (!isAboveRightAvail) ? false : CU::isSameSubPic(*cuCurr, *cuAboveRight);
742 isBelowLeftAvail = (!isBelowLeftAvail) ? false : CU::isSameSubPic(*cuCurr, *cuBelowLeft);
743 isBelowRightAvail = (!isBelowRightAvail) ? false : CU::isSameSubPic(*cuCurr, *cuBelowRight);
744 }

748{

763 verVirBndryPos[numVerVirBndry++] = picHeader->getVirtualBoundariesPosX(i);
764 }
765 }
