60public:
61 rbsp_decoder(const uint8_t *d, size_t l, bool e) : data(d), len(l), emulatiom_prevention(e), emitted(0xFFFFFFFF), offs(0){};

74uint8_t rbsp_decoder::get_byte() {
75 if (offs >= len)
76 throw out_of_range("No more RBSP");
77
78 emitted <<= 8;
79 emitted |= data[offs++];
80 if (emulatiom_prevention && (emitted & 0xFFFFFF) == 0x000003) {
81 emitted <<= 8;
82 if (offs >= len)
83 throw out_of_range("No more RBSP");
84
85 emitted |= data[offs++];
86 }
87 return emitted & 0xFF;
88}

90uint8_t *rbsp_decoder::copy_sei(uint8_t *d, size_t l) {
91 for (size_t i = 0; i < l; i++)
92 d[i] = get_byte();
93
94 return d;
95}

97void rbsp_decoder::copy(PacketBuilder *dst) {
98 size_t i = 0;
99 dst->reserve((uint32_t)len - (uint32_t)offs);
100 while (true) {
101 try {
102 dst->data()[i] = get_byte();
103 } catch (const std::out_of_range &) {
104 dst->resize((uint32_t)i -
105 1); // 0x80 is added during encapsulation process but does not belong to the actual data - stop bit
106 return;
107 }
108 i++;
109 }
110}

112// Recognize the NALU marker (used by LCEVC,AVC,HEVC,VVC, but not EVC)
113//
114static inline bool is_nal_marker(const uint8_t *data) { return data[0] == 0 && data[1] == 0 && data[2] == 1; }
115
116// Check for valid LCEVC NAL uint type
117//
118static inline bool is_lcevc_nal_unit_type(const uint8_t b) { return (b >= LCEVC_NON_IDR) && (b <= LCEVC_RSV); }

120// Extract enhancement data and pass to callback
121//
122static void extract_enhancement_nal(const uint8_t *data, size_t size, bool emulation_prevention, uint64_t pts, const bool is_idr,
123 std::function<void(const Packet &pkt, const bool is_lcevc_idr)> callback) {
124
125 auto pss = Packet::build();
126 pss.timestamp(pts);
127
128 if (emulation_prevention) {
129 rbsp_decoder rbsp(data, size, true);
130 rbsp.copy(&pss);
131 } else {
132 pss.contents(data, (unsigned)size);
133 }
134 callback(pss.finish(), is_idr);
135}

137// Find any enhancement data in native LCEVC NAL Unit and pass to callback
138//
139static size_t scan_enhancement_nal(uint8_t *data, size_t data_size, uint64_t pts,
140 std::function<void(const Packet &pkt, const bool is_lcevc_idr)> callback) {
141 // Scan for Enhancement NALU
142 for (unsigned i = 0; i < data_size - 4; ++i) {
143 if (is_nal_marker(data + i) && (data[i + 3] & 0xc0) == 0x40 && data[i + 4] == 0xff) {
144 const unsigned nal_unit_type = (data[i + 3] & 0x3e) >> 1;
145
146 if (is_lcevc_nal_unit_type(nal_unit_type)) {
147 const bool is_idr = (nal_unit_type == NalUnitType::LCEVC_IDR ? true : false);
148
149 // Figure size of NALU
150 unsigned e = (unsigned)data_size;
151 for (unsigned j = i + 4; j < data_size - 3; ++j) {
152 if (is_nal_marker(data + j)) {
153 // Handle [0, 0, 0, 1] case
154 if (data[j - 1] == 0x00)
155 j--;
156 e = j;
157 break;
158 }
159 }
160 extract_enhancement_nal(data + (i + 5), e - (i + 5), true, pts, is_idr, callback);
161
162 // Remove enhancement data from bitstream leaving base codec bitstream
163 memmove(data + i, data + e, data_size - e);
164 data_size = data_size + i - e;
165 }
166 }
167 }
168
169 return data_size;
170}

172// Find any enhancement data in native EVC NAL Unit (u32 length-prefix) and pass to callback
173//
174static size_t scan_enhancement_nal_evc(uint8_t *data, size_t data_size, uint64_t pts,
175 std::function<void(const Packet &pkt, const bool is_lcevc_idr)> callback) {
176 unsigned offset = 0;
177
178 while (offset < data_size - 6) {
179 uint32_t nal_length = 0;
180 memcpy(&nal_length, data + offset, 4);
181 uint32_t total_length = nal_length + 4;
182
183 if ((data[offset + 4] & 0xfe) == 0x3c && (data[offset + 5] & 0x3f) == 0x00) {
184 const unsigned nal_unit_type = (data[offset + 4] & 0xf8) >> 3;
185 const bool is_idr = (nal_unit_type == NalUnitType::LCEVC_IDR ? true : false);
186
187 extract_enhancement_nal(data + offset + 6, nal_length - 2, false, pts, is_idr, callback);
188
189 memmove(data + offset, data + offset + total_length, data_size - (offset + total_length));
190 data_size -= total_length;
191 } else {
192 offset += total_length;
193 }
194 }
195
196 return data_size;
197}

229// Find any enhancement data in registered SEI NALU and pass to callback
230// Used as a template parameter to scan_enhancement_sei_XXX()
231//
232void RegisteredSEI::extract_enhancement_sei(const uint8_t *data, size_t size, bool emulation_prevention, uint64_t pts,
233 bool is_base_idr,
234 std::function<void(const Packet &pkt, const bool is_lcevc_idr)> callback) {
235 const unsigned USER_DATA_REGISTERED = 4;
236 static const uint8_t sei_code[4] = {0xb4, 0x00, 0x50, 0x00};
237
238 rbsp_decoder rbsp(data, size, emulation_prevention);
239
240 size_t sei_type = 0;
241 uint8_t byte;
242
243 do {
244 byte = rbsp.get_byte();
245 sei_type += byte;
246 } while (byte == 0xFF);
247
248 if (sei_type == USER_DATA_REGISTERED) {
249 size_t sei_length = 0;
250 do {
251 byte = rbsp.get_byte();
252 sei_length += byte;
253 } while (byte == 0xFF);
254
255 if (sei_length > size) {
256 WARN("SEI length overflow");
257 } else {
258 uint8_t tmp[sizeof(sei_code)];
259 rbsp.copy_sei(tmp, sizeof(tmp));
260
261 if (memcmp(sei_code, tmp, sizeof(sei_code)) == 0) {
262 int64_t enhancement_len = sei_length - sizeof(sei_code);
263
264 vector<uint8_t> buf(enhancement_len);
265 rbsp.copy_sei(buf.data(), buf.size());
266
267 // Additional NAL Encapsulation (LCEVC NALU Type)
268 scan_enhancement_nal_prod(buf.data(), buf.size(), pts, callback);
269 }
270 }
271 }

274// Find any enhancement data in unregistered SEI NALU and pass to callback
275// Used as a template parameter to scan_enhancement_sei_XXX()
276//
277void UnregisteredSEI::extract_enhancement_sei(const uint8_t *data, size_t size, bool emulation_prevention, uint64_t pts,
278 bool is_base_idr,
279 std::function<void(const Packet &pkt, const bool is_lcevc_idr)> callback) {
280 const unsigned USER_DATA_UNREGISTERED = 5;
281 static const uint8_t uuid[16] = {0xa7, 0xc4, 0x6d, 0xed, 0x49, 0xd8, 0x38, 0xeb,
282 0x9a, 0xad, 0x6d, 0xa6, 0x84, 0x97, 0xa7, 0x54};
283
284 rbsp_decoder rbsp(data, size, emulation_prevention);
285
286 size_t sei_type = 0;
287 uint8_t byte;
288
289 do {
290 byte = rbsp.get_byte();
291 sei_type += byte;
292 } while (byte == 0xFF);
293
294 if (sei_type == USER_DATA_UNREGISTERED) {
295 size_t sei_length = 0;
296 do {
297 byte = rbsp.get_byte();
298 sei_length += byte;
299 } while (byte == 0xFF);
300
301 if (sei_length > size) {
302 WARN("SEI length overflow");
303 } else {
304 uint8_t tmp[sizeof(uuid)];
305 rbsp.copy_sei(tmp, sizeof(tmp));
306
307 if (memcmp(uuid, tmp, sizeof(uuid)) == 0) {
308 int64_t enhancement_len = sei_length - sizeof(uuid);
309
310 auto pss = Packet::build();
311 pss.reserve((unsigned)enhancement_len);
312 pss.timestamp(pts);
313 rbsp.copy_sei(pss.data(), enhancement_len);
314 callback(pss.finish(), is_base_idr);
315 }
316 }
317 }

320// AVC NALUs headers have one byte for type
321template <typename SEI>
322static size_t scan_enhancement_sei_avc(uint8_t *data, size_t data_size, uint64_t pts, bool is_base_idr,
323 std::function<void(const Packet &pkt, const bool is_lcevc_idr)> callback) {
324 // Scan for SEI NALU
325 for (unsigned i = 0; i < data_size - 4; ++i) {
326 if (is_nal_marker(data + i) && data[i + 3] == 0x6) {
327 // Figure size of NALU
328 unsigned e = (unsigned)data_size;
329 for (unsigned j = i + 4; j < data_size - 3; ++j) {
330 if (is_nal_marker(data + j)) {
331 e = j;
332 break;
333 }
334 }
335 SEI::extract_enhancement_sei(data + (i + 4), e - (i + 4), true, pts, is_base_idr, callback);
336 }
337 }
338
339 // SEI can survive base decoder - don't remove
340 return data_size;
341}

411// Scan for enhancement data in buffer - may remove data and returns new size
412//
413size_t scan_enhancement(uint8_t *data, size_t data_size, Encapsulation encapsulation, BaseCoding base_coding, uint64_t pts,
414 bool is_base_idr, std::function<void(const Packet &pkt, const bool is_lcevc_idr)> callback) {
415 switch (encapsulation) {
416 case Encapsulation_SEI_Registered:
417 switch (base_coding) {
418 case BaseCoding_AVC:
419 return scan_enhancement_sei_avc<RegisteredSEI>(data, data_size, pts, is_base_idr, callback);
420 case BaseCoding_HEVC:
421 return scan_enhancement_sei_hevc<RegisteredSEI>(data, data_size, pts, is_base_idr, callback);
422 case BaseCoding_VVC:
423 return scan_enhancement_sei_vvc<RegisteredSEI>(data, data_size, pts, is_base_idr, callback);
424 case BaseCoding_EVC:
425 return scan_enhancement_sei_evc<RegisteredSEI>(data, data_size, pts, is_base_idr, callback);
426 default:
427 CHECK(0);
428 }
429 break;
430 case Encapsulation_SEI_Unregistered:
431 switch (base_coding) {
432 case BaseCoding_AVC:
433 return scan_enhancement_sei_avc<UnregisteredSEI>(data, data_size, pts, is_base_idr, callback);
434 case BaseCoding_HEVC:
435 return scan_enhancement_sei_hevc<UnregisteredSEI>(data, data_size, pts, is_base_idr, callback);
436 case BaseCoding_VVC:
437 return scan_enhancement_sei_vvc<UnregisteredSEI>(data, data_size, pts, is_base_idr, callback);
438 case BaseCoding_EVC:
439 return scan_enhancement_sei_evc<UnregisteredSEI>(data, data_size, pts, is_base_idr, callback);
440 default:
441 CHECK(0);
442 }
443 break;
444 case Encapsulation_NAL:
445 switch (base_coding) {
446 case BaseCoding_AVC:
447 return scan_enhancement_nal(data, data_size, pts, callback);
448 case BaseCoding_HEVC:
449 return scan_enhancement_nal(data, data_size, pts, callback);
450 case BaseCoding_VVC:
451 return scan_enhancement_nal(data, data_size, pts, callback);
452 case BaseCoding_EVC:
453 return scan_enhancement_nal_evc(data, data_size, pts, callback);
454 case BaseCoding_YUV:
455 return scan_enhancement_nal(data, data_size, pts, callback);
456 default:
457 CHECK(0);
458 }
459 break;
460 case Encapsulation_None:
461 return scan_enhancement_nal(data, data_size, pts, callback);
462
463 default:
464 CHECK(0);
465 }
466
467 return 0;
