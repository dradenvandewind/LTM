518template <typename T> inline T Clip3 (const T minVal, const T maxVal, const T a) { return std::min<T> (std::max<T> (minVal, a) , maxVal); }
519template <typename T> inline T ClipBD( const T x, const int bitDepth ) { return Clip3( T( 0 ), T( ( 1 << bitDepth ) - 1 ), x ); }
520template <typename T> inline T ClipPel (const T a, const ClpRng& clpRng) { return std::min<T> (std::max<T> (clpRng.min, a) , clpRng.max); }

522template <typename T> inline void Check3( T minVal, T maxVal, T a)
523{
524 CHECK( ( a > maxVal ) || ( a < minVal ), "ERROR: Range check " << minVal << " >= " << a << " <= " << maxVal << " failed" );
525}

640static inline int floorLog2(uint32_t x)
641{
642 if (x == 0)
643 {
644 // note: ceilLog2() expects -1 as return value
645 return -1;
646 }
647#ifdef __GNUC__
648 return 31 - __builtin_clz(x);
649#else
650#ifdef _MSC_VER
651 unsigned long r = 0;
652 _BitScanReverse(&r, x);
653 return r;
654#else
655 int result = 0;
656 if (x & 0xffff0000)
657 {
658 x >>= 16;
659 result += 16;
660 }
661 if (x & 0xff00)
662 {
663 x >>= 8;
664 result += 8;
665 }
666 if (x & 0xf0)
667 {
668 x >>= 4;
669 result += 4;
670 }
671 if (x & 0xc)
672 {
673 x >>= 2;
674 result += 2;
675 }
676 if (x & 0x2)
677 {
678 x >>= 1;
679 result += 1;
680 }
681 return result;
682#endif
683#endif
684}
