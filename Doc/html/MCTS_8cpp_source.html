81void MCTSHelper::clipMvToArea( Mv& rcMv, const Area& block, const Area& clipArea, const SPS& sps, const int mvFracBits )
82{
83 const int iHorMax = ( clipArea.x + clipArea.width - (int)block.x - (int)block.width ) << mvFracBits;
84 const int iHorMin = ( clipArea.x - (int)block.x ) << mvFracBits;
85
86 const int iVerMax = ( clipArea.y + clipArea.height - (int)block.y - (int)block.height ) << mvFracBits;
87 const int iVerMin = ( clipArea.y - (int)block.y ) << mvFracBits;
88
89 rcMv.setHor( Clip3( iHorMin, iHorMax, rcMv.getHor() ) );
90 rcMv.setVer( Clip3( iVerMin, iVerMax, rcMv.getVer() ) );

93Area MCTSHelper::getTileArea( const CodingStructure* cs, const int ctuAddr )
94{
95 const PPS *pps = cs->pps;
96 const int maxCUWidth = cs->pcv->maxCUWidth;
97 const int maxCUHeight = cs->pcv->maxCUHeight;
98
99 const uint32_t tileIdx = pps->getTileIdx( (uint32_t)ctuAddr );
100 const uint32_t tileX = tileIdx % pps->getNumTileColumns();
101 const uint32_t tileY = tileIdx / pps->getNumTileColumns();
102
103 const int tileWidthtInCtus = pps->getTileColumnWidth( tileX );
104 const int tileHeightInCtus = pps->getTileRowHeight ( tileY );
105 const int tileXPosInCtus = pps->getTileColumnBd( tileX );
106 const int tileYPosInCtus = pps->getTileRowBd( tileY );
107
108 const int tileLeftTopPelPosX = maxCUWidth * tileXPosInCtus;
109 const int tileLeftTopPelPosY = maxCUHeight * tileYPosInCtus;
110 const int tileRightBottomPelPosX = std::min<int>( ( ( tileWidthtInCtus + tileXPosInCtus ) * maxCUWidth ), (int)cs->picture->lwidth() ) - 1;
111 const int tileRightBottomPelPosY = std::min<int>( ( ( tileHeightInCtus + tileYPosInCtus ) * maxCUHeight ), (int)cs->picture->lheight() ) - 1;
112
113 return Area( tileLeftTopPelPosX, tileLeftTopPelPosY, tileRightBottomPelPosX - tileLeftTopPelPosX + 1, tileRightBottomPelPosY - tileLeftTopPelPosY + 1 );
114}

116bool MCTSHelper::isRefBlockAtRestrictedTileBoundary( const PredictionUnit &pu )
117{
118 const Area& tileArea = pu.cs->picture->mctsInfo.getTileArea();
119 const int mvPrecBits = MV_FRACTIONAL_BITS_INTERNAL;
120
121 for( int refList = 0; refList < NUM_REF_PIC_LIST_01; refList++ )
122 {
123 if( pu.refIdx[refList] >= 0 )
124 {
125 const Mv& mv = pu.mv[refList];
126 Area targetBlock( pu.Y().offset( mv.getHor() >> mvPrecBits, mv.getVer() >> mvPrecBits ), pu.Y().size() );
127 // NOTE: at boundary take sub-pel filter samples into account
128 if(
129 targetBlock.x < tileArea.x + 3 ||
130 targetBlock.y < tileArea.y + 3 ||
131 targetBlock.bottomRight().x > tileArea.bottomRight().x - 4 ||
132 targetBlock.bottomRight().y > tileArea.bottomRight().y - 4 )
133 {
134 return true;
135 }
136 }
137 }
138 return false;
139}

141static void getMotInfoBlockPartPos( const PredictionUnit& pu, int xOff, int yOff, const Mv& mv, int& ruiPredXLeft, int& ruiPredYTop, int& ruiPredXRight, int& ruiPredYBottom )
142{
143 const int mvFracBits = MV_FRACTIONAL_BITS_INTERNAL;
144
145 ruiPredXLeft = ( mv.getHor() >> mvFracBits ) + pu.lx() + xOff;
146 ruiPredYTop = ( mv.getVer() >> mvFracBits ) + pu.ly() + yOff;
147 ruiPredXRight = ruiPredXLeft + ( 1 << g_miScaling.posx ) - 1;
148 ruiPredYBottom = ruiPredYTop + ( 1 << g_miScaling.posy ) - 1;
149}

151static bool checkMVRange( const Mv& mv, const Area& tileArea, int predXLeft, int predXRight, int predYTop, int predYBottom, ChromaFormat chromaFormat, bool isLuma, bool msgFlag )
152{
153 // Filter length of sub-sample generation filter to be considered
154 int sampleOffsetLT, sampleOffsetRB, fracDiv;
155
156 if( isLuma )
157 {
158 sampleOffsetLT = 3;
159 sampleOffsetRB = 4;
160 fracDiv = 1 << MV_FRACTIONAL_BITS_INTERNAL;
161 }
162 else
163 {
164 sampleOffsetLT = 2; //1; // 2 in luma coordinates!
165 sampleOffsetRB = 3; //2; // 3 in luma coordinates!
166 fracDiv = 1 << ( MV_FRACTIONAL_BITS_INTERNAL + 1 );
167 }
168
169 const int leftTopPelPosX = tileArea.topLeft().x;
170 const int leftTopPelPosY = tileArea.topLeft().y;
171 const int rightBottomPelPosX = tileArea.bottomRight().x;
172 const int rightBottomPelPosY = tileArea.bottomRight().y;
173
174 const bool isFullPelHor = ( mv.getHor() % fracDiv == 0 );
175 const bool isFullPelVer = ( mv.getVer() % fracDiv == 0 );
176
177 const int rangeXLeft = leftTopPelPosX + ( isFullPelHor ? 0 : sampleOffsetLT );
178 const int rangeYTop = leftTopPelPosY + ( isFullPelVer ? 0 : sampleOffsetLT );
179 const int rangeXRight = rightBottomPelPosX - ( isFullPelHor ? 0 : sampleOffsetRB );
180 const int rangeYBottom = rightBottomPelPosY - ( isFullPelVer ? 0 : sampleOffsetRB );
181
182 if( !( predXLeft >= rangeXLeft && predXLeft <= rangeXRight ) || !( predXRight >= rangeXLeft && predXRight <= rangeXRight ) )
183 {
184 if( msgFlag )
185 {
186 msg( WARNING, "%s: pu motion vector across tile boundaries MV(%d,%d) RangeLR(%d,%d) PredLR(%d,%d)\n", (isLuma ? "LUMA": "CHROMA"), mv.getHor(), mv.getVer(), rangeXLeft, rangeXRight, predXLeft, predXRight );
187 }
188 return false;
189 }
190 else if( !( predYTop >= rangeYTop && predYTop <= rangeYBottom ) || !( predYBottom >= rangeYTop && predYBottom <= rangeYBottom ) )
191 {
192 if( msgFlag )
193 {
194 msg( WARNING, "%s: pu motion vector across tile boundaries MV(%d,%d) RangeTB(%d,%d) PredTB(%d,%d)\n", (isLuma ? "LUMA" : "CHROMA"), mv.getHor(), mv.getVer(), rangeYTop, rangeYBottom, predYTop, predYBottom );
195 }
196 return false;
197 }
198
199 return true;
200}

202bool MCTSHelper::checkMvBufferForMCTSConstraint( const PredictionUnit &pu, bool msgFlag )
203{
204 const Area& tileArea = pu.cs->picture->mctsInfo.getTileArea();
205 const ChromaFormat chromaFormat = pu.cs->sps->getChromaFormatIdc();
206
207 CMotionBuf mb = pu.getMotionBuf();
208
209 const int blkW = 1 << ( g_miScaling.posx );
210 const int blkH = 1 << ( g_miScaling.posy );
211
212 for( int y = 0; y < mb.height; y++ )
213 {
214 for( int x = 0; x < mb.width; x++ )
215 {
216 const MotionInfo &mi = mb.at( x, y );
217 int xOff = x << g_miScaling.posx;
218 int yOff = y << g_miScaling.posy;
219
220 for( int refList = 0; refList < NUM_REF_PIC_LIST_01; refList++ )
221 {
222 if( pu.refIdx[refList] >= 0 || mi.refIdx[refList] >= 0 )
223 {
224 const Mv& mv = mi.mv[refList];
225 int predXLeft, predYTop, predXRight, predYBottom;
226 getMotInfoBlockPartPos( pu, xOff, yOff, mv, predXLeft, predYTop, predXRight, predYBottom );
227 if( !checkMVRange( mv, tileArea, predXLeft, predXRight, predYTop, predYBottom, chromaFormat, true, msgFlag ) )
228 {
229 return false;
230 }
231
232 // Chroma
233 if( !pu.cu->affine )
234 {
235 if( !checkMVRange( mv, tileArea, predXLeft, predXRight, predYTop, predYBottom, chromaFormat, false, msgFlag ) )
236 {
237 return false;
238 }
239 }
240 else
241 {
242 if( x % 2 == 0 && y % 2 == 0 )
243 {
244 const MotionInfo &miA = mi;
245 const MotionInfo &miB = mb.at( x + 1, y + 1 );
246 Mv mvAff = miA.mv[refList] + miB.mv[refList];
247 roundAffineMv( mvAff.hor, mvAff.ver, 1 );
248 getMotInfoBlockPartPos( pu, xOff, yOff, mvAff, predXLeft, predYTop, predXRight, predYBottom );
249 if( !checkMVRange( mvAff, tileArea, predXLeft, predXRight + blkW, predYTop, predYBottom + blkH, chromaFormat, false, msgFlag ) )
250 {
251 return false;
252 }
253 }
254 }
255 }
256 }
257 }
258 }
259
260 return true;

262bool MCTSHelper::checkMvIsNotInRestrictedArea( const PredictionUnit &pu, const Mv& mv, const Area& restrArea, const MvPrecision mvPrec )
263{
264 CHECKD( mvPrec < MV_PRECISION_INT, "Wrong MV precision!" );
265 Mv testMv = mv;
266 testMv >>= mvPrec - MV_PRECISION_INT;
267 Area targetArea = pu.Y();
268 targetArea.repositionTo( targetArea.offset( testMv.getHor(), testMv.getVer() ) );
269 if( !restrArea.contains( targetArea ) )
270 {
271 // Skip this pos
272 return false;
273 }
274 return true;
275}
276bool MCTSHelper::checkMvForMCTSConstraint( const PredictionUnit &pu, const Mv& mv, const MvPrecision mvPrec )

278 return checkMvIsNotInRestrictedArea( pu, mv, pu.cs->picture->mctsInfo.getTileAreaSubPelRestricted( pu ), mvPrec );
