74struct MvField
75{
76 Mv mv;
77 int16_t refIdx;
78
79 MvField() : refIdx( NOT_VALID ) {}
80 MvField( Mv const & cMv, const int iRefIdx ) : mv( cMv ), refIdx( iRefIdx ) {}
81
82 void setMvField( Mv const & cMv, const int iRefIdx )
83 {
84 CHECK( iRefIdx == -1 && cMv != Mv(0,0), "Must not happen." );
85 mv = cMv;
86 refIdx = iRefIdx;
87 }
88
89 bool operator==( const MvField& other ) const
90 {
91 CHECK( refIdx == -1 && mv != Mv(0,0), "Error in operator== of MvField." );
92 CHECK( other.refIdx == -1 && other.mv != Mv(0,0), "Error in operator== of MvField." );
93 return refIdx == other.refIdx && mv == other.mv;
94 }
95 bool operator!=( const MvField& other ) const
96 {
97 CHECK( refIdx == -1 && mv != Mv(0,0), "Error in operator!= of MvField." );
98 CHECK( other.refIdx == -1 && other.mv != Mv(0,0), "Error in operator!= of MvField." );
99 return refIdx != other.refIdx || mv != other.mv;
100 }
101};

103struct MotionInfo
104{
105 bool isInter;
106 bool isIBCmot;
107 char interDir;
108 bool useAltHpelIf;
109 uint16_t sliceIdx;
110 Mv mv [ NUM_REF_PIC_LIST_01 ];
111 int16_t refIdx [ NUM_REF_PIC_LIST_01 ];
112 uint8_t BcwIdx;
113 Mv bv;
114 MotionInfo() : isInter(false), isIBCmot(false), interDir(0), useAltHpelIf(false), sliceIdx(0), refIdx{ NOT_VALID, NOT_VALID }, BcwIdx(0) { }
115 // ensure that MotionInfo(0) produces '\x000....' bit pattern - needed to work with AreaBuf - don't use this constructor for anything else
116 MotionInfo(int i) : isInter(i != 0), isIBCmot(false), interDir(0), useAltHpelIf(false), sliceIdx(0), refIdx{ 0, 0 }, BcwIdx(0) { CHECKD(i != 0, "The argument for this constructor has to be '0'"); }
117
118 bool operator==( const MotionInfo& mi ) const
119 {
120 if( isInter != mi.isInter ) return false;
121 if (isIBCmot != mi.isIBCmot) return false;
122 if( isInter )
123 {
124 if( sliceIdx != mi.sliceIdx ) return false;
125 if( interDir != mi.interDir ) return false;
126
127 if( interDir != 2 )
128 {
129 if( refIdx[0] != mi.refIdx[0] ) return false;
130 if( mv[0] != mi.mv[0] ) return false;
131 }
132
133 if( interDir != 1 )
134 {
135 if( refIdx[1] != mi.refIdx[1] ) return false;
136 if( mv[1] != mi.mv[1] ) return false;
137 }
138 }
139
140 return true;
141 }
142
143 bool operator!=( const MotionInfo& mi ) const
144 {
145 return !( *this == mi );
146 }
147};

162 void reset()
163 {
164 Mv* pMv = &(m_mv[0][0]);
165 for (int ui = 0; ui < 1 * 2 * 33; ++ui, ++pMv)

167 pMv->set(std::numeric_limits<int16_t>::max(), std::numeric_limits<int16_t>::max());
168 }
169
170 Mv* pAffineMv = &(m_mvAffine[0][0][0][0]);

172 {
173 pAffineMv->set(0, 0);
174 }
175
176 memset(m_readOnly, false, 2 * 33 * sizeof(bool));
177 memset(m_dist, -1, 2 * 33 * sizeof(Distortion));
178 memset(m_readOnlyAffine, false, 2 * 2 * 33 * sizeof(bool));

180 memset( m_mvpIdx, 0, 2 * 2 * 33 * sizeof( int ) );
181 }
