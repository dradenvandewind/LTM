107template <typename T> ParametersBuilder &ParametersBuilder::set(const std::string &name, const T &value) {
108 Parameters::ParametersMap::iterator i = parameters_.find(name);
109 if (i == parameters_.end())
110 parameters_.insert(Parameters::ParametersMap::value_type(name, new Parameter<T>(name, value)));
111 else {
112 i->second = std::shared_ptr<ParameterBase>(new Parameter<T>(name, value));
113 }
114 return *this;
115}

193 // Get a string value as a vector
194 // Expects 'std::istream & operator >> (std::istream &in, T &v)' to exist via extract<T>()
195
196 template <typename T> T *get_vector(T *dest, unsigned size) const {
197 if (!base_)
198 return dest;
199
200 const Parameter<std::string> *p = dynamic_cast<const Parameter<std::string> *>(base_);
201 if (p) {
202 std::istringstream ss(p->value_);
203 T r;
204 for (unsigned i = 0; i < size; i++) {
205 ss >> r;
206 dest[i] = r;
207 }
208 return dest;
209 } else {
210 ERR("Parameter type mismatch: %s", name_.c_str());
211 return dest;
212 }
213 }
