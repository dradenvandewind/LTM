90// Does this layer has user_data embedded?
91//
92bool Decoder::is_user_data_layer(unsigned loq, unsigned layer) const {
93 if (loq == LOQ_LEVEL_1 && configuration_.global_configuration.user_data_enabled != UserData_None) {
94 if (configuration_.global_configuration.transform_block_size == 4)
95 return (layer == 5);
96 else
97 return (layer == 1);
98 }
99 return false;
100}

102// Derive and return temporal mask
103//
104Surface Decoder::get_temporal_mask(Surface temporal_symbols) {
105 if (!configuration_.global_configuration.temporal_enabled)
106 return Surface();
107 else if (configuration_.picture_configuration.temporal_signalling_present)
108 return temporal_symbols;
109 else if (configuration_.picture_configuration.temporal_refresh)
110 return Surface::build_from<uint8_t>()
111 .fill(TEMPORAL_INTR, configuration_.global_configuration.resolution_width / transform_block_size(),
112 configuration_.global_configuration.resolution_height / transform_block_size())
113 .finish();
114 else
115 return Surface::build_from<uint8_t>()
116 .fill(TEMPORAL_PRED, configuration_.global_configuration.resolution_width / transform_block_size(),
117 configuration_.global_configuration.resolution_height / transform_block_size())
118 .finish();
119}

121// Decode residuals of an enhancement sub-layer
122//
123Surface Decoder::decode_residuals(unsigned plane, unsigned loq, Surface &temporal_mask, Surface symbols[MAX_NUM_LAYERS]) {
124#if defined __OPT_INPLACE__
125#else
126 Surface coefficients[MAX_NUM_LAYERS];
127#endif
128 int32_t dirq_step_width[MAX_NUM_LAYERS][2];
129 int32_t invq_step_width[MAX_NUM_LAYERS][2];
130 int32_t invq_offset[MAX_NUM_LAYERS][2];
131 int32_t invq_deadzone[MAX_NUM_LAYERS][2];
132 int32_t invq_applied_offset[MAX_NUM_LAYERS][2];
133 const bool horizontal_only = (configuration_.global_configuration.scaling_mode[loq] == ScalingMode_1D ? true : false);
134 unsigned passes = 1;
135
136 // Get temporal mask
137 if (loq == LOQ_LEVEL_2 && configuration_.global_configuration.temporal_enabled) {
138 temporal_mask = get_temporal_mask(symbols[num_residual_layers()]);
139 }
140
141 // Apply temporal step width modifier, if required
142 {
143 int16_t step_width[2];
144 int16_t calculated_step_width = configuration_.picture_configuration.step_width_loq[loq];
145
146 if (loq == LOQ_LEVEL_2 && plane > 0) {
147 calculated_step_width =
148 clamp((int16_t)((calculated_step_width * configuration_.global_configuration.chroma_step_width_multiplier) >> 6),
149 (int16_t)MIN_STEP_WIDTH, (int16_t)MAX_STEP_WIDTH);
150 }
151
152 if (loq == LOQ_LEVEL_2 && configuration_.global_configuration.temporal_enabled &&
153 !configuration_.picture_configuration.temporal_refresh) {
154 step_width[0] =
155 clamp((int16_t)(calculated_step_width *
156 (1 - clamp(((float)configuration_.global_configuration.temporal_step_width_modifier / (float)255),
157 (float)0, (float)0.5))),
158 (int16_t)MIN_STEP_WIDTH, (int16_t)MAX_STEP_WIDTH);
159 step_width[1] = calculated_step_width;
160 passes = 2;
161 } else {
162 step_width[0] = calculated_step_width;
163 }
164
165 for (unsigned pass = 0; pass < passes; pass++) {
166 for (unsigned layer = 0; layer < num_residual_layers(); ++layer) {
167 dirq_step_width[layer][pass] = find_dirq_step_width(step_width[pass], quant_matrix_coeffs_[plane][loq][layer]);
168 invq_offset[layer][pass] =
169 find_invq_offset(configuration_.picture_configuration, step_width[pass], dirq_step_width[layer][pass]);
170 invq_step_width[layer][pass] = find_invq_step_width(configuration_.picture_configuration,
171 dirq_step_width[layer][pass], invq_offset[layer][pass]);
172 invq_deadzone[layer][pass] = find_layer_deadzone(step_width[pass], invq_step_width[layer][pass]);
173 invq_applied_offset[layer][pass] = find_invq_applied_offset(configuration_.picture_configuration,
174 invq_offset[layer][pass], invq_deadzone[layer][pass]);
175 }
176 }
177 }
178
179 for (unsigned layer = 0; layer < num_residual_layers(); ++layer) {
180#if defined __OPT_INPLACE__
181 auto view = symbols[layer].view_as<int16_t>();
182 // Extract UserData if it is embedded in the coefficients
183 if (is_user_data_layer(loq, layer)) {
184 unsigned size;
185 switch (configuration_.global_configuration.user_data_enabled) {
186 case lctm::UserData_2bits:
187 size = 2;
188 break;
189 case lctm::UserData_6bits:
190 size = 6;
191 break;
192 default:
193 CHECK(0);
194 break;
195 }
196
197 {
198#if USER_DATA_EXTRACTION
199 FILE *file = fopen("userdata_dec.bin", "ab");
200#endif
201 int16_t *__restrict pview = (int16_t *)view.data(0, 0);
202 for (unsigned y = 0; y < view.height() * view.width(); ++y) {
203 uint16_t value = *pview;
204#if USER_DATA_EXTRACTION
205 unsigned char data = value & (size == 6 ? 0x3f : 0x03);
206 fwrite(&data, 1, 1, file);
207#endif
208 value >>= size;
209 bool sign = (value & 0x01) == 0x00 ? false : true;
210 value >>= 1;
211 *pview++ = (int16_t)(sign ? (-value) : (value));
212 }
213#if USER_DATA_EXTRACTION
214 fclose(file);
215#endif
216 }
217 }
218#else
219 Surface syms;
220 // Extract UserData if it is embedded in the coefficients
221 if (is_user_data_layer(loq, layer)) {
222 syms = UserDataClear().process(symbols[layer], configuration_.global_configuration.user_data_enabled);
223 } else {
224 syms = symbols[layer];
225 }
226#endif
227
228 // Dequantize
229 if (passes == 1) {
230#if defined __OPT_INPLACE__
231 {
232 int16_t *__restrict pview = (int16_t *)view.data(0, 0);
233 for (unsigned y = 0; y < view.height() * view.width(); ++y) {
234 int16_t coef = *pview;
235 int32_t res = (coef * invq_step_width[layer][0]) +
236 (coef > 0 ? invq_applied_offset[layer][0] : (coef < 0 ? -invq_applied_offset[layer][0] : 0));
237 *pview++ = clamp_int16(res);
238 }
239 }
240#else
241 coefficients[layer] = InverseQuantize().process(syms, invq_step_width[layer][0], invq_applied_offset[layer][0]);
242#endif
243 } else {
244#if defined __OPT_INPLACE__
245 {
246 auto mask = temporal_mask.view_as<uint8_t>();
247 int16_t *__restrict pview = (int16_t *)view.data(0, 0);
248 for (unsigned y = 0; y < view.height(); ++y) {
249 for (unsigned x = 0; x < view.width(); ++x) {
250 unsigned sw_index = (mask.read(x, y) == TEMPORAL_PRED) ? 0 : 1;
251 int16_t coef = *pview;
252
253 *pview++ = clamp_int16(coef * invq_step_width[layer][sw_index] +
254 (coef > 0 ? invq_applied_offset[layer][sw_index]
255 : (coef < 0 ? -invq_applied_offset[layer][sw_index] : 0)));
256 }
257 }
258 }
259#else
260 coefficients[layer] = InverseQuantize_SWM().process(syms, transform_block_size(), invq_step_width[layer],
261 invq_applied_offset[layer], temporal_mask);
262#endif
263 }
264 }
265
266 Surface residuals;
267
268#if defined __OPT_INPLACE__
269 if (!horizontal_only) {
270 // Inverse Transform Horizontal and Vertical
271 //
272 if (configuration_.global_configuration.transform_block_size == 4)
273 residuals =
274 InverseTransformDDS().process(dimensions_.plane_width(plane, loq), dimensions_.plane_height(plane, loq), symbols);
275 else if (configuration_.global_configuration.transform_block_size == 2)
276 residuals =
277 InverseTransformDD().process(dimensions_.plane_width(plane, loq), dimensions_.plane_height(plane, loq), symbols);
278 } else {
279 // Inverse Transform Horizontal only
280 //
281 if (configuration_.global_configuration.transform_block_size == 4)
282 residuals = InverseTransformDDS_1D().process(dimensions_.plane_width(plane, loq), dimensions_.plane_height(plane, loq),
283 symbols);
284 else if (configuration_.global_configuration.transform_block_size == 2)
285 residuals =
286 InverseTransformDD_1D().process(dimensions_.plane_width(plane, loq), dimensions_.plane_height(plane, loq), symbols);
287 }
288#else
289 if (!horizontal_only) {
290 // Inverse Transform Horizontal and Vertical
291 //
292 if (configuration_.global_configuration.transform_block_size == 4)
293 residuals = InverseTransformDDS().process(dimensions_.plane_width(plane, loq), dimensions_.plane_height(plane, loq),
294 coefficients);
295 else if (configuration_.global_configuration.transform_block_size == 2)
296 residuals = InverseTransformDD().process(dimensions_.plane_width(plane, loq), dimensions_.plane_height(plane, loq),
297 coefficients);
298 } else {
299 // Inverse Transform Horizontal only
300 //
301 if (configuration_.global_configuration.transform_block_size == 4)
302 residuals = InverseTransformDDS_1D().process(dimensions_.plane_width(plane, loq), dimensions_.plane_height(plane, loq),
303 coefficients);
304 else if (configuration_.global_configuration.transform_block_size == 2)
305 residuals = InverseTransformDD_1D().process(dimensions_.plane_width(plane, loq), dimensions_.plane_height(plane, loq),
306 coefficients);
307 }
308#endif
309
310 return residuals;
311}

313void Decoder::initialize_decode(const Packet &enhancement_data, Surface (&symbols)[MAX_NUM_PLANES][MAX_NUM_LOQS][MAX_NUM_LAYERS]) {
314 // Parse the bitstream -- XXX check for seeing blocks in correct order
315 // Deserializer will popluate configuration_ and symbols during parsing
316
317 // Initialize quantization matrix
318 std::fill_n(&quant_matrix_coeffs_[0][0][0], MAX_NUM_PLANES * MAX_NUM_LOQS * MAX_NUM_LAYERS, -1);
319
320 Deserializer deserializer(enhancement_data, configuration_, symbols);
321
322 while (deserializer.has_more()) {
323 const unsigned block = deserializer.parse_block();
324
325 // Has there been a new global config?
326 if (block == SyntaxBlock_Global) {
327 // Resize surfaces
328 dimensions_.set(configuration_, configuration_.global_configuration.resolution_width,
329 configuration_.global_configuration.resolution_height);
330
331 // Copy coordinates and sizes into layer surface info
332 for (unsigned plane = 0; plane < configuration_.global_configuration.num_image_planes; ++plane) {
333 for (unsigned loq = 0; loq < 2; ++loq) {
334 for (unsigned layer = 0; layer < num_residual_layers(); ++layer) {
335 configuration_.surface_configuration[plane][loq][layer].width = dimensions_.layer_width(plane, loq);
336 configuration_.surface_configuration[plane][loq][layer].height = dimensions_.layer_height(plane, loq);
337 }
338 }
339 // Is there a temporal layer?
340 if (configuration_.global_configuration.temporal_enabled &&
341 plane < configuration_.global_configuration.num_processed_planes) {
342
343 configuration_.surface_configuration[plane][LOQ_LEVEL_2][num_residual_layers()].width =
344 dimensions_.layer_width(plane, LOQ_LEVEL_2);
345 configuration_.surface_configuration[plane][LOQ_LEVEL_2][num_residual_layers()].height =
346 dimensions_.layer_height(plane, LOQ_LEVEL_2);
347
348 if (temporal_buffer_[plane].empty() ||
349 dimensions_.plane_width(plane, LOQ_LEVEL_2) != temporal_buffer_[plane].width() ||
350 dimensions_.plane_height(plane, LOQ_LEVEL_2) != temporal_buffer_[plane].height())
351 temporal_buffer_[plane] =
352 Surface::build_from<int16_t>()
353 .reserve(dimensions_.plane_width(plane, LOQ_LEVEL_2), dimensions_.plane_height(plane, LOQ_LEVEL_2))
354 .finish();
355 }
356 }
357 }
358 }

361Image Decoder::decode(const Image &ext_base, Surface (&symbols)[MAX_NUM_PLANES][MAX_NUM_LOQS][MAX_NUM_LAYERS],
362 const Image &src_image, bool report, bool dithering_switch, bool dithering_fixed, bool apply_enhancement) {
363
364 CHECK(configuration_.global_configuration.transform_block_size == 4 ||
365 configuration_.global_configuration.transform_block_size == 2);
366
367 if (dithering_.getInitialised() == false) {
368 INFO("Dither init %4d bitdepth %d", configuration_.picture_configuration.dithering_strength,
369 configuration_.global_configuration.enhancement_depth);
370 dithering_.make_buffer(configuration_.picture_configuration.dithering_strength,
371 configuration_.global_configuration.enhancement_depth, dithering_fixed);
372 dithering_.setInitialised(true);
373 }
374
375 const bool is_idr = configuration_.picture_configuration.coding_type == CodingType::CodingType_IDR;
376
377 // Verify dimensions of base image
378 CHECK(ext_base.description().width() == dimensions_.base_width());
379 CHECK(ext_base.description().height() == dimensions_.base_height());
380
381 // Process each plane...
382 //
383 Surface base_reco[MAX_NUM_PLANES];
384 Surface full_reco[MAX_NUM_PLANES];
385 Surface outp_reco[MAX_NUM_PLANES];
386
387 for (unsigned plane = 0; plane < ext_base.description().num_planes(); ++plane) {
388 const bool enhancement_enabled = configuration_.picture_configuration.enhancement_enabled &&
389 plane < configuration_.global_configuration.num_processed_planes;
390
391 // Convert between base and enhancement bit depth
392 Surface base_plane;
393 unsigned base_bit_depth = configuration_.global_configuration.base_depth;
394 if (configuration_.global_configuration.enhancement_depth > configuration_.global_configuration.base_depth &&
395 configuration_.global_configuration.level1_depth_flag) {
396 base_plane = ConvertBitShift().process(ext_base.plane(plane), configuration_.global_configuration.base_depth,
397 configuration_.global_configuration.enhancement_depth);
398 base_bit_depth = configuration_.global_configuration.enhancement_depth;
399 }
400
401 // Base + Correction
402 base_plane = ConvertToInternal().process(
403 base_bit_depth == configuration_.global_configuration.base_depth ? ext_base.plane(plane) : base_plane, base_bit_depth);
404
406 //
407 Surface base_upsampled;
408 switch (configuration_.global_configuration.scaling_mode[LOQ_LEVEL_1]) {
409 case ScalingMode_1D:
410 base_upsampled = Upsampling_1D().process(base_plane, configuration_.global_configuration.upsample,
411 configuration_.global_configuration.upsampling_coefficients);
412 if (configuration_.global_configuration.predicted_residual_enabled) {
413 base_upsampled = PredictedResidualAdjust_1D().process(base_plane, base_upsampled,
414 PredictedResidualSum_1D().process(base_upsampled));
415 }
416 base_plane.dump(format("dec_base_deco_P%1d", plane));
417 break;
418 case ScalingMode_2D:
419 base_upsampled = Upsampling().process(base_plane, configuration_.global_configuration.upsample,
420 configuration_.global_configuration.upsampling_coefficients);
421 if (configuration_.global_configuration.predicted_residual_enabled) {
422 base_upsampled =
423 PredictedResidualAdjust().process(base_plane, base_upsampled, PredictedResidualSum().process(base_upsampled));
424 }
425 base_plane.dump(format("dec_base_deco_P%1d", plane));
426 break;
427 case ScalingMode_None:
428 base_upsampled = base_plane;
429 break;
430 default:
431 CHECK(0);
432 }
433
434 base_upsampled.dump(format("dec_base_pred_P%1d", plane));
435
436 // Work out quantization matrix for each LoQ
437 for (unsigned loq = 0; loq < MAX_NUM_LOQS; ++loq) {
438 const bool horizontal_only = (configuration_.global_configuration.scaling_mode[loq] == ScalingMode_1D ? true : false);
439 for (unsigned layer = 0; layer < num_residual_layers(); ++layer) {
440 quant_matrix_coeffs_[plane][loq][layer] =
441 find_quant_matrix_coeff(configuration_.picture_configuration, num_residual_layers(), horizontal_only, loq,
442 layer, is_idr, quant_matrix_coeffs_[plane][loq][layer]);
443 }
444 }
445
447 //
448 if (enhancement_enabled && apply_enhancement) {
449 // Base residuals
450 Surface unused_mask;
451 Surface residuals = decode_residuals(plane, LOQ_LEVEL_1, unused_mask, symbols[plane][LOQ_LEVEL_1]);
452
453 // Deblocking
454 if (configuration_.picture_configuration.level_1_filtering_enabled &&
455 configuration_.global_configuration.transform_block_size == 4) {
456 residuals = Deblocking().process(residuals, configuration_.global_configuration.level_1_filtering_first_coefficient,
457 configuration_.global_configuration.level_1_filtering_second_coefficient);
458 }
459
460 // Add to (upsampled) base
461 residuals.dump(format("dec_base_resi_reco_P%1d", plane));
462#if defined __OPT_INPLACE__
463 {
464 auto viewa = base_upsampled.view_as<int16_t>();
465 auto viewb = residuals.view_as<int16_t>();
466 const int16_t *__restrict psrcb = viewb.data(0, 0);
467 int16_t *__restrict pdst = (int16_t *)viewa.data(0, 0);
468 for (unsigned y = 0; y < base_upsampled.height() * base_upsampled.width(); ++y) {
469 *pdst++ += (*psrcb++);
470 }
471 }
472 base_reco[plane] = base_upsampled;
473#else
474 base_reco[plane] = Add().process(base_upsampled, residuals);
475#endif
476 } else {
477 base_reco[plane] = base_upsampled;
478 }
479 }
480
482 //
483 Surface upsampled_planes[MAX_NUM_PLANES];
484 for (unsigned plane = 0; plane < ext_base.description().num_planes(); ++plane) {
485 switch (configuration_.global_configuration.scaling_mode[LOQ_LEVEL_2]) {
486 case ScalingMode_1D:
487 upsampled_planes[plane] = Upsampling_1D().process(base_reco[plane], configuration_.global_configuration.upsample,
488 configuration_.global_configuration.upsampling_coefficients);
489 if (configuration_.global_configuration.predicted_residual_enabled) {
490 upsampled_planes[plane] = PredictedResidualAdjust_1D().process(
491 base_reco[plane], upsampled_planes[plane], PredictedResidualSum_1D().process(upsampled_planes[plane]));
492 }
493 break;
494 case ScalingMode_2D:
495 upsampled_planes[plane] = Upsampling().process(base_reco[plane], configuration_.global_configuration.upsample,
496 configuration_.global_configuration.upsampling_coefficients);
497 if (configuration_.global_configuration.predicted_residual_enabled) {
498 upsampled_planes[plane] = PredictedResidualAdjust().process(
499 base_reco[plane], upsampled_planes[plane], PredictedResidualSum().process(upsampled_planes[plane]));
500 }
501 break;
502 case ScalingMode_None:
503 upsampled_planes[plane] = base_reco[plane];
504 break;
505 default:
506 CHECK(0);
507 }
508 upsampled_planes[plane].dump(format("dec_full_pred_P%1d", plane));
509 }
510
511 for (unsigned plane = 0; plane < ext_base.description().num_planes(); ++plane) {
512 const bool enhancement_enabled = configuration_.picture_configuration.enhancement_enabled &&
513 plane < configuration_.global_configuration.num_processed_planes;
514
516 //
517 if (enhancement_enabled && apply_enhancement) {
518 // Enhacement residuals
519 Surface temporal_mask;
520 Surface residuals = decode_residuals(plane, LOQ_LEVEL_2, temporal_mask, symbols[plane][LOQ_LEVEL_2]);
521 residuals.dump(format("dec_full_resi_reco_P%1d", plane));
522
523 if (configuration_.global_configuration.temporal_enabled) {
524 // Apply temporal map (intra / pred)
525 CHECK(!temporal_mask.empty());
526 if (temporal_buffer_[plane].empty()) {
527 temporal_buffer_[plane] = Surface::build_from<int16_t>()
528 .generate(upsampled_planes[plane].width(), upsampled_planes[plane].height(),
529 [&](unsigned x, unsigned y) -> int16_t { return 0; })
530 .finish();
531 }
532 temporal_buffer_[plane] =
533 ApplyTemporalMap().process(temporal_buffer_[plane], temporal_mask, transform_block_size());
534#if defined __OPT_INPLACE__
535 {
536 auto viewa = temporal_buffer_[plane].view_as<int16_t>();
537 auto viewb = residuals.view_as<int16_t>();
538 const int16_t *__restrict psrcb = viewb.data(0, 0);
539 int16_t *__restrict pdst = (int16_t *)viewa.data(0, 0);
540 for (unsigned y = 0; y < temporal_buffer_[plane].height() * temporal_buffer_[plane].width(); ++y) {
541 *pdst++ += (*psrcb++);
542 }
543 }
544#else
545 temporal_buffer_[plane] = Add().process(temporal_buffer_[plane], residuals);
546#endif
547 temporal_buffer_[plane].dump(format("dec_full_temp_buff_P%1d", plane));
548 temporal_mask.dump(format("dec_full_temp_mask_P%1d", plane));
549
550#if defined __OPT_INPLACE__
551 {
552 auto viewa = upsampled_planes[plane].view_as<int16_t>();
553 auto viewb = temporal_buffer_[plane].view_as<int16_t>();
554 const int16_t *__restrict psrcb = viewb.data(0, 0);
555 int16_t *__restrict pdst = (int16_t *)viewa.data(0, 0);
556 for (unsigned y = 0; y < upsampled_planes[plane].height() * upsampled_planes[plane].width(); ++y) {
557 *pdst++ += (*psrcb++);
558 }
559 }
560 full_reco[plane] = upsampled_planes[plane];
561#else
562 full_reco[plane] = Add().process(upsampled_planes[plane], temporal_buffer_[plane]);
563#endif
564 } else {
565 // Temporal disabled
566#if defined __OPT_INPLACE__
567 {
568 auto viewa = upsampled_planes[plane].view_as<int16_t>();
569 auto viewb = residuals.view_as<int16_t>();
570 const int16_t *__restrict psrcb = viewb.data(0, 0);
571 int16_t *__restrict pdst = (int16_t *)viewa.data(0, 0);
572 for (unsigned y = 0; y < upsampled_planes[plane].height() * upsampled_planes[plane].width(); ++y) {
573 *pdst++ += (*psrcb++);
574 }
575 }
576 full_reco[plane] = upsampled_planes[plane];
577#else

585 Surface temporal_mask = get_temporal_mask(symbols[plane][LOQ_LEVEL_2][num_residual_layers()]);
586 CHECK(!temporal_mask.empty());

588 temporal_buffer_[plane] = Surface::build_from<int16_t>()

602 auto viewb = temporal_buffer_[plane].view_as<int16_t>();

605 for (unsigned y = 0; y < upsampled_planes[plane].height() * upsampled_planes[plane].width(); ++y) {
606 *pdst++ += (*psrcb++);
607 }
608 }
609 full_reco[plane] = upsampled_planes[plane];
610#else
611 full_reco[plane] = Add().process(upsampled_planes[plane], temporal_buffer_[plane]);
612#endif
613 } else {

615 full_reco[plane] = upsampled_planes[plane];
616 }
617 } else {
618 full_reco[plane] = upsampled_planes[plane];

624 if (dithering_fixed) {
625 // PSNR calculation after dithering if using fixed seed
626 full_reco[plane] = outp_reco[plane];
627 }
628 } else

630 }

632 Surface output[MAX_NUM_PLANES];
633 for (unsigned p = 0; p < ext_base.description().num_planes(); ++p) {
634
635 Surface o;
636 if (configuration_.sequence_configuration.conformance_window) {
637 // Apply conformance windowing
638 const unsigned cw = dimensions_.crop_unit_width(p), ch = dimensions_.crop_unit_height(p);
639 o = Conform().process(outp_reco[p], configuration_.sequence_configuration.conf_win_left_offset * cw,

642 configuration_.sequence_configuration.conf_win_bottom_offset * ch);

647 }

663 out.push_back(full_reco[plane].view_as<int16_t>());

669 output_desc.height(plane), goPsnr);
670 }

683 oImageBuffer.h[plane] = full_reco[plane].height();
684 oImageBuffer.w[plane] = full_reco[plane].width();
685 oImageBuffer.s[plane] = full_reco[plane].width() * 2;
686 oImageBuffer.a[plane] = (void *)out[plane].data();
687 }
688 lcevc_md5_imgb(&oImageBuffer, gaucMd5Digest);
689
690 // clang-format off
691 fprintf(stdout, "[MD5Y %02X%02X%02X%02X%02X%02X%02X%02X%02X%02X%02X%02X%02X%02X%02X%02X] ",
692 gaucMd5Digest[0][0], gaucMd5Digest[0][1], gaucMd5Digest[0][2], gaucMd5Digest[0][3],
693 gaucMd5Digest[0][4], gaucMd5Digest[0][5], gaucMd5Digest[0][6], gaucMd5Digest[0][7],
694 gaucMd5Digest[0][8], gaucMd5Digest[0][9], gaucMd5Digest[0][10], gaucMd5Digest[0][11],
695 gaucMd5Digest[0][12], gaucMd5Digest[0][13], gaucMd5Digest[0][14], gaucMd5Digest[0][15]);
696 if (output_desc.num_planes() > 1) {
697 fprintf(stdout, "[MD5U %02X%02X%02X%02X%02X%02X%02X%02X%02X%02X%02X%02X%02X%02X%02X%02X] ",
698 gaucMd5Digest[1][0], gaucMd5Digest[1][1], gaucMd5Digest[1][2], gaucMd5Digest[1][3],

701 gaucMd5Digest[1][12], gaucMd5Digest[1][13], gaucMd5Digest[1][14], gaucMd5Digest[1][15]);
