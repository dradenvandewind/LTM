64QpParam::QpParam(const int qpy,
65 const ComponentID compID,
66 const int qpBdOffset,
67 const int internalMinusInputBitDepth,
68 const int chromaQPOffset,
69 const ChromaFormat chFmt,
70 const int dqp
71 , const SPS *sps
72 , const bool applyACTQpoffset
73)
74{
75 int baseQp;
76 if (isLuma(compID))
77 {
78 baseQp = qpy + qpBdOffset;
79 }
80 else
81 {
82 int qpi = Clip3(-qpBdOffset, MAX_QP, qpy);
83 baseQp = sps->getMappedChromaQpValue(compID, qpi);
84 baseQp = Clip3(-qpBdOffset, MAX_QP, baseQp + chromaQPOffset) + qpBdOffset;
85 }
86
87 if (applyACTQpoffset)
88 {
89 baseQp += DELTA_QP_ACT[compID];
90 }
91 baseQp = Clip3( 0, MAX_QP+qpBdOffset, baseQp + dqp );
92
93 Qps[0] =baseQp;
94 pers[0]=baseQp/6;
95 rems[0]=baseQp%6;
96
97 int baseQpTS = baseQp;
98 baseQpTS = std::max(baseQpTS, 4 + 6 * internalMinusInputBitDepth);
99
100 Qps[1] = baseQpTS;
101 pers[1] = baseQpTS / 6;
102 rems[1] = baseQpTS % 6;
103}

142void invResDPCM( const TransformUnit &tu, const ComponentID &compID, CoeffBuf &dstBuf )
143{
144 const CompArea &rect = tu.blocks[compID];
145 const int wdt = rect.width;
146 const int hgt = rect.height;
147 const CCoeffBuf coeffs = tu.getCoeffs(compID);
148
149 const int maxLog2TrDynamicRange = tu.cs->sps->getMaxLog2TrDynamicRange(toChannelType(compID));
150 const TCoeff inputMinimum = -(1 << maxLog2TrDynamicRange);
151 const TCoeff inputMaximum = (1 << maxLog2TrDynamicRange) - 1;
152
153 const TCoeff* coef = &coeffs.buf[0];
154 TCoeff* dst = &dstBuf.buf[0];
155
156 if( isLuma(compID) ? tu.cu->bdpcmMode == 1 : tu.cu->bdpcmModeChroma == 1)
157 {
158 for( int y = 0; y < hgt; y++ )
159 {
160 dst[0] = coef[0];
161 for( int x = 1; x < wdt; x++ )
162 {
163 dst[x] = Clip3(inputMinimum, inputMaximum, dst[x - 1] + coef[x]);
164 }
165 coef += coeffs.stride;
166 dst += dstBuf.stride;
167 }
168 }
169 else
170 {
171 for( int x = 0; x < wdt; x++ )
172 {
173 dst[x] = coef[x];
174 }
175 for( int y = 0; y < hgt - 1; y++ )
176 {
177 for( int x = 0; x < wdt; x++ )
178 {
179 dst[dstBuf.stride + x] = Clip3(inputMinimum, inputMaximum, dst[x] + coef[coeffs.stride + x]);
180 }
181 coef += coeffs.stride;
182 dst += dstBuf.stride;
183 }
184 }
185}

187void fwdResDPCM( TransformUnit &tu, const ComponentID &compID )
188{
189 const CompArea &rect = tu.blocks[compID];
190 const int wdt = rect.width;
191 const int hgt = rect.height;
192 CoeffBuf coeffs = tu.getCoeffs(compID);
193
194 TCoeff* coef = &coeffs.buf[0];
195
196 if( isLuma(compID) ? tu.cu->bdpcmMode == 1 : tu.cu->bdpcmModeChroma == 1)
197 {
198 for( int y = 0; y < hgt; y++ )
199 {
200 for( int x = wdt - 1; x > 0; x-- )
201 {
202 coef[x] -= coef[x - 1];
203 }
204 coef += coeffs.stride;
205 }
206 }
207 else
208 {
209 coef += coeffs.stride * (hgt - 1);
210 for( int y = 0; y < hgt - 1; y++ )
211 {
212 for ( int x = 0; x < wdt; x++ )
213 {
214 coef[x] -= coef[x - coeffs.stride];
215 }
216 coef -= coeffs.stride;
217 }
218 }

222void Quant::xSignBitHidingHDQ( TCoeff* pQCoef, const TCoeff* pCoef, TCoeff* deltaU, const CoeffCodingContext& cctx, const int maxLog2TrDynamicRange )
223{
224 const uint32_t width = cctx.width();
225 const uint32_t height = cctx.height();
226 const uint32_t groupSize = 1 << cctx.log2CGSize();
227
228 const TCoeff entropyCodingMinimum = -(1 << maxLog2TrDynamicRange);
229 const TCoeff entropyCodingMaximum = (1 << maxLog2TrDynamicRange) - 1;
230
231 int lastCG = -1;
232 int absSum = 0 ;
233 int n ;
234
235 for( int subSet = (width*height-1) >> cctx.log2CGSize(); subSet >= 0; subSet-- )
236 {
237 int subPos = subSet << cctx.log2CGSize();
238 int firstNZPosInCG=groupSize , lastNZPosInCG=-1 ;
239 absSum = 0 ;
240
241 for(n = groupSize-1; n >= 0; --n )
242 {
243 if( pQCoef[ cctx.blockPos( n + subPos ) ] )
244 {
245 lastNZPosInCG = n;
246 break;
247 }
248 }
249
250 for(n = 0; n <groupSize; n++ )
251 {
252 if( pQCoef[ cctx.blockPos( n + subPos ) ] )
253 {
254 firstNZPosInCG = n;
255 break;
256 }
257 }
258
259 for(n = firstNZPosInCG; n <=lastNZPosInCG; n++ )
260 {
261 absSum += int(pQCoef[ cctx.blockPos( n + subPos ) ]);
262 }
263
264 if(lastNZPosInCG>=0 && lastCG==-1)
265 {
266 lastCG = 1 ;
267 }
268
269 if( lastNZPosInCG-firstNZPosInCG>=SBH_THRESHOLD )
270 {
271 uint32_t signbit = (pQCoef[cctx.blockPos(subPos+firstNZPosInCG)]>0?0:1) ;
272 if( signbit!=(absSum&0x1) ) //compare signbit with sum_parity
273 {
274 TCoeff curCost = std::numeric_limits<TCoeff>::max();
275 TCoeff minCostInc = std::numeric_limits<TCoeff>::max();
276 int minPos =-1, finalChange=0, curChange=0;
277
278 for( n = (lastCG==1?lastNZPosInCG:groupSize-1) ; n >= 0; --n )
279 {
280 uint32_t blkPos = cctx.blockPos( n+subPos );
281 if(pQCoef[ blkPos ] != 0 )
282 {
283 if(deltaU[blkPos]>0)
284 {
285 curCost = - deltaU[blkPos];
286 curChange=1 ;
287 }
288 else
289 {
290 //curChange =-1;
291 if(n==firstNZPosInCG && abs(pQCoef[blkPos])==1)
292 {
293 curCost = std::numeric_limits<TCoeff>::max();
294 }
295 else
296 {
297 curCost = deltaU[blkPos];
298 curChange =-1;
299 }
300 }
301 }
302 else
303 {
304 if(n<firstNZPosInCG)
305 {
306 uint32_t thisSignBit = (pCoef[blkPos]>=0?0:1);
307 if(thisSignBit != signbit )
308 {
309 curCost = std::numeric_limits<TCoeff>::max();
310 }
311 else
312 {
313 curCost = - (deltaU[blkPos]) ;
314 curChange = 1 ;
315 }
316 }
317 else
318 {
319 curCost = - (deltaU[blkPos]) ;
320 curChange = 1 ;
321 }
322 }
323
324 if( curCost<minCostInc)
325 {
326 minCostInc = curCost ;
327 finalChange = curChange ;
328 minPos = blkPos ;
329 }
330 } //CG loop
331
332 if(pQCoef[minPos] == entropyCodingMaximum || pQCoef[minPos] == entropyCodingMinimum)
333 {
334 finalChange = -1;
335 }
336
337 if(pCoef[minPos]>=0)
338 {
339 pQCoef[minPos] += finalChange ;
340 }
341 else
342 {
343 pQCoef[minPos] -= finalChange ;
344 }
345 } // Hide
346 }
347 if(lastCG==1)
348 {
349 lastCG=0 ;
350 }
351 } // TU loop
352
353 return;
354}

356void Quant::dequant(const TransformUnit &tu,
357 CoeffBuf &dstCoeff,
358 const ComponentID &compID,
359 const QpParam &cQP)
360{
361 const SPS *sps = tu.cs->sps;
362 const CompArea &area = tu.blocks[compID];
363 const uint32_t uiWidth = area.width;
364 const uint32_t uiHeight = area.height;
365 TCoeff *const piCoef = dstCoeff.buf;
366 const uint32_t numSamplesInBlock = uiWidth * uiHeight;
367 const int maxLog2TrDynamicRange = sps->getMaxLog2TrDynamicRange(toChannelType(compID));
368 const TCoeff transformMinimum = -(1 << maxLog2TrDynamicRange);
369 const TCoeff transformMaximum = (1 << maxLog2TrDynamicRange) - 1;
370 const bool isTransformSkip = (tu.mtsIdx[compID] == MTS_SKIP);
371
372 const bool disableSMForLFNST = tu.cs->slice->getExplicitScalingListUsed() ? tu.cs->slice->getSPS()->getDisableScalingMatrixForLfnstBlks() : false;
373 const bool isLfnstApplied = tu.cu->lfnstIdx > 0 && (tu.cu->isSepTree() ? true : isLuma(compID));
374 const bool disableSMForACT = tu.cs->slice->getSPS()->getScalingMatrixForAlternativeColourSpaceDisabledFlag() && (tu.cs->slice->getSPS()->getScalingMatrixDesignatedColourSpaceFlag() == tu.cu->colorTransform);
375 const bool enableScalingLists = getUseScalingList(uiWidth, uiHeight, isTransformSkip, isLfnstApplied, disableSMForLFNST, disableSMForACT);
376 const int scalingListType = getScalingListType(tu.cu->predMode, compID);
377 const int channelBitDepth = sps->getBitDepth(toChannelType(compID));
378
379 const TCoeff *coef;
380 if ((tu.cu->bdpcmMode && isLuma(compID)) || ( tu.cu->bdpcmModeChroma && isChroma(compID) ))
381 {
382 invResDPCM( tu, compID, dstCoeff );
383 coef = piCoef;
384 }
385 else
386 {
387 coef = tu.getCoeffs(compID).buf;
388 }
389 const TCoeff *const piQCoef = coef;
390 CHECK(scalingListType >= SCALING_LIST_NUM, "Invalid scaling list");
391 CHECK(uiWidth > m_uiMaxTrSize, "Unsupported transformation size");
392
393 // Represents scaling through forward transform
394 const bool bClipTransformShiftTo0 = tu.mtsIdx[compID] != MTS_SKIP && sps->getSpsRangeExtension().getExtendedPrecisionProcessingFlag();
395 const int originalTransformShift = getTransformShift(channelBitDepth, area.size(), maxLog2TrDynamicRange);
396 const bool needSqrtAdjustment = TU::needsBlockSizeTrafoScale( tu, compID );
397 const int iTransformShift = (bClipTransformShiftTo0 ? std::max<int>(0, originalTransformShift) : originalTransformShift) + (needSqrtAdjustment?-1:0);
398
399 const int QP_per = cQP.per(isTransformSkip);
400 const int QP_rem = cQP.rem(isTransformSkip);
401
402 const int rightShift = (IQUANT_SHIFT - ((isTransformSkip ? 0 : iTransformShift) + QP_per)) + (enableScalingLists ? LOG2_SCALING_LIST_NEUTRAL_VALUE : 0);
403
404 if(enableScalingLists)
405 {
406 //from the dequantization equation:
407 //iCoeffQ = ((Intermediate_Int(clipQCoef) * piDequantCoef[deQuantIdx]) + iAdd ) >> rightShift
408 //(sizeof(Intermediate_Int) * 8) = inputBitDepth + dequantCoefBits - rightShift
409 const uint32_t dequantCoefBits = 1 + IQUANT_SHIFT + SCALING_LIST_BITS;
410 const uint32_t targetInputBitDepth = std::min<uint32_t>((maxLog2TrDynamicRange + 1), (((sizeof(Intermediate_Int) * 8) + rightShift) - dequantCoefBits));
411
412 const Intermediate_Int inputMinimum = -(1 << (targetInputBitDepth - 1));
413 const Intermediate_Int inputMaximum = (1 << (targetInputBitDepth - 1)) - 1;
414
415 const uint32_t uiLog2TrWidth = floorLog2(uiWidth);
416 const uint32_t uiLog2TrHeight = floorLog2(uiHeight);
417 int *piDequantCoef = getDequantCoeff(scalingListType, QP_rem, uiLog2TrWidth, uiLog2TrHeight);
418
419 if(rightShift > 0)
420 {
421 const Intermediate_Int iAdd = (Intermediate_Int) 1 << (rightShift - 1);
422
423 for( int n = 0; n < numSamplesInBlock; n++ )
424 {
425 const TCoeff clipQCoef = TCoeff(Clip3<Intermediate_Int>(inputMinimum, inputMaximum, piQCoef[n]));
426 const Intermediate_Int iCoeffQ = ((Intermediate_Int(clipQCoef) * piDequantCoef[n]) + iAdd ) >> rightShift;
427
428 piCoef[n] = TCoeff(Clip3<Intermediate_Int>(transformMinimum,transformMaximum,iCoeffQ));
429 }
430 }
431 else
432 {
433 const int leftShift = -rightShift;
434
435 for( int n = 0; n < numSamplesInBlock; n++ )
436 {
437 const TCoeff clipQCoef = TCoeff(Clip3<Intermediate_Int>(inputMinimum, inputMaximum, piQCoef[n]));
438 const Intermediate_Int iCoeffQ = (Intermediate_Int(clipQCoef) * piDequantCoef[n]) << leftShift;
439
440 piCoef[n] = TCoeff(Clip3<Intermediate_Int>(transformMinimum,transformMaximum,iCoeffQ));
441 }
442 }
443 }
444 else
445 {
446 const int scale = g_invQuantScales[needSqrtAdjustment?1:0][QP_rem];
447 const int scaleBits = ( IQUANT_SHIFT + 1 );
448
449 //from the dequantisation equation:
450 //iCoeffQ = Intermediate_Int((int64_t(clipQCoef) * scale + iAdd) >> rightShift);
451 //(sizeof(Intermediate_Int) * 8) = inputBitDepth + scaleBits - rightShift
452 const uint32_t targetInputBitDepth = std::min<uint32_t>((maxLog2TrDynamicRange + 1), (((sizeof(Intermediate_Int) * 8) + rightShift) - scaleBits));
453 const Intermediate_Int inputMinimum = -(1 << (targetInputBitDepth - 1));
454 const Intermediate_Int inputMaximum = (1 << (targetInputBitDepth - 1)) - 1;
455
456 if (rightShift > 0)
457 {
458 const Intermediate_Int iAdd = (Intermediate_Int) 1 << (rightShift - 1);
459
460 for( int n = 0; n < numSamplesInBlock; n++ )
461 {
462 const TCoeff clipQCoef = TCoeff(Clip3<Intermediate_Int>(inputMinimum, inputMaximum, piQCoef[n]));
463 const Intermediate_Int iCoeffQ = (Intermediate_Int(clipQCoef) * scale + iAdd) >> rightShift;
464
465 piCoef[n] = TCoeff(Clip3<Intermediate_Int>(transformMinimum,transformMaximum,iCoeffQ));
466 }
467 }
468 else
469 {
470 const int leftShift = -rightShift;
471
472 for( int n = 0; n < numSamplesInBlock; n++ )
473 {
474 const TCoeff clipQCoef = TCoeff(Clip3<Intermediate_Int>(inputMinimum, inputMaximum, piQCoef[n]));
475 const Intermediate_Int iCoeffQ = (Intermediate_Int(clipQCoef) * scale) << leftShift;
476
477 piCoef[n] = TCoeff(Clip3<Intermediate_Int>(transformMinimum,transformMaximum,iCoeffQ));
478 }
479 }
480 }

779void Quant::processScalingListEnc( int *coeff, int *quantcoeff, int quantScales, uint32_t height, uint32_t width, uint32_t ratio, int sizuNum, uint32_t dc)
780{
781 if (height != width)
782 {
783 for (uint32_t j = 0; j<height; j++)
784 {
785 for (uint32_t i = 0; i<width; i++)
786 {
787 if (j >= JVET_C0024_ZERO_OUT_TH || i >= JVET_C0024_ZERO_OUT_TH)
788 {
789 quantcoeff[j*width + i] = 0;
790 continue;
791 }
792 int ratioWH = (height>width) ? height / width : width / height;
793 int ratioH = (height / sizuNum) ? (height / sizuNum) : (sizuNum / height); // 32/8 = 4
794 int ratioW = (width / sizuNum) ? (width / sizuNum) : (sizuNum / width); //16/8 = 2 //sizeNum = 8/4
795 if (height > width)
796 {
797 quantcoeff[j*width + i] = quantScales / coeff[sizuNum * (j / ratioH) + ((i * ratioWH) / ratioH)];
798 }
799 else //ratioH < ratioW
800 {
801 quantcoeff[j*width + i] = quantScales / coeff[sizuNum * ((j * ratioWH) / ratioW) + (i / ratioW)];
802 }
803 int largeOne = (width > height) ? width : height;
804 if (largeOne>8)
805 {
806 quantcoeff[0] = quantScales / dc;
807 }
808 }
809 }
810 return;
811 }
812 for(uint32_t j=0;j<height;j++)
813 {
814 for(uint32_t i=0;i<width;i++)
815 {
816 quantcoeff[j*width + i] = quantScales / coeff[sizuNum * (j / ratio) + i / ratio];
817 }
818 }
819
820 if(ratio > 1)
821 {
822 quantcoeff[0] = quantScales / dc;
823 }
824}

826void Quant::processScalingListDec( const int *coeff, int *dequantcoeff, int invQuantScales, uint32_t height, uint32_t width, uint32_t ratio, int sizeNum, uint32_t dc)
827{

829 {
830 int ratioWH = (height > width ) ? (height / width ) : (width / height);
831 int ratioH = (height / sizeNum) ? (height / sizeNum) : (sizeNum / height);
832 int ratioW = (width / sizeNum) ? (width / sizeNum) : (sizeNum / width );
833 if (height > width)
834 {
835 for (uint32_t j = 0; j < height; j++)
836 {
837 int coeffLineSep = (j / ratioH) * sizeNum;

866 }

869 dequantcoeff[0] = invQuantScales * dc;

871 }
872 for (uint32_t j = 0; j<height; j++)
873 {
874 int coeffLineSep = (j / ratio) * sizeNum;
875 int dequantCoeffLineSep = j * width;
876 for (uint32_t i = 0; i<width; i++)
877 {
878 dequantcoeff[dequantCoeffLineSep + i] = invQuantScales * coeff[coeffLineSep + i / ratio];
879 }
880 }
881
882 if (ratio > 1)
883 {
884 dequantcoeff[0] = invQuantScales * dc;

954void Quant::quant(TransformUnit &tu, const ComponentID &compID, const CCoeffBuf &pSrc, TCoeff &uiAbsSum, const QpParam &cQP, const Ctx& ctx)
955{
956 const SPS &sps = *tu.cs->sps;
957 const CompArea &rect = tu.blocks[compID];
958 const uint32_t uiWidth = rect.width;
959 const uint32_t uiHeight = rect.height;
960 const int channelBitDepth = sps.getBitDepth(toChannelType(compID));
961
962 const CCoeffBuf &piCoef = pSrc;
963 CoeffBuf piQCoef = tu.getCoeffs(compID);
964
965 const bool useTransformSkip = (tu.mtsIdx[compID] == MTS_SKIP);
966 const int maxLog2TrDynamicRange = sps.getMaxLog2TrDynamicRange(toChannelType(compID));
967
968 {
969 CoeffCodingContext cctx(tu, compID, tu.cs->slice->getSignDataHidingEnabledFlag());
970 const TCoeff entropyCodingMinimum = -(1 << maxLog2TrDynamicRange);
971 const TCoeff entropyCodingMaximum = (1 << maxLog2TrDynamicRange) - 1;
972
973 TCoeff deltaU[MAX_TB_SIZEY * MAX_TB_SIZEY];
974 int scalingListType = getScalingListType(tu.cu->predMode, compID);
975 CHECK(scalingListType >= SCALING_LIST_NUM, "Invalid scaling list");
976 const uint32_t uiLog2TrWidth = floorLog2(uiWidth);
977 const uint32_t uiLog2TrHeight = floorLog2(uiHeight);
978 int *piQuantCoeff = getQuantCoeff(scalingListType, cQP.rem(useTransformSkip), uiLog2TrWidth, uiLog2TrHeight);
979 const bool disableSMForLFNST = tu.cs->slice->getExplicitScalingListUsed() ? tu.cs->slice->getSPS()->getDisableScalingMatrixForLfnstBlks() : false;
980 const bool isLfnstApplied = tu.cu->lfnstIdx > 0 && (tu.cu->isSepTree() ? true : isLuma(compID));
981 const bool disableSMForACT = tu.cs->slice->getSPS()->getScalingMatrixForAlternativeColourSpaceDisabledFlag() && (tu.cs->slice->getSPS()->getScalingMatrixDesignatedColourSpaceFlag() == tu.cu->colorTransform);
982 const bool enableScalingLists = getUseScalingList(uiWidth, uiHeight, useTransformSkip, isLfnstApplied, disableSMForLFNST, disableSMForACT);
983
984 // for blocks that where width*height != 4^N, the effective scaling applied during transformation cannot be
985 // compensated by a bit-shift (the quantised result will be sqrt(2) * larger than required).
986 // The quantScale table and shift is used to compensate for this.
987 const bool needSqrtAdjustment= TU::needsBlockSizeTrafoScale( tu, compID );
988 const int defaultQuantisationCoefficient = g_quantScales[needSqrtAdjustment?1:0][cQP.rem(useTransformSkip)];
989 int iTransformShift = getTransformShift(channelBitDepth, rect.size(), maxLog2TrDynamicRange) + ( needSqrtAdjustment?-1:0);
990
991 if (useTransformSkip && sps.getSpsRangeExtension().getExtendedPrecisionProcessingFlag())
992 {
993 iTransformShift = std::max<int>(0, iTransformShift);
994 }
995
996 const int iQBits = QUANT_SHIFT + cQP.per(useTransformSkip) + (useTransformSkip ? 0 : iTransformShift);
997 // QBits will be OK for any internal bit depth as the reduction in transform shift is balanced by an increase in Qp_per due to QpBDOffset
998
999 const int64_t iAdd = int64_t(tu.cs->slice->isIRAP() ? 171 : 85) << int64_t(iQBits - 9);
1000 const int qBits8 = iQBits - 8;
1001
1002 const uint32_t lfnstIdx = tu.cu->lfnstIdx;
1003 const int maxNumberOfCoeffs = lfnstIdx > 0 ? ((( uiWidth == 4 && uiHeight == 4 ) || ( uiWidth == 8 && uiHeight == 8) ) ? 8 : 16) : piQCoef.area();
1004 memset( piQCoef.buf, 0, sizeof(TCoeff) * piQCoef.area() );
1005
1006 const ScanElement* scan = g_scanOrder[SCAN_GROUPED_4x4][SCAN_DIAG][gp_sizeIdxInfo->idxFrom(uiWidth)][gp_sizeIdxInfo->idxFrom(uiHeight)];
1007
1008 for (int uiScanPos = 0; uiScanPos < maxNumberOfCoeffs; uiScanPos++)
1009 {
1010 const int uiBlockPos = scan[uiScanPos].idx;
1011 const TCoeff iLevel = piCoef.buf[uiBlockPos];
1012 const TCoeff iSign = (iLevel < 0 ? -1: 1);
1013
1014 const int64_t tmpLevel = (int64_t)abs(iLevel) * (enableScalingLists ? piQuantCoeff[uiBlockPos] : defaultQuantisationCoefficient);
1015
1016 const TCoeff quantisedMagnitude = TCoeff((tmpLevel + iAdd ) >> iQBits);
1017 deltaU[uiBlockPos] = (TCoeff)((tmpLevel - ((int64_t)quantisedMagnitude<<iQBits) )>> qBits8);
1018
1019 uiAbsSum += quantisedMagnitude;
1020 const TCoeff quantisedCoefficient = quantisedMagnitude * iSign;
1021
1022 piQCoef.buf[uiBlockPos] = Clip3<TCoeff>( entropyCodingMinimum, entropyCodingMaximum, quantisedCoefficient );
1023 } // for n
1024 if ((tu.cu->bdpcmMode && isLuma(compID)) || (tu.cu->bdpcmModeChroma && isChroma(compID)) )
1025 {
1026 fwdResDPCM( tu, compID );
1027 }
1028 if( cctx.signHiding() )
1029 {
1030 if(uiAbsSum >= 2) //this prevents TUs with only one coefficient of value 1 from being tested
1031 {
1032 xSignBitHidingHDQ(piQCoef.buf, piCoef.buf, deltaU, cctx, maxLog2TrDynamicRange);
1033 }
1034 }
1035 } //if RDOQ
1036 //return;
1037}
