54VPS* ParameterSetManager::getVPS( int vpsId )
55{
56 if ( vpsId != 0)
57 {

59 }
60 else
61 {

74// activate a PPS and depending on isIRAP parameter also SPS
75// returns true, if activation is successful
76bool ParameterSetManager::activatePPS(int ppsId, bool isIRAP)
77{
78 PPS *pps = m_ppsMap.getPS(ppsId);
79 if (pps)
80 {
81 int spsId = pps->getSPSId();
82 {
83 SPS *sps = m_spsMap.getPS(spsId);
84
85 if (sps)
86 {
87 int vpsId = sps->getVPSId();
88
89 VPS *vps = m_vpsMap.getPS(vpsId);
90 if(vps)
91 {
92 m_activeVPSId = vpsId;
93 m_vpsMap.setActive(vpsId);
94 }
95 else
96 {
97 msg( WARNING, "Warning: tried to activate a PPS that refers to a non-existing VPS." );
98 }
99
100 m_spsMap.clearActive();
101 m_spsMap.setActive(spsId);
102 m_activeSPSId = spsId;
103 m_ppsMap.clearActive();
104 m_ppsMap.setActive(ppsId);
105 return true;
106 }
107 else
108 {
109 msg( WARNING, "Warning: tried to activate a PPS that refers to a non-existing SPS.");
110 }
111 }
112 }
113 else
114 {
115 msg( WARNING, "Warning: tried to activate a non-existing PPS.");
116 }
117
118 // Failed to activate if reach here.
119 m_activeSPSId=-1;
120 return false;
121}

124{
125 APS *aps = m_apsMap.getPS( ( apsId << NUM_APS_TYPE_LEN ) + apsType );

127 {

130 }
131 else
132 {
133 msg(WARNING, "Warning: tried to activate a non-existing APS.");
134 }

174void calculateParameterSetChangedFlag(bool &changed, const std::vector<uint8_t> *oldData, const std::vector<uint8_t> *newData)
175{
176 if (!changed)
177 {
178 if ((oldData==0 && newData!=0) || (oldData!=0 && newData==0))
179 {
180 changed=true;
181 }
182 else if (oldData!=0 && newData!=0)
183 {
184 // compare the two
185 if (oldData->size() != newData->size())
186 {
187 changed=true;
188 }
189 else
190 {
191 const uint8_t *pNewDataArray=&(*newData)[0];
192 const uint8_t *pOldDataArray=&(*oldData)[0];
193 if (memcmp(pOldDataArray, pNewDataArray, oldData->size()))
194 {
195 changed=true;
196 }
197 }
198 }
199 }
200}
