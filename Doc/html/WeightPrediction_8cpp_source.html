47static inline Pel weightBidir( int w0, Pel P0, int w1, Pel P1, int round, int shift, int offset, const ClpRng& clpRng)
48{
49 return ClipPel( ( (w0*(P0 + IF_INTERNAL_OFFS) + w1*(P1 + IF_INTERNAL_OFFS) + round + (offset << (shift-1))) >> shift ), clpRng );
50}
51
52static inline Pel weightUnidir( int w0, Pel P0, int round, int shift, int offset, const ClpRng& clpRng)
53{
54 return ClipPel( ( (w0*(P0 + IF_INTERNAL_OFFS) + round) >> shift ) + offset, clpRng );
55}
56
57static inline Pel noWeightUnidir( Pel P0, int round, int shift, int offset, const ClpRng& clpRng)
58{
59 return ClipPel( ( ((P0 + IF_INTERNAL_OFFS) + round) >> shift ) + offset, clpRng );
60}
61
62static inline Pel noWeightOffsetUnidir( Pel P0, int round, int shift, const ClpRng& clpRng)
63{
64 return ClipPel( ( ((P0 + IF_INTERNAL_OFFS) + round) >> shift ), clpRng );
65}

76void WeightPrediction::getWpScaling(Slice *pcSlice, const int &iRefIdx0, const int &iRefIdx1, WPScalingParam *&wp0,
77 WPScalingParam *&wp1, const ComponentID maxNumComp)
78{
79 CHECK(iRefIdx0 < 0 && iRefIdx1 < 0, "Both picture reference list indizes smaller than '0'");
80
81 const bool wpBiPred = pcSlice->getPPS()->getWPBiPred();
82 const bool bBiPred = (iRefIdx0 >= 0 && iRefIdx1 >= 0);
83 const bool bUniPred = !bBiPred;
84
85 if (bUniPred || wpBiPred)
86 {
87 // explicit --------------------
88 wp0 = pcSlice->getWpScaling(REF_PIC_LIST_0, iRefIdx0);
89 wp1 = pcSlice->getWpScaling(REF_PIC_LIST_1, iRefIdx1);
90 }
91 else
92 {
93 THROW( "Unsupported WP configuration" );
94 }
95
96 if (iRefIdx0 < 0)
97 {
98 wp0 = nullptr;
99 }
100 if (iRefIdx1 < 0)
101 {
102 wp1 = nullptr;
103 }
104
105 const uint32_t numValidComponent = getNumberValidComponents(pcSlice->getSPS()->getChromaFormatIdc());
106 const bool bUseHighPrecisionPredictionWeighting = pcSlice->getSPS()->getSpsRangeExtension().getHighPrecisionOffsetsEnabledFlag();
107
108 if (bBiPred)
109 {
110 // Bi-predictive case
111 for (int yuv = 0; yuv < numValidComponent && yuv <= maxNumComp; yuv++)
112 {
113 const int bitDepth = pcSlice->getSPS()->getBitDepth(toChannelType(ComponentID(yuv)));
114 const int offsetScalingFactor = bUseHighPrecisionPredictionWeighting ? 1 : (1 << (bitDepth - 8));
115
116 wp0[yuv].w = wp0[yuv].codedWeight;
117 wp1[yuv].w = wp1[yuv].codedWeight;
118 wp0[yuv].o = wp0[yuv].codedOffset * offsetScalingFactor;
119 wp1[yuv].o = wp1[yuv].codedOffset * offsetScalingFactor;
120 wp0[yuv].offset = wp0[yuv].o + wp1[yuv].o;
121 wp0[yuv].shift = wp0[yuv].log2WeightDenom + 1;
122 wp0[yuv].round = (1 << wp0[yuv].log2WeightDenom);
123 wp1[yuv].offset = wp0[yuv].offset;
124 wp1[yuv].shift = wp0[yuv].shift;
125 wp1[yuv].round = wp0[yuv].round;
126 }
127 }
128 else
129 {
130 // UniPred
131 WPScalingParam *const pwp = (iRefIdx0 >= 0) ? wp0 : wp1;
132
133 for (int yuv = 0; yuv < numValidComponent && yuv <= maxNumComp; yuv++)
134 {
135 const int bitDepth = pcSlice->getSPS()->getBitDepth(toChannelType(ComponentID(yuv)));
136 const int offsetScalingFactor = bUseHighPrecisionPredictionWeighting ? 1 : (1 << (bitDepth - 8));
137
138 pwp[yuv].w = pwp[yuv].codedWeight;
139 pwp[yuv].offset = pwp[yuv].codedOffset * offsetScalingFactor;
140 pwp[yuv].shift = pwp[yuv].log2WeightDenom;
141 pwp[yuv].round = (pwp[yuv].log2WeightDenom >= 1) ? (1 << (pwp[yuv].log2WeightDenom - 1)) : (0);
142 }
143 }
144}

146void WeightPrediction::addWeightBi(const CPelUnitBuf &pcYuvSrc0,
147 const CPelUnitBuf &pcYuvSrc1,
148 const ClpRngs &clpRngs,
149 const WPScalingParam *const wp0,
150 const WPScalingParam *const wp1,
151 PelUnitBuf &rpcYuvDst,
152 const bool bRoundLuma /*= true*/,
153 const ComponentID maxNumComp
154 , bool lumaOnly
155 , bool chromaOnly
156)
157{
158 const bool enableRounding[MAX_NUM_COMPONENT] = { bRoundLuma, true, true };
159
160 const uint32_t numValidComponent = (const uint32_t)pcYuvSrc0.bufs.size();
161
162 CHECK( lumaOnly && chromaOnly, "Not allowed to have both lumaOnly and chromaOnly selected" );
163 int firstComponent = chromaOnly ? 1 : 0;
164 int lastComponent = lumaOnly ? 0 : maxNumComp;
165 for (int componentIndex = firstComponent; componentIndex < numValidComponent && componentIndex <= lastComponent; componentIndex++)
166 {
167 const ComponentID compID = ComponentID(componentIndex);
168
169 const Pel* pSrc0 = pcYuvSrc0.bufs[compID].buf;
170 const Pel* pSrc1 = pcYuvSrc1.bufs[compID].buf;
171 Pel* pDst = rpcYuvDst.bufs[compID].buf;
172
173 // Luma : --------------------------------------------
174 const ClpRng& clpRng = clpRngs.comp[compID];
175 const int w0 = wp0[compID].w;
176 const int offset = wp0[compID].offset;
177 const int clipBD = clpRng.bd;
178 const int shiftNum = IF_INTERNAL_FRAC_BITS(clipBD);
179 const int shift = wp0[compID].shift + shiftNum;
180 const int round = (enableRounding[compID] && (shift > 0)) ? (1 << (shift - 1)) : 0;
181 const int w1 = wp1[compID].w;
182 const int iHeight = rpcYuvDst.bufs[compID].height;
183 const int iWidth = rpcYuvDst.bufs[compID].width;
184
185 const uint32_t iSrc0Stride = pcYuvSrc0.bufs[compID].stride;
186 const uint32_t iSrc1Stride = pcYuvSrc1.bufs[compID].stride;
187 const uint32_t iDstStride = rpcYuvDst.bufs[compID].stride;
188
189 for (int y = iHeight - 1; y >= 0; y--)
190 {
191 // do it in batches of 4 (partial unroll)
192 int x = iWidth - 1;
193
194 for (; x >= 3; )
195 {
196 pDst[x] = weightBidir(w0, pSrc0[x], w1, pSrc1[x], round, shift, offset, clpRng ); x--;
197 pDst[x] = weightBidir(w0, pSrc0[x], w1, pSrc1[x], round, shift, offset, clpRng ); x--;
198 pDst[x] = weightBidir(w0, pSrc0[x], w1, pSrc1[x], round, shift, offset, clpRng ); x--;
199 pDst[x] = weightBidir(w0, pSrc0[x], w1, pSrc1[x], round, shift, offset, clpRng ); x--;
200 }
201 for (; x >= 0; x--)
202 {
203 pDst[x] = weightBidir(w0, pSrc0[x], w1, pSrc1[x], round, shift, offset, clpRng );
204 }
205
206 pSrc0 += iSrc0Stride;
207 pSrc1 += iSrc1Stride;
208 pDst += iDstStride;
209 } // y loop
210 } // compID loop
211}

245
246 for (int y = height - 1; y >= 0; y--)

248 // do it in batches of 4 (partial unroll)
249 int x = width - 1;

251 for (; x >= 3; )

261 }
262
263 src0 += src0Stride;
264 src1 += src1Stride;
265 dst += dstStride;
266 } // y loop

270 const ClpRngs &clpRngs,
271 const WPScalingParam *const wp0,
272 PelUnitBuf &rpcYuvDst,
273 const ComponentID maxNumComp
274 , bool lumaOnly
275 , bool chromaOnly
276)

279

284 componentIndex++)

287
288 const Pel* pSrc0 = pcYuvSrc0.bufs[compID].buf;

304 {
305 const int round = (shift > 0) ? (1 << (shift - 1)) : 0;

371void WeightPrediction::xWeightedPredictionUni(const PredictionUnit &pu,
372 const CPelUnitBuf &pcYuvSrc,
373 const RefPicList &eRefPicList,
374 PelUnitBuf &pcYuvPred,
375 const int iRefIdx_input/* = -1*/,
376 const ComponentID maxNumComp
377 , bool lumaOnly
378 , bool chromaOnly
379)
380{
381 WPScalingParam *pwp, *pwpTmp;
382
383 int iRefIdx = iRefIdx_input;
384 if (iRefIdx < 0)
385 {
386 iRefIdx = pu.refIdx[eRefPicList];

388
389 CHECK(iRefIdx < 0, "Negative reference picture list index");

402void WeightPrediction::xWeightedPredictionBi(const PredictionUnit &pu,
403 const CPelUnitBuf &pcYuvSrc0,
404 const CPelUnitBuf &pcYuvSrc1,
405 PelUnitBuf &rpcYuvDst,
406 const ComponentID maxNumComp
407 , bool lumaOnly
408 , bool chromaOnly
409)
410{
411 const int iRefIdx0 = pu.refIdx[0];
412 const int iRefIdx1 = pu.refIdx[1];
413 WPScalingParam *pwp0;
414 WPScalingParam *pwp1;
415
416 CHECK( !pu.cs->pps->getWPBiPred(), "Weighted Bi-prediction disabled" );
417
418 if (iRefIdx0 < 0 && iRefIdx1 < 0) return;
419
420 getWpScaling(pu.cu->slice, iRefIdx0, iRefIdx1, pwp0, pwp1, maxNumComp);
421
422 if (iRefIdx0 >= 0 && iRefIdx1 >= 0)
423 {
424 addWeightBi(pcYuvSrc0, pcYuvSrc1, pu.cu->slice->clpRngs(), pwp0, pwp1, rpcYuvDst, true, maxNumComp, lumaOnly, chromaOnly);
425 }
426 else if (iRefIdx0 >= 0 && iRefIdx1 < 0)
427 {
428 addWeightUni(pcYuvSrc0, pu.cu->slice->clpRngs(), pwp0, rpcYuvDst, maxNumComp, lumaOnly, chromaOnly);
429 }
430 else if (iRefIdx0 < 0 && iRefIdx1 >= 0)
431 {
432 addWeightUni(pcYuvSrc1, pu.cu->slice->clpRngs(), pwp1, rpcYuvDst, maxNumComp, lumaOnly, chromaOnly);
433 }
434 else
435 {
436 THROW( "Both reference picture list indizes are negative" );
437 }
438}
