52bool probe_es_file(const string &name, BaseCoding coding, unsigned &width, unsigned &height, unsigned &bit_depth,
53 unsigned &chroma_format_idc) {
54 // Open the file
55 BaseDecoder::Codec es_type;
56 switch (coding) {
57 case BaseCoding_AVC:
58 es_type = BaseDecoder::AVC;
59 break;
60 case BaseCoding_HEVC:
61 es_type = BaseDecoder::HEVC;
62 break;
63 case BaseCoding_VVC:
64 es_type = BaseDecoder::VVC;
65 break;
66 case BaseCoding_EVC:
67 es_type = BaseDecoder::EVC;
68 break;
69 default:
70 CHECK(0);
71 }
72
73 ESFile es_file;
74 CHECK(es_file.Open(name, es_type));
75
76 // Go through AUs until SPS has been seen ...
77 for (int i = 0; i < PROBE_AU_LIMIT; ++i) {
78 ESFile::AccessUnit au;
79 ESFile::Result r = es_file.NextAccessUnit(au);
80
81 // Oh dear ...
82 if (r != ESFile::Success)
83 return false;
84
85 // Have we got a good width, height & depth?
86 unsigned w = es_file.GetPictureWidth();
87 unsigned h = es_file.GetPictureHeight();
88 unsigned d = es_file.GetBitDepth();
89 unsigned c = es_file.GetChromaFormatIDC();
90
91 if (w && h && d) {
92 width = w;
93 height = h;
94 bit_depth = d;
95 chroma_format_idc = c;
96
97 return true;
98 }
99 }
100
101 return false;
102}
