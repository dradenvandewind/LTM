56def parse_vooya_filename(name):
57 """
58 Extract vooya-ish image metadata from filename in form suitable for use as encoder parameters.
59 """
60 params = {}
61 for p in name.split("_"):
62 for n,r in IMAGE_NAME_PARAMS:
63 m = r.match(p)
64 if m:
65 params[n] = m.group(1)
66
67 if 'bit_depth' in params:
68 if 'format' not in params:
69 params['format'] = "420p"
70 params['format'] += params.pop('bit_depth')
71
72 if 'format' in params and params['format'].startswith('4'):
73 params['format'] = 'yuv' + params['format']
74
75 if 'width' in params:
76 params['width'] = int(params['width'])
77 if 'height' in params:
78 params['height'] = int(params['height'])
79 if 'fps' in params:
80 params['fps'] = int(params['fps'])
81
82 return params
