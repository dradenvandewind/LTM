124ESFile::Result ESFile::ReadAccessUnit(AccessUnit &out) {
125 switch (m_decoder->Delimiter()) {
126 case NALDelimiterMarker:
127 return ReadAccessUnitMarker(out);
128
129 case NALDelimiterU32Length:
130 return ReadAccessUnitU32Length(out);

132 CHECK(0);
133 return NoFile;

137ESFile::Result ESFile::ReadAccessUnitMarker(AccessUnit &out) {
138 if (m_file == nullptr)
139 return NoFile;
140
141 utility::DataBuffer buffer;
142 std::vector<NalUnit> nalUnits;
143 int32_t size = 0;
144 uint32_t temporalId = 0;
145
146 while (true) {
147 const int c = fgetc(m_file);
148
149 if (c != EOF)
150 buffer.push_back(c);
151
152 bool foundNalStart =
153 buffer.size() > kNalUnitMarkerSize && std::equal(buffer.end() - kNalUnitMarkerSize, buffer.end(), kNalUnitMarker);
154
155 if (foundNalStart || feof(m_file)) {
156
157 uint64_t nalLength = buffer.size();
158 /* uint64_t */ int markerSize = 0;
159
160 if (foundNalStart) {
161 markerSize = kNalUnitMarkerSize;
162
163 // Handle [0, 0, 0, 1] case
164 if (buffer.size() > kNalUnitMarkerSize && buffer[nalLength - markerSize - 1] == 0x0)
165 ++markerSize;
166 }
167
168 if (nalLength > markerSize) {
169 nalLength -= markerSize;
170
171 unsigned result = m_decoder->ParseNalUnit(buffer.data(), static_cast<uint32_t>(nalLength));
172
173 if (result == 1) {
174 NalUnit unit;
175
176 auto nalEnd = buffer.end() - markerSize;
177 unit.m_type = m_decoder->GetNalType();
178 temporalId = std::max(temporalId, m_decoder->GetTemporalId());
179
180 // Move the next nal unit's marker into unit's buffer so we can just swap the two
181 unit.m_data.insert(unit.m_data.end(), nalEnd, buffer.end());
182 buffer.erase(nalEnd, buffer.end());
183 std::swap(unit.m_data, buffer);
184
185 nalUnits.push_back(std::move(unit));
186 size += (int32_t)(nalLength);
187
188 if (m_decoder->GetBaseNalUnitType() == BaseDecNalUnitType::Slice) {
189
190 out.m_pictureType = m_decoder->GetBasePictureType();
191 out.m_poc = GenerateIncreasingPOC();
192 out.m_qp = m_decoder->GetQP();
193 out.m_temporalId = temporalId;
194 out.m_nalUnits = std::move(nalUnits);
195
196 out.m_size = size;
197
198 fseek(m_file, -static_cast<long>(markerSize), SEEK_CUR);
199
200 return Success;
201 }
202 } else if (result == 2) {
203 NalUnit unit;
204
205 auto nalEnd = buffer.end() - markerSize;
206
207 // Move the next nal unit's marker into unit's buffer so we can just swap the two
208 unit.m_data.insert(unit.m_data.end(), nalEnd, buffer.end());
209 buffer.erase(nalEnd, buffer.end());
210 std::swap(unit.m_data, buffer);
211 nalUnits.push_back(std::move(unit));
212 } else
213 return NalParsingError;
214 }
215 }
216
217 if (c == EOF)
218 break;
219 }
220
221 return EndOfFile;
222}

224ESFile::Result ESFile::ReadAccessUnitU32Length(AccessUnit &out) {
225 if (m_file == nullptr)
226 return NoFile;
227
228 utility::DataBuffer buffer;
229 std::vector<NalUnit> nalUnits;
230 int32_t size = 0;
231
232 while (true) {
233 // Read length prefix
234 uint32_t nal_length = 0;
235 {
236 const size_t result = fread(&nal_length, 1, sizeof(nal_length), m_file);
237 if (result == 0)
238 return EndOfFile;
239
240 if (result != sizeof(nal_length))
241 return NalParsingError;
242 }
243
244 // Read body of NAL unit
245 buffer.resize(nal_length);
246 if (fread(buffer.data(), 1, nal_length, m_file) != nal_length)
247 return NalParsingError;
248
249 if (m_decoder->ParseNalUnit(buffer.data(), nal_length)) {
250 NalUnit unit;
251 unit.m_type = m_decoder->GetNalType();
252 unit.m_data.insert(unit.m_data.end(), (uint8_t *)&nal_length, (uint8_t *)&nal_length + sizeof(nal_length));
253 unit.m_data.insert(unit.m_data.end(), buffer.begin(), buffer.end());
254 nalUnits.push_back(std::move(unit));
255 size += (int32_t)(nal_length + 4); // length + length prefix
256
257 if (m_decoder->GetBaseNalUnitType() == BaseDecNalUnitType::Slice) {
258 out.m_pictureType = m_decoder->GetBasePictureType();
259 out.m_poc = GenerateIncreasingPOC();
260 out.m_qp = m_decoder->GetQP();
261 out.m_nalUnits = std::move(nalUnits);
262
263 out.m_size = size;
264
265 return Success;
266 }
267
268 } else {
269 return NalParsingError;
270 }
271 }
272}

274// Create a POC that always increases across IDR
275//
276uint64_t ESFile::GenerateIncreasingPOC() {
277 uint64_t decoded_poc = m_decoder->GetPictureOrderCount();
278
279 // If we see the start of an IDR and the POC goes backwards, then
280 // offset the generated POC by the highest POC seen so far
281 //
282 if(m_decoder->GetBasePictureType() == BaseDecPictType::IDR && decoded_poc < m_poc_highest) {
283 m_poc_offset = m_poc_highest;
284 }
285
286 const uint64_t poc = decoded_poc + m_poc_offset;
287
288 if(poc > m_poc_highest) {
289 m_poc_highest = poc + m_decoder->GetPictureOrderCountIncrement();
290 }
291
292 return poc;
293}
