75uint32_t GetBitCountFromMax(uint32_t maxValue) {
76 // Bit count is determined by ceil(log2(maxRepresentation);
77 // Because log2 is c99 which is not so widely available we can use standard log with a divide.
78 // log2(x) = logY(x) / logY(2)
79 //
80 // Alternatively we want to find the highest bit set in maxValue.
81 static const float logBase = log(2.0f);
82 float log2Value = log(static_cast<float>(maxValue)) / logBase;
83 return static_cast<uint32_t>(ceil(log2Value));
84}

86BaseDecNalUnitType::Enum uFromAVC(uint8_t type) {
87 switch (type) {
88 case AVCNalType::Unspecified:
89 case AVCNalType::CodedSlice_NonIDR:
90 case AVCNalType::CodedSlice_A:
91 case AVCNalType::CodedSlice_B:
92 case AVCNalType::CodedSlice_C:
93 case AVCNalType::CodedSlice_IDR:
94 case AVCNalType::CodedSlice_Aux:
95 case AVCNalType::CodedSlice_Ext:
96 case AVCNalType::CodedSlice_DepthExt:
97 return BaseDecNalUnitType::Slice;
98 case AVCNalType::SEI:
99 return BaseDecNalUnitType::SEI;
100 case AVCNalType::SPSExt:
101 case AVCNalType::SubsetSPS:
102 case AVCNalType::SPS:
103 return BaseDecNalUnitType::SPS;
104 case AVCNalType::PPS:
105 return BaseDecNalUnitType::PPS;
106 case AVCNalType::AUD:
107 return BaseDecNalUnitType::AUD;
108 case AVCNalType::EndOfSequence:
109 return BaseDecNalUnitType::EOS;
110 case AVCNalType::EndOfStream:
111 return BaseDecNalUnitType::EOB;
112 case AVCNalType::FillerData:
113 return BaseDecNalUnitType::Filler;
114 case AVCNalType::Prefix:
115 case AVCNalType::DPS:
116 break;
117 }
118
119 return BaseDecNalUnitType::Unknown;
120}

122uint32_t uOffsetForNalUnitHeader(const uint8_t *nal) {
123 if (memcmp(nal, kNalHeader, kNalHeaderLength) == 0)
124 return kNalHeaderLength;
125 if (!nal[0] && memcmp(nal + 1, kNalHeader, kNalHeaderLength) == 0)
126 return kNalHeaderLength + 1;
127 return 0;
128}

162bool BaseDecoderAVC::ParseNalUnit(const uint8_t *nal, uint32_t nalLength) {
163 bool bSuccess = true;
164
165 // Account for AnnexB nal unit header ([0],0,0,1)
166 uint32_t offset = uOffsetForNalUnitHeader(nal);
167
168 // [forbidden-zero:1, nal_ref_idc:2, nal_type:5]
169 m_currentNalType = (nal[offset] & 0x1F);
170 m_currentNalRefIDC = ((nal[offset] >> 5) & 0x3);
171
172 offset += 1;
173
174 BaseDecNalUnitType::Enum nalType = uFromAVC(m_currentNalType);
175
176 if ((nalType == BaseDecNalUnitType::SPS) || (nalType == BaseDecNalUnitType::PPS) || (nalType == BaseDecNalUnitType::SEI) ||
177 (nalType == BaseDecNalUnitType::Slice)) {
178 nal += offset;
179 nalLength -= offset;
180
181 Unencapsulate(nal, nalLength, m_currentNalPayload);
182 m_currentBits = *m_currentNalPayload.data();
183 m_remainingBits = 8;
184 m_byteOffset = 0;
185
186 switch (nalType) {
187 case BaseDecNalUnitType::PPS:
188 bSuccess = ParsePPS();
189 break;
190 case BaseDecNalUnitType::SPS:
191 bSuccess = ParseSPS();
192 break;
193 case BaseDecNalUnitType::Slice:
194 if ((bSuccess = ParseSliceHeader())) {
195 HandlePictureOrderCount();
196 }
197 break;
198 default:
199 break;
200 }
201 }
202
203 return bSuccess;
204}

239uint32_t BaseDecoderAVC::GetPictureWidth() const {
240 if (!m_activeSPS) {
241 return 0;
242 }
243
244 uint32_t cropUnitX = 1; // (7-19)
245 if (m_activeSPS->m_chromaArrayType != 0 && m_activeSPS->m_chromaArrayType <= 2) {
246 // subWidthC
247 cropUnitX = 2; // (7-21)
248 }
249
250 return m_activeSPS->m_picWidthInMBs * 16 -
251 cropUnitX * (m_activeSPS->m_frameCropLeftOffset + m_activeSPS->m_frameCropRightOffset); // (D-30)
252}

254uint32_t BaseDecoderAVC::GetPictureHeight() const {
255 if (!m_activeSPS) {
256 return 0;
257 }
258
259 uint32_t twoMinusFrameMBsFlag = 2 - m_activeSPS->m_bFrameMBsOnly;
260 uint32_t cropUnitY = twoMinusFrameMBsFlag; // (7-20)
261 if (m_activeSPS->m_chromaArrayType != 0) {
262 // subHeightC * (2 - frame_mbs_only_flag)
263 cropUnitY = (1 + (m_activeSPS->m_chromaArrayType <= 1)) * (twoMinusFrameMBsFlag); // (7-22)
264 }
265
266 return twoMinusFrameMBsFlag * m_activeSPS->m_picHeightInMapUnits * 16 -
267 cropUnitY * (m_activeSPS->m_frameCropTopOffset + m_activeSPS->m_frameCropBottomOffset); // (D-31)
268}

276uint8_t BaseDecoderAVC::GetMaxNumberOfReorderFrames() const {
277 if (!m_activeSPS) {
278 return kMaxNumRefFramesAllowed;
279 }
280
281 if (m_activeSPS->m_bVUIParametersPresent) {
282 return m_activeSPS->m_vui.m_maxNumReorderFrames;
283 }
284
285 if (m_activeSPS->m_bConstraintSet3 &&
286 (m_activeSPS->m_profileIDC == 44 || m_activeSPS->m_profileIDC == 86 || m_activeSPS->m_profileIDC == 100 ||
287 m_activeSPS->m_profileIDC == 110 || m_activeSPS->m_profileIDC == 122 || m_activeSPS->m_profileIDC == 244)) {
288 return 0;
289 }
290
291 // A.3.1
292 auto levelLimit = GetLevelLimit(m_activeSPS);
293 if (!levelLimit) {
294 return kMaxNumRefFramesAllowed;
295 }
296
297 uint32_t frameHeightInMbs = (2 - m_activeSPS->m_bFrameMBsOnly) * m_activeSPS->m_picHeightInMapUnits;
298 uint8_t calculatedMaxNumRefFrames = levelLimit->m_maxDpbMbs / (m_activeSPS->m_picWidthInMBs * frameHeightInMbs);
299 return std::min(calculatedMaxNumRefFrames, kMaxNumRefFramesAllowed);
300}

334// Section 8.2.1 of Standard
335void BaseDecoderAVC::HandlePictureOrderCount() {
336 assert(!m_activeSlice.m_bFieldPic);
337 assert((m_activeSPS->m_picOrderCntType == 0) || (m_activeSPS->m_picOrderCntType == 2));
338
339 uint32_t topFieldOrderCnt = 0;
340 uint32_t bottomFieldOrderCnt = 0;
341
342 // 8.2.1.1
343 if (m_activeSPS->m_picOrderCntType == 0) {
344 uint32_t prevPicOrderCntMsb = 0;
345 uint32_t prevPicOrderCntLsb = 0;
346 uint32_t picOrderCntMsb = 0;
347 uint32_t maxPicOrderCntLsb = 1 << m_activeSPS->m_log2MaxPicOrderCntLSB;
348
349 if (m_currentNalType == AVCNalType::CodedSlice_IDR) {
350 prevPicOrderCntMsb = 0;
351 prevPicOrderCntLsb = 0;
352 } else if (m_refPic.m_header.m_memManagementControlOp[5]) {
353 prevPicOrderCntMsb = 0;
354 prevPicOrderCntLsb = m_refPic.m_header.m_bBottomField ? 0 : m_refPic.m_topFieldOrderCnt;
355 } else {
356 prevPicOrderCntMsb = m_refPic.m_picOrderCntMsb;
357 prevPicOrderCntLsb = m_refPic.m_header.m_picOrderCntLSB;
358 }
359
360 // (8-3)
361 if (m_activeSlice.m_picOrderCntLSB < prevPicOrderCntLsb &&
362 ((prevPicOrderCntLsb - m_activeSlice.m_picOrderCntLSB) >= (maxPicOrderCntLsb / 2))) {
363 picOrderCntMsb = prevPicOrderCntMsb + maxPicOrderCntLsb;
364 } else if (m_activeSlice.m_picOrderCntLSB > prevPicOrderCntLsb &&
365 ((m_activeSlice.m_picOrderCntLSB - prevPicOrderCntLsb) > (maxPicOrderCntLsb / 2))) {
366 picOrderCntMsb = prevPicOrderCntMsb - maxPicOrderCntLsb;
367 } else
368 picOrderCntMsb = prevPicOrderCntMsb;
369
370 // (8-4), (8-5)
371 bottomFieldOrderCnt = picOrderCntMsb + m_activeSlice.m_picOrderCntLSB;
372 topFieldOrderCnt = bottomFieldOrderCnt;
373
374 if (!m_activeSlice.m_bFieldPic)
375 bottomFieldOrderCnt += m_activeSlice.m_deltaPicOrderCntBottom;
376
377 if (m_currentNalRefIDC != 0) {
378 m_refPic.m_picOrderCntMsb = picOrderCntMsb;
379 m_refPic.m_header = m_activeSlice;
380 m_refPic.m_topFieldOrderCnt = topFieldOrderCnt;
381 m_refPic.m_bottomFieldOrderCnt = bottomFieldOrderCnt;
382 }
383 } else {
384 uint32_t maxFrameNum = 1 << m_sps->m_log2MaxFrameNum;
385 uint32_t frameNumOffset = 0;
386
387 // (8-6), (8-11) Calculate frame num offset.
388 if (m_currentNalType == AVCNalType::CodedSlice_IDR)
389 frameNumOffset = 0;
390 else if (m_refPic.m_frameNum > m_activeSlice.m_frameNum)
391 frameNumOffset = m_refPic.m_frameNumOffset + maxFrameNum;
392 else
393 frameNumOffset = m_refPic.m_frameNumOffset;
394
395 if (m_activeSPS->m_picOrderCntType == 1) {
396 uint32_t absFrameNum = 0;
397 uint32_t expectedPicOrderCnt = 0;
398
399 // (8-7)
400 if (m_activeSPS->m_numRefFramesInPicOrderCntCycle != 0)
401 absFrameNum = frameNumOffset + m_activeSlice.m_frameNum;
402 else
403 absFrameNum = 0;
404
405 if (m_currentNalRefIDC == 0 && absFrameNum > 0)
406 absFrameNum -= 1;
407
408 // (8-9)
409 if (absFrameNum > 0) {
410 // (8-8)
411 uint32_t picOrderCntCycleCnt = (absFrameNum - 1) / m_activeSPS->m_numRefFramesInPicOrderCntCycle;
412 uint32_t frameNumInPicOrderCntCycle = (absFrameNum - 1) % m_activeSPS->m_numRefFramesInPicOrderCntCycle;
413
414 // (7-12)
415 uint32_t expectedDeltaPerPicOrderCntCycle = 0;
416 for (uint32_t i = 0; i < m_activeSPS->m_numRefFramesInPicOrderCntCycle; ++i)
417 expectedDeltaPerPicOrderCntCycle += m_activeSPS->m_offsetForRefFrame[i];
418
419 expectedPicOrderCnt = picOrderCntCycleCnt * expectedDeltaPerPicOrderCntCycle;
420 for (uint32_t i = 0; i <= frameNumInPicOrderCntCycle; ++i)
421 expectedPicOrderCnt += m_activeSPS->m_offsetForRefFrame[i];
422 } else
423 expectedPicOrderCnt = 0;
424
425 if (m_currentNalRefIDC == 0)
426 expectedPicOrderCnt += m_activeSPS->m_offsetForNonRefPic;
427
428 // (8-10)
429 topFieldOrderCnt = expectedPicOrderCnt + m_activeSlice.m_deltaPicOrderCnt[0];
430 bottomFieldOrderCnt = topFieldOrderCnt + m_activeSPS->m_offsetForTopToBottomField;
431
432 if (!m_activeSlice.m_bFieldPic)
433 bottomFieldOrderCnt += m_activeSlice.m_deltaPicOrderCnt[1];
434 } else if (m_activeSPS->m_picOrderCntType == 2) {
435 uint32_t tempPicOrderCnt = 0;
436 // (8-12) Calculate temp pic order count.
437 if (m_currentNalType == AVCNalType::CodedSlice_IDR)
438 tempPicOrderCnt = 0;
439 else if (m_currentNalRefIDC == 0)
440 tempPicOrderCnt = 2 * (frameNumOffset + m_activeSlice.m_frameNum) - 1;
441 else
442 tempPicOrderCnt = 2 * (frameNumOffset + m_activeSlice.m_frameNum);
443
444 // (8-13)
445 topFieldOrderCnt = tempPicOrderCnt;
446 bottomFieldOrderCnt = tempPicOrderCnt;
447 }
448
449 m_refPic.m_frameNum = m_activeSlice.m_frameNum;
450 m_refPic.m_frameNumOffset = m_activeSlice.m_memManagementControlOp[5] ? 0 : frameNumOffset;
451 }
452
453 // (8-1)
454 if (!m_activeSlice.m_bFieldPic)
455 m_currentPictureOrderCount = std::min(topFieldOrderCnt, bottomFieldOrderCnt);
456 else
457 m_currentPictureOrderCount = m_activeSlice.m_bBottomField ? bottomFieldOrderCnt : topFieldOrderCnt;
458}

460bool BaseDecoderAVC::ParseSPS() {
461 uint8_t profileIDC = ReadBits(8);
462 bool bConstraintSet0 = ReadFlag();
463 bool bConstraintSet1 = ReadFlag();
464 bool bConstraintSet2 = ReadFlag();
465 bool bConstraintSet3 = ReadFlag();
466 bool bConstraintSet4 = ReadFlag();
467 bool bConstraintSet5 = ReadFlag();
468 uint8_t reservedZero2Bits = ReadBits(2);
469 uint8_t levelIDC = ReadBits(8);
470 uint32_t spsID = ReadUE();
471
472 if (spsID >= AVCMaxSPSCount) {
473 INFO("ERROR -- spsID out of range");
474 throw std::runtime_error("spsID out of range");
475 return false;
476 }
477
478 if (reservedZero2Bits != 0) {
479 INFO("malformed sps, expected 2 reserved zero bits to be 0");
480 throw std::runtime_error("malformed sps, expected 2 reserved zero bits to be 0\n");
481 return false;
482 }
483
484 m_activeSPS = &m_sps[spsID];
485 m_activeSPS->m_profileIDC = profileIDC;
486 m_activeSPS->m_bConstraintSet0 = bConstraintSet0;
487 m_activeSPS->m_bConstraintSet1 = bConstraintSet1;
488 m_activeSPS->m_bConstraintSet2 = bConstraintSet2;
489 m_activeSPS->m_bConstraintSet3 = bConstraintSet3;
490 m_activeSPS->m_bConstraintSet4 = bConstraintSet4;
491 m_activeSPS->m_bConstraintSet5 = bConstraintSet5;
492 m_activeSPS->m_levelIDC = levelIDC;
493 m_activeSPS->m_spsID = spsID;
494
495 if (m_activeSPS->m_profileIDC == 100 || m_activeSPS->m_profileIDC == 110 || m_activeSPS->m_profileIDC == 122 ||
496 m_activeSPS->m_profileIDC == 244 || m_activeSPS->m_profileIDC == 44 || m_activeSPS->m_profileIDC == 83 ||
497 m_activeSPS->m_profileIDC == 86 || m_activeSPS->m_profileIDC == 118 || m_activeSPS->m_profileIDC == 128 ||
498 m_activeSPS->m_profileIDC == 138 || m_activeSPS->m_profileIDC == 139 || m_activeSPS->m_profileIDC == 134 ||
499 m_activeSPS->m_profileIDC == 135) {
500 m_activeSPS->m_chromaFormatIDC = ReadUE();
501
502 if (m_activeSPS->m_chromaFormatIDC == 3) {
503 m_activeSPS->m_bSeparateColourPlane = ReadFlag();
504 }
505
506 if (m_activeSPS->m_bSeparateColourPlane == false) {
507 m_activeSPS->m_chromaArrayType = m_activeSPS->m_chromaFormatIDC;
508 } else {
509 m_activeSPS->m_chromaArrayType = 0;
510 }
511
512 m_activeSPS->m_bitDepthLuma = ReadUE() + 8;
513 m_activeSPS->m_bitDepthChroma = ReadUE() + 8;
514 m_activeSPS->m_bQPPrimeYZeroTransformBypass = ReadFlag();
515
516 if (ReadFlag()) // seq_scaling_matrix_present_flag
517 {
518 uint32_t loopCount = (m_activeSPS->m_chromaFormatIDC != 3) ? 8 : 12;
519 HandleScalingList(loopCount);
520 }
521 }
522
523 m_activeSPS->m_log2MaxFrameNum = ReadUE() + 4;
524 m_activeSPS->m_picOrderCntType = ReadUE();
525
526 if (m_activeSPS->m_picOrderCntType == 0) {
527 m_activeSPS->m_log2MaxPicOrderCntLSB = ReadUE() + 4;
528 } else if (m_activeSPS->m_picOrderCntType == 1) {
529 m_activeSPS->m_bDeltaPicOrderAlwaysZero = ReadFlag();
530 m_activeSPS->m_offsetForNonRefPic = ReadSE();
531 m_activeSPS->m_offsetForTopToBottomField = ReadSE();
532 m_activeSPS->m_numRefFramesInPicOrderCntCycle = ReadUE();
533
534 for (uint32_t i = 0; i < m_activeSPS->m_numRefFramesInPicOrderCntCycle; ++i) {
535 m_activeSPS->m_offsetForRefFrame[i] = ReadSE();
536 }
537 }
538
539 m_activeSPS->m_numRefFrames = ReadUE();
540 m_activeSPS->m_bGapsInFrameNumValueAllowed = ReadFlag();
541 m_activeSPS->m_picWidthInMBs = ReadUE() + 1;
542 m_activeSPS->m_picHeightInMapUnits = ReadUE() + 1;
543 m_activeSPS->m_bFrameMBsOnly = ReadFlag();
544

601 case 4:

969bool BaseDecoderAVC::HandleScalingListElement(uint32_t listSize) {
970 int32_t lastScale = 8;
971 int32_t nextScale = 8;
972 int32_t deltaScale;
973
974 for (uint32_t j = 0; j < listSize; ++j) {
975 if (nextScale != 0) {
976 deltaScale = ReadSE();
977 nextScale = (lastScale + deltaScale + 256) % 256;
978 }
979 lastScale = (nextScale == 0) ? lastScale : nextScale;
980 }
981
