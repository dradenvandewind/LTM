60struct AlfFilterShape
61{
62 AlfFilterShape( int size )
63 : filterLength( size ),
64 numCoeff( size * size / 4 + 1 ),
65 filterSize( size * size / 2 + 1 )
66 {
67 if( size == 5 )
68 {
69 pattern = {
70 0,
71 1, 2, 3,
72 4, 5, 6, 5, 4,
73 3, 2, 1,
74 0
75 };
76
77 weights = {
78 2,
79 2, 2, 2,
80 2, 2, 1, 1
81 };
82
83 filterType = ALF_FILTER_5;
84 }
85 else if( size == 7 )
86 {
87 pattern = {
88 0,
89 1, 2, 3,
90 4, 5, 6, 7, 8,
91 9, 10, 11, 12, 11, 10, 9,
92 8, 7, 6, 5, 4,
93 3, 2, 1,
94 0
95 };
96
97 weights = {
98 2,
99 2, 2, 2,
100 2, 2, 2, 2, 2,
101 2, 2, 2, 1, 1
102 };
103
104 filterType = ALF_FILTER_7;
105 }
106 else if (size == size_CC_ALF)
107 {
108 size = 4;
109 filterLength = 8;
110 numCoeff = 8;
111 filterSize = 8;
112 filterType = CC_ALF;
113 }
114 else
115 {
116 filterType = ALF_NUM_OF_FILTER_TYPES;
117 CHECK( 0, "Wrong ALF filter shape" );
118 }
119 }
120
121 AlfFilterType filterType;
122 int filterLength;
123 int numCoeff; //TO DO: check whether we need both numCoeff and filterSize
124 int filterSize;
125 std::vector<int> pattern;
126 std::vector<int> weights;
127};

129struct AlfParam
130{
131 bool enabledFlag[MAX_NUM_COMPONENT]; // alf_slice_enable_flag, alf_chroma_idc
132 bool nonLinearFlag[MAX_NUM_CHANNEL_TYPE]; // alf_[luma/chroma]_clip_flag
133 short lumaCoeff[MAX_NUM_ALF_CLASSES * MAX_NUM_ALF_LUMA_COEFF]; // alf_coeff_luma_delta[i][j]
134 Pel lumaClipp[MAX_NUM_ALF_CLASSES * MAX_NUM_ALF_LUMA_COEFF]; // alf_clipp_luma_[i][j]
135 int numAlternativesChroma; // alf_chroma_num_alts_minus_one + 1
136 short chromaCoeff[MAX_NUM_ALF_ALTERNATIVES_CHROMA][MAX_NUM_ALF_CHROMA_COEFF]; // alf_coeff_chroma[i]

138 short filterCoeffDeltaIdx[MAX_NUM_ALF_CLASSES]; // filter_coeff_delta[i]
139 bool alfLumaCoeffFlag[MAX_NUM_ALF_CLASSES]; // alf_luma_coeff_flag[i]
140 int numLumaFilters; // number_of_filters_minus1 + 1
141 bool alfLumaCoeffDeltaFlag; // alf_luma_coeff_delta_flag
142 std::vector<AlfFilterShape>* filterShapes;
143 bool newFilterFlag[MAX_NUM_CHANNEL_TYPE];

147 reset();
148 }

150 void reset()
151 {
152 std::memset( enabledFlag, false, sizeof( enabledFlag ) );
153 std::memset( nonLinearFlag, false, sizeof( nonLinearFlag ) );
154 std::memset( lumaCoeff, 0, sizeof( lumaCoeff ) );
155 std::memset( lumaClipp, 0, sizeof( lumaClipp ) );
156 numAlternativesChroma = 1;
157 std::memset( chromaCoeff, 0, sizeof( chromaCoeff ) );
158 std::memset( chromaClipp, 0, sizeof( chromaClipp ) );
159 std::memset( filterCoeffDeltaIdx, 0, sizeof( filterCoeffDeltaIdx ) );
160 std::memset( alfLumaCoeffFlag, true, sizeof( alfLumaCoeffFlag ) );
161 numLumaFilters = 1;
162 alfLumaCoeffDeltaFlag = false;
163 memset(newFilterFlag, 0, sizeof(newFilterFlag));
164 }

172 numAlternativesChroma = src.numAlternativesChroma;
173 std::memcpy( chromaCoeff, src.chromaCoeff, sizeof( chromaCoeff ) );
174 std::memcpy( chromaClipp, src.chromaClipp, sizeof( chromaClipp ) );
175 std::memcpy( filterCoeffDeltaIdx, src.filterCoeffDeltaIdx, sizeof( filterCoeffDeltaIdx ) );
176 std::memcpy( alfLumaCoeffFlag, src.alfLumaCoeffFlag, sizeof( alfLumaCoeffFlag ) );
177 numLumaFilters = src.numLumaFilters;
178 alfLumaCoeffDeltaFlag = src.alfLumaCoeffDeltaFlag;
179 filterShapes = src.filterShapes;
180 std::memcpy(newFilterFlag, src.newFilterFlag, sizeof(newFilterFlag));
181 return *this;
182 }
