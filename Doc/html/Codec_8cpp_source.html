56Codec* CodecCreate(const string &name, CodecOperation operation, const string &create_options) {
57
58 // Try to load a library of form "<program>/external_codecs/libs/base_<name>.so"
59
60 string library_file = get_program_directory(format("external_codecs/libs/%sbase_%s.%s", SHARED_PREFIX, name.c_str(), SHARED_SUFFIX));
61
62 if(file_size(library_file) < 0) {
63 ERR("Cannot find base codec library: \"%s\"", library_file.c_str());
64 return nullptr;
65 }
66
67 codec_api_version_fn_t version_fn;
68 codec_api_query_fn_t query_fn;
69 codec_api_create_fn_t create_fn;
70
71 shared_handle_t h = CHECK(SHARED_LOAD(library_file.c_str()));
72
73 if(!h)
74 return nullptr;
75
76 version_fn = (codec_api_version_fn_t)CHECK(SHARED_SYMBOL(h, "CodecAPI_Version"));
77
78 const uint32_t version = version_fn();
79 if(version != LOADABLE_CODEC_API_VERSION) {
80 ERR("Base codec library \"%s\" has wrong version: %d", library_file.c_str(), version);
81 return nullptr;
82 }
83
84 query_fn = (codec_api_query_fn_t)CHECK(SHARED_SYMBOL(h, "CodecAPI_Query"));
85 create_fn = (codec_api_create_fn_t)CHECK(SHARED_SYMBOL(h, "CodecAPI_Create"));
86
87 Codec* codec = create_fn(name.c_str(), operation, create_options.c_str());
88
89 return codec;
90}

95// Convert a codec error to a user string
96//
97string CodecErrorToString(Codec* codec, CodecError error) {
98 const size_t buffer_len = codec->get_error_message(error, 0, 0);
99 char *buffer = CHECK((static_cast<char *>(alloca(buffer_len))));
100 codec->get_error_message(error, buffer, buffer_len);
101 codec->release_error(error);
102 return string(buffer, buffer + buffer_len);
103}
