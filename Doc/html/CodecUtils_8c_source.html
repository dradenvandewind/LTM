132Codec* LTMCodecAllocate(const char *name, const char *version_string, CodecOperation operation) {
133 Codec *codec = (Codec *)calloc(1, sizeof(Codec));
134
135 codec->api_version = LOADABLE_CODEC_API_VERSION;
136
137 codec->name = strdup(name);
138 codec->version_string = strdup(name);
139 codec->operation = operation;
140 codec->push_packet = push_packet;
141 codec->pull_packet = pull_packet;
142 codec->push_image = push_image;
143 codec->pull_image = pull_image;
144 codec->create_metadata = create_metadata;
145 codec->release_context = release_context;
146 codec->release_error = release_error;
147 codec->release_metadata = release_metadata;
148 codec->get_metadata_property_name = get_metadata_property_name;
149 codec->get_metadata_property_u32 = get_metadata_property_u32;
150 codec->get_metadata_property_i32 = get_metadata_property_i32;
151 codec->get_metadata_property_u64 = get_metadata_property_u64;
152 codec->get_metadata_property_i64 = get_metadata_property_i64;
153 codec->get_metadata_property_bool = get_metadata_property_bool;
154 codec->set_metadata_property_u32 = set_metadata_property_u32;
155 codec->set_metadata_property_i32 = set_metadata_property_i32;
156 codec->set_metadata_property_u64 = set_metadata_property_u64;
157 codec->set_metadata_property_i64 = set_metadata_property_i64;
158 codec->set_metadata_property_bool = set_metadata_property_bool;
159 codec->get_error_code = get_error_code;
160 codec->get_error_message = get_error_message;
161 codec->get_error_file = get_error_file;
162 codec->get_error_line = get_error_line;
163
164 return codec;
165}
