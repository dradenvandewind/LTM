207template<typename T, size_t N>
208uint32_t updateCandList(T uiMode, double uiCost, static_vector<T, N>& candModeList, static_vector<double, N>& candCostList
209 , size_t uiFastCandNum = N, int* iserttPos = nullptr)
210{
211 CHECK( std::min( uiFastCandNum, candModeList.size() ) != std::min( uiFastCandNum, candCostList.size() ), "Sizes do not match!" );
212 CHECK( uiFastCandNum > candModeList.capacity(), "The vector is to small to hold all the candidates!" );
213
214 size_t i;
215 size_t shift = 0;
216 size_t currSize = std::min( uiFastCandNum, candCostList.size() );
217
218 while( shift < uiFastCandNum && shift < currSize && uiCost < candCostList[currSize - 1 - shift] )
219 {
220 shift++;
221 }
222
223 if( candModeList.size() >= uiFastCandNum && shift != 0 )
224 {
225 for( i = 1; i < shift; i++ )
226 {
227 candModeList[currSize - i] = candModeList[currSize - 1 - i];
228 candCostList[currSize - i] = candCostList[currSize - 1 - i];
229 }
230 candModeList[currSize - shift] = uiMode;
231 candCostList[currSize - shift] = uiCost;
232 if (iserttPos != nullptr)
233 {
234 *iserttPos = int(currSize - shift);
235 }
236 return 1;
237 }
238 else if( currSize < uiFastCandNum )
239 {
240 candModeList.insert( candModeList.end() - shift, uiMode );
241 candCostList.insert( candCostList.end() - shift, uiCost );
242 if (iserttPos != nullptr)
243 {
244 *iserttPos = int(candModeList.size() - shift - 1);
245 }
246 return 1;
247 }
248 if (iserttPos != nullptr)
249 {
250 *iserttPos = -1;
251 }
252 return 0;
253}
