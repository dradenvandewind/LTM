73void IbcHashMap::init(const int picWidth, const int picHeight)
74{
75 if (picWidth != m_picWidth || picHeight != m_picHeight)
76 {
77 destroy();
78 }
79
80 m_picWidth = picWidth;
81 m_picHeight = picHeight;
82 m_pos2Hash = new unsigned int*[m_picHeight];
83 m_pos2Hash[0] = new unsigned int[m_picWidth * m_picHeight];
84 for (int n = 1; n < m_picHeight; n++)
85 {
86 m_pos2Hash[n] = m_pos2Hash[n - 1] + m_picWidth;

171uint32_t IbcHashMap::xxComputeCrc32c16bit(uint32_t crc, const Pel pel)
172{
173 const void *buf = &pel;
174 const uint8_t *p = (const uint8_t *)buf;
175 size_t size = 2;
176
177 while (size--)
178 {
179 crc = crc32Table[(crc ^ *p++) & 0xff] ^ (crc >> 8);
180 }
181
182 return crc;
183}

187unsigned int IbcHashMap::xxCalcBlockHash(const Pel* pel, const int stride, const int width, const int height, unsigned int crc)
188{
189 for (int y = 0; y < height; y++)
190 {
191 for (int x = 0; x < width; x++)
192 {
193 crc = m_computeCrc32c(crc, pel[x]);
194 }
195 pel += stride;
196 }
197 return crc;
198}

200template<ChromaFormat chromaFormat>
201void IbcHashMap::xxBuildPicHashMap(const PelUnitBuf& pic)
202{
203 const int chromaScalingX = getChannelTypeScaleX(CHANNEL_TYPE_CHROMA, chromaFormat);
204 const int chromaScalingY = getChannelTypeScaleY(CHANNEL_TYPE_CHROMA, chromaFormat);
205 const int chromaMinBlkWidth = MIN_PU_SIZE >> chromaScalingX;
206 const int chromaMinBlkHeight = MIN_PU_SIZE >> chromaScalingY;
207 const Pel* pelY = NULL;
208 const Pel* pelCb = NULL;
209 const Pel* pelCr = NULL;
210
211 Position pos;
212 for (pos.y = 0; pos.y + MIN_PU_SIZE <= pic.Y().height; pos.y++)
213 {
214 // row pointer
215 pelY = pic.Y().bufAt(0, pos.y);
216 if (chromaFormat != CHROMA_400)
217 {
218 int chromaY = pos.y >> chromaScalingY;
219 pelCb = pic.Cb().bufAt(0, chromaY);
220 pelCr = pic.Cr().bufAt(0, chromaY);
221 }
222
223 for (pos.x = 0; pos.x + MIN_PU_SIZE <= pic.Y().width; pos.x++)
224 {
225 // 0x1FF is just an initial value
226 unsigned int hashValue = 0x1FF;
227
228 // luma part
229 hashValue = xxCalcBlockHash(&pelY[pos.x], pic.Y().stride, MIN_PU_SIZE, MIN_PU_SIZE, hashValue);
230
231 // chroma part
232 if (chromaFormat != CHROMA_400)
233 {
234 int chromaX = pos.x >> chromaScalingX;
235 hashValue = xxCalcBlockHash(&pelCb[chromaX], pic.Cb().stride, chromaMinBlkWidth, chromaMinBlkHeight, hashValue);
236 hashValue = xxCalcBlockHash(&pelCr[chromaX], pic.Cr().stride, chromaMinBlkWidth, chromaMinBlkHeight, hashValue);
237 }
238
239 // hash table
240 m_hash2Pos[hashValue].push_back(pos);
241 m_pos2Hash[pos.y][pos.x] = hashValue;
242 }
243 }
244}

270bool IbcHashMap::ibcHashMatch(const Area& lumaArea, std::vector<Position>& cand, const CodingStructure& cs, const int maxCand, const int searchRange4SmallBlk)
271{
272 cand.clear();
273
274 // find the block with least candidates
275 size_t minSize = MAX_UINT;
276 unsigned int targetHashOneBlock = 0;
277 Position targetBlockOffsetInCu(0, 0);
278 for (SizeType y = 0; y < lumaArea.height && minSize > 1; y += MIN_PU_SIZE)
279 {
280 for (SizeType x = 0; x < lumaArea.width && minSize > 1; x += MIN_PU_SIZE)
281 {
282 unsigned int hash = m_pos2Hash[lumaArea.pos().y + y][lumaArea.pos().x + x];
283 if (m_hash2Pos[hash].size() < minSize)
284 {
285 minSize = m_hash2Pos[hash].size();
286 targetHashOneBlock = hash;
287 targetBlockOffsetInCu.repositionTo(Position(x, y));
288 }
289 }
290 }
291
292 if (m_hash2Pos[targetHashOneBlock].size() > 1)
293 {
294 std::vector<Position>& candOneBlock = m_hash2Pos[targetHashOneBlock];
295
296 // check whether whole block match
297 for (std::vector<Position>::iterator refBlockPos = candOneBlock.begin(); refBlockPos != candOneBlock.end(); refBlockPos++)
298 {
299 Position topLeft = refBlockPos->offset(-targetBlockOffsetInCu.x, -targetBlockOffsetInCu.y);
300 Position bottomRight = topLeft.offset(lumaArea.width - 1, lumaArea.height - 1);
301 bool wholeBlockMatch = true;
302 if (lumaArea.width > MIN_PU_SIZE || lumaArea.height > MIN_PU_SIZE)
303 {
304 if (!cs.isDecomp(bottomRight, CHANNEL_TYPE_LUMA) || bottomRight.x >= m_picWidth || bottomRight.y >= m_picHeight || topLeft.x < 0 || topLeft.y < 0)
305 {
306 continue;
307 }
308 for (SizeType y = 0; y < lumaArea.height && wholeBlockMatch; y += MIN_PU_SIZE)
309 {
310 for (SizeType x = 0; x < lumaArea.width && wholeBlockMatch; x += MIN_PU_SIZE)
311 {
312 // whether the reference block and current block has the same hash
313 wholeBlockMatch &= (m_pos2Hash[lumaArea.pos().y + y][lumaArea.pos().x + x] == m_pos2Hash[topLeft.y + y][topLeft.x + x]);
314 }
315 }
316 }
317 else
318 {
319 CHECK(topLeft != *refBlockPos, "4x4 target block should not have offset!");
320 if (abs(topLeft.x - lumaArea.x) > searchRange4SmallBlk || abs(topLeft.y - lumaArea.y) > searchRange4SmallBlk || !cs.isDecomp(bottomRight, CHANNEL_TYPE_LUMA))
321 {
322 continue;
323 }
324 }
325 if (wholeBlockMatch)
326 {
327 cand.push_back(topLeft);
328 if (cand.size() > maxCand)
329 {
330 break;
331 }
332 }
333 }
334 }
335
336 return cand.size() > 0;
337}

339int IbcHashMap::getHashHitRatio(const Area& lumaArea)
340{
341 int maxX = std::min((int)(lumaArea.x + lumaArea.width), m_picWidth);
342 int maxY = std::min((int)(lumaArea.y + lumaArea.height), m_picHeight);
343 int hit = 0, total = 0;
344 for (int y = lumaArea.y; y < maxY; y += MIN_PU_SIZE)
345 {
346 for (int x = lumaArea.x; x < maxX; x += MIN_PU_SIZE)
347 {
348 const unsigned int hash = m_pos2Hash[y][x];
349 hit += (m_hash2Pos[hash].size() > 1);
350 total++;
351 }
352 }
353 return 100 * hit / total;
354}

356int IbcHashMap::calHashBlkMatchPerc(const Area& lumaArea)
357{
358 int maxX = std::min((int)(lumaArea.x + lumaArea.width), m_picWidth);
359 int maxY = std::min((int)(lumaArea.y + lumaArea.height), m_picHeight);
360 int maxUsage[100];
361 unsigned int mostSelHash[100];
362
363 static int numExcludedHashValue = 36;
364
365 for (int i = 0; i < numExcludedHashValue; i++)
366 {
367 maxUsage[i] = 0;
368 mostSelHash[i] = 0;
369 }
370
371 for (std::unordered_map<unsigned int, std::vector<Position>>::iterator it = m_hash2Pos.begin(); it != m_hash2Pos.end(); ++it)
372 {
373 unsigned int hash = it->first;
374 int usage = (int)it->second.size();
375 assert(usage == m_hash2Pos[hash].size());
376
377 int insertPos = -1;
378 for (insertPos = 0; insertPos < numExcludedHashValue; insertPos++)
379 {
380 if (usage > maxUsage[insertPos])
381 {
382 break;
383 }
384 }
385 assert(insertPos <= numExcludedHashValue);
386
387 if (insertPos < numExcludedHashValue)
388 {
389 for (int i = (numExcludedHashValue - 1); i >= (insertPos + 1); i--)
390 {
391 maxUsage[i] = maxUsage[i - 1];
392 mostSelHash[i] = mostSelHash[i - 1];
393 }
394 maxUsage[insertPos] = usage;
395 mostSelHash[insertPos] = hash;
396 }
397 }
398
399 int hit = 0, total = 0;
400 for (int y = lumaArea.y; y < maxY; y += MIN_PU_SIZE)
401 {
402 for (int x = lumaArea.x; x < maxX; x += MIN_PU_SIZE)
403 {
404 unsigned int hash = m_pos2Hash[y][x];
405
406 bool excludedHash = false;
407 for (int i = 0; i < numExcludedHashValue && !excludedHash; i++)
408 {
409 if (hash == mostSelHash[i])
410 {
411 excludedHash = true;
412 }
413 }
414
415 if (excludedHash)
416 {
417 continue;
418 }
419
420 hit += (m_hash2Pos[hash].size() > 1);
421 total++;
422 }
423 }
424
425 if (total == 0)
426 {
427 return 0;
428 }
429 else
430 {
431 return 100 * hit / total;
432 }
433}
