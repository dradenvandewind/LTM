74uint32_t GetBitCountFromMax(uint32_t maxValue) {
75 // Bit count is determined by ceil(log2(maxRepresentation);
76 // Because log2 is c99 which is not so widely available we can use standard log with a divide.
77 // log2(x) = logY(x) / logY(2)
78 //
79 // Alternatively we want to find the highest bit set in maxValue.
80 static const float logBase = log(2.0f);
81 float log2Value = log(static_cast<float>(maxValue)) / logBase;
82 return static_cast<uint32_t>(ceil(log2Value));
83}

126uint32_t uOffsetForNalUnitHeader(const uint8_t *nal) {
127 if (memcmp(nal, kNalHeader, kNalHeaderLength) == 0)
128 return kNalHeaderLength;
129 if (!nal[0] && memcmp(nal + 1, kNalHeader, kNalHeaderLength) == 0)
130 return kNalHeaderLength + 1;
131 return 0;

149bool BaseDecoderHEVC::ParseNalUnit(const uint8_t *nal, uint32_t nalLength) {
150 bool bSuccess = true;
151
152 // Account for AnnexB nal unit header ([0],0,0,1)
153 uint32_t offset = uOffsetForNalUnitHeader(nal);
154
155 // [forbidden-zero:1, nal_type:6, layer_id_top:1]
156 m_currentNalType = ((nal[offset] >> 1) & 0x3F);
157 // [layer_id:5, temporal_id_plus1:3]
158 m_currentTemporalId = nal[offset + 1] >> 5;
159
160 offset += 2;
161
162 BaseDecNalUnitType::Enum nalType = uFromHEVC(m_currentNalType);
163
164 // if (nalType != BaseDecNalUnitType::VPS)
165 {
166 // printf("BaseDecoderHEVC::ParseNalUnit [%d] --> [%s] nallenght [%d]\n", m_currentNalType, uToString(nalType),
167 // nalLengthIn);
168 nal += offset;
169 nalLength -= offset;
170
171 Unencapsulate(nal, nalLength, m_currentNalPayload);
172 m_currentBits = *m_currentNalPayload.data();
173 m_remainingBits = 8;
174 m_byteOffset = 0;
175
176 switch (nalType) {
177 case BaseDecNalUnitType::VPS:
178 bSuccess = ParseVPS();
179 break;
180 case BaseDecNalUnitType::PPS:
181 bSuccess = ParsePPS();
182 break;
183 case BaseDecNalUnitType::SPS:
184 bSuccess = ParseSPS();
185 break;
186 case BaseDecNalUnitType::Slice:
187 if ((bSuccess = ParseSliceHeader())) {
188 HandleSliceHeader();
189 HandlePictureOrderCount();
190 m_bFirstPicture = false;
191 }
192 break;
193 case BaseDecNalUnitType::EOS:
194 m_bFirstPicture = true;
195 break;
196 default:
197 break;
198 }
199 }
200 // else
201 // printf("BaseDecoderHEVC::ParseNalUnit [%d] --> [%s] nallenght [%d] ---->> SKIPPED\n", m_currentNalType,
202 // uToString(nalType), nalLengthIn);
203
204 return bSuccess;
205}

302// Section 8.3.1 of Standard
303void BaseDecoderHEVC::HandlePictureOrderCount() {
304 uint32_t picOrderCntMsb = 0;
305 uint32_t maxPicOrderCntLsb = 1 << m_activeSPS->m_log2MaxPicOrderCntLSB;
306 HEVCNalType::Enum nalType = static_cast<HEVCNalType::Enum>(m_currentNalType);
307
308 if (isIdr(nalType)) {
309 // Note 1
310 m_prevTid0Pic.m_picOrderCntLsb = 0;
311 m_prevTid0Pic.m_picOrderCntMsb = 0;
312 m_activeSlice.m_slicePicOrderCntLSB = 0;
313 }
314
315 uint32_t prevPicOrderCntLsb = m_prevTid0Pic.m_picOrderCntLsb;
316 uint32_t prevPicOrderCntMsb = m_prevTid0Pic.m_picOrderCntMsb;
317
318 // (8-1)
319 if (isIrap(nalType) && m_bNoRaslOutput) {
320 picOrderCntMsb = 0;
321 } else if (m_activeSlice.m_slicePicOrderCntLSB < prevPicOrderCntLsb &&
322 (prevPicOrderCntLsb - m_activeSlice.m_slicePicOrderCntLSB) >= (maxPicOrderCntLsb / 2)) {
323 picOrderCntMsb = prevPicOrderCntMsb + maxPicOrderCntLsb;
324 } else if (m_activeSlice.m_slicePicOrderCntLSB > prevPicOrderCntLsb &&
325 (m_activeSlice.m_slicePicOrderCntLSB - prevPicOrderCntLsb) > (maxPicOrderCntLsb / 2)) {
326 picOrderCntMsb = prevPicOrderCntMsb - maxPicOrderCntLsb;
327 } else
328 picOrderCntMsb = prevPicOrderCntMsb;
329
330 if (!isRasl(nalType) && !isRadl(nalType) && !isSubLayerNonReferencePicture(nalType) && m_currentTemporalId == 0) {
331 m_prevTid0Pic.m_picOrderCntLsb = m_activeSlice.m_slicePicOrderCntLSB;
332 m_prevTid0Pic.m_picOrderCntMsb = picOrderCntMsb;
333 }
334
335 // (8-2)
336 m_currentPictureOrderCount = picOrderCntMsb + m_activeSlice.m_slicePicOrderCntLSB;
337}

410bool BaseDecoderHEVC::ParseSPS() {
411 uint32_t vpsID = ReadBits(4);
412 // REPORT("%-40s - %8d", "sps_video_parameter_set_id", vpsID);
413 uint32_t maxSubLayersMinus1 = ReadBits(3);
414 // REPORT("%-40s - %8d", "sps_max_sub_layers_minus1", maxSubLayersMinus1);
415 bool bTemporalIDNesting = ReadFlag();
416 // REPORT("%-40s - %8d", "sps_temporal_id_nesting_flag", bTemporalIDNesting);
417
418 if (!ParseProfileTierLevels(true, maxSubLayersMinus1)) {
419 return false;
420 }
421
422 uint32_t spsID = ReadUE();
423 // REPORT("%-40s - %8d", "sps_seq_parameter_set_id", spsID);
424
425 if (spsID >= HEVCMaxSPSCount) {
426 INFO("spsID out of range %4d", spsID);
427 throw std::runtime_error("spsID out of range");
428 return false;
429 }
430
431 m_activeSPS = &m_sps[spsID];
432 m_activeSPS->m_vpsID = vpsID;
433 m_activeSPS->m_maxSubLayers = maxSubLayersMinus1 + 1;
434 m_activeSPS->m_bTemporalIDNesting = bTemporalIDNesting;
435 m_activeSPS->m_spsID = spsID;
436 m_activeSPS->m_chromaFormatIDC = ReadUE();
437 // REPORT("%-40s - %8d", "chroma_format_idc", m_activeSPS->m_chromaFormatIDC);
438
439 if (m_activeSPS->m_chromaFormatIDC == 3) {
440 m_activeSPS->m_bSeparateColourPlane = ReadFlag();
441 }
442
443 if (m_activeSPS->m_bSeparateColourPlane == false) {
444 m_activeSPS->m_chromaArrayType = m_activeSPS->m_chromaFormatIDC;
445 } else {
446 m_activeSPS->m_chromaArrayType = 0;
447 }
448
449 m_activeSPS->m_picWidthInLumaSamples = ReadUE();
450 // REPORT("%-40s - %8d", "pic_width_in_luma_samples", m_activeSPS->m_picWidthInLumaSamples);
451 m_activeSPS->m_picHeightInLumaSamples = ReadUE();
452 // REPORT("%-40s - %8d", "pic_height_in_luma_samples", m_activeSPS->m_picHeightInLumaSamples);
453 m_activeSPS->m_bConformanceWindow = ReadFlag();
454 // REPORT("%-40s - %8d", "conformance_window_flag", m_activeSPS->m_bConformanceWindow);
455
456 if (m_activeSPS->m_bConformanceWindow) {
457 m_activeSPS->m_confWinLeftOffset = ReadUE();
458 m_activeSPS->m_confWinRightOffset = ReadUE();
459 m_activeSPS->m_confWinTopOffset = ReadUE();
460 m_activeSPS->m_confWinBottomOffset = ReadUE();
461 }
462
463 m_activeSPS->m_bitDepthLuma = ReadUE() + 8;
464 m_activeSPS->m_bitDepthChroma = ReadUE() + 8;
465 m_activeSPS->m_log2MaxPicOrderCntLSB = ReadUE() + 4;
466 // REPORT("%-40s - %8d", "log2_max_pic_order_cnt_lsb_minus4", m_activeSPS->m_log2MaxPicOrderCntLSB - 4);
467
468 m_activeSPS->m_bSubLayerOrderingInfoPresent = ReadFlag();
469 // REPORT("%-40s - %8d", "sps_sub_layer_ordering_info_present_flag", m_activeSPS->m_bSubLayerOrderingInfoPresent);
470
471 for (uint32_t i = (m_activeSPS->m_bSubLayerOrderingInfoPresent ? 0 : maxSubLayersMinus1); i <= maxSubLayersMinus1; ++i) {
472 int sps_max_dec_pic_buffering_minus1 = ReadUE(); // max_dec_pic_buffering_minus1
473 // REPORT("%-40s - %8d - %8d", "sps_max_dec_pic_buffering_minus1", i, sps_max_dec_pic_buffering_minus1);
474 m_activeSPS->m_maxNumReorderPics[i] = ReadUE();
475 // REPORT("%-40s - %8d - %8d", "sps_max_num_reorder_pics", i, m_activeSPS->m_maxNumReorderPics[i]);
476 int max_latency_increase_plus1 = ReadUE(); // max_latency_increase_plus1
477 // REPORT("%-40s - %8d - %8d", "max_latency_increase_plus1", i, max_latency_increase_plus1);
478 }
479
480 m_activeSPS->m_log2MinLumaCodingBlockSize = ReadUE() + 3;
481 m_activeSPS->m_log2DiffMaxMinLumaCodingBlockSize = ReadUE();
482 m_activeSPS->m_log2MinLumaTransformBlockSize = ReadUE() + 2;
483 m_activeSPS->m_log2DiffMaxMinLumaTransformBlockSize = ReadUE();
484 m_activeSPS->m_maxTransformHierarchyDepthInter = ReadUE();
485 // REPORT("%-40s - %8d", "max_transform_hierarchy_depth_inter", m_activeSPS->m_maxTransformHierarchyDepthInter);
486 m_activeSPS->m_maxTransformHierarchyDepthIntra = ReadUE();
487 // REPORT("%-40s - %8d", "max_transform_hierarchy_depth_intra", m_activeSPS->m_maxTransformHierarchyDepthIntra);
488 m_activeSPS->m_bScalingListEnabled = ReadFlag();
489 // REPORT("%-40s - %8d", "scaling_list_enabled_flag", m_activeSPS->m_bScalingListEnabled);
490
491 if (m_activeSPS->m_bScalingListEnabled) {
492 m_activeSPS->m_bScalingListDataPresent = ReadFlag();
493
494 if (m_activeSPS->m_bScalingListDataPresent) {
495 // todo!
496 BaseDecoderHEVC::HandleScalingList();
497 // throw std::runtime_error("Unimplemented HEVC header parsing [scaling list]\n");
498 }
499 }
500
501 m_activeSPS->m_bAMPEnabled = ReadFlag();
502 m_activeSPS->m_bSampleAdaptiveOffsetEnabled = ReadFlag();
503 m_activeSPS->m_bPCMEnabled = ReadFlag();
504
505 if (m_activeSPS->m_bPCMEnabled) {
506 m_activeSPS->m_pcmSampleBitDepthLuma = ReadBits(4) + 1;
507 m_activeSPS->m_pcmSampleBitDepthChroma = ReadBits(4) + 1;
508 m_activeSPS->m_log2MinPCMLumaCodingBlockSize = ReadUE() + 3;
509 m_activeSPS->m_log2DiffMaxMinPCMLumaCodingBlockSize = ReadUE();
510 m_activeSPS->m_bPCMLoopFilterDisabled = ReadFlag();
511 }
512
513 m_activeSPS->m_numShortTermRefPicSets = ReadUE();
514 // REPORT("%-40s - %8d", "num_short_term_ref_pic_sets", m_activeSPS->m_numShortTermRefPicSets);
515
516 // debug0879
517 // Ignore remaining data for now.
518 return true;
519
520 for (uint32_t i = 0; i < m_activeSPS->m_numShortTermRefPicSets; ++i) {
521 ParseShortTermRefSet(i);
522 // printf("Unimplemented HEVC header pasing [short term ref pic sets]\n");
523 // throw std::runtime_error("Unimplemented HEVC header pasing [short term ref pic sets]\n");
524 }
525
526 m_activeSPS->m_bLongTermRefPicsPresent = ReadFlag();
527
528 if (m_activeSPS->m_bLongTermRefPicsPresent) {
529 m_activeSPS->m_numLongTermRefPics = ReadUE();
530
531 for (uint32_t i = 0; i < m_activeSPS->m_numLongTermRefPics; ++i) {
532 m_activeSPS->m_longTermRefPicPOCLSB[i] = ReadBits(m_activeSPS->m_log2MaxPicOrderCntLSB);
533 m_activeSPS->m_bLongTermUsedByCurrentPic[i] = ReadFlag();
534 }
535 }
536
537 m_activeSPS->m_bTemporalMVPEnabled = ReadFlag();
538 m_activeSPS->m_bStrongIntraSmoothingEnabled = ReadFlag();
539#if !defined(NO_VUIPARAMS)
540 if ((m_activeSPS->m_bVUIParametersPresent = ReadFlag())) {
541 HandleVUIParameters(); // Doesn't parse entire VUI
542 }
543#endif
544
545 // Ignore remaining data for now.
546 return true;
547}

549bool BaseDecoderHEVC::ParsePPS() {
550 uint32_t ppsID = ReadUE();
551
552 if (ppsID >= HEVCMaxPPSCount) {
553 INFO("ppsID out of range %4d", ppsID);
554 throw std::runtime_error("ppsID out of range");
555 return false;
556 }
557
558 m_activePPS = &m_pps[ppsID];
559 m_activePPS->m_ppsID = ppsID;
560 m_activePPS->m_spsID = ReadUE();
561 m_activePPS->m_bDependentSliceSegmentsEnabled = ReadFlag();
562 m_activePPS->m_bOutputFlagPresent = ReadFlag();

878bool BaseDecoderHEVC::ParseShortTermRefSet(uint32_t idx) {

888 // printf("Unimplemented HEVC header parsing [short-term ref]");

890
891 // printf("ParseShortTermRefSet:[%d] (bInterRefPicSetPrediction)\n", idx);
892
893 if (idx == m_activeSPS->m_numShortTermRefPicSets) {
894 deltaIdxMinus1 = ReadUE(); // delta_idx_minus1
895 }

897 uint32_t delta_rps_sign = ReadBits(1); // delta_rps_sign
898 // REPORT("%-40s - %8d", "delta_rps_sign", delta_rps_sign);

900 // REPORT("%-40s - %8d", "abs_delta_rps_minus1", abs_delta_rps_minus1);
901
902 uint32_t RefRpsIdx = idx - (deltaIdxMinus1 + 1);
903 int32_t deltaRps = (1 - 2 * delta_rps_sign) * (abs_delta_rps_minus1 + 1);
904
905 bool use_delta_flag[32] = {true};
906
907 // REPORT("**** %4d **** %4d ****", deltaRps, m_activeSPS->m_NumDeltaPocs[RefRpsIdx]);

909 for (uint32_t j = 0; j <= m_activeSPS->m_NumDeltaPocs[RefRpsIdx]; ++j) {

911 // REPORT("%-40s - %8d", "used_by_curr_pic_flag", m_activeSPS->m_used_by_curr_pic_flag[j]);

913 use_delta_flag[j] = ReadFlag();
914 // REPORT("%-40s - %8d", "use_delta_flag", use_delta_flag[j]);
915 }
916 }
917
918 uint32_t u = 0;
919 for (int32_t j = m_activeSPS->m_numPositivePics[RefRpsIdx] - 1; j >= 0; j--) {
920 int32_t dpoc = m_activeSPS->m_DeltaPocS1[RefRpsIdx][j] + deltaRps;
921 uint32_t kdx = m_activeSPS->m_numNegativePics[RefRpsIdx] + j;
922 if (dpoc < 0 && use_delta_flag[kdx]) {
923 m_activeSPS->m_DeltaPocS0[idx][u] = dpoc;
924 m_activeSPS->m_UsedByCurrPicS0[idx][u] = m_activeSPS->m_used_by_curr_pic_flag[kdx];

926 }
927 }

930 m_activeSPS->m_DeltaPocS0[idx][u] = deltaRps;
931 m_activeSPS->m_UsedByCurrPicS0[idx][u] = m_activeSPS->m_used_by_curr_pic_flag[fdx];
932 u++;
933 }
934 for (uint32_t j = 0; j < m_activeSPS->m_numNegativePics[RefRpsIdx]; ++j) {
935 int32_t dpoc = m_activeSPS->m_DeltaPocS0[RefRpsIdx][j] + deltaRps;
936 if (dpoc < 0 && use_delta_flag[j]) {
937 m_activeSPS->m_DeltaPocS0[idx][u] = dpoc;
938 m_activeSPS->m_UsedByCurrPicS0[idx][u] = m_activeSPS->m_used_by_curr_pic_flag[j];
939 u++;
940 }
941 }
942 m_activeSPS->m_numNegativePics[idx] = u;
943 //==========================================================================================
944 u = 0;

946 int32_t dpoc = m_activeSPS->m_DeltaPocS0[RefRpsIdx][j] + deltaRps;
947 if (dpoc > 0 && use_delta_flag[j]) {
948 m_activeSPS->m_DeltaPocS1[idx][u] = dpoc;
949 m_activeSPS->m_UsedByCurrPicS1[idx][u] = m_activeSPS->m_used_by_curr_pic_flag[j];
950 u++;
951 }

964 m_activeSPS->m_UsedByCurrPicS1[idx][u] = m_activeSPS->m_used_by_curr_pic_flag[kdx];

966 }
