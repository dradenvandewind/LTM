70static inline uint32_t getTotalSamples(const uint32_t width, const uint32_t height, const ChromaFormat format)
71{
72 const uint32_t samplesPerChannel = width * height;
73
74 switch (format)
75 {
76 case CHROMA_400: return samplesPerChannel; break;
77 case CHROMA_420: return (samplesPerChannel * 3) >> 1; break;
78 case CHROMA_422: return samplesPerChannel * 2; break;
79 case CHROMA_444: return samplesPerChannel * 3; break;
80 default:
81 EXIT( "ERROR: Unrecognised chroma format in getTotalSamples() " );
82 break;
83 }
84
85 return MAX_UINT;
86}

90static inline uint64_t getTotalFracBits(const uint32_t width, const uint32_t height, const ChromaFormat format, const int bitDepths[MAX_NUM_CHANNEL_TYPE])
91{
92 unsigned bitsPerSampleTimes2 = MAX_UINT;
93 switch (format)
94 {
95 case CHROMA_400: bitsPerSampleTimes2 = 2 * bitDepths[CHANNEL_TYPE_LUMA]; break;
96 case CHROMA_420: bitsPerSampleTimes2 = ( 2 * (bitDepths[CHANNEL_TYPE_LUMA]*2 + bitDepths[CHANNEL_TYPE_CHROMA]) ) >> 1; break;
97 case CHROMA_422: bitsPerSampleTimes2 = 2 * (bitDepths[CHANNEL_TYPE_LUMA] + bitDepths[CHANNEL_TYPE_CHROMA]); break;
98 case CHROMA_444: bitsPerSampleTimes2 = 2 * (bitDepths[CHANNEL_TYPE_LUMA] + 2*bitDepths[CHANNEL_TYPE_CHROMA]); break;
99 default:
100 EXIT( "ERROR: Unrecognised chroma format in getTotalFracBits() " );
101 break;
102 }
103 return uint64_t( width * height * bitsPerSampleTimes2 ) << ( SCALE_BITS - 1 );
104}

113static inline int getTransformShift(const int channelBitDepth, const Size size, const int maxLog2TrDynamicRange)
114{
115 return maxLog2TrDynamicRange - channelBitDepth - ( ( floorLog2(size.width) + floorLog2(size.height) ) >> 1 );
