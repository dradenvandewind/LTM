77bool AdaptiveLoopFilter::isCrossedByVirtualBoundaries( const CodingStructure& cs, const int xPos, const int yPos, const int width, const int height, bool& clipTop, bool& clipBottom, bool& clipLeft, bool& clipRight, int& numHorVirBndry, int& numVerVirBndry, int horVirBndryPos[], int verVirBndryPos[], int& rasterSliceAlfPad )
78{
79 clipTop = false; clipBottom = false; clipLeft = false; clipRight = false;
80 numHorVirBndry = 0; numVerVirBndry = 0;
81 const PPS* pps = cs.pps;
82 const PicHeader* picHeader = cs.picHeader;
83
84 if( picHeader->getVirtualBoundariesPresentFlag() )
85 {
86 for( int i = 0; i < picHeader->getNumHorVirtualBoundaries(); i++ )
87 {
88 if( picHeader->getVirtualBoundariesPosY(i) == yPos )
89 {
90 clipTop = true;
91 }
92 else if( picHeader->getVirtualBoundariesPosY(i) == yPos + height )
93 {
94 clipBottom = true;
95 }
96 else if( yPos < picHeader->getVirtualBoundariesPosY(i) && picHeader->getVirtualBoundariesPosY(i) < yPos + height )
97 {
98 horVirBndryPos[numHorVirBndry++] = picHeader->getVirtualBoundariesPosY(i);
99 }
100 }
101 for( int i = 0; i < picHeader->getNumVerVirtualBoundaries(); i++ )
102 {
103 if( picHeader->getVirtualBoundariesPosX(i) == xPos )
104 {
105 clipLeft = true;
106 }
107 else if( picHeader->getVirtualBoundariesPosX(i) == xPos + width )
108 {
109 clipRight = true;
110 }
111 else if( xPos < picHeader->getVirtualBoundariesPosX(i) && picHeader->getVirtualBoundariesPosX(i) < xPos + width )
112 {
113 verVirBndryPos[numVerVirBndry++] = picHeader->getVirtualBoundariesPosX(i);
114 }
115 }
116 }
117
118 const Slice& slice = *(cs.slice);
119 int ctuSize = slice.getSPS()->getCTUSize();
120 const Position currCtuPos(xPos, yPos);
121 const CodingUnit *currCtu = cs.getCU(currCtuPos, CHANNEL_TYPE_LUMA);
122 const SubPic& curSubPic = slice.getPPS()->getSubPicFromPos(currCtuPos);
123 bool loopFilterAcrossSubPicEnabledFlag = curSubPic.getloopFilterAcrossEnabledFlag();
124 //top
125 if (yPos >= ctuSize && clipTop == false)
126 {
127 const Position prevCtuPos(xPos, yPos - ctuSize);
128 const CodingUnit *prevCtu = cs.getCU(prevCtuPos, CHANNEL_TYPE_LUMA);
129 if ((!pps->getLoopFilterAcrossSlicesEnabledFlag() && !CU::isSameSlice(*currCtu, *prevCtu)) ||
130 (!pps->getLoopFilterAcrossTilesEnabledFlag() && !CU::isSameTile(*currCtu, *prevCtu))
131 || (!loopFilterAcrossSubPicEnabledFlag && !CU::isSameSubPic(*currCtu, *prevCtu))
132 )
133 {
134 clipTop = true;
135 }
136 }
137
138 //bottom
139 if (yPos + ctuSize < cs.pcv->lumaHeight && clipBottom == false)
140 {
141 const Position nextCtuPos(xPos, yPos + ctuSize);
142 const CodingUnit *nextCtu = cs.getCU(nextCtuPos, CHANNEL_TYPE_LUMA);
143 if ((!pps->getLoopFilterAcrossSlicesEnabledFlag() && !CU::isSameSlice(*currCtu, *nextCtu)) ||
144 (!pps->getLoopFilterAcrossTilesEnabledFlag() && !CU::isSameTile(*currCtu, *nextCtu))
145 || (!loopFilterAcrossSubPicEnabledFlag && !CU::isSameSubPic(*currCtu, *nextCtu))
146 )
147 {
148 clipBottom = true;
149 }
150 }
151
152 //left
153 if (xPos >= ctuSize && clipLeft == false)
154 {
155 const Position prevCtuPos(xPos - ctuSize, yPos);
156 const CodingUnit *prevCtu = cs.getCU(prevCtuPos, CHANNEL_TYPE_LUMA);
157 if ((!pps->getLoopFilterAcrossSlicesEnabledFlag() && !CU::isSameSlice(*currCtu, *prevCtu)) ||
158 (!pps->getLoopFilterAcrossTilesEnabledFlag() && !CU::isSameTile(*currCtu, *prevCtu))
159 || (!loopFilterAcrossSubPicEnabledFlag && !CU::isSameSubPic(*currCtu, *prevCtu))
160 )
161 {
162 clipLeft = true;
163 }
164 }
165
166 //right
167 if (xPos + ctuSize < cs.pcv->lumaWidth && clipRight == false)
168 {
169 const Position nextCtuPos(xPos + ctuSize, yPos);
170 const CodingUnit *nextCtu = cs.getCU(nextCtuPos, CHANNEL_TYPE_LUMA);
171 if ((!pps->getLoopFilterAcrossSlicesEnabledFlag() && !CU::isSameSlice(*currCtu, *nextCtu)) ||
172 (!pps->getLoopFilterAcrossTilesEnabledFlag() && !CU::isSameTile(*currCtu, *nextCtu))
173 || (!loopFilterAcrossSubPicEnabledFlag && !CU::isSameSubPic(*currCtu, *nextCtu))
174 )
175 {
176 clipRight = true;
177 }
178 }
179
180 rasterSliceAlfPad = 0;
181 if ( !clipTop && !clipLeft )
182 {
183 //top-left CTU
184 if ( xPos >= ctuSize && yPos >= ctuSize )
185 {
186 const Position prevCtuPos( xPos - ctuSize, yPos - ctuSize );
187 const CodingUnit *prevCtu = cs.getCU( prevCtuPos, CHANNEL_TYPE_LUMA );
188 if ( !pps->getLoopFilterAcrossSlicesEnabledFlag() && !CU::isSameSlice( *currCtu, *prevCtu ) )
189 {
190 rasterSliceAlfPad = 1;
191 }
192 }
193 }
194
195 if ( !clipBottom && !clipRight )
196 {
197 //bottom-right CTU
198 if ( xPos + ctuSize < cs.pcv->lumaWidth && yPos + ctuSize < cs.pcv->lumaHeight )
199 {
200 const Position nextCtuPos( xPos + ctuSize, yPos + ctuSize );
201 const CodingUnit *nextCtu = cs.getCU( nextCtuPos, CHANNEL_TYPE_LUMA );
202 if ( !pps->getLoopFilterAcrossSlicesEnabledFlag() && !CU::isSameSlice( *currCtu, *nextCtu ) )
203 {
204 rasterSliceAlfPad += 2;
205 }
206 }
207 }
208
209 return numHorVirBndry > 0 || numVerVirBndry > 0 || clipTop || clipBottom || clipLeft || clipRight || rasterSliceAlfPad;
210}

300void AdaptiveLoopFilter::applyCcAlfFilter(CodingStructure &cs, ComponentID compID, const PelBuf &dstBuf,
301 const PelUnitBuf &recYuvExt, uint8_t *filterControl,
302 const short filterSet[MAX_NUM_CC_ALF_FILTERS][MAX_NUM_CC_ALF_CHROMA_COEFF],
303 const int selectedFilterIdx)
304{
305 bool clipTop = false, clipBottom = false, clipLeft = false, clipRight = false;
306 int numHorVirBndry = 0, numVerVirBndry = 0;
307 int horVirBndryPos[] = { 0, 0, 0 };
308 int verVirBndryPos[] = { 0, 0, 0 };
309
310 int ctuIdx = 0;
311 for( int yPos = 0; yPos < m_picHeight; yPos += m_maxCUHeight )
312 {
313 for( int xPos = 0; xPos < m_picWidth; xPos += m_maxCUWidth )
314 {
315 int filterIdx =
316 (filterControl == nullptr)
317 ? selectedFilterIdx
318 : filterControl[(yPos >> cs.pcv->maxCUHeightLog2) * cs.pcv->widthInCtus + (xPos >> cs.pcv->maxCUWidthLog2)];
319 bool skipFiltering = (filterControl != nullptr && filterIdx == 0) ? true : false;
320 if (!skipFiltering)
321 {
322 if (filterControl != nullptr)
323 {
324 filterIdx--;
325 }
326
327 const int16_t *filterCoeff = filterSet[filterIdx];
328
329 const int width = (xPos + m_maxCUWidth > m_picWidth) ? (m_picWidth - xPos) : m_maxCUWidth;
330 const int height = (yPos + m_maxCUHeight > m_picHeight) ? (m_picHeight - yPos) : m_maxCUHeight;
331 const int chromaScaleX = getComponentScaleX(compID, m_chromaFormat);
332 const int chromaScaleY = getComponentScaleY(compID, m_chromaFormat);
333
334 int rasterSliceAlfPad = 0;
335 if (isCrossedByVirtualBoundaries(cs, xPos, yPos, width, height, clipTop, clipBottom, clipLeft, clipRight,
336 numHorVirBndry, numVerVirBndry, horVirBndryPos, verVirBndryPos,
337 rasterSliceAlfPad))
338 {
339 int yStart = yPos;
340 for (int i = 0; i <= numHorVirBndry; i++)
341 {
342 const int yEnd = i == numHorVirBndry ? yPos + height : horVirBndryPos[i];
343 const int h = yEnd - yStart;
344 const bool clipT = (i == 0 && clipTop) || (i > 0) || (yStart == 0);
345 const bool clipB = (i == numHorVirBndry && clipBottom) || (i < numHorVirBndry) || (yEnd == m_picHeight);
346 int xStart = xPos;
347 for (int j = 0; j <= numVerVirBndry; j++)
348 {
349 const int xEnd = j == numVerVirBndry ? xPos + width : verVirBndryPos[j];
350 const int w = xEnd - xStart;
351 const bool clipL = (j == 0 && clipLeft) || (j > 0) || (xStart == 0);
352 const bool clipR = (j == numVerVirBndry && clipRight) || (j < numVerVirBndry) || (xEnd == m_picWidth);
353 const int wBuf = w + (clipL ? 0 : MAX_ALF_PADDING_SIZE) + (clipR ? 0 : MAX_ALF_PADDING_SIZE);
354 const int hBuf = h + (clipT ? 0 : MAX_ALF_PADDING_SIZE) + (clipB ? 0 : MAX_ALF_PADDING_SIZE);
355 PelUnitBuf buf = m_tempBuf2.subBuf(UnitArea(cs.area.chromaFormat, Area(0, 0, wBuf, hBuf)));
356 buf.copyFrom(recYuvExt.subBuf(
357 UnitArea(cs.area.chromaFormat, Area(xStart - (clipL ? 0 : MAX_ALF_PADDING_SIZE),
358 yStart - (clipT ? 0 : MAX_ALF_PADDING_SIZE), wBuf, hBuf))));
359 // pad top-left unavailable samples for raster slice
360 if (xStart == xPos && yStart == yPos && (rasterSliceAlfPad & 1))
361 {
362 buf.padBorderPel(MAX_ALF_PADDING_SIZE, 1);
363 }
364
365 // pad bottom-right unavailable samples for raster slice
366 if (xEnd == xPos + width && yEnd == yPos + height && (rasterSliceAlfPad & 2))
367 {
368 buf.padBorderPel(MAX_ALF_PADDING_SIZE, 2);
369 }
370 buf.extendBorderPel(MAX_ALF_PADDING_SIZE);
371 buf = buf.subBuf(UnitArea(
372 cs.area.chromaFormat, Area(clipL ? 0 : MAX_ALF_PADDING_SIZE, clipT ? 0 : MAX_ALF_PADDING_SIZE, w, h)));
373
374 const Area blkSrc(0, 0, w, h);
375
376 const Area blkDst(xStart >> chromaScaleX, yStart >> chromaScaleY, w >> chromaScaleX, h >> chromaScaleY);
377 m_filterCcAlf(dstBuf, buf, blkDst, blkSrc, compID, filterCoeff, m_clpRngs, cs, m_alfVBLumaCTUHeight,
378 m_alfVBLumaPos);
379
380 xStart = xEnd;
381 }
382
383 yStart = yEnd;
384 }
385 }
386 else
387 {
388 const UnitArea area(m_chromaFormat, Area(xPos, yPos, width, height));
389
390 Area blkDst(xPos >> chromaScaleX, yPos >> chromaScaleY, width >> chromaScaleX, height >> chromaScaleY);
391 Area blkSrc(xPos, yPos, width, height);
392
393 m_filterCcAlf(dstBuf, recYuvExt, blkDst, blkSrc, compID, filterCoeff, m_clpRngs, cs, m_alfVBLumaCTUHeight,
394 m_alfVBLumaPos);
395 }
396 }
397 ctuIdx++;
398 }
399 }

402void AdaptiveLoopFilter::ALFProcess(CodingStructure& cs)
403{
404
405 // set clipping range
406 m_clpRngs = cs.slice->getClpRngs();
407
408 // set CTU enable flags
409 for( int compIdx = 0; compIdx < MAX_NUM_COMPONENT; compIdx++ )
410 {
411 m_ctuEnableFlag[compIdx] = cs.picture->getAlfCtuEnableFlag( compIdx );
412 m_ctuAlternative[compIdx] = cs.picture->getAlfCtuAlternativeData( compIdx );
413 }
414 short* alfCtuFilterIndex = nullptr;
415 uint32_t lastSliceIdx = 0xFFFFFFFF;
416
417 PelUnitBuf recYuv = cs.getRecoBuf();
418 m_tempBuf.copyFrom( recYuv );
419 PelUnitBuf tmpYuv = m_tempBuf.getBuf( cs.area );
420 tmpYuv.extendBorderPel( MAX_ALF_FILTER_LENGTH >> 1 );
421
422 const PreCalcValues& pcv = *cs.pcv;
423
424 int ctuIdx = 0;

653void AdaptiveLoopFilter::reconstructCoeff( AlfParam& alfParam, ChannelType channel, const bool isRdo, const bool isRedo )
654{
655 int factor = isRdo ? 0 : (1 << (m_NUM_BITS - 1));
656 AlfFilterType filterType = isLuma( channel ) ? ALF_FILTER_7 : ALF_FILTER_5;
657 int numClasses = isLuma( channel ) ? MAX_NUM_ALF_CLASSES : 1;
658 int numCoeff = filterType == ALF_FILTER_5 ? 7 : 13;
659 int numCoeffMinus1 = numCoeff - 1;
660 const int numAlts = isLuma( channel ) ? 1 : alfParam.numAlternativesChroma;
661
662 for( int altIdx = 0; altIdx < numAlts; ++ altIdx )
663 {
664 int numFilters = isLuma( channel ) ? alfParam.numLumaFilters : 1;
665 short* coeff = isLuma( channel ) ? alfParam.lumaCoeff : alfParam.chromaCoeff[altIdx];
666 Pel* clipp = isLuma( channel ) ? alfParam.lumaClipp : alfParam.chromaClipp[altIdx];
667
668 for( int filterIdx = 0; filterIdx < numFilters; filterIdx++ )
669 {
670 coeff[filterIdx* MAX_NUM_ALF_LUMA_COEFF + numCoeffMinus1] = factor;
671 }
672
673 if( isChroma( channel ) )
674 {
675 for( int coeffIdx = 0; coeffIdx < numCoeffMinus1; ++coeffIdx )
676 {
677 m_chromaCoeffFinal[altIdx][coeffIdx] = coeff[coeffIdx];
678 int clipIdx = alfParam.nonLinearFlag[channel] ? clipp[coeffIdx] : 0;
679 m_chromaClippFinal[altIdx][coeffIdx] = isRdo ? clipIdx : m_alfClippingValues[channel][clipIdx];
680 }
681 m_chromaCoeffFinal[altIdx][numCoeffMinus1] = factor;
682 m_chromaClippFinal[altIdx][numCoeffMinus1] = isRdo ? 0 : m_alfClippingValues[channel][0];
683 continue;
684 }
685 for( int classIdx = 0; classIdx < numClasses; classIdx++ )
686 {
687 int filterIdx = alfParam.filterCoeffDeltaIdx[classIdx];
688
689 CHECK(!(filterIdx >= 0 && filterIdx < alfParam.numLumaFilters), "Bad coeff delta idx in ALF");
690 for (int coeffIdx = 0; coeffIdx < numCoeffMinus1; ++coeffIdx)
691 {
692 m_coeffFinal[classIdx * MAX_NUM_ALF_LUMA_COEFF + coeffIdx] = coeff[filterIdx * MAX_NUM_ALF_LUMA_COEFF + coeffIdx];
693 }
694 m_coeffFinal[classIdx* MAX_NUM_ALF_LUMA_COEFF + numCoeffMinus1] = factor;
695 m_clippFinal[classIdx* MAX_NUM_ALF_LUMA_COEFF + numCoeffMinus1] = isRdo ? 0 : m_alfClippingValues[channel][0];
696 for( int coeffIdx = 0; coeffIdx < numCoeffMinus1; ++coeffIdx )
697 {
698 int clipIdx = alfParam.nonLinearFlag[channel] ? clipp[filterIdx * MAX_NUM_ALF_LUMA_COEFF + coeffIdx] : 0;
699 CHECK(!(clipIdx >= 0 && clipIdx < MaxAlfNumClippingValues), "Bad clip idx in ALF");
700 m_clippFinal[classIdx * MAX_NUM_ALF_LUMA_COEFF + coeffIdx] = isRdo ? clipIdx : m_alfClippingValues[channel][clipIdx];
701 }
702 m_clippFinal[classIdx* MAX_NUM_ALF_LUMA_COEFF + numCoeffMinus1] =
703 isRdo ? 0 :
704 m_alfClippingValues[channel][0];
705 }
706 }

718 m_chromaFormat = format;
719

721 m_numCTUsInHeight = ( m_picHeight / m_maxCUHeight ) + ( ( m_picHeight % m_maxCUHeight ) ? 1 : 0 );

730 m_alfVBLumaCTUHeight = m_maxCUHeight;

732
733 static_assert( AlfNumClippingValues[CHANNEL_TYPE_LUMA] > 0, "AlfNumClippingValues[CHANNEL_TYPE_LUMA] must be at least one" );
734 m_alfClippingValues[CHANNEL_TYPE_LUMA][0] = 1 << m_inputBitDepth[CHANNEL_TYPE_LUMA];

750 }

772 {
773 for (int classIdx = 0; classIdx < MAX_NUM_ALF_CLASSES; classIdx++)
774 {
775 int fixedFilterIdx = m_classToFilterMapping[filterSetIndex][classIdx];
776 for (int i = 0; i < MAX_NUM_ALF_LUMA_COEFF - 1; i++)

781 }
782 }
783 for (int i = 0; i < MAX_NUM_ALF_LUMA_COEFF * MAX_NUM_ALF_CLASSES; i++)

848void AdaptiveLoopFilter::deriveClassificationBlk(AlfClassifier **classifier, int **laplacian[NUM_DIRECTIONS],
849 const CPelBuf &srcLuma, const Area &blkDst, const Area &blk,
850 const int shift, const int vbCTUHeight, int vbPos)
851{
852 CHECK((vbCTUHeight & (vbCTUHeight - 1)) != 0, "vbCTUHeight must be a power of 2");
853
854 static const int th[16] = { 0, 1, 2, 2, 2, 2, 2, 3, 3, 3, 3, 3, 3, 3, 3, 4 };
855 const int stride = srcLuma.stride;
856 const Pel* src = srcLuma.buf;
857 const int maxActivity = 15;
858
859 int fl = 2;
860 int flP1 = fl + 1;
861 int fl2 = 2 * fl;
862
863 int mainDirection, secondaryDirection, dirTempHV, dirTempD;
864
865 int pixY;
866 int height = blk.height + fl2;
867 int width = blk.width + fl2;
868 int posX = blk.pos().x;
869 int posY = blk.pos().y;
870 int startHeight = posY - flP1;
871
872 for( int i = 0; i < height; i += 2 )
873 {
874 int yoffset = ( i + 1 + startHeight ) * stride - flP1;
875 const Pel *src0 = &src[yoffset - stride];
876 const Pel *src1 = &src[yoffset];
877 const Pel *src2 = &src[yoffset + stride];
878 const Pel *src3 = &src[yoffset + stride * 2];
879
880 const int y = blkDst.pos().y - 2 + i;
881 if (y > 0 && (y & (vbCTUHeight - 1)) == vbPos - 2)
882 {
883 src3 = &src[yoffset + stride];
884 }
885 else if (y > 0 && (y & (vbCTUHeight - 1)) == vbPos)
886 {
887 src0 = &src[yoffset];
888 }
889 int* pYver = laplacian[VER][i];
890 int* pYhor = laplacian[HOR][i];
891 int* pYdig0 = laplacian[DIAG0][i];
892 int* pYdig1 = laplacian[DIAG1][i];
893
894 for( int j = 0; j < width; j += 2 )
895 {
896 pixY = j + 1 + posX;
897 const Pel *pY = src1 + pixY;
898 const Pel* pYdown = src0 + pixY;
899 const Pel* pYup = src2 + pixY;
900 const Pel* pYup2 = src3 + pixY;
901
902 const Pel y0 = pY[0] << 1;
903 const Pel yup1 = pYup[1] << 1;
904
905 pYver[j] = abs( y0 - pYdown[0] - pYup[0] ) + abs( yup1 - pY[1] - pYup2[1] );
906 pYhor[j] = abs( y0 - pY[1] - pY[-1] ) + abs( yup1 - pYup[2] - pYup[0] );
907 pYdig0[j] = abs( y0 - pYdown[-1] - pYup[1] ) + abs( yup1 - pY[0] - pYup2[2] );
908 pYdig1[j] = abs( y0 - pYup[-1] - pYdown[1] ) + abs( yup1 - pYup2[0] - pY[2] );
909
910 if( j > 4 && ( j - 6 ) % 4 == 0 )
911 {
912 int jM6 = j - 6;
913 int jM4 = j - 4;
914 int jM2 = j - 2;
915 pYver[jM6] += pYver[jM4] + pYver[jM2] + pYver[j];
916 pYhor[jM6] += pYhor[jM4] + pYhor[jM2] + pYhor[j];
917 pYdig0[jM6] += pYdig0[jM4] + pYdig0[jM2] + pYdig0[j];
918 pYdig1[jM6] += pYdig1[jM4] + pYdig1[jM2] + pYdig1[j];
919 }
920 }
921 }
922
923 // classification block size
924 const int clsSizeY = 4;
925 const int clsSizeX = 4;
926
927 for( int i = 0; i < blk.height; i += clsSizeY )
928 {
929 int* pYver = laplacian[VER][i];
930 int* pYver2 = laplacian[VER][i + 2];
931 int* pYver4 = laplacian[VER][i + 4];
932 int* pYver6 = laplacian[VER][i + 6];
933
934 int* pYhor = laplacian[HOR][i];
935 int* pYhor2 = laplacian[HOR][i + 2];
936 int* pYhor4 = laplacian[HOR][i + 4];
937 int* pYhor6 = laplacian[HOR][i + 6];
938
939 int* pYdig0 = laplacian[DIAG0][i];
940 int* pYdig02 = laplacian[DIAG0][i + 2];
941 int* pYdig04 = laplacian[DIAG0][i + 4];
942 int* pYdig06 = laplacian[DIAG0][i + 6];
943
944 int* pYdig1 = laplacian[DIAG1][i];
945 int* pYdig12 = laplacian[DIAG1][i + 2];
946 int* pYdig14 = laplacian[DIAG1][i + 4];
947 int* pYdig16 = laplacian[DIAG1][i + 6];
948
949 for( int j = 0; j < blk.width; j += clsSizeX )
950 {
951 int sumV = 0; int sumH = 0; int sumD0 = 0; int sumD1 = 0;
952 if (((i + blkDst.pos().y) % vbCTUHeight) == (vbPos - 4))
953 {
954 sumV = pYver[j] + pYver2[j] + pYver4[j];
955 sumH = pYhor[j] + pYhor2[j] + pYhor4[j];
956 sumD0 = pYdig0[j] + pYdig02[j] + pYdig04[j];
957 sumD1 = pYdig1[j] + pYdig12[j] + pYdig14[j];
958 }
959 else if (((i + blkDst.pos().y) % vbCTUHeight) == vbPos)
960 {
961 sumV = pYver2[j] + pYver4[j] + pYver6[j];
962 sumH = pYhor2[j] + pYhor4[j] + pYhor6[j];
963 sumD0 = pYdig02[j] + pYdig04[j] + pYdig06[j];
964 sumD1 = pYdig12[j] + pYdig14[j] + pYdig16[j];
965 }
966 else
967 {
968 sumV = pYver[j] + pYver2[j] + pYver4[j] + pYver6[j];
969 sumH = pYhor[j] + pYhor2[j] + pYhor4[j] + pYhor6[j];
970 sumD0 = pYdig0[j] + pYdig02[j] + pYdig04[j] + pYdig06[j];
971 sumD1 = pYdig1[j] + pYdig12[j] + pYdig14[j] + pYdig16[j];
972 }
973
974 int tempAct = sumV + sumH;
975 int activity = 0;
976
977 const int y = (i + blkDst.pos().y) & (vbCTUHeight - 1);
978 if (y == vbPos - 4 || y == vbPos)
979 {
980 activity = (Pel)Clip3<int>(0, maxActivity, (tempAct * 96) >> shift);
981 }
982 else
983 {
984 activity = (Pel)Clip3<int>(0, maxActivity, (tempAct * 64) >> shift);
985 }
986 int classIdx = th[activity];
987
988 int hv1, hv0, d1, d0, hvd1, hvd0;
989
990 if( sumV > sumH )
991 {
992 hv1 = sumV;
993 hv0 = sumH;
994 dirTempHV = 1;
995 }
996 else
997 {
998 hv1 = sumH;
999 hv0 = sumV;
1000 dirTempHV = 3;
1001 }
1002 if( sumD0 > sumD1 )
1003 {
1004 d1 = sumD0;
1005 d0 = sumD1;
1006 dirTempD = 0;
1007 }
1008 else
1009 {
1010 d1 = sumD1;
1011 d0 = sumD0;
1012 dirTempD = 2;
1013 }
1014 if( (uint32_t)d1 * (uint32_t)hv0 > (uint32_t)hv1 * (uint32_t)d0 )
1015 {
1016 hvd1 = d1;
1017 hvd0 = d0;
1018 mainDirection = dirTempD;
1019 secondaryDirection = dirTempHV;
1020 }
1021 else
1022 {
1023 hvd1 = hv1;
1024 hvd0 = hv0;
1025 mainDirection = dirTempHV;
1026 secondaryDirection = dirTempD;
1027 }
1028
1029 int directionStrength = 0;
1030 if( hvd1 > 2 * hvd0 )
1031 {
1032 directionStrength = 1;
1033 }
1034 if( hvd1 * 2 > 9 * hvd0 )
1035 {
1036 directionStrength = 2;
1037 }
1038
1039 if( directionStrength )
1040 {
1041 classIdx += ( ( ( mainDirection & 0x1 ) << 1 ) + directionStrength ) * 5;
1042 }
1043
1044 static const int transposeTable[8] = { 0, 1, 0, 2, 2, 3, 1, 3 };
1045 int transposeIdx = transposeTable[mainDirection * 2 + ( secondaryDirection >> 1 )];
1046
1047 int yOffset = i + blkDst.pos().y;
1048 int xOffset = j + blkDst.pos().x;
1049
1050 AlfClassifier *cl0 = classifier[yOffset] + xOffset;
1051 AlfClassifier *cl1 = classifier[yOffset + 1] + xOffset;
1052 AlfClassifier *cl2 = classifier[yOffset + 2] + xOffset;
1053 AlfClassifier *cl3 = classifier[yOffset + 3] + xOffset;
1054 cl0[0] = cl0[1] = cl0[2] = cl0[3] = cl1[0] = cl1[1] = cl1[2] = cl1[3] = cl2[0] = cl2[1] = cl2[2] = cl2[3] = cl3[0] = cl3[1] = cl3[2] = cl3[3] = AlfClassifier( classIdx, transposeIdx );
1055 }
1056 }

1312 CHECK( startWidth % clsSizeX, "Wrong startWidth in filtering" );
