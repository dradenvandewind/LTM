59int BitstreamStatistic::Update(char *pcCategory, int iSize) {
60 if (strstr(pcCategory, "data_size") != NULL) {
61 miCounter[STAT_DATA_SIZE]++;
62 miBits[STAT_DATA_SIZE] += iSize;
63 } else if (strstr(pcCategory, "entropy_enab") != NULL) {
64 miCounter[STAT_ENTROPY_ENABLED]++;
65 miBits[STAT_ENTROPY_ENABLED] += iSize;

67 miCounter[STAT_RLE_ONLY]++;
68 miBits[STAT_RLE_ONLY] += iSize;
69 } else if (strstr(pcCategory, "entropy_code") != NULL) {
70 miCounter[STAT_ENTROPY_CODE]++;

73 miCounter[STAT_ENTROPY_SYMBOL]++;
74 miBits[STAT_ENTROPY_SYMBOL] += iSize;
