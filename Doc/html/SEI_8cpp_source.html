55SEIMessages getSeisByType(const SEIMessages &seiList, SEI::PayloadType seiType)
56{
57 SEIMessages result;
58
59 for (SEIMessages::const_iterator it=seiList.begin(); it!=seiList.end(); it++)
60 {
61 if ((*it)->payloadType() == seiType)
62 {
63 result.push_back(*it);
64 }
65 }
66 return result;
67}

69SEIMessages extractSeisByType(SEIMessages &seiList, SEI::PayloadType seiType)
70{
71 SEIMessages result;
72
73 SEIMessages::iterator it=seiList.begin();
74 while ( it!=seiList.end() )
75 {
76 if ((*it)->payloadType() == seiType)
77 {
78 result.push_back(*it);
79 it = seiList.erase(it);
80 }
81 else
82 {
83 it++;
84 }
85 }
86 return result;
87}
