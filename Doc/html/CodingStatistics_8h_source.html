482 void OutputBitStats()
483 {
484 const int64_t es = CODINGSTATISTICS_ENTROPYSCALE;
485
486 int64_t countTotal = 0;
487 int64_t classCounts[STATS__NUM_STATS];
488 std::fill_n( classCounts, ( size_t ) STATS__NUM_STATS, 0 );
489
490#if RExt__DECODER_DEBUG_TOOL_MAX_FRAME_STATS
491 SStat_max &max = GetStatisticMax();
492#endif
493
494 int64_t cr = 0; // CABAC remainder, which is added to "STATS__CABAC_INITIALISATION"
495 {
496 int64_t totalCABACbits = 0, roundedCABACbits = 0;
497 for( int i = STATS__NAL_UNIT_PACKING; i < STATS__NUM_STATS; i++ )
498 {
499 int64_t classCount = 0;
500
501 for( uint32_t c = 0; c < CODING_STATS_NUM_SUBCLASSES; c++ )
502 {
503 totalCABACbits += data.statistics[i][c].bits;
504 roundedCABACbits += data.statistics[i][c].bits / es;
505 classCount += data.statistics[i][c].count;
506 }
507
508 for( uint32_t c = 0; c < CODING_STATS_NUM_SUBCLASSES; c++ )
509 {
510 data.statistics[i][c].classCount = classCount;
511 }
512
513 classCounts[i] = classCount;
514 countTotal += classCount;
515 }
516 int64_t remainder = totalCABACbits - roundedCABACbits * es;
517 cr = ( remainder + es / 2 ) / es;
518 }
519
520 classCounts[0] = countTotal;
521
522 printf( "Note %s will be excluded from the total as it should be the sum of all the other entries (except for %s)\n", getName( STATS__NAL_UNIT_TOTAL_BODY ), getName( STATS__NAL_UNIT_PACKING ) );
523 printf( " %-45s- Width Height Type CABAC quote CABAC count CABAC Sum CABAC bits CABAC ratio EP Count EP Sum EP bits Total bits ( Total bytes)\n", "Decoder statistics" );
524
525 OutputDashedLine( "" );
526 SStat cabacTotalBits, epTotalBits;
527
528 cabacTotalBits.classCount = countTotal;
529 epTotalBits .classCount = countTotal;
530
531 SStat statTotals_cabac[CODING_STATS_NUM_SUBCLASSES];
532 SStat statTotals_ep [CODING_STATS_NUM_SUBCLASSES];
533
534 for( int i = 0; i < STATS__NUM_STATS; i++ )
535 {
536 SStat cabacSubTotal, epSubTotal;
537 cabacSubTotal.classCount = classCounts[i];
538 epSubTotal .classCount = classCounts[i];
