182inline void checkBPSyntaxElementLength(const SEIBufferingPeriod* bp1, const SEIBufferingPeriod* bp2)
183{
184 CHECK(bp1->m_initialCpbRemovalDelayLength != bp2->m_initialCpbRemovalDelayLength ||
185 bp1->m_cpbRemovalDelayLength != bp2->m_cpbRemovalDelayLength ||
186 bp1->m_dpbOutputDelayLength != bp2->m_dpbOutputDelayLength ||
187 bp1->m_duCpbRemovalDelayIncrementLength != bp2->m_duCpbRemovalDelayIncrementLength ||
188 bp1->m_dpbOutputDelayDuLength != bp2->m_dpbOutputDelayDuLength,
189 "All scalable-nested and non-scalable nested BP SEI messages in a CVS shall have the same value for "
190 "each of the syntax elements bp_cpb_initial_removal_delay_length_minus1, bp_cpb_removal_delay_length_minus1, "
191 "bp_dpb_output_delay_length_minus1, bp_du_cpb_removal_delay_increment_length_minus1, "
192 "and bp_dpb_output_delay_du_length_minus1");
193}
