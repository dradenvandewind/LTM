144void YUVDesc::Initialise(YUVFormat::Enum format, uint32_t width, uint32_t height) {
145 const YUVFormatInfo &info = kFormatInfo[format];
146
147 m_format = format;
148 m_planeCount = info.m_planeCount;
149 assert(m_planeCount <= kMaxNumPlanes);
150
151 m_bitDepth = info.m_bitDepth;
152 m_bitDepthContainer = (m_bitDepth + 7) & ~7;
153
154 const uint8_t byteDepth = GetByteDepth();
155
156 for (uint32_t planeIndex = 0; planeIndex < m_planeCount; ++planeIndex) {
157 const uint32_t planeWidth =
158 (planeIndex == 0 || (info.m_horizontalDownsample == 1)) ? width : ((width + 1) / info.m_horizontalDownsample);
159 const uint32_t planeHeight =
160 (planeIndex == 0 || (info.m_verticalDownsample == 1)) ? height : ((height + 1) / info.m_verticalDownsample);
161 m_planeDesc[planeIndex] = PlaneDesc(planeWidth, planeHeight, planeWidth, byteDepth * planeWidth);
162 }
163
164 m_byteSize = (width * height * info.m_sizeMult) / info.m_sizeDiv;
165}

223void YUVDesc::GetPlanePointers(uint8_t *memoryPtr, uint8_t **planePtrs, uint32_t *planePixelStrides) const {
224 // Assumes that the out arrays are the correct size.
225 for (uint32_t planeIndex = 0; planeIndex < m_planeCount; ++planeIndex) {
226 planePtrs[planeIndex] = memoryPtr;
227
228 if (planePixelStrides) {
229 planePixelStrides[planeIndex] = GetPlaneStridePixels(planeIndex);
230 }
231
232 memoryPtr += GetPlaneMemorySize(planeIndex);
233 }
234}
