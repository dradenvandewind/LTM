50EncodeBits_DD::EncodeBits_DD(EncodingMode enc_mode) : EncodeBits(enc_mode) {
51 // Do not allow AA for DD, only for DDS
52 assert(enc_mode != EncodingMode::AA);
53
54 encode_flags_.reset();
55 switch (enc_mode) {
56 case ENCODE_ALL:
57 for (size_t i = 0; i < encode_flags_.size(); ++i) {
58 encode_flags_.set(i, true);
59 }
60 break;
61 case Ax:
62 encode_flags_.set(0, true);
63 break;
64 case ENCODE_NONE:
65 encode_flags_.reset();
66 break;
67 default:
68 break;
69 }
70}

81EncodeBits_DDS::EncodeBits_DDS(EncodingMode enc_mode) : EncodeBits(enc_mode) {
82 encode_flags_.reset();
83 switch (enc_mode) {
84 case ENCODE_ALL:
85 for (size_t i = 0; i < encode_flags_.size(); ++i) {
86 encode_flags_[i] = true;
87 }
88 break;
89 case Ax:
90 encode_flags_[0] = encode_flags_[1] = encode_flags_[2] = encode_flags_[3] = true;
91 break;
92 case AA:
93 encode_flags_[0] = true;
94 break;
95 case NA:
96 encode_flags_[1] = encode_flags_[2] = encode_flags_[3] = true;
97 break;
98 case ENCODE_NONE:
99 encode_flags_.reset();
100 break;
101 default:
102 break;
103 }
104}

115LayerEncodeFlags::LayerEncodeFlags(TransformType transform_type, EncodingMode enc_mode) {
116 assert(transform_type == TransformType::TransformType_DD || transform_type == TransformType::TransformType_DDS);
117 if (transform_type == TransformType::TransformType_DD) {
118 encode_flags_ = std::unique_ptr<EncodeBits_DD>(new EncodeBits_DD(enc_mode));
119 } else if (transform_type == TransformType::TransformType_DDS) {
120 encode_flags_ = std::unique_ptr<EncodeBits_DDS>(new EncodeBits_DDS(enc_mode));
121 }
122}
