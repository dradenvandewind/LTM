24def md5sum(filename, blocksize=65536):
25 """
26 Compute MD5 checksum of named file.
27 """
28 hash = hashlib.md5()
29 with open(filename, "rb") as f:
30 for block in iter(lambda: f.read(blocksize), b""):
31 hash.update(block)
32 return hash.hexdigest()

42def parse_filename(name):
43 """
44 Extract vooya-ish image metadata from filename in form suitable for use as encoder parameters.
45 """
46 params = {}
47 for p in name.split("_"):
48 for n,r in IMAGE_NAME_PARAMS:
49 m = r.match(p)
50 if m:
51 params[n] = m.group(1)
52
53 if 'bit_depth' in params:
54 params['format'] += params.pop('bit_depth')
55
56 if 'format' in params and params['format'].startswith('4'):
57 params['format'] = 'yuv' + params['format']
58
59 return params
