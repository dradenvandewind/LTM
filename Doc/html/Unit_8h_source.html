56inline Position recalcPosition(const ChromaFormat _cf, const ComponentID srcCId, const ComponentID dstCId, const Position &pos)
57{
58 if( toChannelType( srcCId ) == toChannelType( dstCId ) )
59 {
60 return pos;
61 }
62 else if (isLuma(srcCId) && isChroma(dstCId))
63 {
64 return Position(pos.x >> getComponentScaleX(dstCId, _cf), pos.y >> getComponentScaleY(dstCId, _cf));
65 }
66 else
67 {
68 return Position(pos.x << getComponentScaleX(srcCId, _cf), pos.y << getComponentScaleY(srcCId, _cf));
69 }
70}

72inline Position recalcPosition( const ChromaFormat _cf, const ChannelType srcCHt, const ChannelType dstCHt, const Position &pos )
73{
74 if( srcCHt == dstCHt )
75 {
76 return pos;
77 }
78 else if( isLuma( srcCHt ) && isChroma( dstCHt ) )
79 {
80 return Position( pos.x >> getChannelTypeScaleX( dstCHt, _cf ), pos.y >> getChannelTypeScaleY( dstCHt, _cf ) );
81 }
82 else
83 {
84 return Position( pos.x << getChannelTypeScaleX( srcCHt, _cf ), pos.y << getChannelTypeScaleY( srcCHt, _cf ) );
85 }
86}

88inline Size recalcSize( const ChromaFormat _cf, const ComponentID srcCId, const ComponentID dstCId, const Size &size )
89{
90 if( toChannelType( srcCId ) == toChannelType( dstCId ) )
91 {
92 return size;
93 }
94 else if( isLuma( srcCId ) && isChroma( dstCId ) )
95 {
96 return Size( size.width >> getComponentScaleX( dstCId, _cf ), size.height >> getComponentScaleY( dstCId, _cf ) );
97 }
98 else
99 {
100 return Size( size.width << getComponentScaleX( srcCId, _cf ), size.height << getComponentScaleY( srcCId, _cf ) );
101 }
102}

249inline UnitArea clipArea(const UnitArea &area, const UnitArea &boundingBox)
250{
251 UnitArea ret(area.chromaFormat);
252
253 for (uint32_t i = 0; i < area.blocks.size(); i++)
254 {
255 ret.blocks.push_back(clipArea(area.blocks[i], boundingBox.blocks[i]));
256 }
257
258 return ret;
259}
