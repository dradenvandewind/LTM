72static bool compare_packet_timestamps(const lctm::Packet &l, const lctm::Packet &r) {
73 uint64_t lt = l.timestamp();
74 uint64_t rt = r.timestamp();
75 int64_t d = (rt - lt);
76 return d < 0;
77}
78
79struct BufferCompare {
80 bool operator()(const PssPacket &l, const PssPacket &r) { return compare_packet_timestamps(l.packet, r.packet); }
81};

152void BaseVideoDecoderCodecApi::push_au(const uint8_t *data, size_t data_size, uint64_t pts, bool is_base_idr, int iPictureType) {
153 if (data) {
154 // Temp. copy of AU
155 vector<uint8_t> b(data, data + data_size);
156
157 // Pick out enhancement data - may be removed - new size is returned
158 size_t new_size = scan_enhancement(b.data(), data_size, encapsulation_, base_, pts, is_base_idr,
159 [this](const Packet &pkt, const bool is_lcevc_idr) {
160 this->enhancement_queue_.push({pkt, is_lcevc_idr});
161 });
162
163 b.resize(new_size);
164
165 goReportStructure.miTimeStamp = (int)(pts);
166 goReportStructure.miPictureType = iPictureType;
167 goReportStructure.miBaseSize = (int)(new_size);
168 goReportStructure.miEnhancementSize = (int)(data_size - new_size);
169 goReportQueue.push(goReportStructure);
170
171 // Flush any previous buffer
172 if (!buffer_.empty()) {
173 push_es(buffer_.data(), buffer_.size(), pts);
174 }
175
176 buffer_.swap(b);
177 } else {
178 if (!buffer_.empty()) {
179 push_es(buffer_.data(), buffer_.size(), pts);
180 }
181
182 push_es(0, 0, pts);
183 }
184}

186void BaseVideoDecoderCodecApi::push_es(const uint8_t *data, size_t data_size, uint64_t pts) {
187 CHECK(context_);
188
189 // Push packet into codec
190 CodecError error = 0;
191 codec_->push_packet(context_, data, data_size, 0, data == nullptr, &error);
192
193 // Pull images while codec continues to return them
194 int32_t n = 0;
195 int8_t eos = 0;
196 do {
197 CodecImage codec_image = {0};
198 CodecError error = 0;
199 n = codec_->pull_image(context_, &codec_image, 0, &eos, &error);
200 if (n) {
201 // Get next PSS from queue
202 //
203 CHECK(!enhancement_queue_.empty());
204 PssPacket pss = enhancement_queue_.top();
205 enhancement_queue_.pop();
206
207 // Send to enhancement decoder
208 //
209 PacketView enhancement_view(pss.packet);
210 output_.push_base_enhancement(&codec_image, enhancement_view.data(), enhancement_view.size(), pss.packet.timestamp(),
211 pss.is_lcevc_idr);
212 }
213 } while (n);
214}
