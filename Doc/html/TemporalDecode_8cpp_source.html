46// Extract the single-bit-per-transform signalling as a mask
47//
48Surface TemporalExtractMask::process(const Surface &symbols) {
49 const auto src_symbols = symbols.view_as<int16_t>();
50
51#if defined __OPT_MATRIX__
52 auto dest = Surface::build_from<uint8_t>();
53 dest.reserve(src_symbols.width(), src_symbols.height());
54 for (unsigned y = 0; y < src_symbols.height(); ++y) {
55 const int16_t *__restrict psrc = src_symbols.data(0, y);
56 uint8_t *__restrict pdst = dest.data(0, y);
57 for (unsigned x = 0; x < src_symbols.width(); ++x) {
58 // dest.write(x, y, (src_symbols.read(x, y) & 1) ? TemporalType::TEMPORAL_INTR : TemporalType::TEMPORAL_PRED);
59 *pdst++ = ((*psrc++ & 1) ? TemporalType::TEMPORAL_INTR : TemporalType::TEMPORAL_PRED);
60 }
61 }
62 return dest.finish();
63#else
64 return Surface::build_from<uint8_t>()
65 .generate(src_symbols.width(), src_symbols.height(),
66 [&](unsigned x, unsigned y) -> uint8_t {
67 return (src_symbols.read(x, y) & 1) ? TemporalType::TEMPORAL_INTR : TemporalType::TEMPORAL_PRED;
68 })
69 .finish();
70#endif
71}

98// Combine the previous temporal buffer, the current residuals and the signalling mask to make a new temporal buffer
99//
100Surface TemporalUpdate::process(const Surface &temporal_plane, const Surface &residuals_plane, Surface &mask_plane,
101 unsigned transform_block_size, bool per_picture_intra, bool use_reduced_signalling) {
102 if (mask_plane.empty()) {
103 mask_plane = Surface::build_from<uint8_t>()
104 .generate(residuals_plane.width() / transform_block_size, residuals_plane.height() / transform_block_size,
105 [&](unsigned x, unsigned y) -> uint8_t { return TemporalType::TEMPORAL_INTR; })
106 .finish();
107 }
108 const auto src_temporal = temporal_plane.view_as<int16_t>();
109 const auto src_residuals = residuals_plane.view_as<int16_t>();
110 const auto src_mask = mask_plane.view_as<uint8_t>();
111
112 const unsigned tb_shift = log2(transform_block_size);
113 const unsigned d = 32 / transform_block_size;
114
115#if defined __OPT_MATRIX__
116 auto dest = Surface::build_from<int16_t>();
117 dest.reserve(src_temporal.width(), src_temporal.height());
118 for (unsigned y = 0; y < src_temporal.height(); ++y) {
119 const int16_t *__restrict psrc = src_residuals.data(0, y);
120 int16_t *__restrict pdst = dest.data(0, y);
121 for (unsigned x = 0; x < src_temporal.width(); ++x) {
122 const bool per_tile_intra =
123 use_reduced_signalling && (src_mask.read((x >> 5) * d, (y >> 5) * d) == TemporalType::TEMPORAL_INTR);
124 const bool per_block_intra = (src_mask.read(x >> tb_shift, y >> tb_shift) == TemporalType::TEMPORAL_INTR);
125 // int16_t out = 0;
126 if (per_picture_intra || per_tile_intra || per_block_intra) {
127 // out = src_residuals.read(x, y);
128 *pdst++ = *psrc++;
129 } else {
130 // out = src_residuals.read(x, y) + src_temporal.read(x, y);
131 *pdst++ = *psrc++ + src_temporal.read(x, y);
132 }
133 // dest.write(x, y, out);
134 }
135 }
136 return dest.finish();
137#else
138 return Surface::build_from<int16_t>()
139 .generate(src_temporal.width(), src_temporal.height(),
140 [&](unsigned x, unsigned y) -> int16_t {
141 const bool per_tile_intra =
142#if defined __OPT_DIVISION__
143 use_reduced_signalling && (src_mask.read((x >> 5) * d, (y >> 5) * d) == TemporalType::TEMPORAL_INTR);
144#else
145 use_reduced_signalling && (src_mask.read((x / 32) * d, (y / 32) * d) == TemporalType::TEMPORAL_INTR);
146#endif
147 const bool per_block_intra = (src_mask.read(x >> tb_shift, y >> tb_shift) == TemporalType::TEMPORAL_INTR);
148
149 if (per_picture_intra || per_tile_intra || per_block_intra) {
150 return src_residuals.read(x, y);
151 } else {
152 return src_residuals.read(x, y) + src_temporal.read(x, y);
153 }
154 })
155 .finish();
156#endif
157}

178Surface TemporalTileMap::process(Surface intra_symbols[MAX_NUM_LAYERS], Surface inter_symbols[MAX_NUM_LAYERS],
179 unsigned transform_block_size) {
180 const unsigned tile_size = 32;
181 const unsigned transforms_per_tile = tile_size / transform_block_size;
182
183 const unsigned level_width = intra_symbols[0].width() * transform_block_size;
184 const unsigned level_height = intra_symbols[0].height() * transform_block_size;
185
186 const unsigned tiles_wide = (level_width + tile_size - 1) / tile_size;
187 const unsigned tiles_high = (level_height + tile_size - 1) / tile_size;
188
189 // These all have S=2 as template param in inverse.
190 const std::vector<SurfaceView<int16_t>> srcs_intra =
191 (transform_block_size == 2) ? collect_surfaces_DD(intra_symbols) : collect_surfaces_DDS(intra_symbols);
192
193 const std::vector<SurfaceView<int16_t>> srcs_inter =
194 (transform_block_size == 2) ? collect_surfaces_DD(inter_symbols) : collect_surfaces_DDS(inter_symbols);
195
196 return Surface::build_from<uint8_t>()
197 .generate(tiles_wide, tiles_high,
198 [&](unsigned x, unsigned y) -> uint8_t {
199 // This can be based on 2x2 regions, even for DDS.
200 unsigned both_z = 0;
201 unsigned inter_nz = 0;
202 unsigned inter_z = 0;
203 unsigned intra_nz = 0;
204 unsigned intra_z = 0;
205 unsigned inter_accum = 0;
206 unsigned intra_accum = 0;
207 unsigned sav_mixed = 0;
208
209 for (unsigned tile_y = y * transforms_per_tile; tile_y < (y + 1) * transforms_per_tile; ++tile_y) {
210 if (tile_y >= intra_symbols[0].height())
211 continue;
212
213 for (unsigned tile_x = x * transforms_per_tile; tile_x < (x + 1) * transforms_per_tile; ++tile_x) {
214 if (tile_x >= intra_symbols[0].width())
215 continue;
216
217 // for each transform in each tile.
218 unsigned intra_sav = 0;
219 unsigned inter_sav = 0;
220
221 for (unsigned l = 0; l < transform_block_size * transform_block_size; ++l) {
222 intra_sav += std::abs(srcs_intra[l].read(tile_x, tile_y));
223 inter_sav += std::abs(srcs_inter[l].read(tile_x, tile_y));
224 }
225
226 if (inter_sav == 0 && intra_sav == 0) {
227 both_z++;
228 } else if (inter_sav == 0) {
229 inter_z++;
230 } else if (intra_sav == 0) {
231 intra_z++;
232 } else if (intra_sav < inter_sav) {
233 intra_nz++;
234 } else {
235 inter_nz++;
236 }
237
238 intra_accum += intra_sav;
239 inter_accum += inter_sav;
240 // sav_mixed += std::min(intra_sav, inter_sav);
241 sav_mixed += LCEVC_MIN(intra_sav, inter_sav);
242 }
243 }
244
245 unsigned num_temporals = intra_z + intra_nz + inter_z + inter_nz;
246 unsigned intra_pct = (100 * (intra_z + intra_nz)) / (num_temporals + 1);
247 unsigned inter_pct = (100 * inter_z) / (num_temporals + 1);
248
249 unsigned intra_accum_75 = intra_accum - (inter_accum >> 2);
250 unsigned inter_accum_25 = inter_accum >> 2;
251
252 if (intra_pct > 38 && inter_pct < 20 && (intra_accum_75 <= sav_mixed || inter_accum_25 > sav_mixed)) {
253 return TemporalType::TEMPORAL_INTR;
254 }
255
256 return TemporalType::TEMPORAL_PRED;
257 })
258 .finish();
259}

261// Perform sweep updating temporal mask based upon tile type with reduced signalling.
262//
263Surface TemporalTileIntraSignal::process(const Surface &temporal_tile_map, const Surface &mask_plane,
264 unsigned transform_block_size) {
265 const unsigned transforms_per_tile = 32 / transform_block_size;
266 const auto src_tile_map = temporal_tile_map.view_as<uint8_t>();
267 const auto src_mask_plane = mask_plane.view_as<uint8_t>();
268
269 return Surface::build_from<uint8_t>()
270 .generate(mask_plane.width(), mask_plane.height(),
271 [&](unsigned x, unsigned y) -> uint8_t {
272 const unsigned tile_x = x / transforms_per_tile;
273 const unsigned tile_y = y / transforms_per_tile;
274
275 const bool tile_intra = (src_tile_map.read(tile_x, tile_y) == TemporalType::TEMPORAL_INTR);
276 const bool tile_start = ((x % transforms_per_tile) == 0) && ((y % transforms_per_tile) == 0);
277
278 if (tile_intra)
279 return TemporalType::TEMPORAL_INTR;
280 else if (tile_start)
281 return TemporalType::TEMPORAL_PRED;
282
283 // Pass through the mask based on cost
284 return src_mask_plane.read(x, y);
285 })
286 .finish();
287}

301 .finish();
302 }
303 const auto src_tile_map = temporal_tile_map.view_as<uint8_t>();
304 const auto src_syms = temporal_signal_syms.view_as<int16_t>();
305
306 return Surface::build_from<int16_t>()
307 .generate(temporal_signal_syms.width(), temporal_signal_syms.height(),

310 const unsigned tile_y = y / transforms_per_tile;

324// For each block in map - zero prediction if corresponding block is set to INTRA in map
325//
326Surface ApplyTemporalMap::process(const Surface &src_plane, Surface &map_plane, unsigned transform_block_size) {
327
328 // Scale the residual map to cover source image
329 //
330 const unsigned block_w = tile_size(map_plane.width(), src_plane.width());
331 const unsigned block_h = tile_size(map_plane.height(), src_plane.height());
332 const unsigned shift_bw = log2(block_w);
333 const unsigned shift_bh = log2(block_h);
334
335 CHECK(src_plane.width() == map_plane.width() * block_w);
336 CHECK(src_plane.height() == map_plane.height() * block_h);
337
338 const auto src = src_plane.view_as<int16_t>();
339 const auto map = map_plane.view_as<uint8_t>();
340
341#if defined __OPT_MATRIX__
342 auto dest = Surface::build_from<int16_t>();
343 dest.reserve(src.width(), src.height());
344 for (unsigned y = 0; y < src.height(); ++y) {
345 const int16_t *__restrict psrc = src.data(0, y);
346 int16_t *__restrict pdst = dest.data(0, y);
347 for (unsigned x = 0; x < src.width(); ++x) {
348 // apply the "buffer refresh" per block
349 if (map.read(x >> shift_bw, y >> shift_bh) == TemporalType::TEMPORAL_INTR)
350 // dest.write(x, y, 0);
351 *pdst++ = 0;
352 else
353 // dest.write(x, y, src.read(x, y));
354 *pdst++ = *psrc;
355 psrc++;
356 }
357 }
358 return dest.finish();
359#else
360 return Surface::build_from<int16_t>()
361 .generate(src.width(), src.height(),
362 [&](unsigned x, unsigned y) -> int16_t {
363 // apply the "buffer refresh" per block
364 if (map.read(x >> shift_bw, y >> shift_bh) == TemporalType::TEMPORAL_INTR)
365 return 0;
366 else
367 return src.read(x, y);
368 })
369 .finish();
370#endif
371}

373// Filter out the embedded user_data
374//
375Surface UserDataClear::process(const Surface &symbols, UserDataMode user_data) {
376 unsigned size;
377 switch (user_data) {
378 case lctm::UserData_2bits:
379 size = 2;
380 break;
381 case lctm::UserData_6bits:
382 size = 6;
383 break;
384 default:
385 CHECK(0);
386 break;
387 }
388 const auto src_symbols = symbols.view_as<int16_t>();
389
390#if defined __OPT_MATRIX__
391 auto dest = Surface::build_from<int16_t>();
392 dest.reserve(src_symbols.width(), src_symbols.height());
393 for (unsigned y = 0; y < src_symbols.height(); ++y) {
394 const int16_t *__restrict psrc = src_symbols.data(0, y);
395 int16_t *__restrict pdst = dest.data(0, y);
396 for (unsigned x = 0; x < src_symbols.width(); ++x) {
397 uint16_t value = *psrc++;
398 value >>= size;
399 bool sign = (value & 0x01) == 0x00 ? false : true;
400 value >>= 1;
401 *pdst++ = (int16_t)(sign ? (-value) : (value));
402 }
403 }
404 return dest.finish();
405#else
406 return Surface::build_from<int16_t>()
407 .generate(src_symbols.width(), src_symbols.height(),
408 [&](unsigned x, unsigned y) -> int16_t {
409 uint16_t value = src_symbols.read(x, y);
410 value >>= size;
411 bool sign = (value & 0x01) == 0x00 ? false : true;
412 value >>= 1;
413 return (int16_t)(sign ? (-value) : (value));
414 })
415 .finish();
416#endif
417}
