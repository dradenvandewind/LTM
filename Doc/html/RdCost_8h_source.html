210 unsigned int getBitsMultiplePreds(int x, int y, bool useIMV)
211 {
212 int rmvH[2];
213 int rmvV[2];
214 rmvH[0] = x - m_bvPredictors[0].getHor();
215 rmvH[1] = x - m_bvPredictors[1].getHor();
216
217 rmvV[0] = y - m_bvPredictors[0].getVer();
218 rmvV[1] = y - m_bvPredictors[1].getVer();
219 int absCand[2];
220 absCand[0] = abs(rmvH[0]) + abs(rmvV[0]);
221 absCand[1] = abs(rmvH[1]) + abs(rmvV[1]);
222
223 int rmvHQP[2];
224 int rmvVQP[2];
225 if (x % 4 == 0 && y % 4 == 0 && useIMV)
226 {
227 int imvShift = 2;
228 int offset = 1 << (imvShift - 1);
229
230 rmvHQP[0] = (x >> 2) - ((m_bvPredictors[0].getHor() + offset) >> 2);
231 rmvHQP[1] = (x >> 2) - ((m_bvPredictors[1].getHor() + offset) >> 2);
232 rmvVQP[0] = (y >> 2) - ((m_bvPredictors[0].getVer() + offset) >> 2);
233 rmvVQP[1] = (y >> 2) - ((m_bvPredictors[1].getVer() + offset) >> 2);
234
235 int absCandQP[2];
236 absCandQP[0] = abs(rmvHQP[0]) + abs(rmvVQP[0]);
237 absCandQP[1] = abs(rmvHQP[1]) + abs(rmvVQP[1]);
238 unsigned int candBits0QP, candBits1QP;
239 if (absCand[0] < absCand[1])
240 {
241 unsigned int candBits0 = getIComponentBits(rmvH[0]) + getIComponentBits(rmvV[0]);
242 if (absCandQP[0] < absCandQP[1])
243 {
244 candBits0QP = getIComponentBits(rmvHQP[0]) + getIComponentBits(rmvVQP[0]);
245 return candBits0QP <candBits0 ? candBits0QP : candBits0;
246 }
247 else
248 {
249 candBits1QP = getIComponentBits(rmvHQP[1]) + getIComponentBits(rmvVQP[1]);
250 return candBits1QP < candBits0 ? candBits1QP : candBits0;
251 }
252 }
253 else
254 {
255 unsigned int candBits1 = getIComponentBits(rmvH[1]) + getIComponentBits(rmvV[1]);
256 if (absCandQP[0] < absCandQP[1])
257 {
258 candBits0QP = getIComponentBits(rmvHQP[0]) + getIComponentBits(rmvVQP[0]);
259 return candBits0QP < candBits1 ? candBits0QP : candBits1;
260 }
261 else
262 {
263 candBits1QP = getIComponentBits(rmvHQP[1]) + getIComponentBits(rmvVQP[1]);
264 return candBits1QP < candBits1 ? candBits1QP : candBits1;
265 }
266 }
267 }
268 else
269
270 {
271 if (absCand[0] < absCand[1])
272 {
273 return getIComponentBits(rmvH[0]) + getIComponentBits(rmvV[0]);
274 }
275 else
276 {
277 return getIComponentBits(rmvH[1]) + getIComponentBits(rmvV[1]);
278 }
279 }
280 }

282 unsigned int getIComponentBits(int val)
283 {
284 if (!val) return 1;
285
286 unsigned int length = 1;
287 unsigned int temp = (val <= 0) ? (-val << 1) + 1 : (val << 1);
288
289 while (1 != temp)
290 {
291 temp >>= 1;
292 length += 2;
293 }
294
295 return length;
296 }

303 static uint32_t xGetExpGolombNumberOfBits( int iVal )
304 {
305 CHECKD( iVal == std::numeric_limits<int>::min(), "Wrong value" );
306 unsigned uiLength2 = 1, uiTemp2 = ( iVal <= 0 ) ? ( unsigned( -iVal ) << 1 ) + 1 : unsigned( iVal << 1 );
307
308 while( uiTemp2 > MAX_CU_SIZE )
309 {
310 uiLength2 += ( MAX_CU_DEPTH << 1 );
311 uiTemp2 >>= MAX_CU_DEPTH;
312 }
313
314 return uiLength2 + ( floorLog2(uiTemp2) << 1 );
315 }
