46def hash_md5(filename, blocksize=65536):
47 """
48 Compute md5 hash of named file.
49 """
50 hash = hashlib.md5()
51 with open(filename, "rb") as f:
52 for block in iter(lambda: f.read(blocksize), b""):
53 hash.update(block)
54 return hash.hexdigest()
