62void MatrixIntraPrediction::prepareInputForPred(const CPelBuf &pSrc, const Area &block, const int bitDepth,
63 const ComponentID compId)
64{
65 m_component = compId;
66
67 // Step 1: Save block size and calculate dependent values
68 initPredBlockParams(block);
69
70 // Step 2: Get the input data (left and top reference samples)
71 m_refSamplesTop.resize(block.width);
72 for (int x = 0; x < block.width; x++)
73 {
74 m_refSamplesTop[x] = pSrc.at(x + 1, 0);
75 }
76
77 m_refSamplesLeft.resize(block.height);
78 for (int y = 0; y < block.height; y++)
79 {
80 m_refSamplesLeft[y] = pSrc.at(y + 1, 1);
81 }
82
83 // Step 3: Compute the reduced boundary via Haar-downsampling (input for the prediction)
84 const int inputSize = 2 * m_reducedBdrySize;
85 m_reducedBoundary .resize( inputSize );
86 m_reducedBoundaryTransposed.resize( inputSize );
87
88 int* const topReduced = m_reducedBoundary.data();
89 boundaryDownsampling1D( topReduced, m_refSamplesTop.data(), block.width, m_reducedBdrySize );
90
91 int* const leftReduced = m_reducedBoundary.data() + m_reducedBdrySize;
92 boundaryDownsampling1D( leftReduced, m_refSamplesLeft.data(), block.height, m_reducedBdrySize );
93
94 int* const leftReducedTransposed = m_reducedBoundaryTransposed.data();
95 int* const topReducedTransposed = m_reducedBoundaryTransposed.data() + m_reducedBdrySize;
96 for( int x = 0; x < m_reducedBdrySize; x++ )
97 {
98 topReducedTransposed[x] = topReduced[x];
99 }
100 for( int y = 0; y < m_reducedBdrySize; y++ )
101 {
102 leftReducedTransposed[y] = leftReduced[y];
103 }
104
105 // Step 4: Rebase the reduced boundary
106
107 m_inputOffset = m_reducedBoundary[0];
108 m_inputOffsetTransp = m_reducedBoundaryTransposed[0];
109
110 const bool hasFirstCol = (m_sizeId < 2);
111 m_reducedBoundary [0] = hasFirstCol ? ((1 << (bitDepth - 1)) - m_inputOffset ) : 0; // first column of matrix not needed for large blocks
112 m_reducedBoundaryTransposed[0] = hasFirstCol ? ((1 << (bitDepth - 1)) - m_inputOffsetTransp) : 0;
113 for (int i = 1; i < inputSize; i++)
114 {
115 m_reducedBoundary [i] -= m_inputOffset;
116 m_reducedBoundaryTransposed[i] -= m_inputOffsetTransp;
117 }
118}

120void MatrixIntraPrediction::predBlock(int *const result, const int modeIdx, const bool transpose, const int bitDepth,
121 const ComponentID compId)
122{
123 CHECK(m_component != compId, "Boundary has not been prepared for this component.");
124
125 const bool needUpsampling = ( m_upsmpFactorHor > 1 ) || ( m_upsmpFactorVer > 1 );
126
127 const uint8_t* matrix = getMatrixData(modeIdx);
128
129 static_vector<int, MIP_MAX_REDUCED_OUTPUT_SAMPLES> bufReducedPred( m_reducedPredSize * m_reducedPredSize );
130 int* const reducedPred = needUpsampling ? bufReducedPred.data() : result;
131 const int* const reducedBoundary = transpose ? m_reducedBoundaryTransposed.data() : m_reducedBoundary.data();
132 computeReducedPred(reducedPred, reducedBoundary, matrix, transpose, bitDepth);
133 if( needUpsampling )
134 {
135 predictionUpsampling( result, reducedPred );
136 }
137}

140void MatrixIntraPrediction::initPredBlockParams(const Size& block)
141{
142 m_blockSize = block;
143 // init size index
144 m_sizeId = getMipSizeId( m_blockSize );
145
146 // init reduced boundary size
147 m_reducedBdrySize = (m_sizeId == 0) ? 2 : 4;
148
149 // init reduced prediction size
150 m_reducedPredSize = ( m_sizeId < 2 ) ? 4 : 8;
151
152
153 // init upsampling factors
154 m_upsmpFactorHor = m_blockSize.width / m_reducedPredSize;
155 m_upsmpFactorVer = m_blockSize.height / m_reducedPredSize;
156
157 CHECKD( (m_upsmpFactorHor < 1) || ((m_upsmpFactorHor & (m_upsmpFactorHor - 1)) != 0), "Need power of two horizontal upsampling factor." );
158 CHECKD( (m_upsmpFactorVer < 1) || ((m_upsmpFactorVer & (m_upsmpFactorVer - 1)) != 0), "Need power of two vertical upsampling factor." );
159}

163void MatrixIntraPrediction::boundaryDownsampling1D(int* reducedDst, const int* const fullSrc, const SizeType srcLen, const SizeType dstLen)
164{
165 if (dstLen < srcLen)
166 {
167 // Create reduced boundary by downsampling
168 const SizeType downsmpFactor = srcLen / dstLen;
169 const int log2DownsmpFactor = floorLog2(downsmpFactor);
170 const int roundingOffset = (1 << (log2DownsmpFactor - 1));
171
172 SizeType srcIdx = 0;
173 for( SizeType dstIdx = 0; dstIdx < dstLen; dstIdx++ )
174 {
175 int sum = 0;
176 for( int k = 0; k < downsmpFactor; k++ )
177 {
178 sum += fullSrc[srcIdx++];
179 }
180 reducedDst[dstIdx] = (sum + roundingOffset) >> log2DownsmpFactor;
181 }
182 }
183 else
184 {
185 // Copy boundary if no downsampling is needed
186 for (SizeType i = 0; i < dstLen; ++i)
187 {
188 reducedDst[i] = fullSrc[i];
189 }
190 }
191}

194void MatrixIntraPrediction::predictionUpsampling1D(int* const dst, const int* const src, const int* const bndry,
195 const SizeType srcSizeUpsmpDim, const SizeType srcSizeOrthDim,
196 const SizeType srcStep, const SizeType srcStride,
197 const SizeType dstStep, const SizeType dstStride,
198 const SizeType bndryStep,
199 const unsigned int upsmpFactor )
200{
201 const int log2UpsmpFactor = floorLog2( upsmpFactor );
202 CHECKD( upsmpFactor <= 1, "Upsampling factor must be at least 2." );
203 const int roundingOffset = 1 << (log2UpsmpFactor - 1);
204
205 SizeType idxOrthDim = 0;
206 const int* srcLine = src;
207 int* dstLine = dst;
208 const int* bndryLine = bndry + bndryStep - 1;
209 while( idxOrthDim < srcSizeOrthDim )
210 {
211 SizeType idxUpsmpDim = 0;
212 const int* before = bndryLine;
213 const int* behind = srcLine;
214 int* currDst = dstLine;
215 while( idxUpsmpDim < srcSizeUpsmpDim )
216 {
217 SizeType pos = 1;
218 int scaledBefore = ( *before ) << log2UpsmpFactor;
219 int scaledBehind = 0;
220 while( pos <= upsmpFactor )
221 {
222 scaledBefore -= *before;
223 scaledBehind += *behind;
224 *currDst = (scaledBefore + scaledBehind + roundingOffset) >> log2UpsmpFactor;
225
226 pos++;
227 currDst += dstStep;
228 }
229
230 idxUpsmpDim++;
231 before = behind;
232 behind += srcStep;
233 }
234
235 idxOrthDim++;
236 srcLine += srcStride;
237 dstLine += dstStride;
238 bndryLine += bndryStep;
239 }
240}

243void MatrixIntraPrediction::predictionUpsampling( int* const dst, const int* const src ) const
244{
245 const int* verSrc = src;
246 SizeType verSrcStep = m_blockSize.width;
247
248 if( m_upsmpFactorHor > 1 )
249 {
250 int* const horDst = dst + (m_upsmpFactorVer - 1) * m_blockSize.width;
251 verSrc = horDst;
252 verSrcStep *= m_upsmpFactorVer;
253
254 predictionUpsampling1D( horDst, src, m_refSamplesLeft.data(),
255 m_reducedPredSize, m_reducedPredSize,
256 1, m_reducedPredSize, 1, verSrcStep,
257 m_upsmpFactorVer, m_upsmpFactorHor );
258 }
259
260 if( m_upsmpFactorVer > 1 )
261 {
262 predictionUpsampling1D( dst, verSrc, m_refSamplesTop.data(),
263 m_reducedPredSize, m_blockSize.width,
264 verSrcStep, 1, m_blockSize.width, 1,
265 1, m_upsmpFactorVer );
266 }

283void MatrixIntraPrediction::computeReducedPred( int*const result, const int* const input,
284 const uint8_t* matrix,
285 const bool transpose, const int bitDepth )
286{
287 const int inputSize = 2 * m_reducedBdrySize;
288
289 // use local buffer for transposed result
290 static_vector<int, MIP_MAX_REDUCED_OUTPUT_SAMPLES> resBufTransposed( m_reducedPredSize * m_reducedPredSize );
291 int*const resPtr = (transpose) ? resBufTransposed.data() : result;
292
293 int sum = 0;
294 for( int i = 0; i < inputSize; i++ ) { sum += input[i]; }
295 const int offset = (1 << (MIP_SHIFT_MATRIX - 1)) - MIP_OFFSET_MATRIX * sum;
296 CHECK( inputSize != 4 * (inputSize >> 2), "Error, input size not divisible by four" );
297
298 const uint8_t *weight = matrix;
299 const int inputOffset = transpose ? m_inputOffsetTransp : m_inputOffset;
300
301 const bool redSize = (m_sizeId == 2);
302 int posRes = 0;
303 for( int y = 0; y < m_reducedPredSize; y++ )
304 {
305 for( int x = 0; x < m_reducedPredSize; x++ )
306 {
307 if( redSize ) weight -= 1;
308 int tmp0 = redSize ? 0 : (input[0] * weight[0]);
309 int tmp1 = input[1] * weight[1];
310 int tmp2 = input[2] * weight[2];
311 int tmp3 = input[3] * weight[3];
312 for (int i = 4; i < inputSize; i += 4)
313 {
314 tmp0 += input[i] * weight[i];
315 tmp1 += input[i + 1] * weight[i + 1];
316 tmp2 += input[i + 2] * weight[i + 2];
317 tmp3 += input[i + 3] * weight[i + 3];
318 }
319 resPtr[posRes++] = ClipBD<int>(((tmp0 + tmp1 + tmp2 + tmp3 + offset) >> MIP_SHIFT_MATRIX) + inputOffset, bitDepth);
320
321 weight += inputSize;
322 }
323 }
324
325 if( transpose )
326 {
327 for( int y = 0; y < m_reducedPredSize; y++ )
328 {
329 for( int x = 0; x < m_reducedPredSize; x++ )
330 {
331 result[ y * m_reducedPredSize + x ] = resPtr[ x * m_reducedPredSize + y ];
332 }
333 }
334 }
