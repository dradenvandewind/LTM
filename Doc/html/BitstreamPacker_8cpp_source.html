59// Bit shifts that are defined for shift >= bit-width
60//
61inline uint32_t shl32(uint32_t v, unsigned c) { return (c < 32) ? (v << c) : 0; }
62inline uint32_t shr32(uint32_t v, unsigned c) { return (c < 32) ? (v >> c) : 0; }
63
64// 'width' bitmask
65inline uint32_t mask32(unsigned width) { return (1 << width) - 1; }

81unsigned BitstreamPacker::emit(PacketBuilder &builder) {
82 unsigned size = (bit_offset_ + (BITS_PER_BYTE - 1)) / BITS_PER_BYTE;
83
84#if BITSTREAM_DEBUG
85 fprintf(goBits, "======== ======== ======== ======== ======== ======== ======== ======== ======== ======== \n");

87 fprintf(goBits, "%s", hex_dump(data_.data(), size, 0).c_str());
88 fprintf(goBits, "======== ======== ======== ======== ======== ======== ======== ======== ======== ======== \n");
89 fflush(goBits);
90#endif
91
92 builder.contents(data_.data(), size);
93 return size;
94}

105void BitstreamPacker::u(unsigned num_bits, uint32_t value) {
106 assert(num_bits <= 32);
107
108 value &= mask32(num_bits);
109
110 while (num_bits) {
111 // Index and bit
112 const unsigned bit = bit_offset_ % BITS_PER_BYTE;
113 unsigned idx = bit_offset_ / BITS_PER_BYTE;
114
115 // Expand temp buffer as needed
116 if (idx >= data_.size())
117 data_.resize(data_.size() * 2, (uint8_t)0);
118
119 // How many bits can be written into this byte?
120 const unsigned n = std::min(BITS_PER_BYTE - bit, num_bits);
121
122 // Insert bits into data
123 const unsigned v = shr32(value, num_bits - n);
124 data_[idx] |= shl32(v, BITS_PER_BYTE - (bit + n));
125
126 bit_offset_ += n;
127 num_bits -= n;
128 }
129}

146// Write a sequence of bytes
147void BitstreamPacker::bytes(const uint8_t *data, unsigned size) {
148 assert((bit_offset_ % BITS_PER_BYTE) == 0);
149 const unsigned idx = bit_offset_ / BITS_PER_BYTE;
150
151 while (idx + size >= data_.size())
152 data_.resize(data_.size() * 2, 0);
153
154 for (unsigned i = 0; i < size; ++i)
155 data_[idx + i] = data[i];
156
157 bit_offset_ += size * 8;
158}
