45const uint32_t kNalHeaderLength = 3;
46const uint8_t kNalHeader[kNalHeaderLength] = {0, 0, 1};
47
48BaseDecNalUnitType::Enum uFromLCEVC(uint8_t type) {
49 switch (type) {
50 case LCEVCNalType::LCEVC_IDR:
51 case LCEVCNalType::LCEVC_NonIDR:
52 return BaseDecNalUnitType::Slice;
53 case LCEVCNalType::Unspecified:
54 break;
55 }
56
57 return BaseDecNalUnitType::Unknown;
58}

60uint32_t uOffsetForNalUnitHeader(const uint8_t *nal) {
61 if (memcmp(nal, kNalHeader, kNalHeaderLength) == 0)
62 return kNalHeaderLength;
63 if (!nal[0] && memcmp(nal + 1, kNalHeader, kNalHeaderLength) == 0)
64 return kNalHeaderLength + 1;
65 return 0;
66}

72bool BaseDecoderYUV::ParseNalUnit(const uint8_t *nal, uint32_t nalLength) {
73 bool bSuccess = true;
74
75 // Account for AnnexB nal unit header ([0],0,0,1)
76 uint32_t offset = uOffsetForNalUnitHeader(nal);
77
78 // [forbidden-zero:1, forbidden-one:1, nal_type:5, reserved:9]
79 m_currentNalTypeNumber = (nal[offset] & 0x3e) >> 1;
80 if (m_currentNalTypeNumber == 28)
81 m_currentNalType = LCEVCNalType::LCEVC_NonIDR;
82 else if (m_currentNalTypeNumber == 29)
83 m_currentNalType = LCEVCNalType::LCEVC_IDR;
84 else if (m_currentNalTypeNumber < 28 || m_currentNalTypeNumber == 31)
85 m_currentNalType = LCEVCNalType::Unspecified;
86 else
87 return false; // Reserved NAL Unit (30)
88
89 if (m_currentNalType == LCEVCNalType::LCEVC_IDR || m_currentNalType == LCEVCNalType::LCEVC_NonIDR)
90 m_currentPictureOrderCountPlus1++;
91
92 offset += 1;
93
94 return bSuccess;
95}
