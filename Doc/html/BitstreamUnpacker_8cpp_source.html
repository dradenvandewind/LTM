58// Bit shifts that are defined for shift >= bit-width
59//
60inline uint32_t shl32(uint32_t v, unsigned c) { return (c < 32) ? (v << c) : 0; }
61inline uint32_t shr32(uint32_t v, unsigned c) { return (c < 32) ? (v >> c) : 0; }
62
63// 'width' bitmask
64inline uint32_t mask32(unsigned width) { return (1 << width) - 1; }

86// Read 0..32 bits into unsigned integer
87//
88uint32_t BitstreamUnpacker::u(unsigned num_bits) {
89 assert(num_bits <= 32);
90 assert((bit_offset_ + num_bits) <= bit_size());
91
92 uint32_t r = 0;
93
94 while (num_bits) {
95 // Index and bit
96#if defined __OPT_MODULO__
97 const unsigned bit_used = bit_offset_ & BITS_COUNT_MASK;
98#else
99 const unsigned bit_used = bit_offset_ % BITS_PER_BYTE;
100#endif
101#if defined __OPT_DIVISION__
102 unsigned idx = bit_offset_ >> BITS_COUNT_SHIFT;
103#else
104 unsigned idx = bit_offset_ / BITS_PER_BYTE;
105#endif
106
107 const unsigned data = view_.data()[idx];
108 const unsigned bit_left = BITS_PER_BYTE - bit_used;
109
110 // How many bits can be read from this byte?
111 // const unsigned n = std::min(BITS_PER_BYTE - bit, num_bits);
112 const unsigned n = LCEVC_MIN(bit_left, num_bits);
113
114 // Extract bits and merge into result
115 // r = shl32(r, n) | (shr32(data, (BITS_PER_BYTE - bit) - n) & mask32(n));
116 r = (r << n) | ((data >> (bit_left - n)) & ((1 << n) - 1));
117
118 bit_offset_ += n;
119 num_bits -= n;
120 }
121
122 return r;
123}

125// Read a single of byte from stream - expects offset byte aligned
126//
127uint8_t BitstreamUnpacker::byte() {
128#if defined __OPT_MODULO__
129 assert((bit_offset_ & BITS_COUNT_MASK) == 0);
130#else
131 assert((bit_offset_ % BITS_PER_BYTE) == 0);
132#endif
133 assert(BITS_PER_BYTE <= (bit_size() - bit_offset_));
134#if defined __OPT_DIVISION__
135 const unsigned idx = bit_offset_ >> BITS_COUNT_SHIFT;
136#else
137 const unsigned idx = bit_offset_ / BITS_PER_BYTE;
138#endif
139 bit_offset_ += BITS_PER_BYTE;
140
141// CHECK(idx < view_.size());
142 if(idx < view_.size()) {
143 return view_.data()[idx];
144 } else {
145 WARN("Read beyond end of packet.");
146 return 0;
147 }
148}

150// Read a sequence of bytes from stream - expects offset byte aligned
151//
152Packet BitstreamUnpacker::bytes(unsigned num_bytes) {
153#if defined __OPT_MODULO__
154 assert((bit_offset_ & BITS_COUNT_MASK) == 0);
155#else
156 assert((bit_offset_ % BITS_PER_BYTE) == 0);
157#endif
158 assert(num_bytes * BITS_PER_BYTE <= (bit_size() - bit_offset_));
159#if defined __OPT_DIVISION__
160 const unsigned idx = bit_offset_ >> BITS_COUNT_SHIFT;
161#else
162 const unsigned idx = bit_offset_ / BITS_PER_BYTE;
163#endif
164
165 auto b = Packet::build().reserve(num_bytes);
166 for (unsigned i = 0; i < num_bytes; ++i)
167 b.data()[i] = view_.data()[idx + i];
168
169 bit_offset_ += num_bytes * BITS_PER_BYTE;
170
171 return b.finish();
172}
