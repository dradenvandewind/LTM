83// From a view into another packet
84PacketBuilder &PacketBuilder::contents(const PacketView &view, unsigned offset, unsigned size) {
85
86 // Start with all the same metadata
87 packet_ = view.packet_;
88
89 assert(offset <= packet_.size());
90 assert(offset + size <= packet_.size());
91
92 // Adjust this offset and size
93 packet_.offset_ += offset;
94 packet_.size_ = size;
95
96 return *this;
97}

138void PacketBuilder::resize(unsigned size) {
139 assert(mapped_data_ != nullptr);
140
141 packet_.buffer_ = std::shared_ptr<Buffer>(CreateBufferVector(mapped_data_, size));
