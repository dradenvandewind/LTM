104// Shift int64 down and clamp to int16
105//
106static inline int16_t shift_clamp_s16(int64_t v, int32_t shift) {
107 const int64_t half = (int64_t)1 << (shift - 1);
108 return (int16_t)clamp((v + half) >> shift, (int64_t)-32768, (int64_t)32767);
109}

130// Work out scale to make (a << s) >= b
131//
132static inline unsigned tile_size(unsigned a, unsigned b) {
133
134 unsigned s = 0, t = 1;
135 while ((a << s) < b) {
136 t *= 2;
137 ++s;
138 }
139
140 return t;
141}

154// Wrap a string in a istringstream and extract into type
155//
156template <typename T> T extract(const std::string &s) {
157 T r;
158 try {
159 std::istringstream ss(s);
160 ss >> r;
161 } catch (const std::ios_base::failure &) {
162 ERR("Cannot extract %s", s.c_str());
163 }
164 return r;
165}
