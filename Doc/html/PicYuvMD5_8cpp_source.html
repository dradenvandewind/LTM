45template<uint32_t OUTPUT_BITDEPTH_DIV8>
46static void md5_block(MD5& md5, const Pel* plane, uint32_t n)
47{
48 /* create a 64 byte buffer for packing Pel's into */
49 uint8_t buf[64/OUTPUT_BITDEPTH_DIV8][OUTPUT_BITDEPTH_DIV8];
50 for (uint32_t i = 0; i < n; i++)
51 {
52 Pel pel = plane[i];
53 /* perform bitdepth and endian conversion */
54 for (uint32_t d = 0; d < OUTPUT_BITDEPTH_DIV8; d++)
55 {
56 buf[i][d] = pel >> (d*8);
57 }
58 }
59 md5.update((uint8_t*)buf, n * OUTPUT_BITDEPTH_DIV8);
60}

66template<uint32_t OUTPUT_BITDEPTH_DIV8>
67static void md5_plane(MD5& md5, const Pel* plane, uint32_t width, uint32_t height, uint32_t stride)
68{
69 /* N is the number of samples to process per md5 update.
70 * All N samples must fit in buf */
71 uint32_t N = 32;
72 uint32_t width_modN = width % N;
73 uint32_t width_less_modN = width - width_modN;
74
75 for (uint32_t y = 0; y < height; y++)
76 {
77 /* convert pels into unsigned chars in little endian byte order.
78 * NB, for 8bit data, data is truncated to 8bits. */
79 for (uint32_t x = 0; x < width_less_modN; x += N)
80 {
81 md5_block<OUTPUT_BITDEPTH_DIV8>(md5, &plane[y*stride + x], N);
82 }
83
84 /* mop up any of the remaining line */
85 md5_block<OUTPUT_BITDEPTH_DIV8>(md5, &plane[y*stride + width_less_modN], width_modN);
86 }
87}

90uint32_t compCRC(int bitdepth, const Pel* plane, uint32_t width, uint32_t height, uint32_t stride, PictureHash &digest)
91{
92 uint32_t crcMsb;
93 uint32_t bitVal;
94 uint32_t crcVal = 0xffff;
95 uint32_t bitIdx;
96 for (uint32_t y = 0; y < height; y++)
97 {
98 for (uint32_t x = 0; x < width; x++)
99 {
100 // take CRC of first pictureData byte
101 for(bitIdx=0; bitIdx<8; bitIdx++)
102 {
103 crcMsb = (crcVal >> 15) & 1;
104 bitVal = (plane[y*stride+x] >> (7 - bitIdx)) & 1;
105 crcVal = (((crcVal << 1) + bitVal) & 0xffff) ^ (crcMsb * 0x1021);
106 }
107 // take CRC of second pictureData byte if bit depth is greater than 8-bits
108 if(bitdepth > 8)
109 {
110 for(bitIdx=0; bitIdx<8; bitIdx++)
111 {
112 crcMsb = (crcVal >> 15) & 1;
113 bitVal = (plane[y*stride+x] >> (15 - bitIdx)) & 1;
114 crcVal = (((crcVal << 1) + bitVal) & 0xffff) ^ (crcMsb * 0x1021);
115 }
116 }
117 }
118 }
119 for(bitIdx=0; bitIdx<16; bitIdx++)
120 {
121 crcMsb = (crcVal >> 15) & 1;
122 crcVal = ((crcVal << 1) & 0xffff) ^ (crcMsb * 0x1021);
123 }
124
125 digest.hash.push_back((crcVal>>8) & 0xff);
126 digest.hash.push_back( crcVal & 0xff);
127 return 2;
128}

143uint32_t compChecksum(int bitdepth, const Pel* plane, uint32_t width, uint32_t height, uint32_t stride, PictureHash &digest, const BitDepths &/*bitDepths*/)
144{
145 uint32_t checksum = 0;
146 uint8_t xor_mask;
147
148 for (uint32_t y = 0; y < height; y++)
149 {
150 for (uint32_t x = 0; x < width; x++)
151 {
152 xor_mask = (x & 0xff) ^ (y & 0xff) ^ (x >> 8) ^ (y >> 8);
153 checksum = (checksum + ((plane[y*stride+x] & 0xff) ^ xor_mask)) & 0xffffffff;
154
155 if(bitdepth > 8)
156 {
157 checksum = (checksum + ((plane[y*stride+x]>>8) ^ xor_mask)) & 0xffffffff;
158 }
159 }
160 }
161
162 digest.hash.push_back((checksum>>24) & 0xff);
163 digest.hash.push_back((checksum>>16) & 0xff);
164 digest.hash.push_back((checksum>>8) & 0xff);
165 digest.hash.push_back( checksum & 0xff);
166 return 4;
167}

193uint32_t calcMD5WithCropping(const CPelUnitBuf &pic, PictureHash &digest, const BitDepths &bitDepths,
194 const int leftOffset, const int rightOffset, const int topOffset, const int bottomOffset)
195{
196 /* choose an md5_plane packing function based on the system bitdepth */
197 typedef void (*MD5PlaneFunc)(MD5&, const Pel*, uint32_t, uint32_t, uint32_t);
198 MD5PlaneFunc md5_plane_func;
199
200 MD5 md5[MAX_NUM_COMPONENT];
201
202 digest.hash.clear();
203
204 for (uint32_t chan = 0; chan< (uint32_t)pic.bufs.size(); chan++)
205 {
206 const ComponentID compID = ComponentID(chan);
207 const CPelBuf area = pic.get(compID);
208 const int chromaScaleX = getComponentScaleX(compID, pic.chromaFormat);
209 const int chromaScaleY = getComponentScaleY(compID, pic.chromaFormat);
210 const int compLeftOffset = leftOffset >> chromaScaleX;
211 const int compRightOffset = rightOffset >> chromaScaleX;
212 const int compTopOffset = topOffset >> chromaScaleY;
213 const int compBottomOffset = bottomOffset >> chromaScaleY;
214 md5_plane_func = bitDepths.recon[toChannelType(compID)] <= 8 ? (MD5PlaneFunc)md5_plane<1> : (MD5PlaneFunc)md5_plane<2>;
215 uint8_t tmp_digest[MD5_DIGEST_STRING_LENGTH];
216 md5_plane_func(md5[compID], area.bufAt(compLeftOffset, compTopOffset),
217 area.width - compRightOffset - compLeftOffset, area.height - compTopOffset - compBottomOffset,
218 area.stride);
219 md5[compID].finalize(tmp_digest);
220 for (uint32_t i = 0; i < MD5_DIGEST_STRING_LENGTH; i++)
221 {
222 digest.hash.push_back(tmp_digest[i]);
223 }
224 }
225
226 return 16;
227}

229std::string hashToString(const PictureHash &digest, int numChar)
230{
231 static const char* hex = "0123456789abcdef";
232 std::string result;
233
234 for(int pos=0; pos<int(digest.hash.size()); pos++)
235 {
236 if ((pos % numChar) == 0 && pos!=0 )
237 {
238 result += ',';
239 }
240 result += hex[digest.hash[pos] >> 4];
241 result += hex[digest.hash[pos] & 0xf];
242 }
243
244 return result;
245}

247int calcAndPrintHashStatus(const CPelUnitBuf& pic, const SEIDecodedPictureHash* pictureHashSEI, const BitDepths &bitDepths, const MsgLevel msgl)
248{
249 /* calculate MD5sum for entire reconstructed picture */
250 PictureHash recon_digest;
251 int numChar=0;
252 const char* hashType = "\0";
253
254 if (pictureHashSEI)
255 {
256#if FIX_TICKET_1405
257 CHECK ((uint32_t)pic.bufs.size() != ( pictureHashSEI->singleCompFlag ? 1 : 3 ), "The value of dph_sei_single_component_flag shall be equal to (ChromaFormatIdc == 0).");
258#endif
259 switch (pictureHashSEI->method)
260 {
261 case HASHTYPE_MD5:
262 {
263 hashType = "MD5";
264 numChar = calcMD5(pic, recon_digest, bitDepths);
265 break;
266 }
267 case HASHTYPE_CRC:
268 {
269 hashType = "CRC";
270 numChar = calcCRC(pic, recon_digest, bitDepths);
271 break;
272 }
273 case HASHTYPE_CHECKSUM:
274 {
275 hashType = "Checksum";
276 numChar = calcChecksum(pic, recon_digest, bitDepths);
277 break;
278 }
279 default:
280 {
281 THROW("Unknown hash type");
282 break;
283 }
284 }
285 }
286
287 /* compare digest against received version */
288 const char* ok = "(unk)";
289 bool mismatch = false;
290
291 if (pictureHashSEI)
292 {
293 ok = "(OK)";
294 if (recon_digest != pictureHashSEI->m_pictureHash)
295 {
296 ok = "(***ERROR***)";
297 mismatch = true;
298 }
299 }
300
301 msg( msgl, "[%s:%s,%s] ", hashType, hashToString(recon_digest, numChar).c_str(), ok);
302
303 if (mismatch)
304 {
305 msg( msgl, "[rx%s:%s] ", hashType, hashToString(pictureHashSEI->m_pictureHash, numChar).c_str());
306 }
307 return mismatch;
308}
