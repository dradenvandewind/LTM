46template <class T> class ParameterSetMap
47{
48public:
49 template <class Tm>
50 struct MapData
51 {
52 bool bChanged;
53 std::vector<uint8_t> *pNaluData; // Can be null
54 Tm* parameterSet;
55 };

57 ParameterSetMap(int maxId)
58 :m_maxId (maxId)
59 ,m_lastActiveParameterSet(NULL)
60 {
61 m_activePsId.clear();
62 }
63
64 ~ParameterSetMap()
65 {
66 for (typename std::map<int,MapData<T> >::iterator i = m_paramsetMap.begin(); i!= m_paramsetMap.end(); i++)
67 {
68 delete (*i).second.pNaluData;
69 delete (*i).second.parameterSet;
70 }
71 delete m_lastActiveParameterSet; m_lastActiveParameterSet = NULL;
72 }
73
74 T *allocatePS(const int psId)
75 {
76 CHECK( psId >= m_maxId, "Invalid PS id" );
77 if ( m_paramsetMap.find(psId) == m_paramsetMap.end() )
78 {
79 m_paramsetMap[psId].bChanged = true;
80 m_paramsetMap[psId].pNaluData=0;
81 m_paramsetMap[psId].parameterSet = new T;
82 setID(m_paramsetMap[psId].parameterSet, psId);
83 }
84 return m_paramsetMap[psId].parameterSet;
85 }
86
87 void clearMap()
88 {
89 m_paramsetMap.clear();
90 }
91 void storePS( int psId, T *ps )
92 {
93 CHECK( psId >= m_maxId, "Invalid PS id" );
94 if( m_paramsetMap.find( psId ) != m_paramsetMap.end() )
95 {
96 delete m_paramsetMap[psId].parameterSet;
97 }
98 m_paramsetMap[psId].parameterSet = ps;
99 m_paramsetMap[psId].bChanged = true;
100 }
101 void storePS(int psId, T *ps, const std::vector<uint8_t> *pNaluData)
102 {
103 CHECK( psId >= m_maxId, "Invalid PS id" );
104 if ( m_paramsetMap.find(psId) != m_paramsetMap.end() )
105 {
106 MapData<T> &mapData=m_paramsetMap[psId];
107
108 // work out changed flag
109 calculateParameterSetChangedFlag(mapData.bChanged, mapData.pNaluData, pNaluData);
110
111 if( ! mapData.bChanged )
112 {
113 // just keep the old one
114 delete ps;
115 return;
116 }
117
118 if (find(m_activePsId.begin(), m_activePsId.end(), psId) != m_activePsId.end())
119 {
120 std::swap( m_paramsetMap[psId].parameterSet, m_lastActiveParameterSet );
121 }
122 delete m_paramsetMap[psId].pNaluData;
123 delete m_paramsetMap[psId].parameterSet;
124
125 m_paramsetMap[psId].parameterSet = ps;
126 }
127 else
128 {
129 m_paramsetMap[psId].parameterSet = ps;
130 m_paramsetMap[psId].bChanged = false;
131 }
132 if (pNaluData != 0)
133 {
134 m_paramsetMap[psId].pNaluData=new std::vector<uint8_t>;
135 *(m_paramsetMap[psId].pNaluData) = *pNaluData;
136 }
137 else
138 {
139 m_paramsetMap[psId].pNaluData=0;
140 }
141 }

143 void checkAuApsContent( APS *aps, std::vector<int>& accessUnitApsNals )
144 {
145 int apsId = ( aps->getAPSId() << NUM_APS_TYPE_LEN ) + aps->getAPSType();
146
147 if( std::find( accessUnitApsNals.begin(), accessUnitApsNals.end(), apsId ) != accessUnitApsNals.end() )
148 {
149 CHECK( m_paramsetMap.find( apsId ) == m_paramsetMap.end(), "APS does not exist" );
150 APS* existedAPS = m_paramsetMap[apsId].parameterSet;
151 bool sameNalUnitType = aps->getHasPrefixNalUnitType() == existedAPS->getHasPrefixNalUnitType();
152 bool samePU = aps->getLayerId() == existedAPS->getLayerId();
153 if( aps->getAPSType() == LMCS_APS )
154 {
155 CHECK( samePU && sameNalUnitType && aps->getReshaperAPSInfo() != existedAPS->getReshaperAPSInfo(), "All APS NAL units with a particular value of nal_unit_type, a particular value of aps_adaptation_parameter_set_id, and a particular value of aps_params_type within a PU shall have the same content" );
156 }
157 else if( aps->getAPSType() == ALF_APS )
158 {
159 CHECK( samePU && sameNalUnitType && aps->getAlfAPSParam() != existedAPS->getAlfAPSParam(), "All APS NAL units with a particular value of nal_unit_type, a particular value of aps_adaptation_parameter_set_id, and a particular value of aps_params_type within a PU shall have the same content" );
160 }
161 else if( aps->getAPSType() == SCALING_LIST_APS )
162 {
163 CHECK( samePU && sameNalUnitType && aps->getScalingList() != existedAPS->getScalingList(), "All APS NAL units with a particular value of nal_unit_type, a particular value of aps_adaptation_parameter_set_id, and a particular value of aps_params_type within a PU shall have the same content" );
164 }
165 else
166 {
167 CHECK( true, "Wrong APS type" );
168 }
169 }
170 else
171 {
172 accessUnitApsNals.push_back( apsId );
173 }
174 }

177 void setChangedFlag(int psId, bool bChanged=true)
178 {
179 if ( m_paramsetMap.find(psId) != m_paramsetMap.end() )
180 {
181 m_paramsetMap[psId].bChanged=bChanged;
182 }
183 }
184
185 void clearChangedFlag(int psId)
186 {
187 if ( m_paramsetMap.find(psId) != m_paramsetMap.end() )
188 {
189 m_paramsetMap[psId].bChanged=false;
190 }
191 }
192
193 bool getChangedFlag(int psId) const
194 {
195 const typename std::map<int,MapData<T> >::const_iterator constit=m_paramsetMap.find(psId);
196 if ( constit != m_paramsetMap.end() )
197 {
198 return constit->second.bChanged;
199 }
200 return false;
201 }
202
203 T* getPS(int psId)
204 {
205 typename std::map<int,MapData<T> >::iterator it=m_paramsetMap.find(psId);
206 return ( it == m_paramsetMap.end() ) ? NULL : (it)->second.parameterSet;
207 }
208
209 const T* getPS(int psId) const
210 {
211 typename std::map<int,MapData<T> >::const_iterator it=m_paramsetMap.find(psId);
212 return ( it == m_paramsetMap.end() ) ? NULL : (it)->second.parameterSet;
213 }
214
215 T* getFirstPS()
216 {
217 return (m_paramsetMap.begin() == m_paramsetMap.end() ) ? NULL : m_paramsetMap.begin()->second.parameterSet;
218 }
219
220 void setActive(int psId) { m_activePsId.push_back(psId); }
221 void clearActive() { m_activePsId.clear(); }
222
223private:
224 std::map<int,MapData<T> > m_paramsetMap;
225 int m_maxId;
226 std::vector<int> m_activePsId;
227 T* m_lastActiveParameterSet;
228 static void setID(T* parameterSet, const int psId);
229};
