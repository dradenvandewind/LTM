66// Dump surface as 8bit p420, with dummy chroma planes
67//
68void Surface::dump_p420(const std::string &name) const {
69 if (!dump_surfaces)
70 return;
71
72 ImageDescription desc(IMAGE_FORMAT_YUV420P8, width(), height());
73
74 if (yuv_writers.find(name) == yuv_writers.end())
75 yuv_writers[name] = CreateYUVWriter(name, desc, true);
76
77 Surface planes[3];
78
79 if (bpp() == 1)
80 // Just use base as bytes
81 planes[0] = *this;
82 else
83 // Assume S87
84 planes[0] = ConvertToU8().process(*this, 7);
85
86 // Create a dummy surface for U&V
87 auto sb = Surface::build_from<uint8_t>();
88 // sb.reserve(width() / 2, height() / 2);
89 sb.reserve(width() >> 1, height() >> 1);
90 memset(sb.data(0, 0), 0x80, sb.width() * sb.height());
91 planes[1] = planes[2] = sb.finish();
92
93 yuv_writers[name]->write(Image(name, desc, 0, planes));
94}

96// Dump surface as 8bpp, 10bpp, 12bpp, or 14bpp with correct chroma subsampling and proper chroma planes
97//
98void Surface::dump_yuv(const std::string &name, Surface (&planes)[3], ImageDescription desc, bool decorate) const {
99 if (desc.bit_depth() == 8) {
100 Surface converted[3];
101 if (yuv_writers.find(name) == yuv_writers.end())
102 yuv_writers[name] = CreateYUVWriter(name, desc, decorate);
103 for (unsigned p = 0; p < desc.num_planes(); p++)
104 converted[p] = ConvertToU8().process(planes[p], 7);
105 yuv_writers[name]->write(Image(name, desc, 0, converted));
106 } else if (desc.bit_depth() == 10) {
107 Surface converted[3];
108 if (yuv_writers.find(name) == yuv_writers.end())
109 yuv_writers[name] = CreateYUVWriter(name, desc, decorate);
110 for (unsigned p = 0; p < desc.num_planes(); p++)
111 converted[p] = ConvertToU16().process(planes[p], 5);
112 yuv_writers[name]->write(Image(name, desc, 0, converted));
113 } else if (desc.bit_depth() == 12) {
114 Surface converted[3];
115 if (yuv_writers.find(name) == yuv_writers.end())
116 yuv_writers[name] = CreateYUVWriter(name, desc, decorate);
117 for (unsigned p = 0; p < desc.num_planes(); p++)
118 converted[p] = ConvertToU16().process(planes[p], 3);
119 yuv_writers[name]->write(Image(name, desc, 0, converted));
120 } else if (desc.bit_depth() == 14) {
121 Surface converted[3];
122 if (yuv_writers.find(name) == yuv_writers.end())
123 yuv_writers[name] = CreateYUVWriter(name, desc, decorate);
124 for (unsigned p = 0; p < desc.num_planes(); p++)
125 converted[p] = ConvertToU16().process(planes[p], 1);
126 yuv_writers[name]->write(Image(name, desc, 0, converted));
127 } else {
128 INFO("Image Format not supported %4d bpp", desc.bit_depth());
129 }
130}

161// Dump array of layers/surfaces as 16 bit Y
162//
163void Surface::dump_layers(const Surface surface[], const std::string &name, const unsigned transform_block_size) {
164 if (!dump_surfaces)
165 return;
166
167 CHECK(surface[0].bpp() == 2); // only 16 bit supported at the moment
168 ImageDescription desc(IMAGE_FORMAT_Y16, surface[0].width() * transform_block_size, surface[0].height() * transform_block_size);
169
170 if (yuv_writers.find(name) == yuv_writers.end())
171 yuv_writers[name] = CreateYUVWriter(name, desc, true);
172
173 Surface planes[3];
174
175 if (transform_block_size == 2) {
176 const auto src00_view = surface[0].view_as<int16_t>();
177 const auto src01_view = surface[1].view_as<int16_t>();
178 const auto src02_view = surface[2].view_as<int16_t>();
179 const auto src03_view = surface[3].view_as<int16_t>();
180
181 auto dst = Surface::build_from<int16_t>();
182 dst.reserve(surface[0].width() * transform_block_size, surface[0].height() * transform_block_size);
183
184 for (unsigned x = 0; x < surface[0].width(); x++) {
185 for (unsigned y = 0; y < surface[0].height(); y++) {
186 dst.write(x * 2 + 0, y * 2 + 0, src00_view.read(x, y));
187 dst.write(x * 2 + 1, y * 2 + 0, src01_view.read(x, y));
188 dst.write(x * 2 + 0, y * 2 + 1, src02_view.read(x, y));
189 dst.write(x * 2 + 1, y * 2 + 1, src03_view.read(x, y));
190 }
191 }
192 planes[0] = dst.finish();
193 } else {
194 const auto src00_view = surface[0].view_as<int16_t>();
195 const auto src01_view = surface[1].view_as<int16_t>();
196 const auto src02_view = surface[2].view_as<int16_t>();
197 const auto src03_view = surface[3].view_as<int16_t>();
198 const auto src04_view = surface[4].view_as<int16_t>();
199 const auto src05_view = surface[5].view_as<int16_t>();
200 const auto src06_view = surface[6].view_as<int16_t>();
201 const auto src07_view = surface[7].view_as<int16_t>();
202 const auto src08_view = surface[8].view_as<int16_t>();
203 const auto src09_view = surface[9].view_as<int16_t>();
204 const auto src10_view = surface[10].view_as<int16_t>();
205 const auto src11_view = surface[11].view_as<int16_t>();
206 const auto src12_view = surface[12].view_as<int16_t>();
207 const auto src13_view = surface[13].view_as<int16_t>();
208 const auto src14_view = surface[14].view_as<int16_t>();
209 const auto src15_view = surface[15].view_as<int16_t>();
210
211 auto dst = Surface::build_from<int16_t>();
212 dst.reserve(surface[0].width() * transform_block_size, surface[0].height() * transform_block_size);
213
214 for (unsigned x = 0; x < surface[0].width(); x++) {
215 for (unsigned y = 0; y < surface[0].height(); y++) {
216 dst.write(x * 4 + 0, y * 4 + 0, src00_view.read(x, y));
217 dst.write(x * 4 + 1, y * 4 + 0, src01_view.read(x, y));
218 dst.write(x * 4 + 2, y * 4 + 0, src02_view.read(x, y));
219 dst.write(x * 4 + 3, y * 4 + 0, src03_view.read(x, y));
220 dst.write(x * 4 + 0, y * 4 + 1, src04_view.read(x, y));
221 dst.write(x * 4 + 1, y * 4 + 1, src05_view.read(x, y));
222 dst.write(x * 4 + 2, y * 4 + 1, src06_view.read(x, y));
223 dst.write(x * 4 + 3, y * 4 + 1, src07_view.read(x, y));
224 dst.write(x * 4 + 0, y * 4 + 2, src08_view.read(x, y));
225 dst.write(x * 4 + 1, y * 4 + 2, src09_view.read(x, y));
226 dst.write(x * 4 + 2, y * 4 + 2, src10_view.read(x, y));
227 dst.write(x * 4 + 3, y * 4 + 2, src11_view.read(x, y));
228 dst.write(x * 4 + 0, y * 4 + 3, src12_view.read(x, y));
229 dst.write(x * 4 + 1, y * 4 + 3, src13_view.read(x, y));
230 dst.write(x * 4 + 2, y * 4 + 3, src14_view.read(x, y));
231 dst.write(x * 4 + 3, y * 4 + 3, src15_view.read(x, y));
232 }
233 }
234 planes[0] = dst.finish();
235 }
236
237 yuv_writers[name]->write(Image(name, desc, 0, planes));
238}
