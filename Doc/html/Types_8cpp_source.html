45std::istream &operator>>(std::istream &in, Upsample &v) {
46 std::string s;
47 in >> s;
48 s = lowercase(s);
49
50 if (s == "nearest")
51 v = Upsample_Nearest;
52 else if (s == "linear")
53 v = Upsample_Linear;
54 else if (s == "cubic")
55 v = Upsample_Cubic;
56 else if (s == "modifiedcubic")
57 v = Upsample_ModifiedCubic;
58 else if (s == "adaptivecubic")
59 v = Upsample_AdaptiveCubic;
60 else
61 ERR("not an upsampler: %s", s.c_str());
62 return in;
63}
