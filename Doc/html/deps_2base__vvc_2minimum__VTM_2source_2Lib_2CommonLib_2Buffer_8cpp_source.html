47void applyPROFCore(Pel *dst, int dstStride, const Pel *src, int srcStride, int width, int height, const Pel *gradX,
48 const Pel *gradY, int gradStride, const int *dMvX, const int *dMvY, int dMvStride, const bool &bi, int shiftNum,
49 Pel offset, const ClpRng &clpRng) {
50 int idx = 0;
51
52 const int dILimit = 1 << std::max<int>(clpRng.bd + 1, 13);
53 for (int h = 0; h < height; h++)
54 {
55 for (int w = 0; w < width; w++)
56 {
57 int32_t dI = dMvX[idx] * gradX[w] + dMvY[idx] * gradY[w];
58 dI = Clip3(-dILimit, dILimit - 1, dI);
59 dst[w] = src[w] + dI;
60 if (!bi)
61 {
62 dst[w] = (dst[w] + offset) >> shiftNum;
63 dst[w] = ClipPel(dst[w], clpRng);
64 }
65
66 idx++;
67 }
68 gradX += gradStride;
69 gradY += gradStride;
70 dst += dstStride;
71 src += srcStride;
72 }
73}

91void addBIOAvgCore(const Pel* src0, int src0Stride, const Pel* src1, int src1Stride, Pel *dst, int dstStride, const Pel *gradX0, const Pel *gradX1, const Pel *gradY0, const Pel*gradY1, int gradStride, int width, int height, int tmpx, int tmpy, int shift, int offset, const ClpRng& clpRng)
92{
93 int b = 0;
94
95 for (int y = 0; y < height; y++)
96 {
97 for (int x = 0; x < width; x += 4)
98 {
99 b = tmpx * (gradX0[x] - gradX1[x]) + tmpy * (gradY0[x] - gradY1[x]);
100 dst[x] = ClipPel(rightShift((src0[x] + src1[x] + b + offset), shift), clpRng);
101
102 b = tmpx * (gradX0[x + 1] - gradX1[x + 1]) + tmpy * (gradY0[x + 1] - gradY1[x + 1]);
103 dst[x + 1] = ClipPel(rightShift((src0[x + 1] + src1[x + 1] + b + offset), shift), clpRng);
104
105 b = tmpx * (gradX0[x + 2] - gradX1[x + 2]) + tmpy * (gradY0[x + 2] - gradY1[x + 2]);
106 dst[x + 2] = ClipPel(rightShift((src0[x + 2] + src1[x + 2] + b + offset), shift), clpRng);
107
108 b = tmpx * (gradX0[x + 3] - gradX1[x + 3]) + tmpy * (gradY0[x + 3] - gradY1[x + 3]);
109 dst[x + 3] = ClipPel(rightShift((src0[x + 3] + src1[x + 3] + b + offset), shift), clpRng);
110 }
111 dst += dstStride; src0 += src0Stride; src1 += src1Stride;
112 gradX0 += gradStride; gradX1 += gradStride; gradY0 += gradStride; gradY1 += gradStride;
113 }

116template<bool PAD = true>
117void gradFilterCore(Pel* pSrc, int srcStride, int width, int height, int gradStride, Pel* gradX, Pel* gradY, const int bitDepth)
118{
119 Pel* srcTmp = pSrc + srcStride + 1;
120 Pel* gradXTmp = gradX + gradStride + 1;
121 Pel* gradYTmp = gradY + gradStride + 1;
122 int shift1 = 6;
123
124 for (int y = 0; y < (height - 2 * BIO_EXTEND_SIZE); y++)
125 {
126 for (int x = 0; x < (width - 2 * BIO_EXTEND_SIZE); x++)
127 {
128 gradYTmp[x] = ( srcTmp[x + srcStride] >> shift1 ) - ( srcTmp[x - srcStride] >> shift1 );
129 gradXTmp[x] = ( srcTmp[x + 1] >> shift1 ) - ( srcTmp[x - 1] >> shift1 );
130 }
131 gradXTmp += gradStride;
132 gradYTmp += gradStride;
133 srcTmp += srcStride;
134 }
135
136 if (PAD)
137 {
138 gradXTmp = gradX + gradStride + 1;
139 gradYTmp = gradY + gradStride + 1;
140 for (int y = 0; y < (height - 2 * BIO_EXTEND_SIZE); y++)
141 {
142 gradXTmp[-1] = gradXTmp[0];
143 gradXTmp[width - 2 * BIO_EXTEND_SIZE] = gradXTmp[width - 2 * BIO_EXTEND_SIZE - 1];
144 gradXTmp += gradStride;
145
146 gradYTmp[-1] = gradYTmp[0];
147 gradYTmp[width - 2 * BIO_EXTEND_SIZE] = gradYTmp[width - 2 * BIO_EXTEND_SIZE - 1];
148 gradYTmp += gradStride;
149 }
150
151 gradXTmp = gradX + gradStride;
152 gradYTmp = gradY + gradStride;
153 ::memcpy(gradXTmp - gradStride, gradXTmp, sizeof(Pel)*(width));
154 ::memcpy(gradXTmp + (height - 2 * BIO_EXTEND_SIZE)*gradStride, gradXTmp + (height - 2 * BIO_EXTEND_SIZE - 1)*gradStride, sizeof(Pel)*(width));
155 ::memcpy(gradYTmp - gradStride, gradYTmp, sizeof(Pel)*(width));
156 ::memcpy(gradYTmp + (height - 2 * BIO_EXTEND_SIZE)*gradStride, gradYTmp + (height - 2 * BIO_EXTEND_SIZE - 1)*gradStride, sizeof(Pel)*(width));
157 }

160void calcBIOSumsCore(const Pel* srcY0Tmp, const Pel* srcY1Tmp, Pel* gradX0, Pel* gradX1, Pel* gradY0, Pel* gradY1, int xu, int yu, const int src0Stride, const int src1Stride, const int widthG, const int bitDepth, int* sumAbsGX, int* sumAbsGY, int* sumDIX, int* sumDIY, int* sumSignGY_GX)
161{
162 int shift4 = 4;
163 int shift5 = 1;
164
165 for (int y = 0; y < 6; y++)
166 {
167 for (int x = 0; x < 6; x++)
168 {
169 int tmpGX = (gradX0[x] + gradX1[x]) >> shift5;
170 int tmpGY = (gradY0[x] + gradY1[x]) >> shift5;
171 int tmpDI = (int)((srcY1Tmp[x] >> shift4) - (srcY0Tmp[x] >> shift4));
172 *sumAbsGX += (tmpGX < 0 ? -tmpGX : tmpGX);
173 *sumAbsGY += (tmpGY < 0 ? -tmpGY : tmpGY);
174 *sumDIX += (tmpGX < 0 ? -tmpDI : (tmpGX == 0 ? 0 : tmpDI));
175 *sumDIY += (tmpGY < 0 ? -tmpDI : (tmpGY == 0 ? 0 : tmpDI));
176 *sumSignGY_GX += (tmpGY < 0 ? -tmpGX : (tmpGY == 0 ? 0 : tmpGX));
177
178 }
179 srcY1Tmp += src1Stride;
180 srcY0Tmp += src0Stride;
181 gradX0 += widthG;
182 gradX1 += widthG;
183 gradY0 += widthG;
184 gradY1 += widthG;
185 }

327void paddingCore(Pel *ptr, int stride, int width, int height, int padSize)
328{
329 /*left and right padding*/
330 Pel *ptrTemp1 = ptr;
331 Pel *ptrTemp2 = ptr + (width - 1);
332 int offset = 0;
333 for (int i = 0; i < height; i++)
334 {
335 offset = stride * i;
336 for (int j = 1; j <= padSize; j++)
337 {
338 *(ptrTemp1 - j + offset) = *(ptrTemp1 + offset);
339 *(ptrTemp2 + j + offset) = *(ptrTemp2 + offset);
340 }
341 }
342 /*Top and Bottom padding*/
343 int numBytes = (width + padSize + padSize) * sizeof(Pel);
344 ptrTemp1 = (ptr - padSize);
345 ptrTemp2 = (ptr + (stride * (height - 1)) - padSize);
346 for (int i = 1; i <= padSize; i++)
347 {
348 memcpy(ptrTemp1 - (i * stride), (ptrTemp1), numBytes);
349 memcpy(ptrTemp2 + (i * stride), (ptrTemp2), numBytes);
350 }

398template<>
399void AreaBuf<Pel>::scaleSignal(const int scale, const bool dir, const ClpRng& clpRng)
400{
401 Pel* dst = buf;
402 Pel* src = buf;
403 int sign, absval;
404 int maxAbsclipBD = (1<<clpRng.bd) - 1;
405
406 if (dir) // forward
407 {
408 if (width == 1)
409 {
410 THROW("Blocks of width = 1 not supported");
411 }
412 else
413 {
414 for (unsigned y = 0; y < height; y++)
415 {
416 for (unsigned x = 0; x < width; x++)
417 {
418 sign = src[x] >= 0 ? 1 : -1;
419 absval = sign * src[x];
420 dst[x] = (Pel)Clip3(-maxAbsclipBD, maxAbsclipBD, sign * (((absval << CSCALE_FP_PREC) + (scale >> 1)) / scale));
421 }
422 dst += stride;
423 src += stride;
424 }
425 }
426 }
427 else // inverse
428 {
429 for (unsigned y = 0; y < height; y++)
430 {
431 for (unsigned x = 0; x < width; x++)
432 {
433 src[x] = (Pel)Clip3((Pel)(-maxAbsclipBD - 1), (Pel)maxAbsclipBD, src[x]);
434 sign = src[x] >= 0 ? 1 : -1;
435 absval = sign * src[x];
436 int val = sign * ((absval * scale + (1 << (CSCALE_FP_PREC - 1))) >> CSCALE_FP_PREC);
437 if (sizeof(Pel) == 2) // avoid overflow when storing data
438 {
439 val = Clip3<int>(-32768, 32767, val);
440 }
441 dst[x] = (Pel)val;
442 }
443 dst += stride;
444 src += stride;
445 }
446 }

493
494 const int clipbd = clpRng.bd;

501 }
502 else if (width&2)
503 {
504 for ( int y = 0; y < height; y++ )
505 {
506 for (int x=0 ; x < width; x+=2 )

509 src[x + 1] = ClipPel( rightShift( ( src[x + 1] + offset ), shiftNum ), clpRng );

521 src[x + 1] = ClipPel( rightShift( ( src[x + 1] + offset ), shiftNum ), clpRng );
522 src[x + 2] = ClipPel( rightShift( ( src[x + 2] + offset ), shiftNum ), clpRng );
523 src[x + 3] = ClipPel( rightShift( ( src[x + 3] + offset ), shiftNum ), clpRng );

688void PelStorage::create( const ChromaFormat &_chromaFormat, const Area& _area, const unsigned _maxCUSize, const unsigned _margin, const unsigned _alignment, const bool _scaleChromaMargin )
689{
690 CHECK( !bufs.empty(), "Trying to re-create an already initialized buffer" );
691
692 chromaFormat = _chromaFormat;
693
694 const uint32_t numCh = getNumberValidComponents( _chromaFormat );
695
696 unsigned extHeight = _area.height;
697 unsigned extWidth = _area.width;
698
699 if( _maxCUSize )
700 {
701 extHeight = ( ( _area.height + _maxCUSize - 1 ) / _maxCUSize ) * _maxCUSize;
702 extWidth = ( ( _area.width + _maxCUSize - 1 ) / _maxCUSize ) * _maxCUSize;
703 }
704
705 for( uint32_t i = 0; i < numCh; i++ )
706 {
707 const ComponentID compID = ComponentID( i );
708 const unsigned scaleX = vtm::getComponentScaleX( compID, _chromaFormat );
709 const unsigned scaleY = vtm::getComponentScaleY( compID, _chromaFormat );
710
711 unsigned scaledHeight = extHeight >> scaleY;
712 unsigned scaledWidth = extWidth >> scaleX;
713 unsigned ymargin = _margin >> (_scaleChromaMargin?scaleY:0);
714 unsigned xmargin = _margin >> (_scaleChromaMargin?scaleX:0);
715 unsigned totalWidth = scaledWidth + 2*xmargin;
716 unsigned totalHeight = scaledHeight +2*ymargin;
717
718 if( _alignment )
719 {
720 // make sure buffer lines are align
721 CHECK( _alignment != MEMORY_ALIGN_DEF_SIZE, "Unsupported alignment" );
722 totalWidth = ( ( totalWidth + _alignment - 1 ) / _alignment ) * _alignment;
723 }
724 uint32_t area = totalWidth * totalHeight;
725 CHECK( !area, "Trying to create a buffer with zero area" );
726
727 m_origin[i] = ( Pel* ) xMalloc( Pel, area );
728 Pel* topLeft = m_origin[i] + totalWidth * ymargin + xmargin;
729 bufs.push_back( PelBuf( topLeft, totalWidth, _area.width >> scaleX, _area.height >> scaleY ) );
730 }

814template<>
815void UnitBuf<Pel>::colorSpaceConvert(const UnitBuf<Pel> &other, const bool forward, const ClpRng& clpRng)
816{
817 const Pel* pOrg0 = bufs[COMPONENT_Y].buf;
818 const Pel* pOrg1 = bufs[COMPONENT_Cb].buf;
819 const Pel* pOrg2 = bufs[COMPONENT_Cr].buf;
820 const int strideOrg = bufs[COMPONENT_Y].stride;
821
822 Pel* pDst0 = other.bufs[COMPONENT_Y].buf;
823 Pel* pDst1 = other.bufs[COMPONENT_Cb].buf;
824 Pel* pDst2 = other.bufs[COMPONENT_Cr].buf;
825 const int strideDst = other.bufs[COMPONENT_Y].stride;
826
827 int width = bufs[COMPONENT_Y].width;
828 int height = bufs[COMPONENT_Y].height;
829 int maxAbsclipBD = (1 << (clpRng.bd + 1)) - 1;
830 int r, g, b;
831 int y0, cg, co;
832
833 CHECK(bufs[COMPONENT_Y].stride != bufs[COMPONENT_Cb].stride || bufs[COMPONENT_Y].stride != bufs[COMPONENT_Cr].stride, "unequal stride for 444 content");
834 CHECK(other.bufs[COMPONENT_Y].stride != other.bufs[COMPONENT_Cb].stride || other.bufs[COMPONENT_Y].stride != other.bufs[COMPONENT_Cr].stride, "unequal stride for 444 content");
835 CHECK(bufs[COMPONENT_Y].width != other.bufs[COMPONENT_Y].width || bufs[COMPONENT_Y].height != other.bufs[COMPONENT_Y].height, "unequal block size")
836
837 if (forward)
838 {
839 for (int y = 0; y < height; y++)
840 {
841 for (int x = 0; x < width; x++)
842 {
843 r = pOrg2[x];
844 g = pOrg0[x];
845 b = pOrg1[x];
846
847 co = r - b;
848 int t = b + (co >> 1);
849 cg = g - t;
850 pDst0[x] = t + (cg >> 1);
851 pDst1[x] = cg;
852 pDst2[x] = co;
853 }
854 pOrg0 += strideOrg;
855 pOrg1 += strideOrg;
856 pOrg2 += strideOrg;
857 pDst0 += strideDst;
858 pDst1 += strideDst;
859 pDst2 += strideDst;
860 }
861 }
862 else
863 {
864 for (int y = 0; y < height; y++)
865 {
866 for (int x = 0; x < width; x++)
867 {
868 y0 = pOrg0[x];
869 cg = pOrg1[x];
870 co = pOrg2[x];
871
872 y0 = Clip3((-maxAbsclipBD - 1), maxAbsclipBD, y0);
873 cg = Clip3((-maxAbsclipBD - 1), maxAbsclipBD, cg);
874 co = Clip3((-maxAbsclipBD - 1), maxAbsclipBD, co);
875
876 int t = y0 - (cg >> 1);
877 pDst0[x] = cg + t;
878 pDst1[x] = t - (co >> 1);
879 pDst2[x] = co + pDst1[x];
880 }
881
882 pOrg0 += strideOrg;
883 pOrg1 += strideOrg;
884 pOrg2 += strideOrg;
885 pDst0 += strideDst;
886 pDst1 += strideDst;
887 pDst2 += strideDst;
888 }
889 }
