70HuffmanEncoder HuffmanEncoderBuilder::finish() {
71 std::vector<std::unique_ptr<Node>> nodes;
72 int num_symbols;
73
74 // Build vector of uncoded nodes, sorted by frequency
75 auto node_compare = [](const Node *l, const Node *r) {
76 if (l->count == r->count) {
77 // Equal counts - order t by symbol or node number - leaves are lower than inter nodes
78 return l->symbol < r->symbol;
79 } else
80 return l->count > r->count;
81 };
82 std::priority_queue<Node *, std::vector<Node *>, decltype(node_compare)> heap(node_compare);
83
84 for (unsigned s = 0; s < HuffmanEncoder::MAX_SYMBOL; ++s) {
85 if (symbol_counts_[s]) {
86 nodes.emplace_back(new Node(s, symbol_counts_[s]));
87 heap.emplace(nodes.back().get());
88 }
89 }
90
91 if (heap.empty())
92 // No symbols at all
93 return HuffmanEncoder();
94
95 num_symbols = static_cast<int>(nodes.size());
96
98 //
99
100 // While there is more than 1 pending node:
101 // take 2 smallest by frequency and combine them to make a new internal node
102 while (heap.size() > 1) {
103 Node *l = heap.top();
104 heap.pop();
105 Node *r = heap.top();
106 heap.pop();
107 nodes.emplace_back(new Node(l, r, (unsigned int)nodes.size()));
108 heap.emplace(nodes.back().get());
109 }
110
111 // Starting with the root (last entry in nodes) - walk backwards filling in code lengths
112 assert(heap.top() == nodes.back().get());
113 for (auto n = nodes.rbegin(); n != nodes.rend(); ++n) {
114 if ((*n)->symbol >= HuffmanEncoder::MAX_SYMBOL) {
115 (*n)->left->bits = (*n)->bits + 1;
116 (*n)->right->bits = (*n)->bits + 1;
117 }
118 }
119
120 // The symbol nodes will now have their final code lengths.
121 // Sort leaf nodes by (decending coded length, ascending symbol)
122 std::sort(nodes.begin(), nodes.begin() + num_symbols, [](const std::unique_ptr<Node> &a, const std::unique_ptr<Node> &b) {
123 if (a->bits == b->bits)
124 return a->symbol > b->symbol;
125 else
126 return a->bits < b->bits;
127 });
128
129 // Generate code table for encoder
130 //
131 std::vector<HuffmanEncoder::HuffmanCode> codes;
132 for (auto n = nodes.begin(); n != nodes.begin() + num_symbols; ++n)
133 codes.emplace_back((*n)->symbol, (*n)->bits);
134
135 // Assign values to codes
136 unsigned current_length = codes.back().bits;
137 unsigned current_value = 0;
138
139 for (auto c = codes.rbegin(); c != codes.rend(); ++c) {
140 if (c->bits < current_length) {
141 current_value >>= (current_length - c->bits);
142 current_length = c->bits;
143 }
144 c->value = current_value++;
145 }
146
147 return HuffmanEncoder(codes);
148}

150// Resolve symbol codes and write codes+lengths to bitstream
151void HuffmanEncoder::write_codes(BitstreamPacker &b) {
152 BitstreamPacker::ScopedContextLabel label(b, "entropy_code");
153
154 if (codes_.empty()) {
155 // No symbols
156 b.u(5, 31, "min_code_length");
157 b.u(5, 31, "max_code_length");
158 return;
159 }
160
161 if (codes_.size() == 1) {
162 // Single symbol
163 b.u(5, 0, "min_code_length");
164 b.u(5, 0, "max_code_length");
165 b.u(8, codes_.front().symbol, "single_symbol");
166 return;
167 }
168
169 const unsigned min_code_length = codes_.front().bits;
170 const unsigned max_code_length = codes_.back().bits;
171 unsigned length_bits = HuffmanDecoder::bit_width(max_code_length - min_code_length);
172
173 b.u(5, min_code_length, "min_code_length");
174 b.u(5, max_code_length, "max_code_length");
175
176 if (codes_.size() > 31) {
177 b.u(1, 1, "presence_bitmap");
178
179 // More than 31 coded symbols - write a 'presence' bitmap
180 //
181 unsigned lengths[MAX_SYMBOL] = {0};
182 for (const auto &c : codes_)
183 lengths[c.symbol] = c.bits;
184
185 for (unsigned s = 0; s < MAX_SYMBOL; ++s) {
186 if (lengths[s]) {
187 b.u(1, 1, "presence");
188 b.u(length_bits, lengths[s] - min_code_length, "length");
189 } else {
190 b.u(1, 0, "presence");
191 }
192 }
193 } else {
194 b.u(1, 0, "presence_bitmap");
195
196 // 31 or less coded symbols - Write symbol/length pairs
197 b.u(5, (uint32_t)codes_.size(), "count");
198 for (const auto &c : codes_) {
199 b.u(8, c.symbol, "symbol");
200 b.u(length_bits, c.bits - min_code_length, "length");
201 }
202 }
203}

205// Write a coded symbol to the bitstream
206void HuffmanEncoder::write_symbol(BitstreamPacker &b, unsigned symbol) {
207 BitstreamPacker::ScopedContextLabel label(b, "entropy_symb");
208 // Find symbols in codes
209 for (auto &c : codes_) {
210 if (c.symbol == symbol) {
211 b.u(c.bits, c.value, "codebits");
212 return;
213 }
214 }
215
216 FATAL("Uncoded symbol");
217}
