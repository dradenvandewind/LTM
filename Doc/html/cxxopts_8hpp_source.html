300template <typename R, typename T> R checked_negate(T &&t, const std::string &, std::true_type) {
301 // if we got to here, then `t` is a positive number that fits into
302 // `R`. So to avoid MSVC C4146, we first cast it to `R`.
303 // See https://github.com/jarro2783/cxxopts/issues/62 for more details.
304 return -static_cast<R>(t);
305}

307template <typename R, typename T> T checked_negate(T &&, const std::string &text, std::false_type) {
308 throw argument_incorrect_type(text);
309}

311template <typename T> void integer_parser(const std::string &text, T &value) {
312 std::smatch match;
313 std::regex_match(text, match, integer_pattern);
314
315 if (match.length() == 0) {
316 throw argument_incorrect_type(text);
317 }
318
319 if (match.length(4) > 0) {
320 value = 0;
321 return;
322 }
323
324 using US = typename std::make_unsigned<T>::type;
325
326 constexpr auto umax = std::numeric_limits<US>::max();
327 constexpr bool is_signed = std::numeric_limits<T>::is_signed;
328 const bool negative = match.length(1) > 0;
329 const uint8_t base = match.length(2) > 0 ? 16 : 10;
330
331 auto value_match = match[3];
332
333 US result = 0;
334
335 for (auto iter = value_match.first; iter != value_match.second; ++iter) {
336 US digit = 0;
337
338 if (*iter >= '0' && *iter <= '9') {
339 digit = *iter - '0';
340 } else if (base == 16 && *iter >= 'a' && *iter <= 'f') {
341 digit = *iter - 'a' + 10;
342 } else if (base == 16 && *iter >= 'A' && *iter <= 'F') {
343 digit = *iter - 'A' + 10;
344 } else {
345 throw argument_incorrect_type(text);
346 }
347
348 if (umax - digit < result * base) {
349 throw argument_incorrect_type(text);
350 }
351
352 result = result * base + digit;
353 }
354
355 detail::check_signed_range<T>(negative, result, text);
356
357 if (negative) {
358 value = checked_negate<T>(result, text, std::integral_constant<bool, is_signed>());
359 } else {
360 value = result;
361 }
362}

388inline void parse_value(const std::string &text, bool &value) {
389 std::smatch result;
390 std::regex_match(text, result, truthy_pattern);
391
392 if (!result.empty()) {
393 value = true;
394 return;
395 }
396
397 std::regex_match(text, result, falsy_pattern);
398 if (!result.empty()) {
399 value = false;
400 return;
401 }
402
403 throw argument_incorrect_type(text);
404}

782String format_option(const HelpOptionDetails &o) {
783 auto &s = o.s;
784 auto &l = o.l;
785
786 String result = " ";
787
788 if (s.size() > 0) {
789 result += "-" + toLocalString(s) + ",";
790 } else {
791 result += " ";
792 }
793
794 if (l.size() > 0) {
795 result += " --" + toLocalString(l);
796 }
797
798 auto arg = o.arg_help.size() > 0 ? toLocalString(o.arg_help) : "arg";
799
800 if (!o.is_boolean) {
801 if (o.has_implicit) {
802 result += " [=" + arg + "(=" + toLocalString(o.implicit_value) + ")]";
803 } else {
804 result += " " + arg;
805 }
806 }
807
808 return result;
809}
810
811String format_description(const HelpOptionDetails &o, size_t start, size_t width) {
812 auto desc = o.desc;
813
814 if (o.has_default && (!o.is_boolean || o.default_value != "false")) {
815 desc += toLocalString(" (default: " + o.default_value + ")");
816 }
817
818 String result;
819
820 auto current = std::begin(desc);
821 auto startLine = current;
822 auto lastSpace = current;
823
824 auto size = size_t{};
825
826 while (current != std::end(desc)) {
827 if (*current == ' ') {
828 lastSpace = current;
829 }
830
831 if (size > width) {
832 if (lastSpace == startLine) {
833 stringAppend(result, startLine, current + 1);
834 stringAppend(result, "\n");
835 stringAppend(result, start, ' ');
836 startLine = current + 1;
837 lastSpace = startLine;
838 } else {
839 stringAppend(result, startLine, lastSpace);
840 stringAppend(result, "\n");
841 stringAppend(result, start, ' ');
842 startLine = lastSpace + 1;
843 }
844 size = 0;
845 } else {
846 ++size;
847 }
848
849 ++current;
850 }
851
852 // append whatever is left
853 stringAppend(result, startLine, current);
854
855 return result;
856}

868inline OptionAdder &OptionAdder::operator()(const std::string &opts, const std::string &desc, std::shared_ptr<const Value> value,
869 std::string arg_help) {
870 std::match_results<const char *> result;
871 std::regex_match(opts.c_str(), result, option_specifier);
872
873 if (result.empty()) {
874 throw invalid_option_format_error(opts);
875 }
876
877 const auto &short_match = result[2];
878 const auto &long_match = result[3];
879
880 if (!short_match.length() && !long_match.length()) {
881 throw invalid_option_format_error(opts);
882 } else if (long_match.length() == 1 && short_match.length()) {
883 throw invalid_option_format_error(opts);
884 }
885
886 auto option_names = [](const std::sub_match<const char *> &short_, const std::sub_match<const char *> &long_) {
887 if (long_.length() == 1) {
888 return std::make_tuple(long_.str(), short_.str());
889 } else {
890 return std::make_tuple(short_.str(), long_.str());
891 }
892 }(short_match, long_match);
893
894 m_options.add_option(m_group, std::get<0>(option_names), std::get<1>(option_names), desc, value, std::move(arg_help));
895
896 return *this;
897}

979inline void ParseResult::parse(int &argc, char **&argv) {
980 int current = 1;
981
982 int nextKeep = 1;
983
984 bool consume_remaining = false;
985
986 while (current != argc) {
987 if (strcmp(argv[current], "--") == 0) {
988 consume_remaining = true;
989 ++current;
990 break;
991 }
992
993 std::match_results<const char *> result;
994 std::regex_match(argv[current], result, option_matcher);
995
996 if (result.empty()) {
997 // not a flag
998
999 // if true is returned here then it was consumed, otherwise it is
1000 // ignored
1001 if (consume_positional(argv[current])) {
1002 } else {
1003 argv[nextKeep] = argv[current];
1004 ++nextKeep;
1005 }
1006 // if we return from here then it was parsed successfully, so continue
1007 } else {
1008 // short or long option?
1009 if (result[4].length() != 0) {
1010 const std::string &s = result[4];
1011
1012 for (std::size_t i = 0; i != s.size(); ++i) {
1013 std::string name(1, s[i]);
1014 auto iter = m_options.find(name);
1015
1016 if (iter == m_options.end()) {
1017 if (m_allow_unrecognised) {
1018 continue;
1019 } else {
1020 // error
1021 throw option_not_exists_exception(name);
1022 }
1023 }
1024
1025 auto value = iter->second;
1026
1027 if (i + 1 == s.size()) {
1028 // it must be the last argument
1029 checked_parse_arg(argc, argv, current, value, name);
1030 } else if (value->value().has_implicit()) {
1031 parse_option(value, name, value->value().get_implicit_value());
1032 } else {
1033 // error
1034 throw option_requires_argument_exception(name);
1035 }
1036 }
1037 } else if (result[1].length() != 0) {
1038 const std::string &name = result[1];
1039
1040 auto iter = m_options.find(name);
1041
1042 if (iter == m_options.end()) {
1043 if (m_allow_unrecognised) {
1044 // keep unrecognised options in argument list, skip to next argument
1045 argv[nextKeep] = argv[current];
1046 ++nextKeep;
1047 ++current;
1048 continue;
1049 } else {
1050 // error
1051 throw option_not_exists_exception(name);
1052 }
1053 }
1054
1055 auto opt = iter->second;
1056
1057 // equals provided for long option?
1058 if (result[2].length() != 0) {
1059 // parse the option given
1060
1061 parse_option(opt, name, result[3]);
1062 } else {
1063 // parse the next argument
1064 checked_parse_arg(argc, argv, current, opt, name);
1065 }
1066 }
1067 }
1068
1069 ++current;
1070 }
1071
1072 for (auto &opt : m_options) {
1073 auto &detail = opt.second;
1074 auto &value = detail->value();
1075
1076 auto &store = m_results[detail];
1077
1078 if (!store.count() && value.has_default()) {
1079 parse_default(detail);
1080 }
1081 }
1082
1083 if (consume_remaining) {
1084 while (current < argc) {
1085 if (!consume_positional(argv[current])) {
1086 break;
1087 }
1088 ++current;
1089 }
1090
1091 // adjust argv for any that couldn't be swallowed
1092 while (current != argc) {
1093 argv[nextKeep] = argv[current];
1094 ++nextKeep;
1095 ++current;
1096 }
1097 }
1098
1099 argc = nextKeep;
1100}

1131inline String Options::help_one_group(const std::string &g) const {
1132 typedef std::vector<std::pair<String, String>> OptionHelp;
1133
1134 auto group = m_help.find(g);
1135 if (group == m_help.end()) {
1136 return "";
1137 }
1138
1139 OptionHelp format;
1140
1141 size_t longest = 0;
1142
1143 String result;
1144
1145 if (!g.empty()) {
1146 result += toLocalString(" " + g + " options:\n");
1147 }
1148
1149 for (const auto &o : group->second.options) {
1150 if (o.is_container && m_positional_set.find(o.l) != m_positional_set.end() && !m_show_positional) {
1151 continue;
1152 }
1153
1154 auto s = format_option(o);
1155 longest = std::max(longest, stringLength(s));
1156 format.push_back(std::make_pair(s, String()));
1157 }
1158
1159 longest = std::min(longest, static_cast<size_t>(OPTION_LONGEST));
1160
1161 // widest allowed description
1162 auto allowed = size_t{999} - longest - OPTION_DESC_GAP;
1163
1164 auto fiter = format.begin();
1165 for (const auto &o : group->second.options) {
1166 if (o.is_container && m_positional_set.find(o.l) != m_positional_set.end() && !m_show_positional) {
1167 continue;
1168 }
1169
1170 auto d = format_description(o, longest + OPTION_DESC_GAP, allowed);
1171
1172 result += fiter->first;
1173 if (stringLength(fiter->first) > longest) {
1174 result += '\n';
1175 result += toLocalString(std::string(longest + OPTION_DESC_GAP, ' '));
1176 } else {
1177 result += toLocalString(std::string(longest + OPTION_DESC_GAP - stringLength(fiter->first), ' '));
1178 }
1179 result += d;
1180 result += '\n';
1181
1182 ++fiter;
1183 }
1184
1185 return result;
1186}
