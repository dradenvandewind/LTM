132 // clang-format off
133 options_description.add_options()
134 ("i,input_file", "Input elementary stream filename", cxxopts::value<string>()->default_value("input.lvc"))
135 ("o,output_file", "Output filename for decoded YUV data", cxxopts::value<string>()->default_value("output.yuv"))
136 ("b,base", "Base codec (avc, hevc, evc, vvc, or yuv)", cxxopts::value<string>()->default_value("avc"))
137 ("base_encoder", "Base codec (same as --base)", cxxopts::value<string>()->default_value("avc"))
138 ("base_external", "Use an external base codec executable (select for decoding of monochrome output)", cxxopts::value<bool>()->default_value("false")->implicit_value("true"))
139 ("y,base_yuv", "Prepared YUV data for base decode", cxxopts::value<string>()->default_value(""))
140 ("input_yuv", "Original YUV data for PSNR computation", cxxopts::value<string>()->default_value(""))
141 ("l,limit", "Number of frames to decode", cxxopts::value<unsigned>()->default_value("1000000"))
142 ("dump_surfaces", "Dump intermediate surfaces to yuv files", cxxopts::value<bool>()->default_value("false")->implicit_value("true"))
143 ("encapsulation", "Wrap enhancement as SEI or NAL", cxxopts::value<string>()->default_value("nal"))
144 ("dithering_switch", "Disable decoder dithering independent of configuration in bitstream", cxxopts::value<bool>()->default_value("true"))
145 ("dithering_fixed", "Use a fixed seed for dithering", cxxopts::value<bool>()->default_value("false")->implicit_value("true"))
146 ("report", "Calculate PSNR and checksums", cxxopts::value<bool>()->default_value("false")->implicit_value("true"))
147 ("keep_base", "Keep the base + enhancement bitstreams and base decoded yuv file", cxxopts::value<bool>()->default_value("false")->implicit_value("true"))
148 ("apply_enhancement", "Apply LCEVC enhancement data (residuals) on output YUV", cxxopts::value<bool>()->default_value("true"))
149
150 // Retain additional arguments for backwards compatibility (to be removed in future release)
151 ("w,width", "Placeholder (to be removed in future release)", cxxopts::value<unsigned>()->default_value("1920"))
152 ("h,height", "Placeholder (to be removed in future release)", cxxopts::value<unsigned>()->default_value("1080"))
153 ("f,format", "Placeholder (to be removed in future release)", cxxopts::value<string>()->default_value("yuv420p"))
154 ("base_depth", "Placeholder (to be removed in future release)", cxxopts::value<unsigned>()->default_value("0"))
155
156 ("version", "Show version")
157 ("help", "Show help");
158 // clang-format on
159
160 const bool show_help = argc == 1;
161
162 // Parse command line
163 auto options = options_description.parse(argc, argv);
164
165 // Help
166 if (show_help || options.count("help")) {
167 cout << options_description.help({""}) << std::endl;
168 exit(0);
169 }
170
171 // Version
172 if (options.count("version")) {
173 INFO(GIT_VERSION);
174 exit(0);
175 }
176
177 // Debugging
178 Surface::set_dump_surfaces(options["dump_surfaces"].as<bool>());
179
180 // Base codec
181 //
182 base_video_type = extract<BaseCoding>(options["base"].as<string>());
183 if (options["base_encoder"].count())
184 base_video_type = extract<BaseCoding>(options["base_encoder"].as<string>());
185
186 switch (base_video_type) {
187 case BaseCoding_AVC:
188 file_base = BaseDecoder::AVC;
189 break;
190 case BaseCoding_HEVC:
191 file_base = BaseDecoder::HEVC;
192 break;
193 case BaseCoding_VVC:
194 file_base = BaseDecoder::VVC;
195 break;
196 case BaseCoding_EVC:
197 file_base = BaseDecoder::EVC;
198 break;
199 case BaseCoding_YUV:
200 file_base = BaseDecoder::None;
201 break;
202 default:
203 CHECK(0);
204 }
205
206 base_external = options["base_external"].as<bool>();
207 keep_base = options["keep_base"].as<bool>();
208 apply_enhancement = options["apply_enhancement"].as<bool>();
209
210 // Encapsulation
211 encapsulation = extract<Encapsulation>(options["encapsulation"].as<string>());
212
213 input_es = options["input_file"].as<string>();
214 output_yuv = options["output_file"].as<string>();
215 base_yuv = options["base_yuv"].as<string>();
216 input_yuv = options["input_yuv"].as<string>();
217
218 report = options["report"].as<bool>();
219 dithering_switch = options["dithering_switch"].as<bool>();
220 dithering_fixed = options["dithering_fixed"].as<bool>();
221 limit = options["limit"].as<unsigned>();
222
223 if (base_video_type == BaseCoding_YUV && base_yuv.empty())
224 ERR("No base codec selected and no base yuv file provided.");
225
226 } catch (const cxxopts::OptionException &e) {
227 std::cout << "error parsing options: " << e.what() << std::endl;
228 exit(1);
229 }
230
231 // Input
232 //
233 ESFile es_file;
234 if (!es_file.Open(input_es, file_base)) {
235 ERR("Cannot open file: %s\n", input_es.c_str());
236 }
237
238 // Dummy Output
239 //
240 unique_ptr<YUVWriter> yuv_writer(CreateYUVWriter(output_yuv));
241 // additional Input for PSNR
242 unique_ptr<YUVReader> yuv_reader(CreateYUVReader(input_yuv, 60));
243
244 // Create Base Video Decpder
245 DecoderApp app(*yuv_writer, *yuv_reader);
246 std::unique_ptr<BaseVideoDecoder> base_video_decoder(
247 CreateBaseVideoDecoder(app, base_video_type, encapsulation, base_external, base_yuv, keep_base));
248
249 app.report_ = report;
250 app.dithering_switch_ = dithering_switch;
251 app.dithering_fixed_ = dithering_fixed;
252 app.apply_enhancement_ = apply_enhancement;
253
254 const float start = (float)(system_timestamp() / 1000000.0);
255 INFO("-- Starting: %.3f", start);
256
257 base_video_decoder->start();
258
259#if BITSTREAM_DEBUG
260 goStat.Reset();
261 goBits = fopen("BitstreamDec.txt", "w");
262#endif
263
264 ESFile::AccessUnit au;
265
266 clock_t EnhaClock0;
267 EnhaClock0 = clock();
268 INFO("**** Enh. start %16d", EnhaClock0);
269
270 unsigned count = 0;
271 while (count < limit) {
272 ESFile::Result rc = ESFile::EndOfFile;
273
274 try {
275 rc = es_file.NextAccessUnit(au);
276 } catch (const runtime_error &e) {
277 ERR("End of file: %s\n", e.what());
278 rc = ESFile::NalParsingError;
279 }
280
281 if (rc != ESFile::Success)
282 break;
283
284 // Reassemble AU from NALUs
285 vector<uint8_t> bytes;
286 for (const auto &n : au.m_nalUnits)
287 bytes.insert(bytes.end(), n.m_data.begin(), n.m_data.end());
288
289 // Use derived Picture Order Count to construct a plausible PTS
290 const uint64_t pts = au.m_poc + 1000;
291 const bool is_base_idr = au.m_pictureType == BaseDecPictType::IDR;
292
293 // Push AU into base decoder
294 // INFO("AU: size %8d - pts. %8d", bytes.size(), pts);
295 // REPORT("BASE AU -- pts. %8d size %8d", pts, bytes.size());
296 base_video_decoder->push_au(bytes.data(), bytes.size(), pts, is_base_idr, au.m_pictureType);
297
298 ++count;
299 }
300 INFO("-- Flushing: %.3f", system_timestamp() / 1000000.0);
301
302 base_video_decoder->push_au(0, 0, 0, false, 0);
303
304 clock_t EnhaClock1;
305 EnhaClock1 = clock();
306 INFO("**** Enh. stop. %16d", EnhaClock1);
307 INFO("@@@@ Enh. delta %16d", EnhaClock1 - EnhaClock0);
308
309 INFO("-- Flushed: %.3f", system_timestamp() / 1000000.0);
310
311 const float finish = (float)(system_timestamp() / 1000000.0);
312 INFO("-- Finished: %.3f", finish);
313 INFO("-- FPS: %.3f", (float)count / (finish - start));
314
315 base_video_decoder->StatisticsComputation();
316
317#if BITSTREAM_DEBUG
318 goStat.Dump();
319 fflush(goBits);
320 fclose(goBits);
321#endif
322}

326void DecoderApp::push_base_enhancement(const BaseVideoDecoder::BasePicture *base_picture, const uint8_t *enhancement_data,
327 size_t enhancement_data_size, uint64_t pts, bool is_lcevc_idr) {
328 if (count_ == 0)
329 INFO("-- Decoding: %.3f", system_timestamp() / 1000000.0);
330
331 // Deserilize LCEVC data
332 Surface symbols[MAX_NUM_PLANES][MAX_NUM_LOQS][MAX_NUM_LAYERS];
333 decoder_.initialize_decode(Packet::build().contents(enhancement_data, (unsigned)enhancement_data_size).finish(), symbols);
334 const SignaledConfiguration configuration = decoder_.get_configuration();
335 decoder_.set_idr(is_lcevc_idr);
336
337 if (count_ == 0) {
338 // Set-up Output before decoding first frame
339 unsigned width = configuration.global_configuration.resolution_width;
340 unsigned height = configuration.global_configuration.resolution_height;
341 if (configuration.sequence_configuration.conformance_window) {
342 // Figure out output size after conformance window
343 Dimensions dimensions = decoder_.get_dimensions();
344 const unsigned cw = dimensions.crop_unit_width(0), ch = dimensions.crop_unit_height(0);
345 width -= (configuration.sequence_configuration.conf_win_left_offset * cw +
346 configuration.sequence_configuration.conf_win_right_offset * cw);
347 height -= (configuration.sequence_configuration.conf_win_top_offset * ch +
348 configuration.sequence_configuration.conf_win_bottom_offset * ch);
349 }
350 const ImageDescription image_description(configuration.global_configuration.image_format, width, height);
351 writer_.update_data(image_description);
352 // additional Input for PSNR
353 if (&reader_)
354 reader_.update_data(image_description);
355 if (image_description.colourspace() == Colourspace_Y)
356 ERR("Please use argument '--base_external=true' for decoding with a monochrome output");
357 }
358
359 // Base image
360 //
361 std::vector<Surface> base_planes;
362 const ImageDescription base_desc = writer_.image_description()
363 .with_depth(configuration.global_configuration.base_depth)
364 .with_size(base_picture->width_y, base_picture->height_y);
365 if (configuration.global_configuration.base_depth == 8) {
366 base_planes.push_back(Surface::build_from<int8_t>()
367 .contents((int8_t *)base_picture->data_y, base_picture->width_y, base_picture->height_y)
368 .finish());
369 base_planes.push_back(Surface::build_from<int8_t>()
370 .contents((int8_t *)base_picture->data_u, base_picture->width_uv, base_picture->height_uv)
371 .finish());
372 base_planes.push_back(Surface::build_from<int8_t>()
373 .contents((int8_t *)base_picture->data_v, base_picture->width_uv, base_picture->height_uv)
374 .finish());
375 } else {
376 base_planes.push_back(Surface::build_from<int16_t>()
377 .contents((int16_t *)base_picture->data_y, base_picture->width_y, base_picture->height_y)
378 .finish());
379 base_planes.push_back(Surface::build_from<int16_t>()
380 .contents((int16_t *)base_picture->data_u, base_picture->width_uv, base_picture->height_uv)
381 .finish());
382 base_planes.push_back(Surface::build_from<int16_t>()
383 .contents((int16_t *)base_picture->data_v, base_picture->width_uv, base_picture->height_uv)
384 .finish());
385 }
386
387 // INFO("PTS: %" PRIu64 " PSS: %zu", pts, enhancement_data_size);
388 // REPORT("ENHA AU -- pts. %8d size %8d ", pts, enhancement_data_size);
389
390 if (&reader_) {
391 auto reference_image = ExpandImage(reader_.read(count_), reader_.description());
392 auto full_image = decoder_.decode(Image("base", base_desc, pts, base_planes), symbols, reference_image,
393 report_, dithering_switch_, dithering_fixed_, apply_enhancement_);
394 writer_.write(full_image);
395 } else {
396 Image reference_image;
397 auto full_image = decoder_.decode(Image("base", base_desc, pts, base_planes), symbols, reference_image,
398 report_, dithering_switch_, dithering_fixed_, apply_enhancement_);
399 writer_.write(full_image);
400 }
401
402 count_++;
403}

405void DecoderApp::push_base_enhancement(const uint8_t *base_data, size_t base_data_size,
406 Surface (&symbols)[MAX_NUM_PLANES][MAX_NUM_LOQS][MAX_NUM_LAYERS], uint64_t pts,
407 bool is_lcevc_idr) {
408
409 Dimensions dimensions = decoder_.get_dimensions();
410 const SignaledConfiguration configuration = decoder_.get_configuration();
411 decoder_.set_idr(is_lcevc_idr);
412
413 if (count_ == 0) {
414 INFO("-- Decoding: %.3f", system_timestamp() / 1000000.0);
415
416 // Set-up Output before decoding first frame
417 unsigned width = configuration.global_configuration.resolution_width;
418 unsigned height = configuration.global_configuration.resolution_height;
419 if (configuration.sequence_configuration.conformance_window) {
420 // Figure out output size after conformance window
421 const unsigned cw = dimensions.crop_unit_width(0), ch = dimensions.crop_unit_height(0);
422 width -= (configuration.sequence_configuration.conf_win_left_offset * cw +
423 configuration.sequence_configuration.conf_win_right_offset * cw);
424 height -= (configuration.sequence_configuration.conf_win_top_offset * ch +
425 configuration.sequence_configuration.conf_win_bottom_offset * ch);
426 }
427 const ImageDescription image_description(configuration.global_configuration.image_format, width, height);
428 writer_.update_data(image_description);
429 // additional Input for PSNR
430 if (&reader_)
431 reader_.update_data(image_description);
432 }
433
434 // Base image
435 //
436 std::vector<Surface> base_planes;
437 const ImageDescription base_desc = writer_.image_description()
438 .with_depth(configuration.global_configuration.base_depth)
439 .with_size(dimensions.base_width(), dimensions.base_height());
440
441 if (configuration.global_configuration.base_depth == 8) {
442 int offset = 0;
443 for (unsigned plane = 0; plane < base_desc.num_planes(); plane++) {
444 base_planes.push_back(Surface::build_from<int8_t>()
445 .contents((int8_t *)(base_data + offset), base_desc.width(plane), base_desc.height(plane))
446 .finish());
447 offset += base_desc.width(plane) * base_desc.height(plane);
448 }
449 } else {
450 int offset = 0;
451 for (unsigned plane = 0; plane < base_desc.num_planes(); plane++) {
452 base_planes.push_back(Surface::build_from<int16_t>()
453 .contents((int16_t *)(base_data + offset), base_desc.width(plane), base_desc.height(plane))
454 .finish());
455 offset += base_desc.width(plane) * base_desc.height(plane) * sizeof(int16_t);
456 }
457 }
458
459 // INFO("PTS: %" PRIu64 " PSS: %zu", pts, enhancement_data_size);
460 // REPORT("ENHA AU -- pts. %8d size %8d ", pts, enhancement_data_size);
461
462 Image base("base", base_desc, pts, base_planes);
463
464 if (&reader_) {
465 auto reference_image = ExpandImage(reader_.read(count_), reader_.description());
466 auto full_image = decoder_.decode(base, symbols, reference_image, report_, dithering_switch_, dithering_fixed_,
467 apply_enhancement_);
468 writer_.write(full_image);
469 } else {
470 Image reference_image;
471 auto full_image = decoder_.decode(base, symbols, reference_image, report_, dithering_switch_, dithering_fixed_,
472 apply_enhancement_);
473 writer_.write(full_image);
474 }
475
476 count_++;
477}
