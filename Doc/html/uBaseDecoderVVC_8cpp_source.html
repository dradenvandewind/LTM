59BaseDecNalUnitType::Enum uFromVVC(uint8_t type) {
60 switch (type) {
61 case VVCNalType::CODED_SLICE_TRAIL:
62 case VVCNalType::CODED_SLICE_STSA:
63 case VVCNalType::CODED_SLICE_RADL:
64 case VVCNalType::CODED_SLICE_RASL:
65 case VVCNalType::RESERVED_VCL_4:
66 case VVCNalType::RESERVED_VCL_5:
67 case VVCNalType::RESERVED_VCL_6:
68 case VVCNalType::CODED_SLICE_IDR_W_RADL:
69 case VVCNalType::CODED_SLICE_IDR_N_LP:
70 case VVCNalType::CODED_SLICE_CRA:
71 case VVCNalType::CODED_SLICE_GDR:
72 return BaseDecNalUnitType::Slice;
73
74 case VVCNalType::DCI:

76
77 case VVCNalType::VPS:
78 return BaseDecNalUnitType::VPS;
79 case VVCNalType::SPS:
80 return BaseDecNalUnitType::SPS;
81 case VVCNalType::PPS:
82 return BaseDecNalUnitType::PPS;
83
84 case VVCNalType::PREFIX_APS:
85 case VVCNalType::SUFFIX_APS:
86 case VVCNalType::PH:
87 return BaseDecNalUnitType::Unknown;
88
89 case VVCNalType::ACCESS_UNIT_DELIMITER:
90 return BaseDecNalUnitType::AUD;
91
92 case VVCNalType::EOS:
93 return BaseDecNalUnitType::EOS;
94
95 case VVCNalType::EOB:
96 return BaseDecNalUnitType::EOB;
97
98 case VVCNalType::PREFIX_SEI:
99 case VVCNalType::SUFFIX_SEI:
100 return BaseDecNalUnitType::SEI;
101
102 case VVCNalType::FD:
103 return BaseDecNalUnitType::Filler;
104
105 default:
106 return BaseDecNalUnitType::Unknown;
107 }

110static const uint32_t kNalHeaderLength = 3;
111static const uint8_t kNalHeader[kNalHeaderLength] = {0, 0, 1};
112static const uint32_t kNalEscapeLength = 3;
113static const uint8_t kNalEscape[kNalEscapeLength] = {0, 0, 3};
114
115uint32_t uOffsetForNalUnitHeader(const uint8_t *nal) {
116 if (memcmp(nal, kNalHeader, kNalHeaderLength) == 0)
117 return kNalHeaderLength;
118 if (!nal[0] && memcmp(nal + 1, kNalHeader, kNalHeaderLength) == 0)
119 return kNalHeaderLength + 1;
120 return 0;
121}

132bool BaseDecoderVVC::ParseNalUnit(const uint8_t *nal, uint32_t nalLength) {
133 bool bSuccess = true;
134
135#if defined __BASE_VVC__
136 // Account for AnnexB nal unit header ([0],0,0,1)
137 uint32_t offset = uOffsetForNalUnitHeader(nal);
138
139 m_nuhLayerId = nal[offset+0] & 0x3f;
140 m_nalUnitType = ((nal[offset+1] >> 3) & 0x1F);
141 m_temporalId = (nal[offset + 1] & 0x07) - 1;
142
143 offset += 2;
144
145 nal += offset;
146 nalLength -= offset;
147
148 InputBitstream bitstream;
149 Unencapsulate(nal, nalLength, bitstream.getFifo());
150
151 switch (m_nalUnitType) {
152 case NAL_UNIT_VPS:
153 bSuccess = ParseVPS(bitstream);
154 break;
155 case NAL_UNIT_PPS:
156 bSuccess = ParsePPS(bitstream);
157 break;
158 case NAL_UNIT_SPS:
159 bSuccess = ParseSPS(bitstream);
160 break;
161 case NAL_UNIT_PH:
162 bSuccess = ParsePH(bitstream);
163 break;
164 case NAL_UNIT_PREFIX_APS:
165 case NAL_UNIT_SUFFIX_APS:
166 bSuccess = ParseAPS(bitstream);
167 break;
168
169 case NAL_UNIT_CODED_SLICE_TRAIL:
170 case NAL_UNIT_CODED_SLICE_STSA:
171 case NAL_UNIT_CODED_SLICE_IDR_W_RADL:
172 case NAL_UNIT_CODED_SLICE_IDR_N_LP:
173 case NAL_UNIT_CODED_SLICE_CRA:
174 case NAL_UNIT_CODED_SLICE_GDR:
175 case NAL_UNIT_CODED_SLICE_RADL:
176 case NAL_UNIT_CODED_SLICE_RASL:
177 bSuccess = ParseSliceHeader(bitstream);
178 break;
179
180 case NAL_UNIT_EOS:
181 break;
182 default:
183 break;
184 }
185#endif // defined __BASE_VVC__
186
187 return bSuccess;
188}

300bool BaseDecoderVVC::ParseSliceHeader(InputBitstream &bitstream) {
301#if defined __BASE_VVC__
302 Slice *slice = new Slice();
303 slice->setPicHeader(&m_picHeader);
304 slice->initSlice();
305
306 slice->setNalUnitType((NalUnitType)m_nalUnitType);
307 slice->setNalUnitLayerId(m_nuhLayerId);
308 slice->setTLayer(m_temporalId);
309
310 m_reader.setBitstream(&bitstream);
311 // m_reader.parseSliceHeader(slice, &m_picHeader, &m_parameterSetManager, m_prevTid0POC);
312 m_reader.parseSliceHeader(slice, &m_picHeader, &m_parameterSetManager, m_prevTid0POC, m_prevPOC);
313
314 PPS *pps = m_parameterSetManager.getPPS(m_picHeader.getPPSId());
315 CHECK(pps);
316 SPS *sps = m_parameterSetManager.getSPS(pps->getSPSId());
317 CHECK(sps);
318
319 // Taking in consideration the conformance window
320 m_width = pps->getPicWidthInLumaSamples() -
321 sps->getWinUnitX(sps->getChromaFormatIdc()) *
322 (pps->getConformanceWindow().getWindowLeftOffset() + pps->getConformanceWindow().getWindowRightOffset());
323 m_height = pps->getPicHeightInLumaSamples() -
324 sps->getWinUnitY(sps->getChromaFormatIdc()) *
325 (pps->getConformanceWindow().getWindowTopOffset() + pps->getConformanceWindow().getWindowBottomOffset());
326
327 m_bitdepthLuma = sps->getBitDepth(CHANNEL_TYPE_LUMA);
328 m_bitdepthChroma = sps->getBitDepth(CHANNEL_TYPE_CHROMA);
329 m_chromaFormatIDC = sps->getChromaFormatIdc();
330
331 if (slice->isIDRorBLA())
332 m_basePictureType = BaseDecPictType::IDR;
333 else {
334 switch (slice->getSliceType()) {
335 case B_SLICE:
336 m_basePictureType = BaseDecPictType::B;
337 break;
338 case P_SLICE:
339 m_basePictureType = BaseDecPictType::P;
340 break;
341 case I_SLICE:
342 m_basePictureType = BaseDecPictType::I;
343 break;
344 default:
345 CHECK(0);
346 }
347 }
348
349 m_pictureOrderCount = slice->getPOC();
350
351 m_QP = slice->getSliceQp();
352
353 if (slice->getTLayer() == 0 &&
354 slice->getNalUnitType() != NAL_UNIT_CODED_SLICE_RASL &&
355 slice->getNalUnitType() != NAL_UNIT_CODED_SLICE_RADL) {
356 m_prevTid0POC = slice->getPOC();
357 }
358#endif // defined __BASE_VVC__
359 return true;
360}
