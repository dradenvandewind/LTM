151std::string read_file(const std::string &name) {
152 struct stat stat_buf = {0};
153
154 if (stat(name.c_str(), &stat_buf) < 0)
155 return "";
156
157 UniquePtrFile f(fopen(name.c_str(), "rb"));
158
159 if (!f)
160 return "";
161
162 std::vector<char> tmp(stat_buf.st_size + 1);
163
164 if (fread(tmp.data(), 1, stat_buf.st_size, f.get()) != stat_buf.st_size)
165 return "";
166
167 tmp[stat_buf.st_size] = '\0';
168 return std::string(tmp.begin(), tmp.end());
169}

191unsigned split(std::vector<std::string> &output, const std::string &src, const char *seperators) {
192 bool in_token = false;
193
194 std::string s;
195
196 for (const auto c : src) {
197 if (strchr(seperators, c)) {
198 if (in_token) {
199 // End of token - push accumulated string to output
200 output.push_back(s);
201 in_token = false;
202 } else {
203 // In seperators - do nothing
204 }
205 } else {
206 if (in_token) {
207 // Still in token - accumulate
208 s.push_back(c);
209 } else {
210 // Start of token - start new string
211 s.clear();
212 s.push_back(c);
213 in_token = true;
214 }
215 }
216 }
217
218 // Add any unfinished token
219 if (!s.empty())
220 output.push_back(s);
221
222 return (unsigned)output.size();
223}

233std::string file_extension(const std::string &filename) {
234 size_t p = filename.find_last_of(PATH_SEPERATORS ".");

237 return "";
238
239 return filename.substr(p + 1);
