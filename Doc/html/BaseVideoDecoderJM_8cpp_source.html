82static bool compare_packet_timestamps(const lctm::Packet &l, const lctm::Packet &r) {
83 uint64_t lt = l.timestamp();
84 uint64_t rt = r.timestamp();
85 int64_t d = (rt - lt);
86 return d < 0;
87}
88
89struct BufferCompare {
90 bool operator()(const PssPacket &l, const PssPacket &r) { return compare_packet_timestamps(l.packet, r.packet); }
91};

161void BaseVideoDecoderJM::push_au(const uint8_t *data, size_t data_size, uint64_t pts, bool is_base_idr, int iPictureType) {
162 if (data) {
163 // Temp. copy of AU
164 vector<uint8_t> b(data, data + data_size);
165
166 // Pick out enhancement data - may be removed - new size is returned
167 size_t new_size =
168 scan_enhancement(b.data(), data_size, encapsulation_, BaseCoding_AVC, pts, [is_base_idr, this](const Packet &pkt) {
169 this->enhancement_queue_.push({pkt, is_base_idr});
170 });
171
172 b.resize(new_size);
173
174 goReportStructure.miTimeStamp = (int)(pts);
175 goReportStructure.miPictureType = iPictureType;
176 goReportStructure.miBaseSize = (int)(new_size);
177 goReportStructure.miEnhancementSize = (int)(data_size - new_size);
178 goReportQueue.push(goReportStructure);
179
180 // Flush any previous buffer
181 if (!buffer_.empty()) {
182 push_es(buffer_.data(), buffer_.size(), pts, is_base_idr);
183 }
184
185 buffer_.swap(b);
186 } else {
187 // this comment is specifically to trigger the pipeline (and for Florian) and serves no other purpose
188 if (!buffer_.empty()) {
189 push_es(buffer_.data(), buffer_.size(), pts, is_base_idr);
190 }
191
192 push_es(0, 0, pts, is_base_idr);
193 }
194}

196void BaseVideoDecoderJM::push_es(const uint8_t *data, size_t data_size, uint64_t pts, bool is_base_idr) {
197 CHECK(context_);
198
199 // Push packet into codec
200 CodecError error = 0;
201 codec_->push_packet(context_, data, data_size, 0, data == nullptr, &error);
202
203 // Pull images while codec continues to return them
204 int32_t n = 0;
205 int8_t eos = 0;
206 do {
207 CodecImage codec_image = {0};
208 CodecError error = 0;
209 n = codec_->pull_image(context_, &codec_image, 0, &eos, &error);
210
211 if(n) {
212 // Get next PSS from queue
213 //
214 CHECK(!enhancement_queue_.empty());
215 PssPacket pss = enhancement_queue_.top();
216 enhancement_queue_.pop();
217
218 // Send to enhancement decoder
219 //
220 PacketView enhancement_view(pss.packet);
221 output_.push_base_enhancement(&codec_image, enhancement_view.data(), enhancement_view.size(), pss.packet.timestamp(),
222 pss.is_base_idr);
223 }
224 } while(n);
225}

227void BaseVideoDecoderJM::flush() {
228
229 // Poke bitstream into our JM hooks
230 inject_annex_b_bitstream(base_bitstream_.data(), (unsigned)base_bitstream_.size());
231
232 InputParameters input_parameters;
233 memset(&input_parameters, 0, sizeof(input_parameters));
234 strcpy(input_parameters.infile, "");
235 strcpy(input_parameters.outfile, "");
236 strcpy(input_parameters.reffile, "");
237
238 const char *const decoder_args[] = {"ldecod"};
239 ParseCommand(&input_parameters, ARRAY_SIZE(decoder_args), (char **)decoder_args);
240
241 // Open decoder
242 if (OpenDecoder(&input_parameters) != DEC_OPEN_NOERR) {
243 ERR("Cannot open base decoder");
244 }
245
246 p_Dec->p_Vid->p_out = -2;
247
248 // Run decoder
249 //
250 DecodedPicList *pDecPicList = 0;
251 int frame_count = 0;
252
253 int r = 0;
254 do {
255 r = DecodeOneFrame(&pDecPicList);
256
257 if (r == DEC_EOS || r == DEC_SUCCEED) {
258 decode_enhanced_frames(pDecPicList, false);
259 frame_count++;
260 } else {
261 ERR("Error in decoding process: 0x%x", r);
262 }
263 } while ((r == DEC_SUCCEED) && ((p_Dec->p_Inp->iDecFrmNum == 0) || (frame_count < p_Dec->p_Inp->iDecFrmNum)));
264
265 FinitDecoder(&pDecPicList);
266
267 // Write last frame
268 decode_enhanced_frames(pDecPicList, true);
269
270 CloseDecoder();
271}

273void BaseVideoDecoderJM::decode_enhanced_frames(DecodedPicList *pic, bool all_frames) {
274 // Check picture and plane pointers - valid flag seems to be wrong sometimes!?
275 if (!pic || !pic->pY || !pic->pU || !pic->pV) {
276 return;
277 }
278
279 if (!pic->bValid) {
280 INFO("Picture not marked valid?");
281 }
282
283 do {
284 BasePicture base_picture = {0};
285 base_picture.bpp = ((pic->iBitDepth + 7) >> 3);
286 base_picture.width_y = pic->iWidth;
287 base_picture.height_y = pic->iHeight;
288 base_picture.stride_y = pic->iYBufStride;
289
290 base_picture.width_uv = (pic->iYUVFormat != YUV444) ? (pic->iWidth / 2) : pic->iWidth;
291 base_picture.height_uv = (pic->iYUVFormat == YUV420) ? (pic->iHeight / 2) : pic->iHeight;
292 base_picture.stride_uv = pic->iUVBufStride;
293
294 base_picture.data_y = pic->pY;
295 base_picture.data_u = pic->pU;
296 base_picture.data_v = pic->pV;
297
298 // Get next PSS from queue
299 //
300 CHECK(!enhancement_queue_.empty());
301 PssPacket pss = enhancement_queue_.top();
302 enhancement_queue_.pop();
303
304 // Send to enhancement decoder
305 //
306 PacketView enhancement_view(pss.packet);
307 output_.push_base_enhancement(&base_picture, enhancement_view.data(), enhancement_view.size(), pss.packet.timestamp(),
308 pss.is_base_idr);
309
310 // Mark as used and move to next
311 pic->bValid = 0;
312 pic = pic->pNext;
313 } while (pic != NULL && pic->bValid && all_frames);
314}
