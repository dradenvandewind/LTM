91static bool compare_packet_timestamps(const lctm::Packet &l, const lctm::Packet &r) {
92 uint64_t lt = l.timestamp();
93 uint64_t rt = r.timestamp();
94 int64_t d = (rt - lt);
95 return d < 0;
96}

193void BaseVideoDecoderExternal::push_au(const uint8_t *data, size_t data_size, uint64_t pts, bool is_base_idr, int iPictureType) {
194 if (data) {
195 // Temp. copy of AU
196 vector<uint8_t> b(data, data + data_size);
197
198 // Pick out enhancement data - may be removed - new size is returned
199 size_t new_size = scan_enhancement(b.data(), data_size, encapsulation_, base_coding(), pts, is_base_idr,
200 [this](const Packet &pkt, const bool is_lcevc_idr) {
201 PacketView v(pkt);
202 this->enhancement_queue_.push({pkt, is_lcevc_idr});
203 });
204
205 b.resize(new_size);
206
207 goReportStructure.miTimeStamp = (int)(pts);
208 goReportStructure.miPictureType = iPictureType;
209 goReportStructure.miBaseSize = (int)(new_size);
210 goReportStructure.miEnhancementSize = (int)(data_size - new_size);
211 goReportQueue.push(goReportStructure);
212
213 // Flush any previous buffer
214 if (!buffer_.empty())
215 push_es(buffer_.data(), buffer_.size(), pts, is_base_idr);
216
217 buffer_.swap(b);
218 } else {
219 if (!buffer_.empty()) {
220 push_es(buffer_.data(), buffer_.size(), pts, is_base_idr);
221 }
222
223 flush();
224 }
225}

227void BaseVideoDecoderExternal::push_es(const uint8_t *data, size_t data_size, uint64_t pts, bool is_base_idr) {
228 // Fixup NALU start code - if first is 3 bytes, make it 4
229 if (data[0] == '\0' && data[1] == '\0' && data[2] == '\1') {
230 char const zero = '\0';
231 CHECK(fwrite(&zero, 1, 1, es_file_) == 1);
232 }
233
234 // Accumulate AUs into a temporary file
235 CHECK(fwrite(data, 1, data_size, es_file_) == data_size);
236}

238void BaseVideoDecoderExternal::flush() {
239
240 int iPictureCount = 0;
241
242 if (base_coding() != BaseCoding::BaseCoding_YUV) {
243 // Got end of input - finish up ES file
244 fflush(es_file_);
245 fclose(es_file_);
246 es_file_ = nullptr;
247 }
248
249 while (!enhancement_queue_.empty()) {
250 enhancement_vector_.push_back(std::move(const_cast<PssPacket &>(enhancement_queue_.top())));
251 enhancement_queue_.pop();
252 }
253
254 if (keep_base_) {
255 string es_enhancement_file_name_ = make_temporary_filename("_enhancement.es");
256 FILE *es_enhancement_file_ = CHECK(fopen(es_enhancement_file_name_.c_str(), "wb"));
257 INFO("Using temporary file for enhancement data: '%s'", es_enhancement_file_name_.c_str());
258
259 for (auto const &element : enhancement_vector_) {
260 const uint8_t nal_type = 28 + element.is_lcevc_idr;
261 const uint8_t nalu_header[5] = {0x00, 0x00, 0x01, static_cast<uint8_t>(0x41 | (nal_type << 1)), 0xff};
262 CHECK(fwrite(&nalu_header, 1, sizeof(nalu_header), es_enhancement_file_) == sizeof(nalu_header));
263
264 Packet rbsp = rbsp_encapsulate(element.packet);
265 const PacketView v(rbsp);
266 CHECK(fwrite(v.data(), 1, v.size(), es_enhancement_file_) == v.size());
267 }
268
269 fflush(es_enhancement_file_);
270 fclose(es_enhancement_file_);
271 es_enhancement_file_ = nullptr;
272 }
273
274 Surface symbols_initial[MAX_NUM_PLANES][MAX_NUM_LOQS][MAX_NUM_LAYERS];
275 PssPacket pss_initial;
276 if (!enhancement_vector_.empty()) {
277 // Get PSS from queue
278 pss_initial = enhancement_vector_.front();
279 enhancement_vector_.erase(enhancement_vector_.begin());
280
281 PacketView enhancement_view(pss_initial.packet);
282 output_.deserialize_enhancement(enhancement_view.data(), enhancement_view.size(), symbols_initial);
283 iPictureCount++;
284 } else
285 INFO("No LCEVC Data found in bitstream!");
286
287 // Find base size
288 //
289 unsigned width = 0, height = 0, bit_depth = 0, chroma_format_idc = 0;
290 if (prepared_yuv_file_name_.empty()) {
291 if (!probe_es_file(es_file_name_, base_coding(), width, height, bit_depth, chroma_format_idc)) {
292 ERR("Cannot get base size, depth, and chroma_format_idc");
293 return;
294 }
295 if (iPictureCount > 0) {
296 // Verify base dimensions with signalled LCEVC configuration
297 Dimensions dimensions = output_.get_dimensions();
298 CHECK(bit_depth == output_.get_base_bitdepth());
299 CHECK(width == dimensions.base_width());
300 CHECK(height == dimensions.base_height());
301 CHECK(chroma_format_idc == output_.get_colourspace());
302 }
303 } else {
304 // Get base dimensions from singalled LCEVC configuration
305 if (iPictureCount == 0)
306 ERR("No LCEVC Data and no base bitstream -> Cannot get width/height/format");
307 Dimensions dimensions = output_.get_dimensions();
308 width = dimensions.base_width();
309 height = dimensions.base_height();
310 bit_depth = output_.get_base_bitdepth();
311 chroma_format_idc = output_.get_colourspace();
312 }
313
314 // Figure ESFile format
315 YUVFormat::Enum format;
316
317 switch (bit_depth) {
318 case 8:
319 format = static_cast<YUVFormat::Enum>(0 + (chroma_format_idc == 0 ? 3 : chroma_format_idc - 1));
320 break;
321
322 case 10:
323 format = static_cast<YUVFormat::Enum>(4 + (chroma_format_idc == 0 ? 3 : chroma_format_idc - 1));
324 break;
325
326 case 12:
327 format = static_cast<YUVFormat::Enum>(8 + (chroma_format_idc == 0 ? 3 : chroma_format_idc - 1));
328 break;
329
330 case 14:
331 format = static_cast<YUVFormat::Enum>(12 + (chroma_format_idc == 0 ? 3 : chroma_format_idc - 1));
332 break;
333 default:
334 CHECK(0);
335 break;
336 }
337
338 YUVDesc yuv_desc;
339 yuv_desc.Initialise(format, width, height);
340 size_t base_size = yuv_desc.GetMemorySize();
341
342 const char *fmt = "";
343 YUVFormat::ToString(&fmt, format);
344 INFO("Base is %s %dx%d %d bit (%zu bytes)\n", fmt, width, height, bit_depth, base_size);
345
346 base_width_ = width;
347 base_height_ = height;
348 base_bit_depth_ = bit_depth;
349
350 UniquePtrFile yuv_file;
351
352 if (!prepared_yuv_file_name_.empty()) {
353 // We have been given decoded YUV already
354 yuv_file.reset(CHECK(fopen(prepared_yuv_file_name_.c_str(), "rb")));
355 } else {
356 // Run the decoder esfile->yuvfile
357 //
358 run_decoder(es_file_name_, yuv_file_name_);
359
360 // Open temp. file
361 yuv_file.reset(CHECK(fopen(yuv_file_name_.c_str(), "rb")));
362 }
363
364 // Buffer for the YUV frames
365 vector<uint8_t> base(base_size);
366
367 if (iPictureCount > 0) {
368 // First frame: Copy data through, already deserialized previously
369 CHECK(fread(base.data(), 1, base.size(), yuv_file.get()) == base.size());
370 output_.push_base_enhancement(base.data(), base.size(), symbols_initial, pss_initial.packet.timestamp(),
371 pss_initial.is_lcevc_idr);
372
373 // Consume queue in PTS order
374 while (!enhancement_vector_.empty()) {
375 // Read base
376 CHECK(fread(base.data(), 1, base.size(), yuv_file.get()) == base.size());
377
378 // Get PSS from queue
379 Surface symbols[MAX_NUM_PLANES][MAX_NUM_LOQS][MAX_NUM_LAYERS];
380 PssPacket pss = enhancement_vector_.front();
381 enhancement_vector_.erase(enhancement_vector_.begin());
382 PacketView enhancement_view(pss.packet);
383 output_.deserialize_enhancement(enhancement_view.data(), enhancement_view.size(), symbols);
384 output_.push_base_enhancement(base.data(), base.size(), symbols, pss.packet.timestamp(), pss.is_lcevc_idr);
385 iPictureCount++;
386 }
387
388 int iFrames = iPictureCount;
389
390 float fAccMse[3];
391 float fPsnr[3];
392 for (unsigned plane = 0; plane < yuv_desc.GetPlaneCount(); plane++) {
393 fAccMse[plane] = goPsnr.mfAccMse[plane] / iFrames;
394 fPsnr[plane] = (float)(10.0f * log10((32767.0f * 32767.0f) / fAccMse[plane]));
395 }
396 REPORT("========= ========= ========= ========= ========= ========= ========= ========= ");
397 if (yuv_desc.GetPlaneCount() > 1)
398 REPORT("PSNR -- YUV %8.4f -- Y %8.4f U %8.4f V %8.4f", (6 * fPsnr[0] + fPsnr[1] + fPsnr[2]) / 8, fPsnr[0], fPsnr[1],
399 fPsnr[2]);
400 else
401 REPORT("PSNR -- Y %8.4f", fPsnr[0]);
402 REPORT("========= ========= ========= ========= ========= ========= ========= ========= ");
403 REPORT("BITS -- base %8d bps -- enha %8d bps ", (goPsnr.miBaseBytes * 8 * /* this->fps_ */ 60) / iFrames,
404 (goPsnr.miEnhancementBytes * 8 * /* this->fps_ */ 60) / iFrames);
405 REPORT("========= ========= ========= ========= ========= ========= ========= ========= ");
406 }
407}

422// RBSP encapsualtion (0b00000000 0b00000000 0b000000xx -> 0b00000000 0b00000000 0b00000011 0b000000xx)
423//
424Packet BaseVideoDecoder::rbsp_encapsulate(const Packet &src) const {
425 PacketView in(src);
426
427 vector<uint8_t> out;
428
429 unsigned zeros = 0;
430 for (unsigned int i = 0; i < in.size(); ++i) {
431 uint8_t b = in.data()[i];
432
433 if (zeros == 2 && (b & ~3) == 0) {
434 out.push_back(0x03);
435 zeros = 0;
436 }
437
438 if (b == 0)
439 ++zeros;
440 else
441 zeros = 0;
442
443 out.push_back(b);
444 }
445
446 out.push_back(0x80);
447
448 return Packet::build().contents(out).finish();
449}

563// Factory function
564//
565unique_ptr<BaseVideoDecoder> CreateBaseVideoDecoder(BaseVideoDecoder::Output &output, BaseCoding base, Encapsulation encapsulation,
566 bool external, const std::string &yuv_file, bool keep_base) {
567
568 try {
569 if (external || keep_base || !yuv_file.empty()) {
570 if (base == BaseCoding_AVC)
571 return unique_ptr<BaseVideoDecoder>(new BaseVideoDecoderExternalAVC(output, encapsulation, yuv_file, keep_base));
572 else if (base == BaseCoding_HEVC)
573 return unique_ptr<BaseVideoDecoder>(new BaseVideoDecoderExternalHEVC(output, encapsulation, yuv_file, keep_base));
574 else if (base == BaseCoding_VVC)
575 return unique_ptr<BaseVideoDecoder>(new BaseVideoDecoderExternalVVC(output, encapsulation, yuv_file, keep_base));
576 else if (base == BaseCoding_EVC)
577 return unique_ptr<BaseVideoDecoder>(new BaseVideoDecoderExternalEVC(output, encapsulation, yuv_file, keep_base));
578 else if (base == BaseCoding_YUV)
579 return unique_ptr<BaseVideoDecoder>(new BaseVideoDecoderExternalYUV(output, encapsulation, yuv_file, keep_base));
580 else
581 ERR("Unknown base");
582 } else {
583 if (base == BaseCoding_AVC) {
584#if defined(LTM_ENABLE_CODECAPI_AVC)
585 return CreateBaseVideoDecoderCodecApi(output, encapsulation, base, "avc","","");
586#else
587 return unique_ptr<BaseVideoDecoder>(new BaseVideoDecoderExternalAVC(output, encapsulation, yuv_file, keep_base));
588#endif
589 } else if (base == BaseCoding_HEVC) {
590#if defined(LTM_ENABLE_CODECAPI_HEVC)
591 return CreateBaseVideoDecoderCodecApi(output, encapsulation,base, "hevc","","");
592#else
593 return unique_ptr<BaseVideoDecoder>(new BaseVideoDecoderExternalHEVC(output, encapsulation, yuv_file, keep_base));
594#endif
595 } else if (base == BaseCoding_VVC) {
596#if defined(LTM_ENABLE_CODECAPI_VVC)
597 return CreateBaseVideoDecoderCodecApi(output, encapsulation, base, "vvc","","");
598#else
599 return unique_ptr<BaseVideoDecoder>(new BaseVideoDecoderExternalVVC(output, encapsulation, yuv_file, keep_base));
600#endif
601 } else if (base == BaseCoding_EVC) {
602#if defined(LTM_ENABLE_CODECAPI_EVC)
603 return CreateBaseVideoDecoderCodecApi(output, encapsulation, base, "evc","","");
604#else
605 return unique_ptr<BaseVideoDecoder>(new BaseVideoDecoderExternalEVC(output, encapsulation, yuv_file, keep_base));
606#endif
607 } else {
608 ERR("Unknown base");
609 }
610 }
611 } catch (string e) {
612 ERR("CreateBaseVideoDecoder: Caught: %s\n", e.c_str());
613 }
614 return nullptr;
615}
