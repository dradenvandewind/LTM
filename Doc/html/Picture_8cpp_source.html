362void Picture::finalInit( const VPS* vps, const SPS& sps, const PPS& pps, PicHeader *picHeader, APS** alfApss, APS* lmcsAps, APS* scalingListAps )
363{
364 for( auto &sei : SEIs )
365 {
366 delete sei;
367 }
368 SEIs.clear();
369 clearSliceBuffer();
370
371 const ChromaFormat chromaFormatIDC = sps.getChromaFormatIdc();
372 const int iWidth = pps.getPicWidthInLumaSamples();
373 const int iHeight = pps.getPicHeightInLumaSamples();
374
375 if( cs )
376 {
377 cs->initStructData();
378 }
379 else
380 {
381 cs = new CodingStructure( g_globalUnitCache.cuCache, g_globalUnitCache.puCache, g_globalUnitCache.tuCache );
382 cs->sps = &sps;
383 cs->create(chromaFormatIDC, Area(0, 0, iWidth, iHeight), true, (bool)sps.getPLTMode());
384 }
385
386 cs->vps = vps;
387 cs->picture = this;
388 cs->slice = nullptr; // the slices for this picture have not been set at this point. update cs->slice after swapSliceObject()
389 cs->pps = &pps;
390 picHeader->setSPSId( sps.getSPSId() );
391 picHeader->setPPSId( pps.getPPSId() );
392 cs->picHeader = picHeader;
393 memcpy(cs->alfApss, alfApss, sizeof(cs->alfApss));
394 cs->lmcsAps = lmcsAps;
395 cs->scalinglistAps = scalingListAps;
396 cs->pcv = pps.pcv;
397 m_conformanceWindow = pps.getConformanceWindow();
398 m_scalingWindow = pps.getScalingWindow();
399
400 if (m_spliceIdx == NULL)
401 {
402 m_ctuNums = cs->pcv->sizeInCtus;
403 m_spliceIdx = new int[m_ctuNums];
404 memset(m_spliceIdx, 0, m_ctuNums * sizeof(int));
405 }
406}

424void Picture::fillSliceLossyLosslessArray(std::vector<uint16_t> sliceLosslessIndexArray, bool mixedLossyLossless)
425{
426 uint16_t numElementsinsliceLosslessIndexArray = (uint16_t)sliceLosslessIndexArray.size();
427 uint32_t numSlices = this->cs->pps->getNumSlicesInPic();
428 m_lossylosslessSliceArray.assign(numSlices, true); // initialize to all slices are lossless
429 if (mixedLossyLossless)
430 {
431 m_lossylosslessSliceArray.assign(numSlices, false); // initialize to all slices are lossless
432 CHECK(numElementsinsliceLosslessIndexArray == 0 , "sliceLosslessArray is empty, must need to configure for mixed lossy/lossless");
433
434 // mixed lossy/lossless slices, set only lossless slices;
435 for (uint16_t i = 0; i < numElementsinsliceLosslessIndexArray; i++)
436 {
437 CHECK(sliceLosslessIndexArray[i] >= numSlices || sliceLosslessIndexArray[i] < 0, "index of lossless slice is out of slice index bound");
438 m_lossylosslessSliceArray[sliceLosslessIndexArray[i]] = true;
439 }
440 }
441 CHECK(m_lossylosslessSliceArray.size() < numSlices, "sliceLosslessArray size is less than number of slices");
442}

640void Picture::sampleRateConv( const std::pair<int, int> scalingRatio, const std::pair<int, int> compScale,
641 const CPelBuf& beforeScale, const int beforeScaleLeftOffset, const int beforeScaleTopOffset,
642 const PelBuf& afterScale, const int afterScaleLeftOffset, const int afterScaleTopOffset,
643 const int bitDepth, const bool useLumaFilter, const bool downsampling,
644 const bool horCollocatedPositionFlag, const bool verCollocatedPositionFlag )
645{
646 const Pel* orgSrc = beforeScale.buf;
647 const int orgWidth = beforeScale.width;
648 const int orgHeight = beforeScale.height;
649 const int orgStride = beforeScale.stride;
650
651 Pel* scaledSrc = afterScale.buf;
652 const int scaledWidth = afterScale.width;
653 const int scaledHeight = afterScale.height;
654 const int scaledStride = afterScale.stride;
655
656 if( orgWidth == scaledWidth && orgHeight == scaledHeight && scalingRatio == SCALE_1X && !beforeScaleLeftOffset && !beforeScaleTopOffset && !afterScaleLeftOffset && !afterScaleTopOffset )
657 {
658 for( int j = 0; j < orgHeight; j++ )
659 {
660 memcpy( scaledSrc + j * scaledStride, orgSrc + j * orgStride, sizeof( Pel ) * orgWidth );
661 }
662
663 return;
664 }
665
666 const TFilterCoeff* filterHor = useLumaFilter ? &InterpolationFilter::m_lumaFilter[0][0] : &InterpolationFilter::m_chromaFilter[0][0];
667 const TFilterCoeff* filterVer = useLumaFilter ? &InterpolationFilter::m_lumaFilter[0][0] : &InterpolationFilter::m_chromaFilter[0][0];
668 const int numFracPositions = useLumaFilter ? 15 : 31;
669 const int numFracShift = useLumaFilter ? 4 : 5;
670 const int posShiftX = SCALE_RATIO_BITS - numFracShift + compScale.first;
671 const int posShiftY = SCALE_RATIO_BITS - numFracShift + compScale.second;
672 int addX = ( 1 << ( posShiftX - 1 ) ) + ( beforeScaleLeftOffset << SCALE_RATIO_BITS ) + ( ( int( 1 - horCollocatedPositionFlag ) * 8 * ( scalingRatio.first - SCALE_1X.first ) + ( 1 << ( 2 + compScale.first ) ) ) >> ( 3 + compScale.first ) );
673 int addY = ( 1 << ( posShiftY - 1 ) ) + ( beforeScaleTopOffset << SCALE_RATIO_BITS ) + ( ( int( 1 - verCollocatedPositionFlag ) * 8 * ( scalingRatio.second - SCALE_1X.second ) + ( 1 << ( 2 + compScale.second ) ) ) >> ( 3 + compScale.second ) );
674
675 if( downsampling )
676 {
677 int verFilter = 0;
678 int horFilter = 0;
679
680 if (scalingRatio.first > (15 << SCALE_RATIO_BITS) / 4)
681 {
682 horFilter = 7;
683 }
684 else if (scalingRatio.first > (20 << SCALE_RATIO_BITS) / 7)
685 {
686 horFilter = 6;
687 }
688 else if (scalingRatio.first > (5 << SCALE_RATIO_BITS) / 2)
689 {
690 horFilter = 5;
691 }
692 else if (scalingRatio.first > (2 << SCALE_RATIO_BITS))
693 {
694 horFilter = 4;
695 }
696 else if (scalingRatio.first > (5 << SCALE_RATIO_BITS) / 3)
697 {
698 horFilter = 3;
699 }
700 else if (scalingRatio.first > (5 << SCALE_RATIO_BITS) / 4)
701 {
702 horFilter = 2;
703 }
704 else if (scalingRatio.first > (20 << SCALE_RATIO_BITS) / 19)
705 {
706 horFilter = 1;
707 }
708
709 if (scalingRatio.second > (15 << SCALE_RATIO_BITS) / 4)
710 {
711 verFilter = 7;
712 }
713 else if (scalingRatio.second > (20 << SCALE_RATIO_BITS) / 7)
714 {
715 verFilter = 6;
716 }
717 else if (scalingRatio.second > (5 << SCALE_RATIO_BITS) / 2)
718 {
719 verFilter = 5;
720 }
721 else if (scalingRatio.second > (2 << SCALE_RATIO_BITS))
722 {
723 verFilter = 4;
724 }
725 else if (scalingRatio.second > (5 << SCALE_RATIO_BITS) / 3)
726 {
727 verFilter = 3;
728 }
729 else if (scalingRatio.second > (5 << SCALE_RATIO_BITS) / 4)
730 {
731 verFilter = 2;
732 }
733 else if (scalingRatio.second > (20 << SCALE_RATIO_BITS) / 19)
734 {
735 verFilter = 1;
736 }
737
738 filterHor = &DownsamplingFilterSRC[horFilter][0][0];
739 filterVer = &DownsamplingFilterSRC[verFilter][0][0];
740 }
741
742 const int filterLength = downsampling ? 12 : ( useLumaFilter ? NTAPS_LUMA : NTAPS_CHROMA );
743 const int log2Norm = downsampling ? 14 : 12;
744
745 int *buf = new int[orgHeight * scaledWidth];
746 int maxVal = ( 1 << bitDepth ) - 1;
747
748 CHECK( bitDepth > 17, "Overflow may happen!" );
749
750 for( int i = 0; i < scaledWidth; i++ )
751 {
752 const Pel* org = orgSrc;
753 int refPos = ( ( ( i << compScale.first ) - afterScaleLeftOffset ) * scalingRatio.first + addX ) >> posShiftX;
754 int integer = refPos >> numFracShift;
755 int frac = refPos & numFracPositions;
756 int* tmp = buf + i;
757
758 for( int j = 0; j < orgHeight; j++ )
759 {
760 int sum = 0;
761 const TFilterCoeff* f = filterHor + frac * filterLength;
762
763 for( int k = 0; k < filterLength; k++ )
764 {
765 int xInt = std::min<int>( std::max( 0, integer + k - filterLength / 2 + 1 ), orgWidth - 1 );
766 sum += f[k] * org[xInt]; // postpone horizontal filtering gain removal after vertical filtering
767 }
768
769 *tmp = sum;
770
771 tmp += scaledWidth;
772 org += orgStride;
773 }
774 }
775
776 Pel* dst = scaledSrc;
777
778 for( int j = 0; j < scaledHeight; j++ )
779 {
780 int refPos = ( ( ( j << compScale.second ) - afterScaleTopOffset ) * scalingRatio.second + addY ) >> posShiftY;
781 int integer = refPos >> numFracShift;
782 int frac = refPos & numFracPositions;
783
784 for( int i = 0; i < scaledWidth; i++ )
785 {
786 int sum = 0;
787 int* tmp = buf + i;
788 const TFilterCoeff* f = filterVer + frac * filterLength;
789
790 for( int k = 0; k < filterLength; k++ )
791 {
792 int yInt = std::min<int>( std::max( 0, integer + k - filterLength / 2 + 1 ), orgHeight - 1 );
793 sum += f[k] * tmp[yInt*scaledWidth];
794 }
795
796 dst[i] = std::min<int>( std::max( 0, ( sum + ( 1 << ( log2Norm - 1 ) ) ) >> log2Norm ), maxVal );
797 }
798
799 dst += scaledStride;
800 }
801
802 delete[] buf;
803}
