52static const UpsampleKernel upsample_kernels[] = {
53 {0, 16384, 0, 0}, // Upsample_Nearest
54 {0, 12288, 4096, 0}, // Upsample_Linear
55 {-1382, 14285, 3942, -461}, // Upsample_Cubic
56 {-2360, 15855, 4165, -1276}, // Upsample_ModifiedCubic

59static void make_kernel(UpsampleKernel &kernel, Upsample upsample, const unsigned *coefficients) {
60 CHECK(upsample >= Upsample_Nearest && upsample <= Upsample_AdaptiveCubic);
61
62 if (upsample <= Upsample_ModifiedCubic)
63 for (unsigned i = 0; i < ARRAY_SIZE(kernel); ++i)
64 kernel[i] = upsample_kernels[upsample][i];
65 else {
66 CHECK(upsample == Upsample_AdaptiveCubic);
67 CHECK(coefficients != nullptr);
68
69 kernel[0] = -(int32_t)coefficients[0];
70 kernel[1] = (int32_t)coefficients[1];
71 kernel[2] = (int32_t)coefficients[2];
72 kernel[3] = -(int32_t)coefficients[3];
73 }
74}

89// Specialised for a 4 tap kernel, offset by -1
90//
91static void inline apply_kernel(int16_t *__restrict dest, unsigned dest_stride, const int16_t *__restrict src, unsigned src_stride,
92 unsigned size, const UpsampleKernel &kernel) {
93 if (size == 0)
94 return;
95
96 assert(size >= 4); // Head and tail have been unrolled
97
98 // First sample
99 int32_t ds = 0x2000;
100 for (int k = 0; k < 4; ++k)
101 ds += kernel[3 - k] * src[src_stride * clamp(k - 2, 0, (int)size - 1)];
102
103 *dest = us_shift_clamp_s16(ds);
104 dest += dest_stride;
105
106 // Second pair
107 int32_t ds0 = 0x2000, ds1 = 0x2000;
108 for (int k = 0; k < 4; ++k) {
109 const int32_t sample = src[src_stride * clamp(k - 1, 0, (int)size - 1)];
110 ds0 += kernel[k] * sample;
111 ds1 += kernel[3 - k] * sample;
112 }
113 *dest = us_shift_clamp_s16(ds0);
114 dest += dest_stride;
115 *dest = us_shift_clamp_s16(ds1);
116 dest += dest_stride;
117
118 // Pairs of output samples
119 for (int s = 1; s < (int)(size - 2); ++s) {
120 int32_t d0 = 0x2000, d1 = 0x2000;
121 for (int k = 0; k < 4; ++k) {
122 const int32_t sample = src[src_stride * (s + k - 1)];
123 d0 += kernel[k] * sample;
124 d1 += kernel[3 - k] * sample;
125 }
126 dest[0] = us_shift_clamp_s16(d0);
127 dest[dest_stride] = us_shift_clamp_s16(d1);
128
129 dest += 2 * dest_stride;
130 }
131
132 // Last pair
133 int32_t de0 = 0x2000, de1 = 0x2000;
134 for (int k = 0; k < 4; ++k) {
135 const int32_t sample = src[src_stride * clamp(((int)size - 3) + k, 0, (int)size - 1)];
136 de0 += kernel[k] * sample;
137 de1 += kernel[3 - k] * sample;
138 }
139 *dest = us_shift_clamp_s16(de0);
140 dest += dest_stride;
141 *dest = us_shift_clamp_s16(de1);
142 dest += dest_stride;
143
144 // Last sample
145 int32_t de = 0x2000;
146 for (int k = 0; k < 4; ++k)
147 de += kernel[k] * src[src_stride * clamp(((int)size - 2) + k, 0, (int)size - 1)];
148
149 *dest = us_shift_clamp_s16(de);
150}

152Surface Upsampling::process(const Surface &src_plane, Upsample upsample, const unsigned *coefficients) {
153 const unsigned width = src_plane.width();
154 const unsigned height = src_plane.height();
155
156 UpsampleKernel kernel;
157 make_kernel(kernel, upsample, coefficients);
158
159 // Intermediate is w,2h
160 auto v_src = src_plane.view_as<int16_t>();
161 auto v_dest = Surface::build_from<int16_t>();
162 v_dest.reserve(width, height * 2);
163
164 // Vertical scale
165 //
166 for (unsigned x = 0; x < width; ++x)
167 apply_kernel(v_dest.data(x, 0), width, v_src.data(x, 0), width, height, kernel);
168
169 Surface intermediate = v_dest.finish();
170
171 // Final is 2w,2h
172 auto h_src = intermediate.view_as<int16_t>();
173 auto h_dst = Surface::build_from<int16_t>();
174 h_dst.reserve(width * 2, height * 2);
175
176 // Horizontal scale
177 //
178 for (unsigned y = 0; y < height * 2; ++y)
179 apply_kernel(h_dst.data(0, y), 1, h_src.data(0, y), 1, width, kernel);
180
181 return h_dst.finish();
182}

184Surface Upsampling_1D::process(const Surface &src_plane, Upsample upsample, const unsigned *coefficients) {

193 auto h_dest = Surface::build_from<int16_t>();

201 return h_dest.finish();
202}

204Image UpsampleImage(const Image &src, Upsample upsample, const unsigned upsampling_coefficients[4], ScalingMode scaling_mode) {
205 if (scaling_mode == ScalingMode_None)
206 return src;
207
208 vector<Surface> upsampled_surfaces;
209 for (unsigned p = 0; p < src.description().num_planes(); ++p) {
210 const Surface s = ConvertToInternal().process(src.plane(p), src.description().bit_depth());
211 Surface scaled;
212
213 switch (scaling_mode) {
214 case ScalingMode_1D:
215 scaled = Upsampling_1D().process(s, upsample, upsampling_coefficients);
216 break;
217
218 case ScalingMode_2D:
219 scaled = Upsampling().process(s, upsample, upsampling_coefficients);
220 break;
221
222 default:
223 CHECK(0);
224 break;
225 }
226
227 upsampled_surfaces.push_back(ConvertFromInternal().process(scaled, src.description().bit_depth()));
228 }
229
230 const ImageDescription us_desc(src.description().format(), upsampled_surfaces[0].width(), upsampled_surfaces[0].height());
231
232 return Image("upsampled", us_desc, src.timestamp(), upsampled_surfaces);
233}
