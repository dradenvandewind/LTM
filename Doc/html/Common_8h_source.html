52 Position() : x(0), y(0) { }

54
55 bool operator!=(const Position &other) const { return x != other.x || y != other.y; }
56 bool operator==(const Position &other) const { return x == other.x && y == other.y; }
57
58 Position offset(const Position pos) const { return Position(x + pos.x, y + pos.y); }
59 Position offset(const PosType _x, const PosType _y) const { return Position(x + _x , y + _y ); }

148inline size_t rsAddr(const Position &pos, const uint32_t stride, const UnitScale &unitScale )
149{
150 return (size_t)(stride >> unitScale.posx) * (size_t)(pos.y >> unitScale.posy) + (size_t)(pos.x >> unitScale.posx);
151}
152
153inline size_t rsAddr(const Position &pos, const Position &origin, const uint32_t stride, const UnitScale &unitScale )
154{
155 return (stride >> unitScale.posx) * ((pos.y - origin.y) >> unitScale.posy) + ((pos.x - origin.x) >> unitScale.posx);
156}

168inline Area clipArea(const Area &_area, const Area &boundingBox)
169{
170 Area area = _area;
171
172 if (area.x + area.width > boundingBox.x + boundingBox.width)
173 {
174 area.width = boundingBox.x + boundingBox.width - area.x;
175 }
176
177 if (area.y + area.height > boundingBox.y + boundingBox.height)
178 {
179 area.height = boundingBox.y + boundingBox.height - area.y;
180 }
181
182 return area;
183}

186class SizeIndexInfo
187{
188public:
189 SizeIndexInfo(){}
190 virtual ~SizeIndexInfo(){}
191 SizeType numAllWidths() { return (SizeType)m_idxToSizeTab.size(); }
192 SizeType numAllHeights() { return (SizeType)m_idxToSizeTab.size(); }
193 SizeType numWidths() { return (SizeType)m_numBlkSizes; }
194 SizeType numHeights() { return (SizeType)m_numBlkSizes; }
195 SizeType sizeFrom( SizeType idx ) { return m_idxToSizeTab[idx]; }
196 SizeType idxFrom( SizeType size ) { CHECKD( m_sizeToIdxTab[size] == std::numeric_limits<SizeType>::max(), "Index of given size does NOT EXIST!" ); return m_sizeToIdxTab[size]; }
197 bool isCuSize( SizeType size ) { return m_isCuSize[size]; }
198 virtual void init( SizeType maxSize ) {}
199
200protected:
201
202 void xInit()
203 {
204 m_isCuSize.resize( m_sizeToIdxTab.size(), false );
205
206 std::vector<SizeType> grpSizes;
207
208 for( int i = 0, n = 0; i < m_sizeToIdxTab.size(); i++ )
209 {
210 if( m_sizeToIdxTab[i] != std::numeric_limits<SizeType>::max() )
211 {
212 m_sizeToIdxTab[i] = n;
213 m_idxToSizeTab.push_back( i );
214 n++;
215 }
216
217 if( m_sizeToIdxTab[i] != std::numeric_limits<SizeType>::max() && m_sizeToIdxTab[i >> 1] != std::numeric_limits<SizeType>::max() && i >= 4 )
218 {
219 m_isCuSize[i] = true;
220 }
221
222 // collect group sizes (for coefficient group coding)
223 SizeType grpSize = i >> ( ( i & 3 ) != 0 ? 1 : 2 );
224 if( m_sizeToIdxTab[i] != std::numeric_limits<SizeType>::max() && m_sizeToIdxTab[grpSize] == std::numeric_limits<SizeType>::max() )
225 {
226 grpSizes.push_back( grpSize );
227 }
228 }
229
230 m_numBlkSizes = (SizeType)m_idxToSizeTab.size();
231
232 for( SizeType grpSize : grpSizes )
233 {
234 if( grpSize > 0 && m_sizeToIdxTab[grpSize] == std::numeric_limits<SizeType>::max() )
235 {
236 m_sizeToIdxTab[grpSize] = (SizeType)m_idxToSizeTab.size();
237 m_idxToSizeTab.push_back( grpSize );
238 }
239 }
240 };
241
242 std::vector<bool > m_isCuSize;
243 int m_numBlkSizes; // as opposed to number all sizes, which also contains grouped sizes
244 std::vector<SizeType> m_sizeToIdxTab;
245 std::vector<SizeType> m_idxToSizeTab;
246};
