96SplitSeries Partitioner::getSplitSeries() const
97{
98 SplitSeries splitSeries = 0;
99 SplitSeries depth = 0;
100
101 for( const auto &level : m_partStack )
102 {
103 if( level.split == CTU_LEVEL ) continue;
104 else splitSeries += static_cast< SplitSeries >( level.split ) << ( depth * SPLIT_DMULT );
105
106 depth++;
107 }
108
109 return splitSeries;
110}
111
112ModeTypeSeries Partitioner::getModeTypeSeries() const
113{
114 ModeTypeSeries modeTypeSeries = 0;
115 int depth = 0;
116
117 for( const auto &level : m_partStack )
118 {
119 if( level.split == CTU_LEVEL ) continue;
120 else modeTypeSeries += static_cast<int>(level.modeType) << (depth * 3);
121
122 depth++;
123 }
124
125 return modeTypeSeries;
126}

171void AdaptiveDepthPartitioner::setMaxMinDepth( unsigned& minDepth, unsigned& maxDepth, const CodingStructure& cs ) const
172{
173 unsigned stdMinDepth = 0;
174 unsigned stdMaxDepth = ( floorLog2(cs.sps->getCTUSize()) - floorLog2(cs.sps->getMinQTSize( cs.slice->getSliceType(), chType )));
175 const Position pos = currArea().blocks[chType].pos();
176 const unsigned curSliceIdx = cs.slice->getIndependentSliceIdx();
177 const unsigned curTileIdx = cs.pps->getTileIdx( currArea().lumaPos() );
178
179 const CodingUnit* cuLeft = cs.getCURestricted( pos.offset( -1, 0 ), pos, curSliceIdx, curTileIdx, chType );
180 const CodingUnit* cuBelowLeft = cs.getCURestricted( pos.offset( -1, currArea().blocks[chType].height), pos, curSliceIdx, curTileIdx, chType );
181 const CodingUnit* cuAbove = cs.getCURestricted( pos.offset( 0, -1 ), pos, curSliceIdx, curTileIdx, chType );
182 const CodingUnit* cuAboveRight = cs.getCURestricted( pos.offset( currArea().blocks[chType].width, -1 ), pos, curSliceIdx, curTileIdx, chType );
183
184 minDepth = stdMaxDepth;
185 maxDepth = stdMinDepth;
186
187 if( cuLeft )
188 {
189 minDepth = std::min<unsigned>( minDepth, cuLeft->qtDepth );
190 maxDepth = std::max<unsigned>( maxDepth, cuLeft->qtDepth );
191 }
192 else
193 {
194 minDepth = stdMinDepth;
195 maxDepth = stdMaxDepth;
196 }
197
198 if( cuBelowLeft )
199 {
200 minDepth = std::min<unsigned>( minDepth, cuBelowLeft->qtDepth );
201 maxDepth = std::max<unsigned>( maxDepth, cuBelowLeft->qtDepth );
202 }
203 else
204 {
205 minDepth = stdMinDepth;
206 maxDepth = stdMaxDepth;
207 }
208
209 if( cuAbove )
210 {
211 minDepth = std::min<unsigned>( minDepth, cuAbove->qtDepth );
212 maxDepth = std::max<unsigned>( maxDepth, cuAbove->qtDepth );
213 }
214 else
215 {
216 minDepth = stdMinDepth;
217 maxDepth = stdMaxDepth;
218 }
219
220 if( cuAboveRight )
221 {
222 minDepth = std::min<unsigned>( minDepth, cuAboveRight->qtDepth );
223 maxDepth = std::max<unsigned>( maxDepth, cuAboveRight->qtDepth );
224 }
225 else
226 {
227 minDepth = stdMinDepth;
228 maxDepth = stdMaxDepth;
229 }
230
231 minDepth = ( minDepth >= 1 ? minDepth - 1 : 0 );
232 maxDepth = std::min<unsigned>( stdMaxDepth, maxDepth + 1 );
233}

271void QTBTPartitioner::splitCurrArea( const PartSplit split, const CodingStructure& cs )
272{
273 CHECKD( !canSplit( split, cs ), "Trying to apply a prohibited split!" );
274
275 bool isImplicit = isSplitImplicit( split, cs );
276 bool canQtSplit = canSplit( CU_QUAD_SPLIT, cs );
277 bool qgEnable = currQgEnable();
278 bool qgChromaEnable = currQgChromaEnable();
279
280 switch( split )
281 {
282 case CU_QUAD_SPLIT:
283 m_partStack.push_back( PartLevel( split, PartitionerImpl::getCUSubPartitions( currArea(), cs ) ) );
284 m_partStack.back().modeType = modeType;
285 break;
286 case CU_HORZ_SPLIT:
287 case CU_VERT_SPLIT:
288 m_partStack.push_back( PartLevel( split, PartitionerImpl::getCUSubPartitions( currArea(), cs, split ) ) );
289 m_partStack.back().modeType = modeType;
290 break;
291 case CU_TRIH_SPLIT:
292 case CU_TRIV_SPLIT:
293 m_partStack.push_back( PartLevel( split, PartitionerImpl::getCUSubPartitions( currArea(), cs, split ) ) );
294 m_partStack.back().modeType = modeType;
295 break;
296 case TU_MAX_TR_SPLIT:
297 m_partStack.push_back( PartLevel( split, PartitionerImpl::getMaxTuTiling( currArea(), cs ) ) );
298 break;
299 case SBT_VER_HALF_POS0_SPLIT:
300 case SBT_VER_HALF_POS1_SPLIT:
301 case SBT_HOR_HALF_POS0_SPLIT:
302 case SBT_HOR_HALF_POS1_SPLIT:
303 case SBT_VER_QUAD_POS0_SPLIT:
304 case SBT_VER_QUAD_POS1_SPLIT:
305 case SBT_HOR_QUAD_POS0_SPLIT:
306 case SBT_HOR_QUAD_POS1_SPLIT:
307 m_partStack.push_back( PartLevel( split, PartitionerImpl::getSbtTuTiling( currArea(), cs, split ) ) );
308 break;
309 default:
310 THROW( "Unknown split mode" );
311 break;
312 }
313
314 currDepth++;
315 currSubdiv++;
316#if _DEBUG
317 m_currArea = m_partStack.back().parts.front();
318#endif
319
320 if( split == TU_MAX_TR_SPLIT )
321 {
322 currTrDepth++;
323 }
324 else if( split >= SBT_VER_HALF_POS0_SPLIT && split <= SBT_HOR_QUAD_POS1_SPLIT )
325 {
326 currTrDepth++;
327 }
328 else
329 {
330 currTrDepth = 0;
331 }
332
333 if( split == CU_HORZ_SPLIT || split == CU_VERT_SPLIT || split == CU_TRIH_SPLIT || split == CU_TRIV_SPLIT )
334 {
335 currBtDepth++;
336 if( isImplicit ) currImplicitBtDepth++;
337 currMtDepth++;
338
339 if( split == CU_TRIH_SPLIT || split == CU_TRIV_SPLIT )
340 {
341 // first and last part of triple split are equivalent to double bt split
342 currBtDepth++;
343 currSubdiv++;
344 }
345 m_partStack.back().canQtSplit = canQtSplit;
346 }
347 else if( split == CU_QUAD_SPLIT )
348 {
349 CHECK( currBtDepth > 0, "Cannot split a non-square area other than with a binary split" );
350 CHECK( currMtDepth > 0, "Cannot split a non-square area other than with a binary split" );
351 currMtDepth = 0;
352 currBtDepth = 0;
353 currQtDepth++;
354 currSubdiv++;
355 }
356 qgEnable &= (currSubdiv <= cs.slice->getCuQpDeltaSubdiv());
357 qgChromaEnable &= (currSubdiv <= cs.slice->getCuChromaQpOffsetSubdiv());
358 m_partStack.back().qgEnable = qgEnable;
359 m_partStack.back().qgChromaEnable = qgChromaEnable;
360 if (qgEnable)
361 currQgPos = currArea().lumaPos();
362 if (qgChromaEnable)
363 currQgChromaPos = currArea().chromaPos();
364}

366void QTBTPartitioner::canSplit( const CodingStructure &cs, bool& canNo, bool& canQt, bool& canBh, bool& canBv, bool& canTh, bool& canTv )
367{
368 const PartSplit implicitSplit = m_partStack.back().checkdIfImplicit ? m_partStack.back().implicitSplit : getImplicitSplit( cs );
369
370 const unsigned maxBTD = cs.pcv->getMaxBtDepth( *cs.slice, chType ) + currImplicitBtDepth;
371 const unsigned maxBtSize = cs.pcv->getMaxBtSize ( *cs.slice, chType );
372 const unsigned minBtSize = cs.pcv->getMinBtSize ( *cs.slice, chType );
373 const unsigned maxTtSize = cs.pcv->getMaxTtSize ( *cs.slice, chType );
374 const unsigned minTtSize = cs.pcv->getMinTtSize ( *cs.slice, chType );
375 const unsigned minQtSize = cs.pcv->getMinQtSize ( *cs.slice, chType );
376
377 canNo = canQt = canBh = canTh = canBv = canTv = true;
378 bool canBtt = currMtDepth < maxBTD;
379
380 // the minimal and maximal sizes are given in luma samples
381 const CompArea& area = currArea().Y();
382 const CompArea *areaC = (chType == CHANNEL_TYPE_CHROMA) ? &(currArea().Cb()) : nullptr;
383 PartLevel& level = m_partStack.back();
384
385 const PartSplit lastSplit = level.split;
386 const PartSplit parlSplit = lastSplit == CU_TRIH_SPLIT ? CU_HORZ_SPLIT : CU_VERT_SPLIT;
387
388 // don't allow QT-splitting below a BT split
389 if( lastSplit != CTU_LEVEL && lastSplit != CU_QUAD_SPLIT ) canQt = false;
390 if( area.width <= minQtSize ) canQt = false;
391
392 if( areaC && areaC->width <= MIN_DUALTREE_CHROMA_WIDTH ) canQt = false;
393 if( treeType == TREE_C )
394 {
395 canQt = canBh = canTh = canBv = canTv = false;
396 return;
397 }
398 if( implicitSplit != CU_DONT_SPLIT )
399 {
400 canNo = canTh = canTv = false;
401
402 canBh = implicitSplit == CU_HORZ_SPLIT;
403 canBv = implicitSplit == CU_VERT_SPLIT;
404 if (areaC && areaC->width == 4) canBv = false;
405 if( !canBh && !canBv && !canQt ) canQt = true;
406 return;
407 }
408
409 if( ( lastSplit == CU_TRIH_SPLIT || lastSplit == CU_TRIV_SPLIT ) && currPartIdx() == 1 )
410 {
411 canBh = parlSplit != CU_HORZ_SPLIT;
412 canBv = parlSplit != CU_VERT_SPLIT;
413 }
414
415 if( canBtt && ( area.width <= minBtSize && area.height <= minBtSize )
416 && ( ( area.width <= minTtSize && area.height <= minTtSize ) ) )
417 {
418 canBtt = false;
419 }
420 if( canBtt && ( area.width > maxBtSize || area.height > maxBtSize )
421 && ( ( area.width > maxTtSize || area.height > maxTtSize ) ) )
422 {
423 canBtt = false;
424 }
425
426 if( !canBtt )
427 {
428 canBh = canTh = canBv = canTv = false;
429
430 return;
431 }
432
433 if( area.width > maxBtSize || area.height > maxBtSize )
434 {
435 canBh = canBv = false;
436 }
437
438 // specific check for BT splits
439 if( area.height <= minBtSize ) canBh = false;
440 if( area.width > MAX_TB_SIZEY && area.height <= MAX_TB_SIZEY ) canBh = false;
441 if( areaC && areaC->width * areaC->height <= MIN_DUALTREE_CHROMA_SIZE ) canBh = false;
442 if( area.width <= minBtSize ) canBv = false;
443 if( area.width <= MAX_TB_SIZEY && area.height > MAX_TB_SIZEY ) canBv = false;
444 if (areaC && (areaC->width * areaC->height <= MIN_DUALTREE_CHROMA_SIZE || areaC->width == 4)) canBv = false;
445 if( modeType == MODE_TYPE_INTER && area.width * area.height == 32 ) canBv = canBh = false;
446 if( area.height <= 2 * minTtSize || area.height > maxTtSize || area.width > maxTtSize )
447 canTh = false;
448 if( area.width > MAX_TB_SIZEY || area.height > MAX_TB_SIZEY ) canTh = false;
449 if( areaC && areaC->width * areaC->height <= MIN_DUALTREE_CHROMA_SIZE*2 ) canTh = false;
450 if( area.width <= 2 * minTtSize || area.width > maxTtSize || area.height > maxTtSize )
451 canTv = false;
452 if( area.width > MAX_TB_SIZEY || area.height > MAX_TB_SIZEY ) canTv = false;
453 if (areaC && (areaC->width * areaC->height <= MIN_DUALTREE_CHROMA_SIZE * 2 || areaC->width == 8)) canTv = false;
454 if( modeType == MODE_TYPE_INTER && area.width * area.height == 64 ) canTv = canTh = false;
455
456}

458bool QTBTPartitioner::canSplit( const PartSplit split, const CodingStructure &cs )
459{
460 const CompArea area = currArea().Y();
461 const unsigned maxTrSize = cs.sps->getMaxTbSize();
462
463 bool canNo, canQt, canBh, canTh, canBv, canTv;
464
465 canSplit( cs, canNo, canQt, canBh, canBv, canTh, canTv );
466 switch( split )
467 {
468 case CTU_LEVEL:
469 THROW( "Checking if top level split is possible" );
470 return true;
471 break;
472 case TU_MAX_TR_SPLIT:
473 return area.width > maxTrSize || area.height > maxTrSize;
474 break;
475 case SBT_VER_HALF_POS0_SPLIT:
476 case SBT_VER_HALF_POS1_SPLIT:
477 case SBT_HOR_HALF_POS0_SPLIT:
478 case SBT_HOR_HALF_POS1_SPLIT:
479 case SBT_VER_QUAD_POS0_SPLIT:
480 case SBT_VER_QUAD_POS1_SPLIT:
481 case SBT_HOR_QUAD_POS0_SPLIT:
482 case SBT_HOR_QUAD_POS1_SPLIT:
483 return currTrDepth == 0;
484 break;
485 case CU_QUAD_SPLIT:
486 return canQt;
487 case CU_DONT_SPLIT:
488 return canNo;
489 case CU_HORZ_SPLIT:
490 return canBh;
491 case CU_VERT_SPLIT:
492 return canBv;
493 case CU_TRIH_SPLIT:
494 return canTh;
495 case CU_TRIV_SPLIT:
496 return canTv;
497 case CU_MT_SPLIT:
498 return ( canBh || canTh || canBv || canTv );
499 case CU_BT_SPLIT:
500 return ( canBh || canBv );
501 break;
502 default:
503 THROW( "Unknown split mode" );
504 return false;
505 break;
506 }
507
508 return true;
509}

516PartSplit QTBTPartitioner::getImplicitSplit( const CodingStructure &cs )
517{
518 if( m_partStack.back().checkdIfImplicit )
519 {
520 return m_partStack.back().implicitSplit;
521 }
522
523 PartSplit split = CU_DONT_SPLIT;
524
525 if( split == CU_DONT_SPLIT )
526 {
527 const bool isBlInPic = cs.picture->Y().contains( currArea().Y().bottomLeft() );
528 const bool isTrInPic = cs.picture->Y().contains( currArea().Y().topRight() );
529
530 const CompArea& area = currArea().Y();
531 const unsigned maxBtSize = cs.pcv->getMaxBtSize( *cs.slice, chType );
532 const bool isBtAllowed = area.width <= maxBtSize && area.height <= maxBtSize && currMtDepth < (cs.pcv->getMaxBtDepth(*cs.slice, chType) + currImplicitBtDepth);
533 const unsigned minQtSize = cs.pcv->getMinQtSize( *cs.slice, chType );
534 const bool isQtAllowed = area.width > minQtSize && area.height > minQtSize && currBtDepth == 0;
535
536 if( !isBlInPic && !isTrInPic && isQtAllowed )
537 {
538 split = CU_QUAD_SPLIT;
539 }
540 else if( !isBlInPic && isBtAllowed && area.width <= MAX_TB_SIZEY )
541 {
542 split = CU_HORZ_SPLIT;
543 }
544 else if( !isTrInPic && isBtAllowed && area.height <= MAX_TB_SIZEY )
545 {
546 split = CU_VERT_SPLIT;
547 }
548 else if( !isBlInPic || !isTrInPic )
549 {
550 split = CU_QUAD_SPLIT;
551 }
552 if (CS::isDualITree(cs) && (currArea().Y().width > 64 || currArea().Y().height > 64))
553 {
554 split = CU_QUAD_SPLIT;
555 }
556 if( (!isBlInPic || !isTrInPic) && split == CU_DONT_SPLIT )
557 {
558 split = CU_QUAD_SPLIT;
559 }
560 }
561
562 m_partStack.back().checkdIfImplicit = true;
563 m_partStack.back().isImplicit = split != CU_DONT_SPLIT;
564 m_partStack.back().implicitSplit = split;
565
566 return split;
567}

569void QTBTPartitioner::exitCurrSplit()
570{
571 PartSplit currSplit = m_partStack.back().split;
572 unsigned currIdx = m_partStack.back().idx;
573
574 m_partStack.pop_back();
575
576 CHECK( currDepth == 0, "depth is '0', although a split was performed" );
577 currDepth--;
578 currSubdiv--;
579 if( currQgEnable() )
580 currQgPos = currArea().lumaPos();
581 if( currArea().chromaFormat != CHROMA_400 && currQgChromaEnable() )
582 currQgChromaPos = currArea().chromaPos();
583#if _DEBUG
584 m_currArea = m_partStack.back().parts[m_partStack.back().idx];
585#endif
586
587 if( currSplit == CU_HORZ_SPLIT || currSplit == CU_VERT_SPLIT || currSplit == CU_TRIH_SPLIT || currSplit == CU_TRIV_SPLIT )
588 {
589 CHECK( !m_partStack.back().checkdIfImplicit, "Didn't check if the current split is implicit" );
590 CHECK( currBtDepth == 0, "BT depth is '0', athough a BT split was performed" );
591 CHECK( currMtDepth == 0, "MT depth is '0', athough a BT split was performed" );
592 currMtDepth--;
593 if( m_partStack.back().isImplicit ) currImplicitBtDepth--;
594 currBtDepth--;
595 if( ( currSplit == CU_TRIH_SPLIT || currSplit == CU_TRIV_SPLIT ) && currIdx != 1 )
596 {
597 CHECK( currBtDepth == 0, "BT depth is '0', athough a TT split was performed" );
598 currBtDepth--;
599 currSubdiv--;
600 }
601 }
602 else if( currSplit == TU_MAX_TR_SPLIT )
603 {
604 CHECK( currTrDepth == 0, "TR depth is '0', although a TU split was performed" );
605 currTrDepth--;
606 }
607 else if( currSplit >= SBT_VER_HALF_POS0_SPLIT && currSplit <= SBT_HOR_QUAD_POS1_SPLIT )
608 {
609 CHECK( currTrDepth == 0, "TR depth is '0', although a TU split was performed" );
610 currTrDepth--;
611 }
612 else
613 {
614 CHECK( currTrDepth > 0, "RQT found with QTBT partitioner" );
615
616 CHECK( currQtDepth == 0, "QT depth is '0', although a QT split was performed" );
617 currQtDepth--;
618 currSubdiv--;
619 }
620}

622bool QTBTPartitioner::nextPart( const CodingStructure &cs, bool autoPop /*= false*/ )
623{
624 const Position &prevPos = currArea().blocks[chType].pos();
625
626 unsigned currIdx = ++m_partStack.back().idx;
627
628 m_partStack.back().checkdIfImplicit = false;
629 m_partStack.back().isImplicit = false;
630
631 if( currIdx == 1 )
632 {
633 const CodingUnit* prevCU = cs.getCU( prevPos, chType );
634 m_partStack.back().firstSubPartSplit = prevCU ? CU::getSplitAtDepth( *prevCU, currDepth ) : CU_DONT_SPLIT;
635 }
636
637 if( currIdx < m_partStack.back().parts.size() )
638 {
639 if( m_partStack.back().split == CU_TRIH_SPLIT || m_partStack.back().split == CU_TRIV_SPLIT )
640 {
641 // adapt the current bt depth

771Partitioning PartitionerImpl::getCUSubPartitions( const UnitArea &cuArea, const CodingStructure &cs, const PartSplit _splitType /*= CU_QUAD_SPLIT*/ )
772{
773 const PartSplit splitType = _splitType;
774
775 if( splitType == CU_QUAD_SPLIT )
776 {
777 if( !cs.pcv->noChroma2x2 )
778 {
779 Partitioning sub;
780
781 sub.resize( 4, cuArea );
782
783 for( uint32_t i = 0; i < 4; i++ )
784 {
785 for( auto &blk : sub[i].blocks )
786 {
787 blk.height >>= 1;
788 blk.width >>= 1;
789 if( i >= 2 ) blk.y += blk.height;
790 if( i & 1 ) blk.x += blk.width;
791 }
792
793 CHECK( sub[i].lumaSize().height < MIN_TB_SIZEY, "the split causes the block to be smaller than the minimal TU size" );
794 }
795
796 return sub;
797 }
798 else
799 {
800 const uint32_t minCUSize = 1 << cs.sps->getLog2MinCodingBlockSize();
801
802 bool canSplit = cuArea.lumaSize().width > minCUSize && cuArea.lumaSize().height > minCUSize;
803
804 Partitioning ret;
805
806
807 if( canSplit )
808 {
809 ret.resize( 4 );
810
811 if( cuArea.chromaFormat == CHROMA_400 )
812 {
813 CompArea blkY = cuArea.Y();
814 blkY.width >>= 1;
815 blkY.height >>= 1;
816 ret[0] = UnitArea( cuArea.chromaFormat, blkY );
817 blkY.x += blkY.width;
818 ret[1] = UnitArea( cuArea.chromaFormat, blkY );
819 blkY.x -= blkY.width;
820 blkY.y += blkY.height;
821 ret[2] = UnitArea( cuArea.chromaFormat, blkY );
822 blkY.x += blkY.width;
823 ret[3] = UnitArea( cuArea.chromaFormat, blkY );
824 }
825 else
826 {
827 for( uint32_t i = 0; i < 4; i++ )
828 {
829 ret[i] = cuArea;
830
831 CompArea &blkY = ret[i].Y();
832 CompArea &blkCb = ret[i].Cb();
833 CompArea &blkCr = ret[i].Cr();
834
835 blkY.width /= 2;
836 blkY.height /= 2;
837
838 // TODO: get those params from SPS
839 if( blkCb.width > 4 )
840 {
841 blkCb.width /= 2;
842 blkCb.height /= 2;
843 blkCr.width /= 2;
844 blkCr.height /= 2;
845 }
846 else if( i > 0 )
847 {
848 blkCb = CompArea();
849 blkCr = CompArea();
850 }
851
852 if( ( i & 1 ) == 1 )
853 {
854 blkY.x += blkY .width;
855 blkCb.x += blkCb.width;
856 blkCr.x += blkCr.width;
857 }
858
859 if( i > 1 )
860 {
861 blkY.y += blkY .height;
862 blkCb.y += blkCb.height;
863 blkCr.y += blkCr.height;
864 }
865 }
866 }
867 }
868
869 return ret;
870 }
871 }
872 else if( splitType == CU_HORZ_SPLIT )
873 {
874 Partitioning sub;
875
876 sub.resize(2, cuArea);
877
878 for (uint32_t i = 0; i < 2; i++)
879 {
880 for (auto &blk : sub[i].blocks)
881 {
882 blk.height >>= 1;
883 if (i == 1) blk.y += blk.height;
884 }
885
886 CHECK(sub[i].lumaSize().height < MIN_TB_SIZEY, "the cs split causes the block to be smaller than the minimal TU size");
887 }
888
889 return sub;
890 }
891 else if( splitType == CU_VERT_SPLIT )
892 {
893 Partitioning sub;
894
895 sub.resize( 2, cuArea );
896
897 for( uint32_t i = 0; i < 2; i++ )
898 {
899 for( auto &blk : sub[i].blocks )
900 {
901 blk.width >>= 1;
902 if( i == 1 ) blk.x += blk.width;
903 }
904
905 CHECK( sub[i].lumaSize().width < MIN_TB_SIZEY, "the split causes the block to be smaller than the minimal TU size" );
906 }
907
908 return sub;
909 }
910 else if( splitType == CU_TRIH_SPLIT )
911 {
912 Partitioning sub;
913
914 sub.resize( 3, cuArea );
915
916 for( int i = 0; i < 3; i++ )
917 {
918 for( auto &blk : sub[i].blocks )
919 {
920 blk.height >>= 1;
921 if( ( i + 1 ) & 1 ) blk.height >>= 1;
922 if( i == 1 ) blk.y += blk.height / 2;
923 if( i == 2 ) blk.y += 3 * blk.height;
924 }
925
926 CHECK( sub[i].lumaSize().height < MIN_TB_SIZEY, "the cs split causes the block to be smaller than the minimal TU size" );
927 }
928
929 return sub;
930 }
931 else if( splitType == CU_TRIV_SPLIT )
932 {
933 Partitioning sub;
934
935 sub.resize( 3, cuArea );
936
937 for( int i = 0; i < 3; i++ )
938 {
939 for( auto &blk : sub[i].blocks )
940 {
941 blk.width >>= 1;
942
943 if( ( i + 1 ) & 1 ) blk.width >>= 1;
944 if( i == 1 ) blk.x += blk.width / 2;
945 if( i == 2 ) blk.x += 3 * blk.width;
946 }
947
948 CHECK( sub[i].lumaSize().width < MIN_TB_SIZEY, "the cs split causes the block to be smaller than the minimal TU size" );
949 }
950
951 return sub;
952 }
953 else
954 {
955 THROW( "Unknown CU sub-partitioning" );
956 return Partitioning();
957 }

960void PartitionerImpl::getTUIntraSubPartitions( Partitioning &sub, const UnitArea &tuArea, const CodingStructure &cs, const PartSplit splitType )
961{
962 uint32_t nPartitions;
963 uint32_t splitDimensionSize = CU::getISPSplitDim( tuArea.lumaSize().width, tuArea.lumaSize().height, splitType );
964
965 bool isDualTree = CS::isDualITree( cs ) || cs.treeType != TREE_D;
966
967 if( splitType == TU_1D_HORZ_SPLIT )
968 {
969 nPartitions = tuArea.lumaSize().height >> floorLog2(splitDimensionSize);
970
971 sub.resize( nPartitions );
972
973 for( uint32_t i = 0; i < nPartitions; i++ )
974 {
975 sub[i] = tuArea;
976 CompArea& blkY = sub[i].blocks[COMPONENT_Y];
977
978 blkY.height = splitDimensionSize;
979 blkY.y = i > 0 ? sub[i - 1].blocks[COMPONENT_Y].y + splitDimensionSize : blkY.y;
980
981 CHECK( sub[i].lumaSize().height < 1, "the cs split causes the block to be smaller than the minimal TU size" );
982 }
983 }
984 else if( splitType == TU_1D_VERT_SPLIT )
985 {
986 nPartitions = tuArea.lumaSize().width >> floorLog2(splitDimensionSize);
987
988 sub.resize( nPartitions );
989
990 for( uint32_t i = 0; i < nPartitions; i++ )
991 {
992 sub[i] = tuArea;
993 CompArea& blkY = sub[i].blocks[COMPONENT_Y];
994
995 blkY.width = splitDimensionSize;
996 blkY.x = i > 0 ? sub[i - 1].blocks[COMPONENT_Y].x + splitDimensionSize : blkY.x;
997 CHECK( sub[i].lumaSize().width < 1, "the split causes the block to be smaller than the minimal TU size" );
998 }
999 }
1000 else
1001 {
1002 THROW( "Unknown TU sub-partitioning" );
1003 }
1004 //we only partition luma, so there is going to be only one chroma tu at the end (unless it is dual tree, in which case there won't be any chroma components)
1005 uint32_t partitionsWithoutChroma = (cs.area.chromaFormat == CHROMA_400) ? 0 : (isDualTree ? nPartitions : nPartitions - 1);
1006 for( uint32_t i = 0; i < partitionsWithoutChroma; i++ )
1007 {
1008 CompArea& blkCb = sub[i].blocks[COMPONENT_Cb];
1009 CompArea& blkCr = sub[i].blocks[COMPONENT_Cr];
1010 blkCb = CompArea();
1011 blkCr = CompArea();
1012 }
1013}

1051Partitioning PartitionerImpl::getMaxTuTiling( const UnitArea &cuArea, const CodingStructure &cs )
1052{
1053 static_assert( MAX_LOG2_DIFF_CU_TR_SIZE <= g_maxRtGridSize, "Z-scan tables are only provided for MAX_LOG2_DIFF_CU_TR_SIZE for up to 3 (8x8 tiling)!" );
1054
1055 const Size area = cuArea.lumaSize();
1056 const int maxTrSize = (area.width>64 || area.height>64) ? 64 : cs.sps->getMaxTbSize();
1057 const int numTilesH = std::max<int>( 1, area.width / maxTrSize );
1058 const int numTilesV = std::max<int>( 1, area.height / maxTrSize );
1059 const int numTiles = numTilesH * numTilesV;
1060
1061 CHECK( numTiles > MAX_CU_TILING_PARTITIONS, "CU partitioning requires more partitions than available" );
1062
1063 Partitioning ret;
1064 ret.resize( numTiles, cuArea );
1065
1066 for( int i = 0; i < numTiles; i++ )
1067 {
1068 const int rsy = i / numTilesH;
1069 const int rsx = i % numTilesH;
1070
1071 const int x = g_zScanToX[g_rsScanToZ[( rsy << g_maxRtGridSize ) + rsx]];
1072 const int y = g_zScanToY[g_rsScanToZ[( rsy << g_maxRtGridSize ) + rsx]];
1073
1074 UnitArea& tile = ret[i];
1075
1076 for( CompArea &comp : tile.blocks )
1077 {
1078 if( !comp.valid() ) continue;
1079
1080 comp.width /= numTilesH;
1081 comp.height /= numTilesV;
1082
1083 comp.x += comp.width * x;
1084 comp.y += comp.height * y;
1085 }
1086 }
1087
1088 return ret;
1089}
