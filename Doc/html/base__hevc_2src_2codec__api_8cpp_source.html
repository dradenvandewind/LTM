121static int32_t push_packet(CodecContext c, const uint8_t *data, size_t length, CodecMetadata metadata, int8_t eos, CodecError *error) {
122 Context *context = (Context *)c;
123 if(error)
124 *error = 0;
125
126 switch(context->state_) {
127 case StatePushing:
128 if(!eos) {
129 CHECK(data);
130 CHECK(length >= 3);
131 // Fixup NALU start code - if first is 3 bytes, make it 4
132 if (data[0] == '\0' && data[1] == '\0' && data[2] == '\1') {
133 char const zero = '\0';
134 context->base_bitstream_.put(0);
135 }
136
137 // Accumulate AUs into a temporary buffer
138 context->base_bitstream_.write((const char *)data, length);
139 } else {
141 context->state_ = StateDecoding;
142 }
143 return 0;
144
145 default:
146 ERR("Unexpected decoder state");
147 }
148 return 0;
149}

151static int32_t pull_image(CodecContext c, CodecImage* image, CodecMetadata *metadata, int8_t* eos, CodecError *error){
152 Context *context = (Context *)c;
153 if(error)
154 *error = 0;
155
156 switch(context->state_) {
157 case StatePushing:
158 // Still accumulating input stream
159 *eos = 0;
160 return 0;
161
162 case StateDecoding:
163 context->m_cTDecTop.create();
164 context->m_cTDecTop.init();
165
166 context->decoder_thread_ = std::thread([context] {
167 context->flush();
168 context->decoded_images_.push(nullptr);
169 });
170
171 context->state_ = StateFlushing;
172 /* Fall through */
173
174 case StateFlushing:
175 context->decoded_images_.pop(context->decoded_image_);
176 if(context->decoded_image_) {
177 *image = context->decoded_image_->image_;
178 *eos = 0;
179 return 1;
180 } else {
181 if(context->decoder_thread_.joinable())
182 context->decoder_thread_.join();
183 *eos = 1;
184 return 0;
185 }
186 default:
187 ERR("Unexpected decoder state");
188 }
189 return 0;
190}

193void Context::flush() {
194 Bool openedRecon = false;
195 bool loopFiltered = false;
196 int poc = 0;
197 TComList<TComPic *> *pcListPic = NULL;
198 InputByteStream bytestream(base_bitstream_);
199
200 while (!!base_bitstream_) {
201 streampos location = base_bitstream_.tellg();
202
203 AnnexBStats stats = AnnexBStats();
204 InputNALUnit nalu;
205 byteStreamNALUnit(bytestream, nalu.getBitstream().getFifo(), stats);
206
207 // call actual decoding function
208 Bool bNewPicture = false;
209 if (nalu.getBitstream().getFifo().empty()) {
210 WARN("Attempt to decode an empty NAL unit");
211 } else {
212 read(nalu);
213 int skip = 0;
214 bNewPicture = m_cTDecTop.decode(nalu, skip, m_iPOCLastDisplay);
215
216 if (bNewPicture) {
217 base_bitstream_.clear();
218 base_bitstream_.seekg(location - streamoff(3));
219 bytestream.reset();
220 }
221 }
222
223 if ((bNewPicture || !base_bitstream_ || nalu.m_nalUnitType == NAL_UNIT_EOS) && !m_cTDecTop.getFirstSliceInSequence()) {
224 if (!loopFiltered || base_bitstream_) {
225 m_cTDecTop.executeLoopFilters(poc, pcListPic);
226 }
227 loopFiltered = (nalu.m_nalUnitType == NAL_UNIT_EOS);
228 if (nalu.m_nalUnitType == NAL_UNIT_EOS) {
229 m_cTDecTop.setFirstSliceInSequence(true);
230 }
231 } else if ((bNewPicture || !base_bitstream_ || nalu.m_nalUnitType == NAL_UNIT_EOS) &&
232 m_cTDecTop.getFirstSliceInSequence()) {
233 m_cTDecTop.setFirstSliceInPicture(true);
234 }
235
236 if (pcListPic) {
237 if (!openedRecon) {
238 const BitDepths &bitDepths =
239 pcListPic->front()->getPicSym()->getSPS().getBitDepths(); // use bit depths of first reconstructed picture.
240 for (UInt channelType = 0; channelType < MAX_NUM_CHANNEL_TYPE; channelType++) {
241 if (m_outputBitDepth[channelType] == 0) {
242 m_outputBitDepth[channelType] = bitDepths.recon[channelType];
243 }
244 }
245
246 m_cTVideoIOYuvRecon.open(this, m_outputBitDepth, m_outputBitDepth, bitDepths.recon); // write mode
247 openedRecon = true;
248 }
249
250 // write reconstruction to file
251 if (bNewPicture) {
252 WriteOutput(pcListPic, nalu.m_temporalId);
253 }
254 if ((bNewPicture || nalu.m_nalUnitType == NAL_UNIT_CODED_SLICE_CRA) && m_cTDecTop.getNoOutputPriorPicsFlag()) {
255 m_cTDecTop.checkNoOutputPriorPics(pcListPic);
256 m_cTDecTop.setNoOutputPriorPicsFlag(false);
257 }
258 if (bNewPicture &&
259 (nalu.m_nalUnitType == NAL_UNIT_CODED_SLICE_IDR_W_RADL || nalu.m_nalUnitType == NAL_UNIT_CODED_SLICE_IDR_N_LP ||
260 nalu.m_nalUnitType == NAL_UNIT_CODED_SLICE_BLA_N_LP || nalu.m_nalUnitType == NAL_UNIT_CODED_SLICE_BLA_W_RADL ||
261 nalu.m_nalUnitType == NAL_UNIT_CODED_SLICE_BLA_W_LP)) {
262 FlushOutput(pcListPic);
263 }
264 if (nalu.m_nalUnitType == NAL_UNIT_EOS) {
265 WriteOutput(pcListPic, nalu.m_temporalId);
266 m_cTDecTop.setFirstSliceInPicture(false);
267 }
268 // write reconstruction to file -- for additional bumping as defined in C.5.2.3
269 if (!bNewPicture && nalu.m_nalUnitType >= NAL_UNIT_CODED_SLICE_TRAIL_N &&
270 nalu.m_nalUnitType <= NAL_UNIT_RESERVED_VCL31) {
271 WriteOutput(pcListPic, nalu.m_temporalId);
272 }
273 }
274 }
275
276 FlushOutput(pcListPic);
277
278 // delete buffers
279 m_cTDecTop.deletePicBuffer();
280}

285void Context::WriteOutput(TComList<TComPic *> *pcListPic, unsigned int tId) {
286 if (pcListPic->empty()) {
287 return;
288 }
289
290 TComList<TComPic *>::iterator iterPic = pcListPic->begin();
291 Int numPicsNotYetDisplayed = 0;
292 Int dpbFullness = 0;
293 const TComSPS *activeSPS = &(pcListPic->front()->getPicSym()->getSPS());
294 UInt numReorderPicsHighestTid;
295 UInt maxDecPicBufferingHighestTid;
296 UInt maxNrSublayers = activeSPS->getMaxTLayers();
297
298 numReorderPicsHighestTid = activeSPS->getNumReorderPics(maxNrSublayers - 1);
299 maxDecPicBufferingHighestTid = activeSPS->getMaxDecPicBuffering(maxNrSublayers - 1);
300
301 while (iterPic != pcListPic->end()) {
302 TComPic *pcPic = *(iterPic);
303 if (pcPic->getOutputMark() && pcPic->getPOC() > m_iPOCLastDisplay) {
304 numPicsNotYetDisplayed++;
305 dpbFullness++;
306 } else if (pcPic->getSlice(0)->isReferenced()) {
307 dpbFullness++;
308 }
309 iterPic++;
310 }
311
312 iterPic = pcListPic->begin();
313
314 if (numPicsNotYetDisplayed > 2) {
315 iterPic++;
316 }
317
318 TComPic *pcPic = *(iterPic);
319 if (numPicsNotYetDisplayed > 2 && pcPic->isField()) // Field Decoding
320 {
321 TComList<TComPic *>::iterator endPic = pcListPic->end();
322 endPic--;
323 iterPic = pcListPic->begin();
324 while (iterPic != endPic) {
325 TComPic *pcPicTop = *(iterPic);
326 iterPic++;
327 TComPic *pcPicBottom = *(iterPic);
328
329 if (pcPicTop->getOutputMark() && pcPicBottom->getOutputMark() &&
330 (numPicsNotYetDisplayed > numReorderPicsHighestTid || dpbFullness > maxDecPicBufferingHighestTid) &&
331 (!(pcPicTop->getPOC() % 2) && pcPicBottom->getPOC() == pcPicTop->getPOC() + 1) &&
332 (pcPicTop->getPOC() == m_iPOCLastDisplay + 1 || m_iPOCLastDisplay < 0)) {
333 // write to file
334 numPicsNotYetDisplayed = numPicsNotYetDisplayed - 2;
335 const Window &conf = pcPicTop->getConformanceWindow();
336 const Window defDisp = pcPicTop->getDefDisplayWindow();
337 const Bool isTff = pcPicTop->isTopField();
338
339 m_cTVideoIOYuvRecon.write(pcPicTop->getPicYuvRec(), pcPicBottom->getPicYuvRec(),
340 IPCOLOURSPACE_UNCHANGED /*m_outputColourSpaceConvert*/,
341 conf.getWindowLeftOffset() + defDisp.getWindowLeftOffset(),
342 conf.getWindowRightOffset() + defDisp.getWindowRightOffset(),
343 conf.getWindowTopOffset() + defDisp.getWindowTopOffset(),
344 conf.getWindowBottomOffset() + defDisp.getWindowBottomOffset(), NUM_CHROMA_FORMAT, isTff);
345
346 // update POC of display order
347 m_iPOCLastDisplay = pcPicBottom->getPOC();
348
349 // erase non-referenced picture in the reference picture list after display
350 if (!pcPicTop->getSlice(0)->isReferenced() && pcPicTop->getReconMark() == true) {
351 pcPicTop->setReconMark(false);
352
353 // mark it should be extended later
354 pcPicTop->getPicYuvRec()->setBorderExtension(false);
355 }
356 if (!pcPicBottom->getSlice(0)->isReferenced() && pcPicBottom->getReconMark() == true) {
357 pcPicBottom->setReconMark(false);
358
359 // mark it should be extended later
360 pcPicBottom->getPicYuvRec()->setBorderExtension(false);
361 }
362 pcPicTop->setOutputMark(false);
363 pcPicBottom->setOutputMark(false);
364 }
365 }
366 } else if (!pcPic->isField()) // Frame Decoding
367 {
368 iterPic = pcListPic->begin();
369
370 while (iterPic != pcListPic->end()) {
371 pcPic = *(iterPic);
372
373 if (pcPic->getOutputMark() && pcPic->getPOC() > m_iPOCLastDisplay &&
374 (numPicsNotYetDisplayed > numReorderPicsHighestTid || dpbFullness > maxDecPicBufferingHighestTid)) {
375 // write to file
376 numPicsNotYetDisplayed--;
377 if (pcPic->getSlice(0)->isReferenced() == false) {
378 dpbFullness--;
379 }
380
381 const Window &conf = pcPic->getConformanceWindow();
382 const Window defDisp = true /*m_respectDefDispWindow*/ ? pcPic->getDefDisplayWindow() : Window();
383
384 m_cTVideoIOYuvRecon.write(pcPic->getPicYuvRec(), IPCOLOURSPACE_UNCHANGED /*m_outputColourSpaceConvert*/,
385 conf.getWindowLeftOffset() + defDisp.getWindowLeftOffset(),
386 conf.getWindowRightOffset() + defDisp.getWindowRightOffset(),
387 conf.getWindowTopOffset() + defDisp.getWindowTopOffset(),
388 conf.getWindowBottomOffset() + defDisp.getWindowBottomOffset(), NUM_CHROMA_FORMAT,
389 false /*m_bClipOutputVideoToRec709Range */);
390
391 // update POC of display order
392 m_iPOCLastDisplay = pcPic->getPOC();
393
394 // erase non-referenced picture in the reference picture list after display
395 if (!pcPic->getSlice(0)->isReferenced() && pcPic->getReconMark() == true) {
396 pcPic->setReconMark(false);
397
398 // mark it should be extended later
399 pcPic->getPicYuvRec()->setBorderExtension(false);
400 }
401 pcPic->setOutputMark(false);
402 }
403
404 iterPic++;
405 }
406 }

515void Context::picture_begin(bool is16bit, UInt width, UInt height, ChromaFormat format) {
516 output_frame_++;
517 size_t frame_size = (width * height + 2 * (width / 2 * height / 2)) * (is16bit ? 2 : 1);
518
519 CHECK(!writing_image_);
520
521 writing_image_ = std::shared_ptr<DecodedImage>(new DecodedImage());
522
523 writing_image_->buffer_.reserve(frame_size);
524
525 writing_image_->image_.bpp = is16bit ? 2 : 1;
526 writing_image_->image_.width_y = width;
527 writing_image_->image_.height_y = height;
528 writing_image_->image_.stride_y = width * writing_image_->image_.bpp;
529
530 writing_image_->image_.width_uv = width / 2;
531 writing_image_->image_.height_uv = height / 2;
532 writing_image_->image_.stride_uv = (width / 2) * writing_image_->image_.bpp;
533}
